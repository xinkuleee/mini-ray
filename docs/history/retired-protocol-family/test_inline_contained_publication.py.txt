"""Pure contracts for INLINE contained-result publication identity."""

from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor
from dataclasses import replace

import pytest

from miniray.contained_cycle import (
    ContainedGraphManifest,
    ContainedGraphManifestDisposition,
    ContainedGraphManifestReceipt,
    ContainedGraphTransactionState,
)
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.inline_contained_publication import (
    InlineContainedPublicationReducer,
    InlineEdgeEffect,
    InlinePublicationConflictError,
    InlinePublicationEnvelope,
    InlinePublicationID,
    InlinePublicationManifest,
    InlinePublicationState,
    InlinePublicationStateError,
    UnknownInlinePublicationError,
)
from miniray.protocol import ResultDescriptor, ResultStorage
from miniray.stored_publication import StoredPublicationID


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self, edge_count: int = 2) -> None:
        self.job_id = JobID.random()
        self.task_id = TaskID.derive(
            self.job_id, TaskID.for_driver(self.job_id), 0
        )
        self.attempt_id = AttemptID(self.task_id, 0)
        self.outer_object_id = ObjectID.for_task(self.task_id)
        self.lease_id = LeaseID.random()
        self.executor = WorkerID.random()
        self.outer_owner = WorkerID.random()
        self.node_id = NodeID.random()
        self.publication_id = InlinePublicationID(
            self.lease_id, self.task_id, self.attempt_id,
            self.outer_object_id,
        )
        self.edges = tuple(self._edge(index) for index in range(edge_count))
        self.payload = b"inline outer with contained ObjectRefs"
        self.envelope = InlinePublicationEnvelope.create(
            self.publication_id, self.job_id, self.executor,
            self.outer_owner, self.node_id, self.payload, self.edges,
        )
        self.graph = self.envelope.to_graph_manifest()

    def _edge(self, index: int) -> ContainedReferenceEdge:
        child_task = TaskID.derive(
            self.job_id, TaskID.for_driver(self.job_id), index + 10
        )
        return ContainedReferenceEdge(
            self.outer_object_id, ObjectID.for_task(child_task),
            self.executor, ("127.0.0.1", 28000 + index),
            f"inline-transfer:{index}",
        )


def _pins_ready(
    fixture: _Fixture, reducer: InlineContainedPublicationReducer
) -> None:
    assert reducer.prepare(fixture.envelope, _prepare_receipt(fixture))
    for index in range(len(fixture.edges)):
        assert reducer.mark_pin_installed(
            fixture.envelope.edge_effect(index)
        )


def _committed(
    fixture: _Fixture, reducer: InlineContainedPublicationReducer
) -> None:
    _pins_ready(fixture, reducer)
    assert reducer.begin_commit(fixture.envelope)
    assert reducer.commit(fixture.envelope, _commit_receipt(fixture))


def _prepare_receipt(fixture: _Fixture) -> ContainedGraphManifestReceipt:
    return ContainedGraphManifestReceipt(
        fixture.graph, ContainedGraphTransactionState.PREPARED,
        ContainedGraphManifestDisposition.APPLIED,
    )


def _commit_receipt(fixture: _Fixture) -> ContainedGraphManifestReceipt:
    return ContainedGraphManifestReceipt(
        fixture.graph, ContainedGraphTransactionState.COMMITTED,
        ContainedGraphManifestDisposition.APPLIED,
    )


def _abort_receipt(fixture: _Fixture) -> ContainedGraphManifestReceipt:
    return ContainedGraphManifestReceipt(
        fixture.graph, ContainedGraphTransactionState.ABORTED,
        ContainedGraphManifestDisposition.APPLIED,
    )


def _release_receipt(fixture: _Fixture) -> ContainedGraphManifestReceipt:
    return ContainedGraphManifestReceipt(
        fixture.graph, ContainedGraphTransactionState.COMMITTED,
        ContainedGraphManifestDisposition.RELEASED, fixture.edges,
    )


def test_inline_id_is_deterministic_and_domain_separated_from_stored() -> None:
    fixture = _Fixture(1)
    replay = InlinePublicationID(
        fixture.lease_id, fixture.task_id, fixture.attempt_id,
        fixture.outer_object_id,
    )
    stored = StoredPublicationID(
        fixture.lease_id, fixture.task_id, fixture.attempt_id,
        fixture.outer_object_id,
    )

    assert replay == fixture.publication_id
    assert replay.graph_transaction_id == fixture.publication_id.graph_transaction_id
    assert replay.graph_transaction_id.startswith("inline-contained-graph:")
    assert replay.graph_transaction_id != stored.graph_transaction_id
    assert replay != stored

    other_task = TaskID.random()
    with pytest.raises(ValueError, match="attempt_id belongs"):
        InlinePublicationID(
            fixture.lease_id, fixture.task_id, AttemptID(other_task, 0),
            fixture.outer_object_id,
        )


def test_manifest_binds_payload_identity_owner_route_and_ordered_edges() -> None:
    fixture = _Fixture()
    envelope = fixture.envelope
    manifest = envelope.manifest

    assert envelope.descriptor.storage is ResultStorage.INLINE
    assert envelope.descriptor.inline_data == fixture.payload
    assert manifest.size_bytes == len(fixture.payload)
    assert manifest.ordered_edges == fixture.edges
    assert fixture.graph.publication_id is fixture.publication_id
    assert fixture.graph.ordered_edges == fixture.edges

    for change in (
        {"outer_owner_worker_id": WorkerID.random()},
        {"publishing_node_id": NodeID.random()},
        {"ordered_edges": tuple(reversed(fixture.edges))},
        {"size_bytes": manifest.size_bytes + 1},
    ):
        with pytest.raises(
            (InlinePublicationConflictError, ValueError),
            match="digest|custody|edge slots",
        ):
            InlinePublicationManifest(**{
                **manifest.__dict__,
                **change,
            })


def test_graph_manifest_round_trip_is_exact_and_rejects_stored_identity() -> None:
    fixture = _Fixture()
    restored = InlinePublicationEnvelope.from_graph_manifest(
        fixture.graph, job_id=fixture.job_id,
        executor_worker_id=fixture.executor,
        publishing_node_id=fixture.node_id,
        descriptor=fixture.envelope.descriptor,
    )
    assert restored == fixture.envelope
    assert restored.to_graph_manifest() == fixture.graph

    stored_id = StoredPublicationID(
        fixture.lease_id, fixture.task_id, fixture.attempt_id,
        fixture.outer_object_id,
    )
    stored_graph = ContainedGraphManifest(
        stored_id.graph_transaction_id, stored_id, fixture.outer_owner,
        fixture.envelope.manifest.manifest_digest, fixture.edges,
    )
    with pytest.raises(TypeError, match="InlinePublicationID"):
        InlinePublicationEnvelope.from_graph_manifest(
            stored_graph, job_id=fixture.job_id,
            executor_worker_id=fixture.executor,
            publishing_node_id=fixture.node_id,
            descriptor=fixture.envelope.descriptor,
        )


def test_envelope_rejects_descriptor_drift() -> None:
    fixture = _Fixture()
    descriptor = fixture.envelope.descriptor
    alternatives = (
        replace(descriptor, owner_worker_id=WorkerID.random()),
        replace(descriptor, node_id=NodeID.random()),
        ResultDescriptor(
            descriptor.object_id, ResultStorage.INLINE, 1,
            descriptor.owner_worker_id, descriptor.node_id,
            __import__("hashlib").sha256(b"x").hexdigest(), b"x",
        ),
    )
    for alternative in alternatives:
        with pytest.raises(
            InlinePublicationConflictError, match="descriptor"
        ):
            InlinePublicationEnvelope(fixture.envelope.manifest, alternative)


def test_prepare_exact_replay_and_conflict_are_non_mutating() -> None:
    fixture = _Fixture()
    reducer = InlineContainedPublicationReducer()

    prepared = _prepare_receipt(fixture)
    assert reducer.prepare(fixture.envelope, prepared)
    assert not reducer.prepare(fixture.envelope, prepared)
    before = reducer.snapshot(fixture.publication_id)
    conflict = InlinePublicationEnvelope.create(
        fixture.publication_id, fixture.job_id, fixture.executor,
        fixture.outer_owner, fixture.node_id, b"different bytes",
        fixture.edges,
    )
    with pytest.raises(InlinePublicationConflictError, match="GCS graph|envelope"):
        reducer.prepare(conflict, prepared)
    assert reducer.snapshot(fixture.publication_id) == before

    with pytest.raises(UnknownInlinePublicationError):
        reducer.snapshot(_Fixture(1).publication_id)


def test_prepare_is_thread_safe_and_has_one_first_application() -> None:
    fixture = _Fixture(1)
    reducer = InlineContainedPublicationReducer()
    with ThreadPoolExecutor(max_workers=8) as executor:
        outcomes = tuple(
            executor.map(
                lambda _index: reducer.prepare(
                    fixture.envelope, _prepare_receipt(fixture)
                ),
                range(32),
            )
        )

    assert outcomes.count(True) == 1
    assert outcomes.count(False) == 31
    assert reducer.publication_ids() == (fixture.publication_id,)


def test_pin_bitmap_requires_exact_manifest_effects_before_commit() -> None:
    fixture = _Fixture()
    reducer = InlineContainedPublicationReducer()
    assert reducer.prepare(fixture.envelope, _prepare_receipt(fixture))
    first = fixture.envelope.edge_effect(0)
    second = fixture.envelope.edge_effect(1)

    assert reducer.mark_pin_installed(second)
    assert not reducer.mark_pin_installed(second)
    snapshot = reducer.snapshot(fixture.publication_id)
    assert snapshot.state is InlinePublicationState.PREPARED
    assert snapshot.pin_bitmap == (False, True)
    with pytest.raises(InlinePublicationStateError, match="commit intent"):
        reducer.commit(fixture.envelope, _commit_receipt(fixture))

    wrong = InlineEdgeEffect(
        fixture.publication_id, 0, fixture.edges[1],
        fixture.envelope.manifest.transfers[1],
    )
    before = reducer.snapshot(fixture.publication_id)
    with pytest.raises(InlinePublicationConflictError, match="manifest slot"):
        reducer.mark_pin_installed(wrong)
    assert reducer.snapshot(fixture.publication_id) == before

    assert reducer.mark_pin_installed(first)
    assert reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.PINS_INSTALLED
    assert reducer.begin_commit(fixture.envelope)
    assert reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.COMMITTING


def test_commit_requires_exact_graph_and_is_replay_safe() -> None:
    fixture = _Fixture()
    reducer = InlineContainedPublicationReducer()
    _pins_ready(fixture, reducer)
    wrong_graph = replace(fixture.graph, manifest_digest="00" * 32)
    wrong_receipt = ContainedGraphManifestReceipt(
        wrong_graph, ContainedGraphTransactionState.COMMITTED,
        ContainedGraphManifestDisposition.APPLIED,
    )
    before = reducer.snapshot(fixture.publication_id)

    with pytest.raises(InlinePublicationConflictError, match="GCS graph"):
        reducer.commit(fixture.envelope, wrong_receipt)
    assert reducer.snapshot(fixture.publication_id) == before
    assert reducer.begin_commit(fixture.envelope)
    committed = _commit_receipt(fixture)
    assert reducer.commit(fixture.envelope, committed)
    assert not reducer.commit(fixture.envelope, committed)
    assert reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.COMMITTED
    with pytest.raises(InlinePublicationStateError, match="cannot abort"):
        reducer.begin_abort(fixture.publication_id, "abort-too-late")

def test_abort_compensates_every_slot_in_reverse_order_before_graph_abort() -> None:
    fixture = _Fixture(3)
    reducer = InlineContainedPublicationReducer()
    assert reducer.prepare(fixture.envelope, _prepare_receipt(fixture))
    assert reducer.mark_pin_installed(fixture.envelope.edge_effect(0))
    assert reducer.begin_abort(fixture.publication_id, "abort-1")
    assert not reducer.begin_abort(fixture.publication_id, "abort-1")
    snapshot = reducer.snapshot(fixture.publication_id)
    assert snapshot.state is InlinePublicationState.ABORTING
    assert tuple(
        effect.index for effect in snapshot.pending_compensations
    ) == (2, 1, 0)

    with pytest.raises(InlinePublicationStateError, match="reverse"):
        reducer.acknowledge_compensation(fixture.envelope.edge_effect(0))
    for index in (2, 1, 0):
        effect = fixture.envelope.edge_effect(index)
        assert reducer.next_compensation(fixture.publication_id) == effect
        assert reducer.acknowledge_compensation(effect)
        assert not reducer.acknowledge_compensation(effect)
    assert reducer.next_compensation(fixture.publication_id) is None

    with pytest.raises(
        InlinePublicationConflictError, match="abort identity"
    ):
        reducer.finish_abort(
            fixture.publication_id, "different-abort",
            _abort_receipt(fixture),
        )
    assert reducer.finish_abort(
        fixture.publication_id, "abort-1", _abort_receipt(fixture)
    )
    assert not reducer.finish_abort(
        fixture.publication_id, "abort-1", _abort_receipt(fixture)
    )
    assert reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.ABORTED
    with pytest.raises(InlinePublicationStateError, match="cannot prepare"):
        reducer.prepare(fixture.envelope, _prepare_receipt(fixture))


def test_abort_cannot_finish_before_all_compensation_acks() -> None:
    fixture = _Fixture()
    reducer = InlineContainedPublicationReducer()
    assert reducer.prepare(fixture.envelope, _prepare_receipt(fixture))
    assert reducer.begin_abort(fixture.publication_id, "abort-partial")

    with pytest.raises(InlinePublicationStateError, match="every compensation"):
        reducer.finish_abort(
            fixture.publication_id, "abort-partial",
            _abort_receipt(fixture),
        )
    assert reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.ABORTING


def test_collection_orders_child_releases_then_graph_then_metadata() -> None:
    fixture = _Fixture(3)
    reducer = InlineContainedPublicationReducer()
    _committed(fixture, reducer)
    assert reducer.begin_collection(fixture.publication_id, "collect-1")
    assert not reducer.begin_collection(fixture.publication_id, "collect-1")

    with pytest.raises(InlinePublicationStateError, match="manifest order"):
        reducer.acknowledge_collection_release(
            "collect-1", fixture.envelope.edge_effect(1)
        )
    with pytest.raises(InlinePublicationStateError, match="every child"):
        reducer.mark_graph_released(
            fixture.publication_id, "collect-1",
            _release_receipt(fixture),
        )

    for index in range(3):
        effect = fixture.envelope.edge_effect(index)
        assert reducer.next_collection_release(
            fixture.publication_id
        ) == effect
        assert reducer.acknowledge_collection_release(
            "collect-1", effect
        )
        assert not reducer.acknowledge_collection_release(
            "collect-1", effect
        )
    assert reducer.next_collection_release(fixture.publication_id) is None
    with pytest.raises(InlinePublicationStateError, match="graph release"):
        reducer.finish_collection(fixture.publication_id, "collect-1")

    assert reducer.mark_graph_released(
        fixture.publication_id, "collect-1", _release_receipt(fixture)
    )
    assert not reducer.mark_graph_released(
        fixture.publication_id, "collect-1", _release_receipt(fixture)
    )
    assert reducer.finish_collection(fixture.publication_id, "collect-1")
    assert not reducer.finish_collection(fixture.publication_id, "collect-1")
    snapshot = reducer.snapshot(fixture.publication_id)
    assert snapshot.state is InlinePublicationState.RELEASED
    assert snapshot.graph_released
    assert snapshot.collection_bitmap == (True, True, True)


def test_collection_identity_conflict_is_non_mutating() -> None:
    fixture = _Fixture(1)
    reducer = InlineContainedPublicationReducer()
    _committed(fixture, reducer)
    assert reducer.begin_collection(fixture.publication_id, "collect-a")
    before = reducer.snapshot(fixture.publication_id)

    with pytest.raises(InlinePublicationConflictError, match="collection"):
        reducer.begin_collection(fixture.publication_id, "collect-b")
    with pytest.raises(InlinePublicationConflictError, match="collection"):
        reducer.acknowledge_collection_release(
            "collect-b", fixture.envelope.edge_effect(0)
        )
    assert reducer.snapshot(fixture.publication_id) == before


def test_manifest_rejects_foreign_container_owner_and_duplicate_edges() -> None:
    fixture = _Fixture(1)
    edge = fixture.edges[0]
    cases = (
        (replace(
            edge, container_object_id=ObjectID.for_task(TaskID.random())
        ),),
        (replace(
            edge, contained_owner_worker_id=WorkerID.random()
        ),),
        (edge, edge),
    )
    for edges in cases:
        with pytest.raises((TypeError, ValueError)):
            InlinePublicationManifest.create(
                fixture.publication_id, fixture.job_id, fixture.executor,
                fixture.outer_owner, fixture.node_id, len(fixture.payload),
                fixture.envelope.descriptor.checksum, edges,
            )
