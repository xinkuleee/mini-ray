"""Pure Node orchestration for crash-replayable stored publication.

The reducers in :mod:`miniray.stored_publication` own every lifecycle fact.
This module only orders their durable intents around injected effects: child
owner pin RPCs, the GCS graph/recovery authorities, and local replica sealing.
It deliberately owns no second phase enum or bitmap.
"""

    [StoredPublicationIntent, StoredPublicationRecoveryRecord], None
]


class StoredPublicationRemoteError(StoredPublicationError):
    """A typed owner/GCS rejection, distinct from an ambiguous exception."""

    def __init__(
        self, operation: str,
        error_kind: protocol.StoredPublicationRPCErrorKind, detail: str,
    ) -> None:
        self.operation = operation
        self.error_kind = error_kind
        self.detail = detail
        super().__init__(
            "{} rejected [{}]: {}".format(
                operation, error_kind.value, detail
            )
        )


class StoredPublicationNodeAdapter:
    """Compose one Node journal with its external publication effects.

    Every method is exact-replay safe.  A callback is invoked only after the
    corresponding journal intent exists, and an acknowledgement advances the
    journal only after its complete identity has been checked.
    """

    def __init__(
        self,
        journal: StoredPublicationJournal,
        *,
        prepare_child: PrepareChild,
        prepare_graph: PrepareGraph,
        seal_replica: SealReplica,
        promote_child: PromoteChild,
        report_intent: ReportIntent,
        report_sealed: ReportSealed,
        report_promotions: ReportPromotions,
        report_complete: ReportComplete,
        get_recovery: GetRecovery,
        abort_graph: AbortGraph,
        drop_replica: DropReplica,
        release_child: ReleaseChild,
        after_intent_ack: Optional[AfterIntentAck] = None,
        after_promotions_ack: Optional[AfterPromotionsAck] = None,
    ) -> None:
        if not isinstance(journal, StoredPublicationJournal):
            raise TypeError("journal must be a StoredPublicationJournal")
        callbacks = (
            (prepare_child, "prepare_child"),
            (prepare_graph, "prepare_graph"),
            (seal_replica, "seal_replica"),
            (promote_child, "promote_child"),
            (report_intent, "report_intent"),
            (report_sealed, "report_sealed"),
            (report_promotions, "report_promotions"),
            (report_complete, "report_complete"),
            (get_recovery, "get_recovery"),
            (abort_graph, "abort_graph"),
            (drop_replica, "drop_replica"),
            (release_child, "release_child"),
        )
        for callback, name in callbacks:
            if not callable(callback):
                raise TypeError(f"{name} must be callable")
        if after_intent_ack is not None and not callable(after_intent_ack):
            raise TypeError(
                "after_intent_ack must be callable or None"
            )
        if (
            after_promotions_ack is not None
            and not callable(after_promotions_ack)
        ):
            raise TypeError(
                "after_promotions_ack must be callable or None"
            )
        self.journal = journal
        self._prepare_child_rpc = prepare_child
        self._prepare_graph_rpc = prepare_graph
        self._seal_replica = seal_replica
        self._promote_child_rpc = promote_child
        self._report_intent_rpc = report_intent
        self._report_sealed_rpc = report_sealed
        self._report_promotions_rpc = report_promotions
        self._report_complete_rpc = report_complete
        self._get_recovery_rpc = get_recovery
        self._abort_graph_rpc = abort_graph
        self._drop_replica_rpc = drop_replica
        self._release_child_rpc = release_child
        self._after_intent_ack = after_intent_ack
        self._after_promotions_ack = after_promotions_ack

    def open(
        self, header: StoredPublicationHeader,
        manifest: StoredPublicationManifest, *,
        node_pid: int, registration_epoch: int,
    ) -> bool:
        """Journal and report the complete intent before any other effect.

        ``open_manifest`` installs every potential child-hold identity in one
        local transaction.  Only an exact GCS acknowledgement opens the gate
        used by :meth:`prepare_pin`; a lost response therefore leaves no
        untracked child effect and is recovered by replaying this same report.
        """

        intent = self.journal.open_manifest(
            header, manifest, node_pid, registration_epoch
        )
        snapshot = self.journal.snapshot(intent.publication_id)
        if snapshot.intent_reported:
            self.journal.require_intent_reported(
                intent.publication_id, intent
            )
            self._run_after_intent_ack(intent)
            return False
        request = protocol.ReportStoredPublicationIntent(intent)
        reply = self._report_intent_rpc(request)
        acknowledgement = self._require_recovery_reply(
            reply, request, "stored publication intent report"
        )
        self._validate_recovery_ack(
            acknowledgement, StoredPublicationRecoveryStage.INTENT,
            intent.publication_id, intent=intent, manifest=manifest,
        )
        changed = self.journal.mark_intent_reported(
            intent.publication_id, intent
        )
        self._run_after_intent_ack(intent)
        return changed

    def _run_after_intent_ack(
        self, intent: StoredPublicationIntent
    ) -> None:
        """Run the optional semantic test gate at the exact safe boundary."""

        callback = self._after_intent_ack
        if callback is not None:
            callback(intent)

    def prepare_pin(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        """Persist both release identities, then prepare the child pin."""

        changed = self.journal.append_pin_intent(
            publication_id, index, transfer
        )
        snapshot = self.journal.snapshot(publication_id)
        self._require_reported_intent(snapshot)
        if snapshot.prepared_bitmap[index]:
            return changed
        request = protocol.PrepareStoredContainedPin(
            transfer, transfer.contained_owner_worker_id
        )
        reply = self._prepare_child_rpc(request)
        disposition = self._require_pin_reply(
            reply, request, "stored child-pin prepare"
        )
        if disposition not in (
            StoredContainedReferenceDisposition.PREPARED,
            StoredContainedReferenceDisposition.ALREADY_PREPARED,
        ):
            raise StoredPublicationStateError(
                "child owner did not acknowledge stored pin preparation"
            )
        return self.journal.mark_pin_prepared(
            publication_id, index, transfer
        )

    def freeze(
        self, publication_id: StoredPublicationID,
        manifest: StoredPublicationManifest,
    ) -> bool:
        """Freeze the same pre-reported manifest after every pin ACK."""

        snapshot = self.journal.snapshot(publication_id)
        intent = self._require_reported_intent(snapshot)
        if intent.manifest != manifest:
            raise StoredPublicationStateError(
                "freeze manifest differs from the GCS-acknowledged intent"
            )
        return self.journal.freeze_discovery(publication_id, manifest)

    def prepare_graph(
        self, publication_id: StoredPublicationID,
    ) -> ContainedGraphManifest:
        """Persist the abort identity before reserving the GCS DAG batch."""

        snapshot = self.journal.snapshot(publication_id)
        self._require_reported_intent(snapshot)
        if snapshot.frozen_manifest is None:
            raise StoredPublicationStateError(
                "graph preparation requires frozen discovery"
            )
        graph = snapshot.frozen_manifest.graph_manifest(
            snapshot.header.outer_owner_worker_id
        )
        self.journal.begin_graph_prepare(publication_id, graph)
        snapshot = self.journal.snapshot(publication_id)
        if not snapshot.graph_prepared:
            request = protocol.PrepareStoredPublicationGraph(graph)
            reply = self._prepare_graph_rpc(request)
            receipt = self._require_graph_reply(
                reply, request, "stored graph prepare"
            )
            self._validate_graph_receipt(receipt, graph)
            self.journal.mark_graph_prepared(publication_id, graph)
        return graph

    def seal(
        self, publication_id: StoredPublicationID,
        descriptor: protocol.ResultDescriptor,
        payload: bytes,
    ) -> protocol.ResultDescriptor:
        """Persist the exact drop compensator before sealing local bytes."""

        if not isinstance(payload, bytes):
            raise TypeError("payload must be bytes")
        self._require_reported_intent(
            self.journal.snapshot(publication_id)
        )
        self.journal.begin_replica_seal(publication_id, descriptor)
        snapshot = self.journal.snapshot(publication_id)
        if not snapshot.replica_sealed:
            acknowledgement = self._seal_replica(
                publication_id, descriptor, payload
            )
            if acknowledgement != descriptor:
                raise StoredPublicationStateError(
                    "replica seal acknowledgement changed descriptor identity"
                )
            self.journal.mark_replica_sealed(publication_id, descriptor)
        request = protocol.ReportStoredPublicationReplicaSealed(
            publication_id, descriptor
        )
        reply = self._report_sealed_rpc(request)
        recovery = self._require_recovery_reply(
            reply, request, "stored sealed-replica report"
        )
        self._validate_recovery_ack(
            recovery, StoredPublicationRecoveryStage.REPLICA_SEALED,
            publication_id, descriptor=descriptor,
        )
        return descriptor

    def promote_pin(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        """Persist old->new custody before the child-owner atomic swap."""

        self._require_reported_intent(
            self.journal.snapshot(publication_id)
        )
        self.journal.begin_pin_promotion(publication_id, index, transfer)
        snapshot = self.journal.snapshot(publication_id)
        changed = False
        if not snapshot.promoted_bitmap[index]:
            request = protocol.PromoteStoredContainedPin(
                transfer, transfer.contained_owner_worker_id
            )
            reply = self._promote_child_rpc(request)
            disposition = self._require_pin_reply(
                reply, request, "stored child-pin promotion"
            )
            if disposition not in (
                StoredContainedReferenceDisposition.PROMOTED,
                StoredContainedReferenceDisposition.ALREADY_PROMOTED,
            ):
                raise StoredPublicationStateError(
                    "child owner did not acknowledge stored pin promotion"
                )
            changed = self.journal.mark_pin_promoted(
                publication_id, index, transfer
            )
        promoted = self.journal.snapshot(publication_id).promoted_bitmap
        request = protocol.ReportStoredPublicationPromotions(
            publication_id, promoted
        )
        reply = self._report_promotions_rpc(request)
        recovery = self._require_recovery_reply(
            reply, request, "stored promotions report"
        )
        self._validate_recovery_ack(
            recovery, StoredPublicationRecoveryStage.PROMOTIONS,
            publication_id, promoted=promoted,
        )
        local = self.journal.snapshot(publication_id)
        if promoted and all(promoted):
            if not local.ready_to_complete:
                raise StoredPublicationStateError(
                    "full promotion ACK preceded local completion readiness"
                )
            record = recovery.record
            if (
                not all(record.promoted_bitmap)
                or record.lease_terminal is not None
            ):
                raise StoredPublicationStateError(
                    "promotion checkpoint requires a full pre-Complete GCS "
                    "record"
                )
            callback = self._after_promotions_ack
            if callback is not None:
                callback(self._require_reported_intent(local), record)
        return changed

    def complete(
        self, publication_id: StoredPublicationID,
        terminal: StoredPublicationLeaseTerminal,
    ) -> Optional[StoredPublicationEnvelope]:
        """Re-establish GCS intent, then record the lease terminal.

        A rejected or response-lost initial ``OPEN`` can leave the complete
        local intent bound to the lease while GCS has no record.  Worker then
        reports ``SYSTEM_ERROR`` through this same publication.  Replaying the
        exact intent here closes that hole before either successful Complete
        or rollback mutates its local journal.  If GCS remains unavailable the
        exception propagates and Node retains its exact inflight Complete.
        """

        if not isinstance(terminal, StoredPublicationLeaseTerminal):
            raise TypeError("terminal must be a StoredPublicationLeaseTerminal")
        if terminal.publication_id != publication_id:
            raise StoredPublicationStateError(
                "lease terminal belongs to another publication"
            )
        self._replay_intent_before_complete(publication_id)
        if terminal.status is protocol.TaskReplyStatus.SUCCEEDED:
            # The journal validates first-time readiness and exact terminal
            # replay in one place. A locally completed publication is no longer
            # SEALED/PROMOTING, but must still resend its identical terminal
            # after a report failure or an applied-but-lost GCS ACK.
            envelope = self.journal.complete_lease(publication_id, terminal)
        else:
            reason = self._terminal_rollback_reason(terminal)
            self.journal.begin_rollback(publication_id, reason)
            envelope = None
        request = protocol.ReportStoredPublicationLeaseTerminal(
            publication_id, terminal
        )
        reply = self._report_complete_rpc(request)
        recovery = self._require_recovery_reply(
            reply, request, "stored lease-terminal report"
        )
        self._validate_recovery_ack(
            recovery, StoredPublicationRecoveryStage.LEASE_COMPLETED,
            publication_id, terminal=terminal,
        )
        return envelope

    def _replay_intent_before_complete(
        self, publication_id: StoredPublicationID,
    ) -> StoredPublicationIntent:
        """Fence every terminal transition behind one exact GCS intent."""

        snapshot = self.journal.snapshot(publication_id)
        intent = snapshot.publication_intent
        if intent is None:
            raise StoredPublicationStateError(
                "stored Complete requires a complete publication intent"
            )
        request = protocol.ReportStoredPublicationIntent(intent)
        reply = self._report_intent_rpc(request)
        acknowledgement = self._require_recovery_reply(
            reply, request, "stored Complete intent replay"
        )
        self._validate_recovery_ack(
            acknowledgement, StoredPublicationRecoveryStage.INTENT,
            publication_id, intent=intent, manifest=intent.manifest,
        )
        self.journal.mark_intent_reported(publication_id, intent)
        self.journal.require_intent_reported(publication_id, intent)
        return intent

    def claim(
        self, request: StoredPublicationClaimRequest,
    ) -> StoredPublicationClaimReceipt:
        """Bind the immutable completed hand-off to its logical owner."""

        if not isinstance(request, StoredPublicationClaimRequest):
            raise TypeError(
                "claim requires a StoredPublicationClaimRequest"
            )
        snapshot = self.journal.snapshot(
            request.envelope.publication_id
        )
        self._validate_completed_envelope(snapshot, request.envelope)
        changed = self.journal.claim(
            request.envelope.publication_id, request.claim
        )
        return StoredPublicationClaimReceipt(
            request,
            (
                StoredOwnerEffectDisposition.APPLIED
                if changed
                else StoredOwnerEffectDisposition.ALREADY_APPLIED
            ),
        )

    def replay_promotions(
        self, request: StoredPublicationPromotionReplay,
    ) -> StoredPublicationPromotionReceipt:
        """Reconfirm every final pin after accepting the exact claim.

        Complete already requires every promotion ACK.  This replay is still
        a deliberate hand-off fence: after a lost reply the owner can resend
        the same request and every child authority sees the same old/new hold
        identity.  No second Node bitmap is introduced.
        """

        if not isinstance(request, StoredPublicationPromotionReplay):
            raise TypeError(
                "promotion replay requires a "
                "StoredPublicationPromotionReplay"
            )
        claim_request = request.claim_request
        publication_id = claim_request.envelope.publication_id
        snapshot = self.journal.snapshot(publication_id)
        self._validate_completed_envelope(
            snapshot, claim_request.envelope
        )
        if snapshot.claim != claim_request.claim:
            raise StoredPublicationStateError(
                "promotion replay requires the exact accepted claim"
            )
        if snapshot.state not in (
            StoredPublicationState.ADOPTING,
            StoredPublicationState.ADOPTED,
        ):
            raise StoredPublicationStateError(
                "promotion replay requires an adopting publication"
            )

        changed = False
        for index, transfer in enumerate(snapshot.transfers):
            pin_request = protocol.PromoteStoredContainedPin(
                transfer, transfer.contained_owner_worker_id
            )
            reply = self._promote_child_rpc(pin_request)
            disposition = self._require_pin_reply(
                reply, pin_request, "stored child-pin promotion replay"
            )
            if disposition not in (
                StoredContainedReferenceDisposition.PROMOTED,
                StoredContainedReferenceDisposition.ALREADY_PROMOTED,
            ):
                raise StoredPublicationStateError(
                    "child owner did not acknowledge promotion replay"
                )
            if disposition is StoredContainedReferenceDisposition.PROMOTED:
                changed = True
            # A recovered in-process journal may have recorded the intent but
            # not the reply.  Complete normally makes this branch a replay.
            if not snapshot.promoted_bitmap[index]:
                changed = (
                    self.journal.mark_pin_promoted(
                        publication_id, index, transfer
                    )
                    or changed
                )
                snapshot = self.journal.snapshot(publication_id)

        promoted = self.journal.snapshot(publication_id).promoted_bitmap
        report = protocol.ReportStoredPublicationPromotions(
            publication_id, promoted
        )
        recovery_reply = self._report_promotions_rpc(report)
        acknowledgement = self._require_recovery_reply(
            recovery_reply, report, "stored promotion-replay report"
        )
        self._validate_recovery_ack(
            acknowledgement, StoredPublicationRecoveryStage.PROMOTIONS,
            publication_id, promoted=promoted,
        )
        return StoredPublicationPromotionReceipt(
            request,
            (
                StoredOwnerEffectDisposition.APPLIED
                if changed
                else StoredOwnerEffectDisposition.ALREADY_APPLIED
            ),
        )

    def acknowledge_adopted(
        self, adoption: StoredPublicationAdoption,
    ) -> bool:
        """Record the exact owner commit receipt as the clean gate."""

        if not isinstance(adoption, StoredPublicationAdoption):
            raise TypeError(
                "adoption must be a StoredPublicationAdoption"
            )
        return self.journal.mark_adopted(
            adoption.claim.publication_id, adoption
        )

    def outcome(
        self, request: protocol.GetWorkerLeaseOutcome, *,
        worker_alive: bool = False,
        unresolved_state: protocol.LeaseExecutionState = (
            protocol.LeaseExecutionState.WORKER_LOST
        ),
    ) -> protocol.GetWorkerLeaseOutcomeReply:
        """Project journal/GCS facts into recoverable versus orphan bytes."""

        publication_id = self._publication_for_request(request)
        snapshot = self.journal.snapshot(publication_id)
        recovery = self._query_recovery(publication_id)
        if recovery is None:
            raise StoredPublicationStateError(
                "GCS has no recovery record for the frozen publication"
            )
        terminal = recovery.lease_terminal or snapshot.lease_terminal
        state = unresolved_state
        status = None
        if terminal is not None:
            state = protocol.LeaseExecutionState.COMPLETED
            status = terminal.status
        elif state not in (
            protocol.LeaseExecutionState.RUNNING,
            protocol.LeaseExecutionState.WORKER_LOST,
            protocol.LeaseExecutionState.ABANDONED,
        ):
            raise ValueError(
                "an unresolved stored publication requires a terminal loss state"
            )

        result_descriptor = (
            recovery.sealed_descriptor
            if recovery.sealed_descriptor is not None
            else snapshot.replica_descriptor
        )
        descriptor = (
            self._object_store_descriptor_from_result(
                snapshot, result_descriptor
            )
            if (
                result_descriptor is not None
                and self._replica_is_live(
                    snapshot, recovery.sealed_descriptor is not None
                )
            )
            else None
        )
        ready_success = bool(
            descriptor is not None
            and terminal is not None
            and terminal.status is protocol.TaskReplyStatus.SUCCEEDED
            and recovery.sealed_descriptor == result_descriptor
            and all(recovery.promoted_bitmap)
        )
        recoverable = (descriptor,) if ready_success else ()
        terminal_orphan = state in (
            protocol.LeaseExecutionState.COMPLETED,
            protocol.LeaseExecutionState.WORKER_LOST,
            protocol.LeaseExecutionState.ABANDONED,
        )
        orphan = (
            (descriptor,)
            if descriptor is not None and not ready_success and terminal_orphan
            else ()
        )
        envelope = None
        if ready_success:
            frozen = snapshot.frozen_manifest
            graph = snapshot.graph_manifest
            result = snapshot.replica_descriptor
            assert frozen is not None
            assert graph is not None
            assert result is not None
            envelope = StoredPublicationEnvelope(
                publication_id, frozen.manifest_digest, result, graph,
                snapshot.header,
            )
        return protocol.GetWorkerLeaseOutcomeReply(
            request.lease_id, request.task_id, request.attempt_id,
            request.executor_worker_id, request.owner_worker_id,
            request.object_ids, snapshot.header.node_id, True, worker_alive,
            state=state, completion_status=status, descriptors=recoverable,
            orphan_descriptors=orphan, scheduling_key=request.scheduling_key,
            target_execution=request.target_execution,
            stored_publication=envelope,
        )

    def rollback_precomplete(
        self, publication_id: StoredPublicationID,
        reason: Optional[str] = None,
    ) -> bool:
        """Drive the journal's frozen reverse plan, retaining lost ACKs."""

        snapshot = self.journal.snapshot(publication_id)
        selected_reason = reason or snapshot.rollback_reason
        if selected_reason is None:
            raise ValueError("rollback reason must be supplied initially")
        self.journal.begin_rollback(publication_id, selected_reason)
        while True:
            snapshot = self.journal.snapshot(publication_id)
            if snapshot.state is StoredPublicationState.ROLLED_BACK:
                return True
            pending = snapshot.pending_compensations
            if not pending:
                return snapshot.state is StoredPublicationState.ROLLED_BACK
            obligation = pending[0]
            self._compensate_obligation(obligation)
            self.journal.mark_compensation_acked(
                publication_id, obligation.ordinal, obligation
            )

    def recovery_record(
        self, publication_id: StoredPublicationID,
    ) -> StoredPublicationRecoveryRecord | None:
        """Return the typed GCS recovery fact for death/shutdown policy."""

        if not isinstance(publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        return self._query_recovery(publication_id)

    def shutdown(self, reason: str = "Node shutdown before successful Complete") -> bool:
        """Rollback every pre-success publication; retain completed claims."""

        if not isinstance(reason, str) or not reason:
            raise ValueError("shutdown reason must be a non-empty string")
        clean = True
        for publication_id in self.journal.publication_ids():
            snapshot = self.journal.snapshot(publication_id)
            if (
                snapshot.lease_terminal is not None
                and snapshot.lease_terminal.status
                is protocol.TaskReplyStatus.SUCCEEDED
            ):
                continue
            if snapshot.state is StoredPublicationState.ROLLED_BACK:
                continue
            try:
                self.rollback_precomplete(
                    publication_id, snapshot.rollback_reason or reason
                )
            except Exception:
                clean = False
        return clean

    def _publication_for_request(
        self, request: protocol.GetWorkerLeaseOutcome,
    ) -> StoredPublicationID:
        if not isinstance(request, protocol.GetWorkerLeaseOutcome):
            raise TypeError("request must be a GetWorkerLeaseOutcome")
        if len(request.object_ids) != 1:
            raise ValueError("stored publication adapter owns exactly one outer object")
        publication_id = StoredPublicationID(
            request.lease_id, request.task_id, request.attempt_id,
            request.object_ids[0],
        )
        snapshot = self.journal.snapshot(publication_id)
        header = snapshot.header
        if (
            header.executor_worker_id != request.executor_worker_id
            or header.outer_owner_worker_id != request.owner_worker_id
        ):
            raise ValueError("worker lease outcome identity does not match publication")
        return publication_id

    @staticmethod
    def _validate_completed_envelope(
        snapshot: StoredPublicationSnapshot,
        envelope: StoredPublicationEnvelope,
    ) -> None:
        if not isinstance(envelope, StoredPublicationEnvelope):
            raise TypeError(
                "completed publication hand-off requires an envelope"
            )
        frozen = snapshot.frozen_manifest
        if (
            snapshot.state not in (
                StoredPublicationState.COMPLETED_UNCLAIMED,
                StoredPublicationState.ADOPTING,
                StoredPublicationState.ADOPTED,
            )
            or snapshot.lease_terminal is None
            or snapshot.lease_terminal.status
            is not protocol.TaskReplyStatus.SUCCEEDED
            or not snapshot.replica_sealed
            or not snapshot.graph_prepared
            or not all(snapshot.promoted_bitmap)
            or frozen is None
            or snapshot.graph_manifest is None
            or snapshot.replica_descriptor is None
            or envelope.publication_id != snapshot.publication_id
            or envelope.header != snapshot.header
            or envelope.manifest_digest != frozen.manifest_digest
            or envelope.graph_manifest != snapshot.graph_manifest
            or envelope.descriptor != snapshot.replica_descriptor
        ):
            raise StoredPublicationStateError(
                "claim envelope does not exactly match a completed "
                "publication"
            )

    @staticmethod
    def _object_store_descriptor_from_result(
        snapshot: StoredPublicationSnapshot,
        descriptor: protocol.ResultDescriptor,
    ) -> protocol.ObjectStoreDescriptor:
        return protocol.ObjectStoreDescriptor(
            descriptor.object_id, descriptor.owner_worker_id,
            snapshot.publication_id.attempt_id, descriptor.node_id,
            descriptor.size_bytes, descriptor.checksum,
        )

    @staticmethod
    def _replica_is_live(
        snapshot: StoredPublicationSnapshot, recovery_reports_sealed: bool,
    ) -> bool:
        if not snapshot.replica_sealed and not recovery_reports_sealed:
            return False
        if snapshot.state is StoredPublicationState.ROLLED_BACK:
            return False
        for obligation, acknowledged in zip(
            snapshot.rollback_obligations, snapshot.compensation_bitmap
        ):
            if (
                obligation.kind is StoredRollbackKind.SEALED_REPLICA
                and acknowledged
            ):
                return False
        return True

    @staticmethod
    def _terminal_rollback_reason(
        terminal: StoredPublicationLeaseTerminal,
    ) -> str:
        return "lease terminal before successful Complete: {}".format(
            terminal.status.value
        )

    def _compensate_obligation(
        self, obligation: StoredRollbackObligation,
    ) -> None:
        if obligation.kind is StoredRollbackKind.GRAPH_MANIFEST:
            graph = obligation.identity
            assert isinstance(graph, ContainedGraphManifest)
            request = protocol.AbortStoredPublicationGraph(graph)
            reply = self._abort_graph_rpc(request)
            receipt = self._require_graph_reply(
                reply, request, "stored graph abort"
            )
            if (
                receipt.state is not ContainedGraphTransactionState.ABORTED
                or receipt.disposition not in (
                    ContainedGraphManifestDisposition.APPLIED,
                    ContainedGraphManifestDisposition.ALREADY_ABORTED,
                )
            ):
                raise StoredPublicationStateError(
                    "GCS did not acknowledge the exact graph abort"
                )
            return
        if obligation.kind is StoredRollbackKind.SEALED_REPLICA:
            descriptor = obligation.identity
            assert isinstance(descriptor, protocol.ResultDescriptor)
            request = protocol.DropObjectReplica(
                descriptor.object_id, obligation.publication_id.attempt_id,
                descriptor.owner_worker_id, descriptor.node_id,
                descriptor.checksum,
            )
            reply = self._drop_replica_rpc(request)
            if (
                not isinstance(reply, protocol.DropObjectReplicaReply)
                or reply.object_id != request.object_id
                or reply.producer_attempt_id != request.producer_attempt_id
                or reply.owner_worker_id != request.owner_worker_id
                or reply.node_id != request.node_id
                or reply.checksum != request.checksum
            ):
                raise StoredPublicationStateError(
                    "stored replica drop acknowledgement changed identity"
                )
            if reply.status not in (
                protocol.DropObjectReplicaStatus.DROPPED,
                protocol.DropObjectReplicaStatus.ALREADY_DROPPED,
            ):
                raise StoredPublicationRemoteError(
                    "stored replica drop",
                    protocol.StoredPublicationRPCErrorKind.INVALID_STATE,
                    reply.error or reply.status.value,
                )
            return
        release = obligation.identity
        assert isinstance(release, StoredPinRelease)
        request = protocol.ReleaseContainedReference(
            release.contained_object_id, release.contained_owner_worker_id,
            release.hold,
        )
        reply = self._release_child_rpc(request)
        if (
            not isinstance(reply, protocol.ReleaseContainedReferenceReply)
            or reply.object_id != request.object_id
            or reply.owner_worker_id != request.owner_worker_id
            or reply.hold != request.hold
        ):
            raise StoredPublicationStateError(
                "stored child-pin release acknowledgement changed identity"
            )
        if not reply.accepted:
            raise StoredPublicationRemoteError(
                "stored child-pin release",
                protocol.StoredPublicationRPCErrorKind.INVALID_STATE,
                reply.error or "child owner rejected contained-pin release",
            )

    @staticmethod
    def _validate_graph_receipt(
        receipt: object, graph: ContainedGraphManifest,
    ) -> None:
        if (
            not isinstance(receipt, ContainedGraphManifestReceipt)
            or receipt.manifest != graph
            or receipt.state not in (
                ContainedGraphTransactionState.PREPARED,
                ContainedGraphTransactionState.COMMITTED,
            )
            or receipt.disposition not in (
                ContainedGraphManifestDisposition.APPLIED,
                ContainedGraphManifestDisposition.ALREADY_PREPARED,
                ContainedGraphManifestDisposition.ALREADY_COMMITTED,
            )
        ):
            raise StoredPublicationStateError(
                "GCS did not acknowledge the exact graph preparation"
            )

    @staticmethod
    def _raise_remote_rejection(operation: str, reply: object) -> None:
        kind = getattr(reply, "error_kind", None)
        detail = getattr(reply, "error", None)
        if not isinstance(kind, protocol.StoredPublicationRPCErrorKind) or (
            not isinstance(detail, str) or not detail
        ):
            raise StoredPublicationStateError(
                "{} returned a malformed rejection".format(operation)
            )
        raise StoredPublicationRemoteError(operation, kind, detail)

    @classmethod
    def _require_pin_reply(
        cls, reply: object, request: object, operation: str,
    ) -> StoredContainedReferenceDisposition:
        if not isinstance(reply, protocol.StoredContainedPinReply):
            raise StoredPublicationStateError(
                "{} returned an invalid reply type".format(operation)
            )
        if reply.request != request:
            raise StoredPublicationStateError(
                "{} acknowledgement changed request identity".format(
                    operation
                )
            )
        if not reply.accepted:
            cls._raise_remote_rejection(operation, reply)
        assert reply.disposition is not None
        return reply.disposition

    @classmethod
    def _require_graph_reply(
        cls, reply: object, request: protocol.StoredPublicationGraphRequest,
        operation: str,
    ) -> ContainedGraphManifestReceipt:
        if not isinstance(reply, protocol.StoredPublicationGraphReply):
            raise StoredPublicationStateError(
                "{} returned an invalid reply type".format(operation)
            )
        if reply.request != request:
            raise StoredPublicationStateError(
                "{} acknowledgement changed request identity".format(
                    operation
                )
            )
        if not reply.accepted:
            cls._raise_remote_rejection(operation, reply)
        assert reply.receipt is not None
        return reply.receipt

    @classmethod
    def _require_recovery_reply(
        cls, reply: object, request: object, operation: str,
    ) -> StoredPublicationRecoveryAck:
        if not isinstance(reply, protocol.StoredPublicationRecoveryReply):
            raise StoredPublicationStateError(
                "{} returned an invalid reply type".format(operation)
            )
        if reply.request != request:
            raise StoredPublicationStateError(
                "{} acknowledgement changed request identity".format(
                    operation
                )
            )
        if not reply.accepted:
            cls._raise_remote_rejection(operation, reply)
        assert reply.ack is not None
        return reply.ack

    def _query_recovery(
        self, publication_id: StoredPublicationID,
    ) -> StoredPublicationRecoveryRecord | None:
        request = protocol.GetStoredPublicationRecovery(publication_id)
        reply = self._get_recovery_rpc(request)
        if not isinstance(reply, protocol.GetStoredPublicationRecoveryReply):
            raise StoredPublicationStateError(
                "stored recovery query returned an invalid reply type"
            )
        if reply.request != request:
            raise StoredPublicationStateError(
                "stored recovery query changed request identity"
            )
        if reply.disposition is protocol.StoredPublicationQueryDisposition.REJECTED:
            self._raise_remote_rejection("stored recovery query", reply)
        if reply.disposition is protocol.StoredPublicationQueryDisposition.NOT_FOUND:
            return None
        if (
            reply.disposition is not protocol.StoredPublicationQueryDisposition.FOUND
            or not isinstance(reply.record, StoredPublicationRecoveryRecord)
            or reply.record.publication_id != publication_id
        ):
            raise StoredPublicationStateError(
                "stored recovery query returned a malformed FOUND reply"
            )
        return reply.record

    @staticmethod
    def _validate_recovery_ack(
        acknowledgement: object, stage: StoredPublicationRecoveryStage,
        publication_id: StoredPublicationID, *,
        intent: StoredPublicationIntent | None = None,
        manifest: StoredPublicationManifest | None = None,
        descriptor: protocol.ResultDescriptor | None = None,
        promoted: tuple[bool, ...] | None = None,
        terminal: StoredPublicationLeaseTerminal | None = None,
    ) -> None:
        if (
            not isinstance(acknowledgement, StoredPublicationRecoveryAck)
            or acknowledgement.stage is not stage
            or acknowledgement.record.publication_id != publication_id
        ):
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement changed publication identity"
            )
        record = acknowledgement.record
        if intent is not None and record.intent != intent:
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement changed publication intent"
            )
        if manifest is not None and record.manifest != manifest:
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement changed frozen manifest"
            )
        if descriptor is not None and record.sealed_descriptor != descriptor:
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement changed sealed descriptor"
            )
        if promoted is not None and any(
            expected and not actual
            for expected, actual in zip(promoted, record.promoted_bitmap)
        ):
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement lost a promoted pin"
            )
        if terminal is not None and record.lease_terminal != terminal:
            raise StoredPublicationStateError(
                "GCS recovery acknowledgement changed lease terminal"
            )

    def _require_reported_intent(
        self, snapshot: StoredPublicationSnapshot,
    ) -> StoredPublicationIntent:
        intent = snapshot.publication_intent
        if intent is None:
            raise StoredPublicationStateError(
                "publication has no complete pre-effect intent"
            )
        self.journal.require_intent_reported(
            snapshot.publication_id, intent
        )
        return intent
