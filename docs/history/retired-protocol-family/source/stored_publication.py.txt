"""Crash-replayable authority for one stored outer-object publication.

The serializer and object store perform effects; this module does not.  It is
the Node-side journal reducer that makes those effects recoverable.  Every
possibly-remote effect first records an exact intent, and every acknowledgement
advances a monotonic bitmap or barrier.  Consequently a lost ACK is handled by
replaying an identity, never by guessing whether the effect happened.

The decisive boundary is ``LEASE_COMPLETED``.  Before that fact an executor
failure owns a reverse compensation plan.  After it, rollback is forbidden and
the logical outer owner must claim and adopt the frozen publication.
"""

class StoredPublicationError(RuntimeError):
    """Base class for stored-publication authority failures."""


class StoredPublicationConflictError(StoredPublicationError):
    """A stable operation identity was rebound to different data."""


class StoredPublicationStateError(StoredPublicationError):
    """An event contradicts the publication's monotonic lifecycle."""


def _validate_digest(value: object, label: str) -> str:
    if not isinstance(value, str) or len(value) != 64:
        raise ValueError(f"{label} must be a SHA-256 hex digest")
    try:
        int(value, 16)
    except ValueError as exc:
        raise ValueError(f"{label} must be a SHA-256 hex digest") from exc
    return value.lower()


def _require_hashable(value: object, label: str) -> None:
    try:
        hash(value)
    except TypeError as exc:
        raise TypeError(f"{label} must be hashable") from exc


@dataclass(frozen=True, order=True)
class StoredPublicationID:
    """Stable identity of a stored result within one lease attempt."""

    lease_id: LeaseID
    task_id: TaskID
    attempt_id: AttemptID
    outer_object_id: ObjectID

    def __post_init__(self) -> None:
        if not isinstance(self.lease_id, LeaseID):
            raise TypeError("lease_id must be a LeaseID")
        if not isinstance(self.task_id, TaskID):
            raise TypeError("task_id must be a TaskID")
        if not isinstance(self.attempt_id, AttemptID):
            raise TypeError("attempt_id must be an AttemptID")
        if not isinstance(self.outer_object_id, ObjectID):
            raise TypeError("outer_object_id must be an ObjectID")
        if self.attempt_id.task_id != self.task_id:
            raise ValueError("attempt_id belongs to another task")
        if self.outer_object_id.task_id != self.task_id:
            raise ValueError("outer_object_id belongs to another task")

    @property
    def graph_transaction_id(self) -> str:
        """Deterministic GCS transaction identity; never a random retry ID."""

        digest = hashlib.sha256()
        digest.update(b"miniray-contained-graph-v1\0")
        digest.update(bytes(self.lease_id))
        digest.update(bytes(self.task_id))
        digest.update(self.attempt_id.attempt_number.to_bytes(8, "big"))
        digest.update(bytes(self.outer_object_id.task_id))
        digest.update(self.outer_object_id.return_index.to_bytes(8, "big"))
        return "contained-graph:" + digest.hexdigest()


@dataclass(frozen=True)
class StoredPublicationHeader:
    """Immutable Node journal header, persisted before the first effect."""

    publication_id: StoredPublicationID
    job_id: JobID
    executor_worker_id: WorkerID
    outer_owner_worker_id: WorkerID
    node_id: NodeID

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        if not isinstance(self.job_id, JobID):
            raise TypeError("job_id must be a JobID")
        if not isinstance(self.executor_worker_id, WorkerID):
            raise TypeError("executor_worker_id must be a WorkerID")
        if not isinstance(self.outer_owner_worker_id, WorkerID):
            raise TypeError("outer_owner_worker_id must be a WorkerID")
        if not isinstance(self.node_id, NodeID):
            raise TypeError("node_id must be a NodeID")


@dataclass(frozen=True)
class StoredPublicationManifest:
    """Frozen serialization boundary and ordered transfer manifest."""

    publication_id: StoredPublicationID
    transfers: Tuple[PreparedContainedTransfer, ...]
    size_bytes: int
    checksum: str
    manifest_digest: str

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        transfers = tuple(self.transfers)
        if any(
            not isinstance(value, PreparedContainedTransfer)
            for value in transfers
        ):
            raise TypeError(
                "transfers must contain PreparedContainedTransfer values"
            )
        if len(transfers) != len(set(transfers)):
            raise ValueError("stored transfer manifest must be unique")
        object.__setattr__(self, "transfers", transfers)
        if (
            isinstance(self.size_bytes, bool)
            or not isinstance(self.size_bytes, int)
            or self.size_bytes < 0
        ):
            raise ValueError("size_bytes must be a non-negative integer")
        object.__setattr__(
            self, "checksum", _validate_digest(self.checksum, "checksum")
        )
        object.__setattr__(
            self, "manifest_digest",
            _validate_digest(self.manifest_digest, "manifest_digest"),
        )

    def graph_manifest(
        self, outer_owner_worker_id: WorkerID
    ) -> ContainedGraphManifest:
        return ContainedGraphManifest(
            transaction_id=self.publication_id.graph_transaction_id,
            publication_id=self.publication_id,
            outer_owner_worker_id=outer_owner_worker_id,
            manifest_digest=self.manifest_digest,
            ordered_edges=tuple(value.edge for value in self.transfers),
        )


def _validate_intent_boundary(
    header: StoredPublicationHeader, manifest: StoredPublicationManifest
) -> None:
    """Validate the whole identity needed before any child-side effect."""

    if not isinstance(header, StoredPublicationHeader):
        raise TypeError("header must be a StoredPublicationHeader")
    if not isinstance(manifest, StoredPublicationManifest):
        raise TypeError("manifest must be a StoredPublicationManifest")
    if header.publication_id != manifest.publication_id:
        raise StoredPublicationConflictError(
            "intent header and manifest name different publications"
        )
    outer_object_id = header.publication_id.outer_object_id
    for transfer in manifest.transfers:
        if (
            transfer.provisional_hold.container_object_id != outer_object_id
            or transfer.final_hold.container_object_id != outer_object_id
        ):
            raise StoredPublicationConflictError(
                "intent transfer belongs to another outer object"
            )
        if (
            transfer.provisional_hold.container_owner_worker_id
            != header.executor_worker_id
            or transfer.final_hold.container_owner_worker_id
            != header.outer_owner_worker_id
        ):
            raise StoredPublicationConflictError(
                "intent transfer custody does not match its header"
            )


@dataclass(frozen=True)
class StoredPublicationIntent:
    """Complete, death-recoverable publication fact.

    GCS must durably reduce this value before the publishing Node contacts a
    child owner.  Header, ordered manifest, and physical Node incarnation are
    one identity so recovery never has to infer an omitted side effect.
    """

    header: StoredPublicationHeader
    manifest: StoredPublicationManifest
    node_pid: int
    registration_epoch: int

    def __post_init__(self) -> None:
        _validate_intent_boundary(self.header, self.manifest)
        _require_positive_int(self.node_pid, "node_pid")
        _require_positive_int(
            self.registration_epoch, "registration_epoch"
        )

    @property
    def publication_id(self) -> StoredPublicationID:
        return self.header.publication_id

    @property
    def node_incarnation(self) -> StoredPublicationNodeIncarnation:
        return StoredPublicationNodeIncarnation(
            self.header.node_id, self.node_pid, self.registration_epoch
        )

@dataclass(frozen=True)
class StoredPublicationRecoveryRecord:
    """Minimal GCS fact set needed after loss of the publishing Node."""

    manifest: StoredPublicationManifest
    sealed_descriptor: Optional[ResultDescriptor]
    promoted_bitmap: Tuple[bool, ...]
    lease_terminal: Optional[StoredPublicationLeaseTerminal] = None
    intent: Optional[StoredPublicationIntent] = None

    def __post_init__(self) -> None:
        if not isinstance(self.manifest, StoredPublicationManifest):
            raise TypeError("manifest must be a StoredPublicationManifest")
        if self.sealed_descriptor is not None and not isinstance(
            self.sealed_descriptor, ResultDescriptor
        ):
            raise TypeError("sealed_descriptor must be a ResultDescriptor")
        bitmap = tuple(self.promoted_bitmap)
        if (
            len(bitmap) != len(self.manifest.transfers)
            or any(type(value) is not bool for value in bitmap)
        ):
            raise ValueError(
                "promoted_bitmap must contain one bool per manifest transfer"
            )
        object.__setattr__(self, "promoted_bitmap", bitmap)
        if self.lease_terminal is not None and not isinstance(
            self.lease_terminal, StoredPublicationLeaseTerminal
        ):
            raise TypeError(
                "lease_terminal must be a StoredPublicationLeaseTerminal"
            )
        if self.intent is not None:
            if not isinstance(self.intent, StoredPublicationIntent):
                raise TypeError(
                    "intent must be a StoredPublicationIntent or None"
                )
            if self.intent.manifest != self.manifest:
                raise StoredPublicationConflictError(
                    "recovery intent and manifest do not match"
                )

    @property
    def publication_id(self) -> StoredPublicationID:
        return self.manifest.publication_id

    @property
    def header(self) -> Optional[StoredPublicationHeader]:
        return None if self.intent is None else self.intent.header

    @property
    def node_incarnation(
        self,
    ) -> Optional[StoredPublicationNodeIncarnation]:
        return (
            None if self.intent is None else self.intent.node_incarnation
        )

    @property
    def is_node_loss_safe(self) -> bool:
        """Whether Node-loss recovery has complete pre-effect evidence."""

        return self.intent is not None

    @property
    def recovery_action(self) -> StoredPublicationRecoveryAction:
        return (
            StoredPublicationRecoveryAction.CLAIM
            if (
                self.lease_terminal is not None
                and self.lease_terminal.status is TaskReplyStatus.SUCCEEDED
            )
            else StoredPublicationRecoveryAction.ROLLBACK
        )

class StoredPublicationRecoveryAuthority:
    """Small in-memory GCS reducer for the publication recovery boundary."""

    def __init__(self) -> None:
        self._records: dict[
            StoredPublicationID, StoredPublicationRecoveryRecord
        ] = {}
        self._by_node_incarnation: dict[
            StoredPublicationNodeIncarnation, set[StoredPublicationID]
        ] = {}
        self._node_deaths: dict[
            StoredPublicationNodeIncarnation, NodeDeathRecord
        ] = {}
        self._death_frozen_records: dict[
            StoredPublicationNodeIncarnation,
            Tuple[StoredPublicationRecoveryRecord, ...],
        ] = {}
        self._by_outer_owner: dict[
            WorkerID, set[StoredPublicationID]
        ] = {}
        self._owner_deaths: dict[WorkerID, WorkerDeathRecord] = {}
        self._owner_death_frozen_records: dict[
            WorkerID, Tuple[StoredPublicationRecoveryRecord, ...]
        ] = {}
        self._owner_death_resolutions: dict[
            StoredPublicationID, PublicationOwnerDeathResolution
        ] = {}
        self._lock = RLock()

    def report_intent(
        self, intent: StoredPublicationIntent
    ) -> StoredPublicationRecoveryAck:
        """Durably bind the complete intent before any child-owner RPC."""

        if not isinstance(intent, StoredPublicationIntent):
            raise TypeError("intent must be a StoredPublicationIntent")
        with self._lock:
            publication_id = intent.publication_id
            incarnation = intent.node_incarnation
            previous = self._records.get(publication_id)
            if previous is not None and previous.intent is not None:
                if previous.intent != intent:
                    raise StoredPublicationConflictError(
                        "recovery intent changed on replay"
                    )
                return self._ack(
                    StoredPublicationRecoveryStage.INTENT, False, previous
                )
            if incarnation in self._node_deaths:
                raise StoredPublicationStateError(
                    "publishing Node incarnation is already death-frozen"
                )
            outer_owner_worker_id = intent.header.outer_owner_worker_id
            if outer_owner_worker_id in self._owner_deaths:
                raise StoredPublicationStateError(
                    "outer owner Worker is already death-frozen"
                )
            if previous is not None:
                if previous.manifest != intent.manifest:
                    raise StoredPublicationConflictError(
                        "recovery intent changed the legacy manifest"
                    )
                record = replace(previous, intent=intent)
            else:
                record = StoredPublicationRecoveryRecord(
                    intent.manifest, None,
                    (False,) * len(intent.manifest.transfers),
                    intent=intent,
                )
            self._records[publication_id] = record
            self._by_node_incarnation.setdefault(incarnation, set()).add(
                publication_id
            )
            self._by_outer_owner.setdefault(
                outer_owner_worker_id, set()
            ).add(publication_id)
            return self._ack(
                StoredPublicationRecoveryStage.INTENT, True, record
            )

    def report_manifest(
        self, manifest: StoredPublicationManifest
    ) -> StoredPublicationRecoveryAck:
        """Compatibility replay after the complete intent is durable.

        A manifest alone omits both the publication header and the physical
        Node incarnation.  It therefore cannot safely create recovery state:
        doing so after a death freeze would produce an unindexed obligation.
        The runtime uses ``report_intent``; this method only accepts an exact
        replay against an already intent-backed record.
        """

        if not isinstance(manifest, StoredPublicationManifest):
            raise TypeError("manifest must be a StoredPublicationManifest")
        with self._lock:
            publication_id = manifest.publication_id
            previous = self._records.get(publication_id)
            if previous is None or previous.intent is None:
                raise StoredPublicationStateError(
                    "legacy manifest report requires a prior complete intent"
                )
            if previous.manifest != manifest:
                raise StoredPublicationConflictError(
                    "recovery manifest changed on replay"
                )
            return self._ack(
                StoredPublicationRecoveryStage.MANIFEST, False, previous
            )

    def report_replica_sealed(
        self, publication_id: StoredPublicationID,
        descriptor: ResultDescriptor,
    ) -> StoredPublicationRecoveryAck:
        if not isinstance(descriptor, ResultDescriptor):
            raise TypeError("descriptor must be a ResultDescriptor")
        with self._lock:
            record = self._record(publication_id)
            self._validate_descriptor(record.manifest, descriptor)
            if record.sealed_descriptor is not None:
                if record.sealed_descriptor != descriptor:
                    raise StoredPublicationConflictError(
                        "sealed recovery descriptor changed on replay"
                    )
                return self._ack(
                    StoredPublicationRecoveryStage.REPLICA_SEALED, False, record
                )
            self._require_not_death_frozen(record)
            record = replace(record, sealed_descriptor=descriptor)
            self._records[publication_id] = record
            return self._ack(
                StoredPublicationRecoveryStage.REPLICA_SEALED, True, record
            )

    def report_promotions(
        self, publication_id: StoredPublicationID,
        promoted_bitmap: Tuple[bool, ...],
    ) -> StoredPublicationRecoveryAck:
        try:
            reported = tuple(promoted_bitmap)
        except TypeError as exc:
            raise TypeError("promoted_bitmap must be iterable") from exc
        with self._lock:
            record = self._record(publication_id)
            if (
                len(reported) != len(record.promoted_bitmap)
                or any(type(value) is not bool for value in reported)
            ):
                raise ValueError(
                    "promoted_bitmap must contain one bool per manifest transfer"
                )
            if record.sealed_descriptor is None:
                raise StoredPublicationStateError(
                    "promotions require a sealed replica"
                )
            merged = tuple(
                current or incoming
                for current, incoming in zip(record.promoted_bitmap, reported)
            )
            changed = merged != record.promoted_bitmap
            if changed:
                self._require_not_death_frozen(record)
                record = replace(record, promoted_bitmap=merged)
                self._records[publication_id] = record
            return self._ack(
                StoredPublicationRecoveryStage.PROMOTIONS, changed, record
            )

    def report_lease_terminal(
        self, publication_id: StoredPublicationID,
        terminal: StoredPublicationLeaseTerminal,
    ) -> StoredPublicationRecoveryAck:
        if not isinstance(terminal, StoredPublicationLeaseTerminal):
            raise TypeError(
                "terminal must be a StoredPublicationLeaseTerminal"
            )
        with self._lock:
            record = self._record(publication_id)
            if record.lease_terminal is not None:
                if record.lease_terminal != terminal:
                    raise StoredPublicationConflictError(
                        "recovery lease terminal changed on replay"
                    )
                return self._ack(
                    StoredPublicationRecoveryStage.LEASE_COMPLETED, False, record
                )
            self._require_not_death_frozen(record)
            if terminal.publication_id != publication_id:
                raise StoredPublicationConflictError(
                    "recovery lease terminal belongs to another publication"
                )
            if (
                terminal.status is TaskReplyStatus.SUCCEEDED
                and (
                    record.sealed_descriptor is None
                    or not all(record.promoted_bitmap)
                )
            ):
                raise StoredPublicationStateError(
                    "lease completion requires a sealed replica and every promotion"
                )
            record = replace(record, lease_terminal=terminal)
            self._records[publication_id] = record
            return self._ack(
                StoredPublicationRecoveryStage.LEASE_COMPLETED, True, record
            )

    def snapshot(
        self, publication_id: StoredPublicationID
    ) -> StoredPublicationRecoveryRecord:
        with self._lock:
            return self._record(publication_id)

    def records_for_node_incarnation(
        self, incarnation: StoredPublicationNodeIncarnation,
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Return deterministic diagnostics for one physical Node."""

        if not isinstance(incarnation, StoredPublicationNodeIncarnation):
            raise TypeError(
                "incarnation must be a StoredPublicationNodeIncarnation"
            )
        with self._lock:
            frozen = self._death_frozen_records.get(incarnation)
            if frozen is not None:
                return frozen
            return tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_node_incarnation.get(incarnation, ())
                )
            )

    def records_for_outer_owner(
        self, outer_owner_worker_id: WorkerID,
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Return the deterministic publication set for one logical owner.

        Once owner death is frozen this returns that immutable snapshot rather
        than a live view.  The distinction prevents diagnostics from silently
        widening the exact cleanup obligation.
        """

        if not isinstance(outer_owner_worker_id, WorkerID):
            raise TypeError(
                "outer_owner_worker_id must be a WorkerID"
            )
        with self._lock:
            frozen = self._owner_death_frozen_records.get(
                outer_owner_worker_id
            )
            if frozen is not None:
                return frozen
            return tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_outer_owner.get(outer_owner_worker_id, ())
                )
            )

    def freeze_owner_death(
        self, death: WorkerDeathRecord,
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Linearize logical-owner death against publication admission.

        ``PROCESS_EXIT`` and ``NODE_EXIT`` are authoritative owner-loss
        proofs.  An orderly ``EXPECTED`` exit is deliberately not a loss
        proof and therefore cannot install a fence.  Exact replay returns the
        same immutable tuple object; a changed proof for the Worker conflicts.
        """

        self._validate_owner_death(death)
        outer_owner_worker_id = death.worker_id
        with self._lock:
            previous = self._owner_deaths.get(outer_owner_worker_id)
            if previous is not None:
                if previous != death:
                    raise StoredPublicationConflictError(
                        "outer owner Worker death proof changed on replay"
                    )
                return self._owner_death_frozen_records[
                    outer_owner_worker_id
                ]
            records = tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_outer_owner.get(outer_owner_worker_id, ())
                )
            )
            self._owner_deaths[outer_owner_worker_id] = death
            self._owner_death_frozen_records[
                outer_owner_worker_id
            ] = records
            return records

    def frozen_records_for_owner_death(
        self, death: WorkerDeathRecord,
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Query an owner workset using the exact committed death proof."""

        self._validate_owner_death(death)
        outer_owner_worker_id = death.worker_id
        with self._lock:
            previous = self._owner_deaths.get(outer_owner_worker_id)
            if previous is None:
                raise StoredPublicationStateError(
                    "outer owner Worker is not death-frozen"
                )
            if previous != death:
                raise StoredPublicationConflictError(
                    "outer owner Worker death proof changed on query"
                )
            return self._owner_death_frozen_records[
                outer_owner_worker_id
            ]

    def acknowledge_owner_death_cleaned(
        self, resolution: PublicationOwnerDeathResolution,
    ) -> PublicationOwnerDeathResolution:
        """Retain the exact outer-owner lifetime terminal independently."""

        from .publication_owner_death import (
            PublicationOwnerDeathResolution, PublicationRecoveryHistory,
            StoredPublicationOwnerDeathKey,
            StoredPublicationOwnerDeathRequest,
        )

        if not isinstance(resolution, PublicationOwnerDeathResolution):
            raise TypeError(
                "resolution must be a PublicationOwnerDeathResolution"
            )
        request = resolution.request
        if (
            not isinstance(request, StoredPublicationOwnerDeathRequest)
            or not isinstance(request.key, StoredPublicationOwnerDeathKey)
        ):
            raise TypeError(
                "resolution must carry a STORED owner-death key"
            )
        publication_id = request.publication_id
        with self._lock:
            self._record(publication_id)
            owner_worker_id = request.owner_worker_id
            frozen_death = self._owner_deaths.get(owner_worker_id)
            if frozen_death is None:
                raise StoredPublicationStateError(
                    "outer owner Worker is not death-frozen"
                )
            if frozen_death != request.owner_death:
                raise StoredPublicationConflictError(
                    "owner-death resolution changed the frozen death proof"
                )
            canonical = next((
                record
                for record in self._owner_death_frozen_records[
                    owner_worker_id
                ]
                if record.publication_id == publication_id
            ), None)
            if canonical is None:
                raise StoredPublicationConflictError(
                    "owner-death resolution is outside the frozen workset"
                )
            if request.recovery != canonical:
                raise StoredPublicationConflictError(
                    "owner-death resolution changed frozen STORED recovery"
                )
            terminal = canonical.lease_terminal
            successful = bool(
                terminal is not None
                and terminal.status is TaskReplyStatus.SUCCEEDED
            )
            compatible_histories = (
                (
                    PublicationRecoveryHistory.COMMIT_UNKNOWN,
                    PublicationRecoveryHistory.ADOPTED,
                )
                if successful
                else (PublicationRecoveryHistory.NOT_COMMITTED,)
            )
            if request.publication_history not in compatible_histories:
                raise StoredPublicationConflictError(
                    "owner-death resolution changed publication history"
                )
            previous = self._owner_death_resolutions.get(publication_id)
            if previous is not None:
                if previous != resolution:
                    raise StoredPublicationConflictError(
                        "owner-death resolution changed on replay"
                    )
                return previous
            self._owner_death_resolutions[publication_id] = resolution
            return resolution

    def owner_death_resolution(
        self, publication_id: StoredPublicationID,
    ) -> Optional[PublicationOwnerDeathResolution]:
        """Return the independent owner-lifetime terminal, if present."""

        with self._lock:
            self._record(publication_id)
            return self._owner_death_resolutions.get(publication_id)

    def freeze_node_death(
        self, death: NodeDeathRecord
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Linearize death against intent admission and freeze its workset.

        Exact replay returns the same immutable tuple.  A conflicting death
        proof or any new intent for this incarnation is rejected.
        """

        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        incarnation = StoredPublicationNodeIncarnation.from_death(death)
        with self._lock:
            previous = self._node_deaths.get(incarnation)
            if previous is not None:
                if previous != death:
                    raise StoredPublicationConflictError(
                        "publishing Node death proof changed on replay"
                    )
                return self._death_frozen_records[incarnation]
            records = tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_node_incarnation.get(incarnation, ())
                )
            )
            self._node_deaths[incarnation] = death
            self._death_frozen_records[incarnation] = records
            return records

    def frozen_records_for_node_death(
        self, death: NodeDeathRecord
    ) -> Tuple[StoredPublicationRecoveryRecord, ...]:
        """Query a previously frozen workset using the exact death proof."""

        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        incarnation = StoredPublicationNodeIncarnation.from_death(death)
        with self._lock:
            previous = self._node_deaths.get(incarnation)
            if previous is None:
                raise StoredPublicationStateError(
                    "publishing Node incarnation is not death-frozen"
                )
            if previous != death:
                raise StoredPublicationConflictError(
                    "publishing Node death proof changed on query"
                )
            return self._death_frozen_records[incarnation]

    def _record(
        self, publication_id: StoredPublicationID
    ) -> StoredPublicationRecoveryRecord:
        if not isinstance(publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        try:
            return self._records[publication_id]
        except KeyError as exc:
            raise StoredPublicationStateError(
                "stored recovery intent or manifest was not reported"
            ) from exc

    def _require_not_death_frozen(
        self, record: StoredPublicationRecoveryRecord
    ) -> None:
        incarnation = record.node_incarnation
        if (
            incarnation is not None
            and incarnation in self._node_deaths
        ):
            raise StoredPublicationStateError(
                "publishing Node incarnation is already death-frozen"
            )
        header = record.header
        if (
            header is not None
            and header.outer_owner_worker_id in self._owner_deaths
        ):
            raise StoredPublicationStateError(
                "outer owner Worker is already death-frozen"
            )

    @staticmethod
    def _validate_owner_death(death: WorkerDeathRecord) -> None:
        if not isinstance(death, WorkerDeathRecord):
            raise TypeError("death must be a WorkerDeathRecord")
        if death.reason is WorkerDeathReason.EXPECTED:
            raise StoredPublicationStateError(
                "EXPECTED Worker exit is not an owner-death proof"
            )

    @staticmethod
    def _validate_descriptor(
        manifest: StoredPublicationManifest, descriptor: ResultDescriptor
    ) -> None:
        if (
            descriptor.storage is not ResultStorage.OBJECT_STORE
            or descriptor.object_id != manifest.publication_id.outer_object_id
            or descriptor.size_bytes != manifest.size_bytes
            or descriptor.checksum != manifest.checksum
        ):
            raise StoredPublicationConflictError(
                "sealed recovery descriptor does not match its manifest"
            )

    @staticmethod
    def _ack(
        stage: StoredPublicationRecoveryStage, changed: bool,
        record: StoredPublicationRecoveryRecord,
    ) -> StoredPublicationRecoveryAck:
        return StoredPublicationRecoveryAck(
            stage,
            (StoredPublicationRecoveryDisposition.APPLIED if changed else
             StoredPublicationRecoveryDisposition.ALREADY_RECORDED),
            record,
        )


@dataclass(frozen=True)
class StoredPublicationEnvelope:
    publication_id: StoredPublicationID
    manifest_digest: str
    descriptor: ResultDescriptor
    graph_manifest: ContainedGraphManifest
    header: StoredPublicationHeader

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        object.__setattr__(
            self, "manifest_digest",
            _validate_digest(self.manifest_digest, "manifest_digest"),
        )
        if not isinstance(self.descriptor, ResultDescriptor):
            raise TypeError("descriptor must be a ResultDescriptor")
        if not isinstance(self.graph_manifest, ContainedGraphManifest):
            raise TypeError("graph_manifest must be a ContainedGraphManifest")
        if not isinstance(self.header, StoredPublicationHeader):
            raise TypeError("header must be a StoredPublicationHeader")
        if self.header.publication_id != self.publication_id:
            raise ValueError("envelope header belongs to another publication")
        if (
            self.descriptor.storage is not ResultStorage.OBJECT_STORE
            or self.descriptor.object_id
            != self.publication_id.outer_object_id
            or self.descriptor.owner_worker_id
            != self.header.outer_owner_worker_id
            or self.descriptor.node_id != self.header.node_id
        ):
            raise ValueError(
                "envelope descriptor does not match its outer owner and node"
            )
        if (
            self.graph_manifest.publication_id != self.publication_id
            or self.graph_manifest.transaction_id
            != self.publication_id.graph_transaction_id
            or self.graph_manifest.outer_owner_worker_id
            != self.header.outer_owner_worker_id
            or self.graph_manifest.manifest_digest != self.manifest_digest
            or any(
                edge.container_object_id
                != self.publication_id.outer_object_id
                for edge in self.graph_manifest.ordered_edges
            )
        ):
            raise ValueError(
                "envelope graph does not match its publication identity"
            )

    def __reduce__(self) -> tuple[object, tuple[object, ...]]:
        """Re-enter validation when this authority crosses pickle."""

        return type(self), (
            self.publication_id, self.manifest_digest, self.descriptor,
            self.graph_manifest, self.header,
        )

    @property
    def edges(self) -> Tuple[ContainedReferenceEdge, ...]:
        return self.graph_manifest.ordered_edges

    publication_id: StoredPublicationID
    ordinal: int

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):

        if (
            isinstance(self.ordinal, bool)
            or not isinstance(self.ordinal, int)
            or self.ordinal < 0
        ):
            raise ValueError("ordinal must be non-negative")

            raise TypeError("rollback identity does not match its kind")

    intent_written: Tuple[bool, ...]

    replica_sealed: bool

    adoption: Optional[StoredPublicationAdoption]

    compensation_bitmap: Tuple[bool, ...]

    events: Tuple[StoredPublicationEvent, ...]

    @property

        return (
            self.state in (StoredPublicationState.SEALED,
                           StoredPublicationState.PROMOTING)
            and self.replica_sealed
            and self.graph_prepared
            and all(self.promoted_bitmap)
        )

    def pending_compensations(self) -> Tuple[StoredRollbackObligation, ...]:
        return tuple(
            obligation
            for obligation, acknowledged in zip(
                self.rollback_obligations, self.compensation_bitmap
            )
            if not acknowledged
        )

class StoredPublicationJournal:
    """Node-process authority for stored publication and crash recovery."""

    def __init__(self) -> None:
        self._records: dict[StoredPublicationID, _PublicationRecord] = {}
        self._lock = RLock()

    def open(self, header: StoredPublicationHeader) -> bool:
        """Persist the immutable header before any child-owner call."""

        if not isinstance(header, StoredPublicationHeader):
            raise TypeError("header must be a StoredPublicationHeader")
        with self._lock:
            publication_id = header.publication_id
            previous = self._records.get(publication_id)
            if previous is not None:
                if previous.header != header:
                    raise StoredPublicationConflictError(
                        "publication_id is already bound to another header"
                    )
                return False
            record = _PublicationRecord(header)
            self._records[publication_id] = record
            self._append_event(
                record, StoredPublicationEventKind.PUBLICATION_OPENED, header
            )
            return True

    def open_manifest(
        self, header: StoredPublicationHeader,
        manifest: StoredPublicationManifest, node_pid: int,
        registration_epoch: int,
    ) -> StoredPublicationIntent:
        """Atomically journal the complete intent before GCS reporting.

        The manifest is intentionally not marked frozen: child pin ACKs still
        have to arrive before ``freeze_discovery`` advances the state.
        """

        intent = StoredPublicationIntent(
            header, manifest, node_pid, registration_epoch
        )
        with self._lock:
            publication_id = intent.publication_id
            previous = self._records.get(publication_id)
            if previous is not None:
                if previous.header != header:
                    raise StoredPublicationConflictError(
                        "publication_id is already bound to another header"
                    )
                if previous.publication_intent != intent:
                    raise StoredPublicationConflictError(
                        "publication_id is already bound to another intent"
                    )
                return intent
            record = _PublicationRecord(
                header=header,
                publication_intent=intent,
                transfers=list(manifest.transfers),
                prepared=[False] * len(manifest.transfers),
                promotion_intents=[False] * len(manifest.transfers),
                promoted=[False] * len(manifest.transfers),
            )
            self._records[publication_id] = record
            self._append_event(
                record, StoredPublicationEventKind.PUBLICATION_OPENED, header
            )
            for index, transfer in enumerate(manifest.transfers):
                self._append_event(
                    record, StoredPublicationEventKind.INTENT_APPENDED,
                    (index, transfer),
                )
            return intent

    def mark_intent_reported(
        self, publication_id: StoredPublicationID,
        intent: StoredPublicationIntent,
    ) -> bool:
        """Record that an exact GCS intent ACK passed adapter validation."""

        if not isinstance(intent, StoredPublicationIntent):
            raise TypeError("intent must be a StoredPublicationIntent")
        with self._lock:
            record = self._record(publication_id)
            if record.publication_intent != intent:
                raise StoredPublicationConflictError(
                    "GCS ACK does not match the journaled intent"
                )
            if record.intent_reported:
                return False
            self._require_state(record, StoredPublicationState.OPEN)
            record.intent_reported = True
            self._append_event(
                record,
                StoredPublicationEventKind.PUBLICATION_INTENT_REPORTED,
                intent,
            )
            return True

    def require_intent_reported(
        self, publication_id: StoredPublicationID,
        intent: StoredPublicationIntent,
    ) -> None:
        """Fence every child-side effect behind the exact GCS ACK."""

        if not isinstance(intent, StoredPublicationIntent):
            raise TypeError("intent must be a StoredPublicationIntent")
        with self._lock:
            record = self._record(publication_id)
            if record.publication_intent != intent:
                raise StoredPublicationConflictError(
                    "effect intent does not match the journaled intent"
                )
            if not record.intent_reported:
                raise StoredPublicationStateError(
                    "publication intent is not acknowledged by GCS"
                )

    def append_pin_intent(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        """Journal a provisional+final release pair before prepare RPC."""

        if not isinstance(transfer, PreparedContainedTransfer):
            raise TypeError("transfer must be a PreparedContainedTransfer")
        with self._lock:
            record = self._record(publication_id)
            self._require_index(index, allow_append=True, size=len(record.transfers))
            if index < len(record.transfers):
                if record.transfers[index] != transfer:
                    raise StoredPublicationConflictError(
                        "pin intent index is already bound to another transfer"
                    )
                return False
            if record.publication_intent is not None:
                raise StoredPublicationConflictError(
                    "complete publication intent cannot append another transfer"
                )
            self._require_state(record, StoredPublicationState.OPEN)
            self._validate_transfer(record.header, transfer)
            if transfer in record.transfers:
                raise StoredPublicationConflictError(
                    "one transfer cannot occupy multiple manifest slots"
                )
            record.transfers.append(transfer)
            record.prepared.append(False)
            record.promotion_intents.append(False)
            record.promoted.append(False)
            self._append_event(
                record, StoredPublicationEventKind.INTENT_APPENDED,
                (index, transfer),
            )
            return True

    def mark_pin_prepared(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        """Record an exact prepare ACK; the intent must already exist."""

        with self._lock:
            record = self._record(publication_id)
            self._matching_transfer(record, index, transfer)
            if record.prepared[index]:
                return False
            self._require_state(record, StoredPublicationState.OPEN)
            record.prepared[index] = True
            self._append_event(
                record, StoredPublicationEventKind.PIN_PREPARED,
                (index, transfer),
            )
            return True

    def freeze_discovery(
        self, publication_id: StoredPublicationID,
        manifest: StoredPublicationManifest,
    ) -> bool:
        """Close serialization only after every provisional pin ACK."""

        if not isinstance(manifest, StoredPublicationManifest):
            raise TypeError("manifest must be a StoredPublicationManifest")
        with self._lock:
            record = self._record(publication_id)
            if record.frozen is not None:
                if record.frozen != manifest:
                    raise StoredPublicationConflictError(
                        "frozen publication manifest changed"
                    )
                return False
            self._require_state(record, StoredPublicationState.OPEN)
            if manifest.publication_id != publication_id:
                raise StoredPublicationConflictError(
                    "manifest belongs to another publication"
                )
            if manifest.transfers != tuple(record.transfers):
                raise StoredPublicationConflictError(
                    "manifest does not exactly match ordered pin intents"
                )
            if not all(record.prepared):
                raise StoredPublicationStateError(
                    "discovery cannot freeze before every pin prepare ACK"
                )
            record.frozen = manifest
            record.state = StoredPublicationState.PINS_PREPARED
            self._append_event(
                record, StoredPublicationEventKind.DISCOVERY_FROZEN, manifest
            )
            return True

    def begin_graph_prepare(
        self, publication_id: StoredPublicationID,
        manifest: ContainedGraphManifest,
    ) -> bool:
        """Persist the exact graph compensator before GCS PREPARE."""

        if not isinstance(manifest, ContainedGraphManifest):
            raise TypeError("manifest must be a ContainedGraphManifest")
        with self._lock:
            record = self._record(publication_id)
            if record.graph_manifest is not None:
                if record.graph_manifest != manifest:
                    raise StoredPublicationConflictError(
                        "graph prepare intent changed"
                    )
                return False
            self._require_state(record, StoredPublicationState.PINS_PREPARED)
            expected = self._expected_graph_manifest(record)
            if manifest != expected:
                raise StoredPublicationConflictError(
                    "graph prepare does not match the frozen publication"
                )
            record.graph_manifest = manifest
            self._append_event(
                record, StoredPublicationEventKind.GRAPH_PREPARE_INTENT, manifest
            )
            return True

    def mark_graph_prepared(
        self, publication_id: StoredPublicationID,
        manifest: ContainedGraphManifest,
    ) -> bool:
        with self._lock:
            record = self._record(publication_id)
            if record.graph_manifest != manifest:
                raise StoredPublicationConflictError(
                    "graph prepare ACK does not match its journaled intent"
                )
            if record.graph_prepared:
                return False
            self._require_state(record, StoredPublicationState.PINS_PREPARED)
            record.graph_prepared = True
            record.state = StoredPublicationState.GRAPH_PREPARED
            self._append_event(
                record, StoredPublicationEventKind.GRAPH_PREPARED, manifest
            )
            return True

    def begin_replica_seal(
        self, publication_id: StoredPublicationID,
        descriptor: ResultDescriptor,
    ) -> bool:
        """Persist exact drop identity before asking the Node store to seal."""

        if not isinstance(descriptor, ResultDescriptor):
            raise TypeError("descriptor must be a ResultDescriptor")
        with self._lock:
            record = self._record(publication_id)
            if record.replica_descriptor is not None:
                if record.replica_descriptor != descriptor:
                    raise StoredPublicationConflictError(
                        "replica seal intent changed"
                    )
                return False
            self._require_state(record, StoredPublicationState.GRAPH_PREPARED)
            self._validate_descriptor(record, descriptor)
            record.replica_descriptor = descriptor
            self._append_event(
                record, StoredPublicationEventKind.REPLICA_SEAL_INTENT,
                descriptor,
            )
            return True

    def mark_replica_sealed(
        self, publication_id: StoredPublicationID,
        descriptor: ResultDescriptor,
    ) -> bool:
        with self._lock:
            record = self._record(publication_id)
            if record.replica_descriptor != descriptor:
                raise StoredPublicationConflictError(
                    "replica seal ACK does not match its journaled intent"
                )
            if record.replica_sealed:
                return False
            self._require_state(record, StoredPublicationState.GRAPH_PREPARED)
            record.replica_sealed = True
            record.state = StoredPublicationState.SEALED
            self._append_event(
                record, StoredPublicationEventKind.REPLICA_SEALED, descriptor
            )
            return True

    def begin_pin_promotion(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        """Persist old->new identity before the child-owner atomic swap."""

        with self._lock:
            record = self._record(publication_id)
            self._matching_transfer(record, index, transfer)
            if record.promotion_intents[index]:
                return False
            if record.state not in (
                StoredPublicationState.SEALED,
                StoredPublicationState.PROMOTING,
            ):
                self._wrong_state(record, "pin promotion requires SEALED")
            if not record.prepared[index]:
                raise StoredPublicationStateError(
                    "an unprepared pin cannot be promoted"
                )
            record.promotion_intents[index] = True
            record.state = StoredPublicationState.PROMOTING
            self._append_event(
                record, StoredPublicationEventKind.PROMOTION_INTENT,
                (index, transfer.provisional_hold, transfer.final_hold),
            )
            return True

    def mark_pin_promoted(
        self, publication_id: StoredPublicationID, index: int,
        transfer: PreparedContainedTransfer,
    ) -> bool:
        with self._lock:
            record = self._record(publication_id)
            self._matching_transfer(record, index, transfer)
            if record.promoted[index]:
                return False
            if not record.promotion_intents[index]:
                raise StoredPublicationStateError(
                    "promotion ACK arrived before its durable intent"
                )
            self._require_state(record, StoredPublicationState.PROMOTING)
            record.promoted[index] = True
            self._append_event(
                record, StoredPublicationEventKind.PIN_PROMOTED,
                (index, transfer.provisional_hold, transfer.final_hold),
            )
            return True

    def complete_lease(
        self, publication_id: StoredPublicationID,
        terminal: StoredPublicationLeaseTerminal,
    ) -> StoredPublicationEnvelope:
        """Linearize SUCCEEDED only after every final hold is authoritative."""

        if not isinstance(terminal, StoredPublicationLeaseTerminal):
            raise TypeError(
                "terminal must be a StoredPublicationLeaseTerminal"
            )
        with self._lock:
            record = self._record(publication_id)
            if terminal.publication_id != publication_id:
                raise StoredPublicationConflictError(
                    "lease terminal belongs to another publication"
                )
            if record.lease_terminal is not None:
                if record.lease_terminal != terminal:
                    raise StoredPublicationConflictError(
                        "lease terminal changed on replay"
                    )
                return self._envelope(record)
            if terminal.status is not TaskReplyStatus.SUCCEEDED:
                raise StoredPublicationStateError(
                    "a stored publication may only complete as SUCCEEDED"
                )
            if record.state not in (
                StoredPublicationState.SEALED,
                StoredPublicationState.PROMOTING,
            ):
                self._wrong_state(record, "lease completion requires SEALED")
            if not (
                record.replica_sealed
                and record.graph_prepared
                and all(record.promoted)
            ):
                raise StoredPublicationStateError(
                    "lease completion requires every final-hold promotion ACK"
                )
            record.lease_terminal = terminal
            record.state = StoredPublicationState.COMPLETED_UNCLAIMED
            self._append_event(
                record, StoredPublicationEventKind.LEASE_COMPLETED, terminal
            )
            return self._envelope(record)

    def claim(
        self, publication_id: StoredPublicationID,
        claim: StoredPublicationClaim,
    ) -> bool:
        if not isinstance(claim, StoredPublicationClaim):
            raise TypeError("claim must be a StoredPublicationClaim")
        with self._lock:
            record = self._record(publication_id)
            if record.claim is not None:
                if record.claim != claim:
                    raise StoredPublicationConflictError(
                        "publication was claimed with another identity"
                    )
                return False
            self._require_state(
                record, StoredPublicationState.COMPLETED_UNCLAIMED
            )
            frozen = record.frozen
            assert frozen is not None
            if (
                claim.publication_id != publication_id
                or claim.claimant_worker_id
                != record.header.outer_owner_worker_id
                or claim.manifest_digest != frozen.manifest_digest
            ):
                raise StoredPublicationConflictError(
                    "claim does not match publication owner and digest"
                )
            record.claim = claim
            record.state = StoredPublicationState.ADOPTING
            self._append_event(
                record, StoredPublicationEventKind.PUBLICATION_CLAIMED, claim
            )
            return True

    def mark_adopted(
        self, publication_id: StoredPublicationID,
        adoption: StoredPublicationAdoption,
    ) -> bool:
        if not isinstance(adoption, StoredPublicationAdoption):
            raise TypeError("adoption must be a StoredPublicationAdoption")
        with self._lock:
            record = self._record(publication_id)
            if record.adoption is not None:
                if record.adoption != adoption:
                    raise StoredPublicationConflictError(
                        "publication adoption receipt changed"
                    )
                return False
            self._require_state(record, StoredPublicationState.ADOPTING)
            if adoption.claim != record.claim:
                raise StoredPublicationConflictError(
                    "adoption does not match the accepted claim"
                )
            record.adoption = adoption
            record.state = StoredPublicationState.ADOPTED
            self._append_event(
                record, StoredPublicationEventKind.PUBLICATION_ADOPTED,
                adoption,
            )
            return True

    def begin_rollback(
        self, publication_id: StoredPublicationID, reason: str
    ) -> Tuple[StoredRollbackObligation, ...]:
        """Freeze graph->replica->final->provisional compensation work."""

        if not isinstance(reason, str) or not reason:
            raise ValueError("rollback reason must be a non-empty string")
        with self._lock:
            record = self._record(publication_id)
            if record.rollback_reason is not None:
                if record.rollback_reason != reason:
                    raise StoredPublicationConflictError(
                        "rollback identity was rebound to another reason"
                    )
                return record.rollback_obligations
            if record.lease_terminal is not None or record.state in (
                StoredPublicationState.COMPLETED_UNCLAIMED,
                StoredPublicationState.ADOPTING,
                StoredPublicationState.ADOPTED,
            ):
                raise StoredPublicationStateError(
                    "a completed publication must be adopted, never rolled back"
                )
            if record.state is StoredPublicationState.ROLLED_BACK:
                self._wrong_state(record, "publication already rolled back")
            obligations = self._build_rollback(record)
            record.rollback_reason = reason
            record.rollback_obligations = obligations
            record.compensation_acked = [False] * len(obligations)
            record.state = (
                StoredPublicationState.ROLLED_BACK
                if not obligations
                else StoredPublicationState.ROLLING_BACK
            )
            self._append_event(
                record, StoredPublicationEventKind.ROLLBACK_STARTED,
                (reason, obligations),
            )
            if not obligations:
                self._append_event(
                    record, StoredPublicationEventKind.ROLLED_BACK, reason
                )
            return obligations

    def mark_compensation_acked(
        self, publication_id: StoredPublicationID, ordinal: int,
        obligation: StoredRollbackObligation,
    ) -> bool:
        """Discharge only the next exact obligation; ACK replay is a no-op."""

        if not isinstance(obligation, StoredRollbackObligation):
            raise TypeError("obligation must be a StoredRollbackObligation")
        with self._lock:
            record = self._record(publication_id)
            self._require_index(
                ordinal, allow_append=False,
                size=len(record.rollback_obligations),
            )
            expected = record.rollback_obligations[ordinal]
            if expected != obligation:
                raise StoredPublicationConflictError(
                    "compensation ACK does not match the frozen obligation"
                )
            if record.compensation_acked[ordinal]:
                return False
            self._require_state(record, StoredPublicationState.ROLLING_BACK)
            first_pending = record.compensation_acked.index(False)
            if ordinal != first_pending:
                raise StoredPublicationStateError(
                    "compensations must acknowledge in frozen rollback order"
                )
            record.compensation_acked[ordinal] = True
            self._append_event(
                record, StoredPublicationEventKind.COMPENSATION_ACKED, obligation
            )
            if all(record.compensation_acked):
                record.state = StoredPublicationState.ROLLED_BACK
                self._append_event(
                    record, StoredPublicationEventKind.ROLLED_BACK,
                    record.rollback_reason,
                )
            return True

    def snapshot(
        self, publication_id: StoredPublicationID
    ) -> StoredPublicationSnapshot:
        with self._lock:
            return self._snapshot(self._record(publication_id))

    def mark_owner_death_cleaned(
        self, publication_id: StoredPublicationID, terminal: object,
    ) -> bool:
        """Install an independent owner-lifetime tombstone exactly once."""

        from .publication_owner_death import (
            FinalizePublicationOwnerDeathEffect,
            StoredPublicationOwnerDeathKey,
        )

        if not isinstance(terminal, FinalizePublicationOwnerDeathEffect):
            raise TypeError(
                "terminal must be a FinalizePublicationOwnerDeathEffect"
            )
        with self._lock:
            record = self._record(publication_id)
            if (
                terminal.key != StoredPublicationOwnerDeathKey(publication_id)
                or terminal.publishing_node_id != record.header.node_id
                or terminal.owner_death.worker_id
                != record.header.outer_owner_worker_id
            ):
                raise StoredPublicationConflictError(
                    "stored owner-death terminal changed publication identity"
                )
            if record.owner_death_cleanup is not None:
                if record.owner_death_cleanup != terminal:
                    raise StoredPublicationConflictError(
                        "stored owner-death terminal changed on replay"
                    )
                return False
            record.owner_death_cleanup = terminal
            record.state = StoredPublicationState.OWNER_DEATH_CLEANED
            self._append_event(
                record, StoredPublicationEventKind.OWNER_DEATH_CLEANED, terminal
            )
            return True

    def publication_ids(self) -> Tuple[StoredPublicationID, ...]:
        with self._lock:
            return tuple(sorted(self._records))

    @staticmethod
    def _validate_transfer(
        header: StoredPublicationHeader, transfer: PreparedContainedTransfer
    ) -> None:
        outer = header.publication_id.outer_object_id
        if (
            transfer.provisional_hold.container_object_id != outer
            or transfer.final_hold.container_object_id != outer
        ):
            raise StoredPublicationConflictError(
                "transfer holds belong to another outer object"
            )
        if (
            transfer.provisional_hold.container_owner_worker_id
            != header.executor_worker_id
        ):
            raise StoredPublicationConflictError(
                "provisional hold must belong to the executor incarnation"
            )
        if (
            transfer.final_hold.container_owner_worker_id
            != header.outer_owner_worker_id
        ):
            raise StoredPublicationConflictError(
                "final hold must belong to the logical outer owner"
            )
        if isinstance(transfer.source, OwnedContainedSource):
            if transfer.source.owner_worker_id != header.executor_worker_id:
                raise StoredPublicationConflictError(
                    "only the owning executor may use a direct source"
                )
        elif transfer.source.borrower_worker_id != header.executor_worker_id:
            raise StoredPublicationConflictError(
                "borrowed source must belong to the executor"
            )

    @staticmethod
    def _validate_descriptor(
        record: _PublicationRecord, descriptor: ResultDescriptor
    ) -> None:
        frozen = record.frozen
        assert frozen is not None
        if descriptor.storage is not ResultStorage.OBJECT_STORE:
            raise StoredPublicationConflictError(
                "stored publication requires an object-store descriptor"
            )
        if (
            descriptor.object_id != record.header.publication_id.outer_object_id
            or descriptor.owner_worker_id
            != record.header.outer_owner_worker_id
            or descriptor.node_id != record.header.node_id
            or descriptor.size_bytes != frozen.size_bytes
            or descriptor.checksum != frozen.checksum
        ):
            raise StoredPublicationConflictError(
                "sealed descriptor does not match frozen publication identity"
            )

    @staticmethod
    def _expected_graph_manifest(
        record: _PublicationRecord,
    ) -> ContainedGraphManifest:
        frozen = record.frozen
        assert frozen is not None
        return frozen.graph_manifest(record.header.outer_owner_worker_id)

    @staticmethod
    def _build_rollback(
        record: _PublicationRecord,
    ) -> Tuple[StoredRollbackObligation, ...]:
        publication_id = record.header.publication_id
        values: list[tuple[StoredRollbackKind, StoredRollbackIdentity]] = []
        # The order is a contract.  Exact abort/drop/release operations are
        # idempotent, so it is also safe when an ACK was lost or promotion was
        # only partially observed.
        if record.graph_manifest is not None:
            values.append((
                StoredRollbackKind.GRAPH_MANIFEST, record.graph_manifest
            ))
        if record.replica_descriptor is not None:
            values.append((
                StoredRollbackKind.SEALED_REPLICA, record.replica_descriptor
            ))
        for index in reversed(range(len(record.transfers))):
            transfer = record.transfers[index]
            values.append((
                StoredRollbackKind.FINAL_HOLD,
                StoredPinRelease(
                    index, transfer.contained_object_id,
                    transfer.contained_owner_worker_id,
                    transfer.contained_owner_address, transfer.final_hold,
                ),
            ))
        for index in reversed(range(len(record.transfers))):
            transfer = record.transfers[index]
            values.append((
                StoredRollbackKind.PROVISIONAL_HOLD,
                StoredPinRelease(
                    index, transfer.contained_object_id,
                    transfer.contained_owner_worker_id,
                    transfer.contained_owner_address,
                    transfer.provisional_hold,
                ),
            ))
        return tuple(
            StoredRollbackObligation(publication_id, ordinal, kind, identity)
            for ordinal, (kind, identity) in enumerate(values)
        )

    @staticmethod
    def _matching_transfer(
        record: _PublicationRecord, index: int,
        transfer: PreparedContainedTransfer,
    ) -> None:
        StoredPublicationJournal._require_index(
            index, allow_append=False, size=len(record.transfers)
        )
        if record.transfers[index] != transfer:
            raise StoredPublicationConflictError(
                "transfer does not match its journaled intent"
            )

    @staticmethod
    def _require_index(index: int, *, allow_append: bool, size: int) -> None:
        if isinstance(index, bool) or not isinstance(index, int) or index < 0:
            raise ValueError("index must be a non-negative integer")
        maximum = size if allow_append else size - 1
        if index > maximum:
            raise StoredPublicationStateError(
                "operation skipped a required journal index"
            )

    def _record(self, publication_id: StoredPublicationID) -> _PublicationRecord:
        if not isinstance(publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        try:
            return self._records[publication_id]
        except KeyError as exc:
            raise StoredPublicationStateError(
                "stored publication was not opened"
            ) from exc

    @staticmethod
    def _require_state(
        record: _PublicationRecord, expected: StoredPublicationState
    ) -> None:
        if record.state is not expected:
            StoredPublicationJournal._wrong_state(
                record, f"operation requires {expected.value}"
            )

    @staticmethod
    def _wrong_state(record: _PublicationRecord, message: str) -> None:
        raise StoredPublicationStateError(
            f"{message}; current state is {record.state.value}"
        )

    @staticmethod
    def _append_event(
        record: _PublicationRecord, kind: StoredPublicationEventKind,
        payload: object,
    ) -> None:
        record.events.append(StoredPublicationEvent(
            len(record.events), kind, record.header.publication_id, payload
        ))

    @staticmethod
    def _envelope(record: _PublicationRecord) -> StoredPublicationEnvelope:
        frozen = record.frozen
        graph = record.graph_manifest
        descriptor = record.replica_descriptor
        assert frozen is not None
        assert graph is not None
        assert descriptor is not None
        return StoredPublicationEnvelope(
            record.header.publication_id, frozen.manifest_digest,
            descriptor, graph, record.header,
        )

    @staticmethod
    def _snapshot(record: _PublicationRecord) -> StoredPublicationSnapshot:
        return StoredPublicationSnapshot(
            header=record.header,
            state=record.state,
            publication_intent=record.publication_intent,
            intent_reported=record.intent_reported,
            transfers=tuple(record.transfers),
            intent_written=(True,) * len(record.transfers),
            prepared_bitmap=tuple(record.prepared),
            promotion_intent_bitmap=tuple(record.promotion_intents),
            promoted_bitmap=tuple(record.promoted),
            frozen_manifest=record.frozen,
            graph_manifest=record.graph_manifest,
            graph_prepared=record.graph_prepared,
            replica_descriptor=record.replica_descriptor,
            replica_sealed=record.replica_sealed,
            lease_terminal=record.lease_terminal,
            claim=record.claim,
            adoption=record.adoption,
            rollback_reason=record.rollback_reason,
            rollback_obligations=record.rollback_obligations,
            compensation_bitmap=tuple(record.compensation_acked),
            owner_death_cleanup=record.owner_death_cleanup,
            events=tuple(record.events),
        )
