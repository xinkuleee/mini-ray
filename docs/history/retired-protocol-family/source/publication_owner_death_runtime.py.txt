"""Executable GCS driver for publication outer-owner death cleanup.

The pure :mod:`miniray.publication_owner_death` coordinator owns ordering and
terminal state.  This adapter performs exactly the currently selected effect,
converts wire replies into the coordinator's domain receipts, and acknowledges
only after every nested identity has been revalidated.  Transport ambiguity
therefore leaves the same immutable effect pending for byte-identical replay.

``drive`` attempts one effect by default.  Passing ``max_effects=None`` is the
explicit converge mode; it still stops when a retryable effect (for example a
PINNED replica) makes no progress, rather than spinning inside GCS.
"""

from __future__ import annotations

from dataclasses import replace
from threading import Lock, RLock
from typing import Callable, Optional

from . import protocol
from .contained_cycle import (
    ContainedGraphManifestDisposition, ContainedGraphManifestReceipt,
    ContainedGraphTransactionState,
)
from .ids import NodeID, WorkerID
from .publication_owner_death import (
    AbortPublicationGraphEffect, DropPublicationReplicaEffect,
    FencePublicationOwnerDeathEffect,
    FinalizePublicationOwnerDeathEffect, PublicationHoldDisposition,
    PublicationHoldReceipt, PublicationNodeFenceDisposition,
    PublicationNodeFenceReceipt, PublicationOwnerDeathCoordinator,
    PublicationOwnerDeathEffect, PublicationOwnerDeathEffectReceipt,
    PublicationOwnerDeathKey, PublicationOwnerDeathRequest,
    PublicationOwnerDeathSnapshot, PublicationPublisherFinalizeDisposition,
    PublicationPublisherFinalizeReceipt, PublicationReplicaDisposition,
    PublicationReplicaFenceStatus, PublicationReplicaReceipt,
    ReleasePublicationGraphEffect, ReleasePublicationHoldEffect,
)


class PublicationOwnerDeathRuntimeError(RuntimeError):
    """Base class for effect-driver failures."""


class PublicationOwnerDeathRuntimeConflictError(
    PublicationOwnerDeathRuntimeError
):
    """A callback rebound an immutable effect or wire identity."""


class PublicationOwnerDeathRuntimeStateError(
    PublicationOwnerDeathRuntimeError
):
    """A callback produced no terminal proof for the selected effect."""


InstallNodeFence = Callable[[protocol.InstallOwnerDeathFence], object]
ReleaseChild = Callable[
    [ReleasePublicationHoldEffect, protocol.ReleaseContainedReference], object
]
AbortGraph = Callable[[protocol.AbortContainedGraph], object]
ReleaseGraph = Callable[[protocol.ReleaseContainedGraphContainer], object]
DropReplica = Callable[[protocol.DropObjectReplica], object]
FinalizePublisher = Callable[[protocol.FinalizePublicationOwnerDeath], object]
NodeDeathLookup = Callable[[NodeID], object | None]


class PublicationOwnerDeathRuntime:
    """Thin, replay-safe side-effect driver around the pure coordinator."""

    def __init__(
        self, coordinator: PublicationOwnerDeathCoordinator, *,
        install_node_fence: InstallNodeFence,
        release_child: ReleaseChild, abort_graph: AbortGraph,
        release_graph: ReleaseGraph, drop_replica: DropReplica,
        finalize_publisher: FinalizePublisher,
        node_death_lookup: NodeDeathLookup,
    ) -> None:
        if not isinstance(coordinator, PublicationOwnerDeathCoordinator):
            raise TypeError(
                "coordinator must be a PublicationOwnerDeathCoordinator"
            )
        callbacks = (
            (install_node_fence, "install_node_fence"),
            (release_child, "release_child"),
            (abort_graph, "abort_graph"),
            (release_graph, "release_graph"),
            (drop_replica, "drop_replica"),
            (finalize_publisher, "finalize_publisher"),
            (node_death_lookup, "node_death_lookup"),
        )
        for callback, name in callbacks:
            if not callable(callback):
                raise TypeError("{} must be callable".format(name))
        self.coordinator = coordinator
        self._install_node_fence = install_node_fence
        self._release_child = release_child
        self._abort_graph = abort_graph
        self._release_graph = release_graph
        self._drop_replica = drop_replica
        self._finalize_publisher = finalize_publisher
        self._node_death_lookup = node_death_lookup
        self._drive_locks: dict[PublicationOwnerDeathKey, Lock] = {}
        self._receipts: dict[
            PublicationOwnerDeathEffect, PublicationOwnerDeathEffectReceipt
        ] = {}
        self._lock = RLock()

    def begin(
        self, request: PublicationOwnerDeathRequest,
    ) -> PublicationOwnerDeathSnapshot:
        """Admit or exactly replay one pure cleanup request."""

        snapshot = self.coordinator.begin(request)
        self._drive_lock(snapshot.request.key)
        return snapshot

    def snapshot(
        self, key: PublicationOwnerDeathKey,
    ) -> PublicationOwnerDeathSnapshot:
        return self.coordinator.snapshot(key)

    def next_effect(
        self, key: PublicationOwnerDeathKey,
    ) -> Optional[PublicationOwnerDeathEffect]:
        return self.coordinator.next_effect(key)

    def drive_one(
        self, key: PublicationOwnerDeathKey,
    ) -> PublicationOwnerDeathSnapshot:
        """Attempt at most one exact effect and persist only a terminal ACK."""

        with self._drive_lock(key):
            effect = self.coordinator.next_effect(key)
            if effect is None:
                return self.coordinator.snapshot(key)
            receipt = self._perform(effect)
            if receipt is None:
                # PINNED/NODE_DRAINING are observations, not deletion ACKs.
                return self.coordinator.snapshot(key)
            snapshot = self.coordinator.acknowledge(effect, receipt)
            self._remember_receipt(effect, receipt)
            return snapshot

    def drive(
        self, key: PublicationOwnerDeathKey, *,
        max_effects: Optional[int] = 1,
    ) -> PublicationOwnerDeathSnapshot:
        """Drive a bounded number of effects, or explicitly converge.

        ``None`` means no fixed effect bound.  A retryable no-progress reply
        always returns to the caller so scheduling/backoff remains external.
        """

        if max_effects is not None and (
            isinstance(max_effects, bool)
            or not isinstance(max_effects, int)
            or max_effects <= 0
        ):
            raise ValueError("max_effects must be a positive integer or None")
        attempted = 0
        snapshot = self.coordinator.snapshot(key)
        while snapshot.next_effect is not None and (
            max_effects is None or attempted < max_effects
        ):
            before = snapshot
            snapshot = self.drive_one(key)
            attempted += 1
            if snapshot.acknowledged_effects == before.acknowledged_effects:
                break
        return snapshot

    def has_active_operations(self) -> bool:
        return self.coordinator.has_active_operations()

    def has_active_obligations(self) -> bool:
        return self.has_active_operations()

    def close_admission(self) -> None:
        self.coordinator.close_admission()

    def _perform(
        self, effect: PublicationOwnerDeathEffect,
    ) -> Optional[PublicationOwnerDeathEffectReceipt]:
        if isinstance(effect, FencePublicationOwnerDeathEffect):
            return self._perform_fence(effect)
        if isinstance(effect, ReleasePublicationHoldEffect):
            return self._perform_hold(effect)
        if isinstance(effect, AbortPublicationGraphEffect):
            return self._perform_abort(effect)
        if isinstance(effect, ReleasePublicationGraphEffect):
            return self._perform_release_graph(effect)
        if isinstance(effect, DropPublicationReplicaEffect):
            return self._perform_drop(effect)
        if isinstance(effect, FinalizePublicationOwnerDeathEffect):
            return self._perform_finalize(effect)
        raise TypeError("effect is not a publication owner-death effect")

    def _perform_fence(
        self, effect: FencePublicationOwnerDeathEffect,
    ) -> PublicationNodeFenceReceipt:
        death = self._lookup_node_death(effect.node_id)
        if death is not None:
            return PublicationNodeFenceReceipt(
                effect, PublicationNodeFenceDisposition.NODE_DEAD,
                node_death=death,
            )
        request = protocol.InstallOwnerDeathFence(
            effect.request_id, effect.owner_death, effect.node_id,
            effect.expected_replicas,
            protocol.OwnerDeathFenceScope.PUBLICATION_EXACT,
        )
        try:
            candidate = self._install_node_fence(request)
        except (KeyboardInterrupt, SystemExit):
            raise
        except Exception:
            death = self._lookup_node_death(effect.node_id)
            if death is not None:
                return PublicationNodeFenceReceipt(
                    effect, PublicationNodeFenceDisposition.NODE_DEAD,
                    node_death=death,
                )
            raise
        if not isinstance(candidate, protocol.InstallOwnerDeathFenceReply):
            raise PublicationOwnerDeathRuntimeStateError(
                "Node fence returned an invalid reply type"
            )
        try:
            reply = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "Node fence returned a malformed reply"
            ) from exc
        if reply.request != request:
            raise PublicationOwnerDeathRuntimeConflictError(
                "Node fence reply changed the exact request"
            )
        if reply.disposition is protocol.OwnerDeathFenceDisposition.CONFLICT:
            raise PublicationOwnerDeathRuntimeConflictError(
                reply.error or "Node rejected the exact owner-death fence"
            )
        statuses = tuple(
            self._replica_fence_status(observation.status)
            for observation in reply.observations
        )
        if any(
            status is PublicationReplicaFenceStatus.CONFLICT
            for status in statuses
        ):
            raise PublicationOwnerDeathRuntimeConflictError(
                "Node fence observed conflicting replica metadata"
            )
        return PublicationNodeFenceReceipt(
            effect, PublicationNodeFenceDisposition.FENCED, statuses,
        )

    def _perform_hold(
        self, effect: ReleasePublicationHoldEffect,
    ) -> PublicationHoldReceipt:
        request = protocol.ReleaseContainedReference(
            effect.edge.contained_object_id,
            effect.edge.contained_owner_worker_id, effect.hold,
        )
        candidate = self._release_child(effect, request)
        if isinstance(candidate, protocol.WorkerDeathRecord):
            death = self._validated_worker_death(
                candidate, effect.edge.contained_owner_worker_id
            )
            return PublicationHoldReceipt(
                effect, PublicationHoldDisposition.OWNER_DEAD, death
            )
        if not isinstance(candidate, protocol.ReleaseContainedReferenceReply):
            raise PublicationOwnerDeathRuntimeStateError(
                "child release returned neither an ACK nor death proof"
            )
        try:
            reply = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "child release returned a malformed reply"
            ) from exc
        if (
            reply.object_id != request.object_id
            or reply.owner_worker_id != request.owner_worker_id
            or reply.hold != request.hold
        ):
            raise PublicationOwnerDeathRuntimeConflictError(
                "child release reply changed object, owner, or hold identity"
            )
        if not reply.accepted or reply.error is not None:
            raise PublicationOwnerDeathRuntimeStateError(
                reply.error or "child owner rejected the exact hold release"
            )
        return PublicationHoldReceipt(
            effect,
            (PublicationHoldDisposition.RELEASED if reply.released else
             PublicationHoldDisposition.ALREADY_RELEASED),
        )

    def _perform_abort(
        self, effect: AbortPublicationGraphEffect,
    ) -> ContainedGraphManifestReceipt:
        request = protocol.AbortContainedGraph(effect.manifest)
        reply = self._validated_graph_reply(
            request, self._abort_graph(request)
        )
        receipt = reply.receipt
        assert isinstance(receipt, ContainedGraphManifestReceipt)
        if (
            receipt.state is not ContainedGraphTransactionState.ABORTED
            or receipt.disposition not in (
                ContainedGraphManifestDisposition.APPLIED,
                ContainedGraphManifestDisposition.ALREADY_ABORTED,
            )
            or receipt.released_edges
        ):
            raise PublicationOwnerDeathRuntimeStateError(
                "graph ABORT did not acknowledge the exact manifest"
            )
        return receipt

    def _perform_release_graph(
        self, effect: ReleasePublicationGraphEffect,
    ) -> ContainedGraphManifestReceipt:
        request = protocol.ReleaseContainedGraphContainer(
            effect.manifest, effect.container_object_id
        )
        reply = self._validated_graph_reply(
            request, self._release_graph(request)
        )
        receipt = reply.receipt
        assert isinstance(receipt, ContainedGraphManifestReceipt)
        expected = tuple(
            edge for edge in effect.manifest.ordered_edges
            if edge.container_object_id == effect.container_object_id
        )
        if (
            receipt.state is not ContainedGraphTransactionState.COMMITTED
            or receipt.disposition not in (
                ContainedGraphManifestDisposition.RELEASED,
                ContainedGraphManifestDisposition.ALREADY_RELEASED,
            )
            or receipt.released_edges != expected
        ):
            raise PublicationOwnerDeathRuntimeStateError(
                "graph RELEASE did not acknowledge every exact edge"
            )
        return receipt

    def _perform_drop(
        self, effect: DropPublicationReplicaEffect,
    ) -> Optional[PublicationReplicaReceipt]:
        fence = self._replica_fence_receipt(effect)
        if fence.disposition is PublicationNodeFenceDisposition.NODE_DEAD:
            assert fence.node_death is not None
            return PublicationReplicaReceipt(
                effect, PublicationReplicaDisposition.NODE_DEAD,
                fence.node_death,
            )
        index = fence.effect.expected_replicas.index(effect.descriptor)
        status = fence.replica_statuses[index]
        if status is PublicationReplicaFenceStatus.ABSENT:
            return PublicationReplicaReceipt(
                effect, PublicationReplicaDisposition.ABSENT_AT_FENCE
            )
        if status is PublicationReplicaFenceStatus.CONFLICT:
            raise PublicationOwnerDeathRuntimeConflictError(
                "conflicting fence observation cannot authorize replica drop"
            )
        death = self._lookup_node_death(effect.descriptor.node_id)
        if death is not None:
            return PublicationReplicaReceipt(
                effect, PublicationReplicaDisposition.NODE_DEAD, death
            )
        descriptor = effect.descriptor
        request = protocol.DropObjectReplica(
            descriptor.object_id, descriptor.producer_attempt_id,
            descriptor.owner_worker_id, descriptor.node_id,
            descriptor.checksum,
        )
        try:
            candidate = self._drop_replica(request)
        except (KeyboardInterrupt, SystemExit):
            raise
        except Exception:
            death = self._lookup_node_death(descriptor.node_id)
            if death is not None:
                return PublicationReplicaReceipt(
                    effect, PublicationReplicaDisposition.NODE_DEAD, death
                )
            raise
        if not isinstance(candidate, protocol.DropObjectReplicaReply):
            raise PublicationOwnerDeathRuntimeStateError(
                "replica drop returned an invalid reply type"
            )
        try:
            reply = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "replica drop returned a malformed reply"
            ) from exc
        if (
            reply.object_id != request.object_id
            or reply.producer_attempt_id != request.producer_attempt_id
            or reply.owner_worker_id != request.owner_worker_id
            or reply.node_id != request.node_id
            or reply.checksum != request.checksum
        ):
            raise PublicationOwnerDeathRuntimeConflictError(
                "replica drop reply changed the exact descriptor identity"
            )
        if reply.status is protocol.DropObjectReplicaStatus.DROPPED:
            return PublicationReplicaReceipt(
                effect, PublicationReplicaDisposition.DROPPED
            )
        if reply.status is protocol.DropObjectReplicaStatus.ALREADY_DROPPED:
            return PublicationReplicaReceipt(
                effect, PublicationReplicaDisposition.ALREADY_DROPPED
            )
        if reply.status in (
            protocol.DropObjectReplicaStatus.PINNED,
            protocol.DropObjectReplicaStatus.NODE_DRAINING,
        ):
            death = self._lookup_node_death(descriptor.node_id)
            if death is not None:
                return PublicationReplicaReceipt(
                    effect, PublicationReplicaDisposition.NODE_DEAD, death
                )
            return None
        if reply.status in (
            protocol.DropObjectReplicaStatus.STALE_EPOCH,
            protocol.DropObjectReplicaStatus.INCONSISTENT,
        ):
            raise PublicationOwnerDeathRuntimeConflictError(
                reply.error or "replica drop observed conflicting metadata"
            )
        raise PublicationOwnerDeathRuntimeStateError(
            reply.error or "replica drop was not acknowledged"
        )

    def _perform_finalize(
        self, effect: FinalizePublicationOwnerDeathEffect,
    ) -> PublicationPublisherFinalizeReceipt:
        death = self._lookup_node_death(effect.publishing_node_id)
        if death is not None:
            return PublicationPublisherFinalizeReceipt(
                effect, PublicationPublisherFinalizeDisposition.NODE_DEAD,
                death,
            )
        try:
            request = protocol.FinalizePublicationOwnerDeath.from_effect(
                effect
            )
            candidate = self._finalize_publisher(request)
        except (KeyboardInterrupt, SystemExit):
            raise
        except Exception:
            death = self._lookup_node_death(effect.publishing_node_id)
            if death is not None:
                return PublicationPublisherFinalizeReceipt(
                    effect,
                    PublicationPublisherFinalizeDisposition.NODE_DEAD, death,
                )
            raise
        if not isinstance(
            candidate, protocol.FinalizePublicationOwnerDeathReply
        ):
            raise PublicationOwnerDeathRuntimeStateError(
                "publisher finalization returned an invalid wire reply"
            )
        try:
            reply = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "publisher finalization returned a malformed reply"
            ) from exc
        if reply.request != request:
            raise PublicationOwnerDeathRuntimeConflictError(
                "publisher finalization changed the exact request"
            )
        if reply.disposition is (
            protocol.PublicationOwnerDeathFinalizeDisposition.CONFLICT
        ):
            raise PublicationOwnerDeathRuntimeConflictError(
                reply.error or "publisher rejected the exact finalization"
            )
        disposition = (
            PublicationPublisherFinalizeDisposition.FINALIZED
            if reply.disposition is (
                protocol.PublicationOwnerDeathFinalizeDisposition
                .OWNER_DEATH_CLEANED
            )
            else PublicationPublisherFinalizeDisposition.ALREADY_FINALIZED
        )
        return PublicationPublisherFinalizeReceipt(effect, disposition)

    @staticmethod
    def _validated_graph_reply(
        request: protocol.ContainedGraphRequest, candidate: object,
    ) -> protocol.ContainedGraphReply:
        if not isinstance(candidate, protocol.ContainedGraphReply):
            raise PublicationOwnerDeathRuntimeStateError(
                "graph mutation returned an invalid reply type"
            )
        try:
            reply = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "graph mutation returned a malformed reply"
            ) from exc
        if reply.request != request:
            raise PublicationOwnerDeathRuntimeConflictError(
                "graph mutation reply changed the exact request"
            )
        if reply.receipt is None:
            raise PublicationOwnerDeathRuntimeStateError(
                reply.error or "graph mutation was not acknowledged"
            )
        return reply

    @staticmethod
    def _replica_fence_status(
        status: protocol.OwnerDeathReplicaStatus,
    ) -> PublicationReplicaFenceStatus:
        try:
            return PublicationReplicaFenceStatus(status.value)
        except (AttributeError, ValueError) as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "Node fence returned an unknown replica observation"
            ) from exc

    def _replica_fence_receipt(
        self, effect: DropPublicationReplicaEffect,
    ) -> PublicationNodeFenceReceipt:
        with self._lock:
            matches = tuple(
                receipt for candidate, receipt in self._receipts.items()
                if (
                    isinstance(candidate, FencePublicationOwnerDeathEffect)
                    and effect.descriptor in candidate.expected_replicas
                    and isinstance(receipt, PublicationNodeFenceReceipt)
                )
            )
        if len(matches) != 1:
            raise PublicationOwnerDeathRuntimeStateError(
                "replica drop lacks one exact acknowledged Node fence"
            )
        return matches[0]

    def _lookup_node_death(
        self, node_id: NodeID,
    ) -> Optional[protocol.NodeDeathRecord]:
        candidate = self._node_death_lookup(node_id)
        if candidate is None:
            return None
        return self._validated_node_death(candidate, node_id)

    @staticmethod
    def _validated_node_death(
        candidate: object, node_id: NodeID,
    ) -> protocol.NodeDeathRecord:
        if not isinstance(candidate, protocol.NodeDeathRecord):
            raise PublicationOwnerDeathRuntimeStateError(
                "node_death_lookup returned an invalid proof type"
            )
        try:
            death = replace(candidate)
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "node_death_lookup returned a malformed proof"
            ) from exc
        if death.node_id != node_id:
            raise PublicationOwnerDeathRuntimeConflictError(
                "Node death proof names another Node"
            )
        if death.reason is not protocol.NodeDeathReason.PROCESS_EXIT:
            raise PublicationOwnerDeathRuntimeStateError(
                "expected Node shutdown is not a replica-loss proof"
            )
        return death

    @staticmethod
    def _validated_worker_death(
        candidate: protocol.WorkerDeathRecord, worker_id: WorkerID,
    ) -> protocol.WorkerDeathRecord:
        incarnation = candidate.incarnation
        try:
            rebuilt = protocol.WorkerIncarnation(
                incarnation.node_id, incarnation.node_pid,
                incarnation.node_registration_epoch, incarnation.worker_id,
                incarnation.worker_pid,
            )
            death = protocol.WorkerDeathRecord(
                candidate.detection_id, rebuilt, candidate.death_epoch,
                candidate.exit_code, candidate.reason,
            )
        except Exception as exc:
            raise PublicationOwnerDeathRuntimeConflictError(
                "child owner returned a malformed death proof"
            ) from exc
        if death.worker_id != worker_id:
            raise PublicationOwnerDeathRuntimeConflictError(
                "child owner death proof names another Worker"
            )
        if death.reason not in (
            protocol.WorkerDeathReason.PROCESS_EXIT,
            protocol.WorkerDeathReason.NODE_EXIT,
        ):
            raise PublicationOwnerDeathRuntimeStateError(
                "expected Worker shutdown is not a child-loss proof"
            )
        return death

    def _remember_receipt(
        self, effect: PublicationOwnerDeathEffect,
        receipt: PublicationOwnerDeathEffectReceipt,
    ) -> None:
        with self._lock:
            previous = self._receipts.get(effect)
            if previous is not None and previous != receipt:
                raise PublicationOwnerDeathRuntimeConflictError(
                    "effect receipt changed on replay"
                )
            self._receipts[effect] = receipt

    def _drive_lock(self, key: PublicationOwnerDeathKey) -> Lock:
        with self._lock:
            return self._drive_locks.setdefault(key, Lock())
