"""Pure GCS saga for a live owner after its publishing Node is lost.

The publishing Node normally owns stored-publication rollback and the final
adoption acknowledgement.  Once that exact Node incarnation has a committed
death record, neither responsibility can remain in the dead process.  This
module is the small durable reducer that lets GCS take over without pretending
that GCS owns user objects or can adopt a result on an owner's behalf.

There is deliberately no RPC callback in this module.  ``next_effect`` returns
one immutable operation, an outer driver performs it without holding a GCS
authority lock, and ``acknowledge`` records only an exact typed receipt.  If an
ACK response is lost, replaying the same effect and receipt is a no-op.

The decisive split remains Node ``Complete``:

* before successful Complete, graph reservations and both possible pin
  identities are compensated and the publication becomes ``ROLLED_BACK``;
* after successful Complete, the live logical owner must first linearize a
  LOST/FENCED retirement CAS.  Only then may this reducer retire the old graph
  and pins.  It never claims, publishes, wakes, or adopts the lost result.

The final resolution stays as a tombstone.  Delayed graph prepares, owner CAS
replays, and pin releases therefore cannot resurrect an already-retired
publication identity.
"""

class StoredNodeLossError(RuntimeError):
    """Base class for publishing-Node-loss recovery failures."""


class StoredNodeLossConflictError(StoredNodeLossError):
    """A stable publication or effect identity was rebound."""


class StoredNodeLossStateError(StoredNodeLossError):
    """An input or acknowledgement contradicts the frozen saga."""


class StoredNodeLossMode(str, Enum):
    PRECOMPLETE_ROLLBACK = "PRECOMPLETE_ROLLBACK"
    POSTCOMPLETE_LOST_RESULT = "POSTCOMPLETE_LOST_RESULT"


class StoredNodeLossPhase(str, Enum):
    OWNER_RETIREMENT = "OWNER_RETIREMENT"
    GRAPH_ABORT = "GRAPH_ABORT"
    HOLDS = "HOLDS"
    GRAPH_RELEASE = "GRAPH_RELEASE"
    REPLICA_PROOF = "REPLICA_PROOF"
    RESOLVED = "RESOLVED"


class StoredNodeLossTerminal(str, Enum):
    ROLLED_BACK = "ROLLED_BACK"
    LOST_RESULT_RETIRED = "LOST_RESULT_RETIRED"
    SUPERSEDED = "SUPERSEDED"


class StoredNodeLossGraphState(str, Enum):
    """Exact GCS graph fact observed after the Node death barrier."""

    NOT_FOUND = "NOT_FOUND"
    PREPARED = "PREPARED"
    COMMITTED = "COMMITTED"
    ABORTED = "ABORTED"


class StoredNodeLossHoldKind(str, Enum):
    FINAL = "FINAL"
    PROVISIONAL = "PROVISIONAL"


class StoredOwnerRetirementDisposition(str, Enum):
    """Atomic result at the live logical outer owner.

    An owner implementation retains the first exact receipt as its retirement
    tombstone.  Replays consequently return the same disposition, not a newly
    interpreted view of owner state.
    """

    PENDING_MARKED_LOST = "PENDING_MARKED_LOST"
    READY_STORED_MARKED_LOST = "READY_STORED_MARKED_LOST"
    FENCED = "FENCED"


class StoredHoldResolutionDisposition(str, Enum):
    RELEASED = "RELEASED"
    ALREADY_RELEASED = "ALREADY_RELEASED"
    OWNER_DEAD = "OWNER_DEAD"


class StoredReplicaResolutionDisposition(str, Enum):
    DEAD_NODE_REPLICA_GONE = "DEAD_NODE_REPLICA_GONE"
    NO_REPLICA_REPORTED = "NO_REPLICA_REPORTED"


@dataclass(frozen=True)
class NodeIncarnationWitness:
    """Bind a publication header to one registered Node incarnation.

    Intent-backed recovery records carry the same PID and registration epoch;
    this normalized witness keeps the saga independent of publication storage
    while preventing a NodeID-only observation from authorizing cleanup of a
    different physical incarnation.
    """

    header: StoredPublicationHeader
    node_pid: int
    registration_epoch: int

    def __post_init__(self) -> None:
        if not isinstance(self.header, StoredPublicationHeader):
            raise TypeError("header must be a StoredPublicationHeader")
        _require_positive_int(self.node_pid, "node_pid")
        _require_positive_int(self.registration_epoch, "registration_epoch")

    def validate_death(self, death: protocol.NodeDeathRecord) -> None:
        if not isinstance(death, protocol.NodeDeathRecord):
            raise TypeError("node_death must be a NodeDeathRecord")
        if (
            death.node_id != self.header.node_id
            or death.node_pid != self.node_pid
            or death.registration_epoch != self.registration_epoch
        ):
            raise StoredNodeLossConflictError(
                "Node death does not match the publishing incarnation witness"
            )

    @classmethod
    def from_intent(
        cls, intent: StoredPublicationIntent
    ) -> "NodeIncarnationWitness":
        if not isinstance(intent, StoredPublicationIntent):
            raise TypeError("intent must be a StoredPublicationIntent")
        return cls(
            intent.header, intent.node_pid, intent.registration_epoch
        )


@dataclass(frozen=True)
class StoredNodeLossGraphWitness:
    """Exact manifest state queried after the publishing Node died.

    ``active_edges`` and ``released_container_ids`` make an already-released
    committed manifest distinguishable from a live committed manifest.  This
    matters after an ACK is lost or GCS reconstructs the coordinator itself.
    """

    manifest: ContainedGraphManifest
    state: StoredNodeLossGraphState
    active_edges: Tuple[ContainedReferenceEdge, ...] = ()
    released_container_ids: Tuple[ObjectID, ...] = ()

    def __post_init__(self) -> None:
        if not isinstance(self.manifest, ContainedGraphManifest):
            raise TypeError("manifest must be a ContainedGraphManifest")
        if not isinstance(self.state, StoredNodeLossGraphState):
            raise TypeError("state must be a StoredNodeLossGraphState")
        active = tuple(self.active_edges)
        released = tuple(self.released_container_ids)
        if any(not isinstance(edge, ContainedReferenceEdge) for edge in active):
            raise TypeError(
                "active_edges must contain ContainedReferenceEdge values"
            )
        if len(active) != len(set(active)):
            raise ValueError("active_edges must be unique")
        if active != tuple(
            edge for edge in self.manifest.ordered_edges if edge in set(active)
        ):
            raise ValueError(
                "active_edges must be an ordered subset of the manifest"
            )
        if any(not isinstance(value, ObjectID) for value in released):
            raise TypeError(
                "released_container_ids must contain ObjectID values"
            )
        if len(released) != len(set(released)):
            raise ValueError("released_container_ids must be unique")
        containers = {
            edge.container_object_id for edge in self.manifest.ordered_edges
        }
        if not set(released).issubset(containers):
            raise ValueError(
                "released_container_ids must belong to the graph manifest"
            )
        if self.state is not StoredNodeLossGraphState.COMMITTED:
            if active or released:
                raise ValueError(
                    "only a committed graph may have active or released edges"
                )
        else:
            expected_active = tuple(
                edge for edge in self.manifest.ordered_edges
                if edge.container_object_id not in set(released)
            )
            if active != expected_active:
                raise ValueError(
                    "committed active_edges must exactly exclude released "
                    "manifest containers"
                )
        object.__setattr__(self, "active_edges", active)
        object.__setattr__(
            self, "released_container_ids", tuple(sorted(released))
        )

    def container_released(self, object_id: ObjectID) -> bool:
        return object_id in self.released_container_ids


@dataclass(frozen=True)
class StoredNodeLossRequest:
    node_death: protocol.NodeDeathRecord
    node_incarnation: NodeIncarnationWitness
    recovery: StoredPublicationRecoveryRecord
    graph: StoredNodeLossGraphWitness

    def __post_init__(self) -> None:
        if not isinstance(self.node_death, protocol.NodeDeathRecord):
            raise TypeError("node_death must be a NodeDeathRecord")
        if not isinstance(self.node_incarnation, NodeIncarnationWitness):
            raise TypeError(
                "node_incarnation must be a NodeIncarnationWitness"
            )
        if not isinstance(self.recovery, StoredPublicationRecoveryRecord):
            raise TypeError(
                "recovery must be a StoredPublicationRecoveryRecord"
            )
        if not isinstance(self.graph, StoredNodeLossGraphWitness):
            raise TypeError("graph must be a StoredNodeLossGraphWitness")
        self.node_incarnation.validate_death(self.node_death)
        header = self.node_incarnation.header
        manifest = self.recovery.manifest
        if header.publication_id != manifest.publication_id:
            raise StoredNodeLossConflictError(
                "incarnation witness belongs to another publication"
            )
        expected_graph = manifest.graph_manifest(
            header.outer_owner_worker_id
        )
        if self.graph.manifest != expected_graph:
            raise StoredNodeLossConflictError(
                "graph witness does not match the recovered publication"
            )
        descriptor = self.recovery.sealed_descriptor
        if descriptor is not None and (
            descriptor.storage is not protocol.ResultStorage.OBJECT_STORE
            or descriptor.object_id != manifest.publication_id.outer_object_id
            or descriptor.owner_worker_id != header.outer_owner_worker_id
            or descriptor.node_id != header.node_id
            or descriptor.size_bytes != manifest.size_bytes
            or descriptor.checksum != manifest.checksum
        ):
            raise StoredNodeLossConflictError(
                "sealed descriptor does not match the dead Node publication"
            )
        terminal = self.recovery.lease_terminal
        if (
            terminal is not None
            and terminal.publication_id != manifest.publication_id
        ):
            raise StoredNodeLossConflictError(
                "lease terminal belongs to another publication"
            )
        for transfer in manifest.transfers:
            if (
                transfer.provisional_hold.container_object_id
                != manifest.publication_id.outer_object_id
                or transfer.final_hold.container_object_id
                != manifest.publication_id.outer_object_id
                or transfer.provisional_hold.container_owner_worker_id
                != header.executor_worker_id
                or transfer.final_hold.container_owner_worker_id
                != header.outer_owner_worker_id
            ):
                raise StoredNodeLossConflictError(
                    "recovered transfer custody does not match the publication"
                )
        successful = bool(
            terminal is not None
            and terminal.status is protocol.TaskReplyStatus.SUCCEEDED
        )
        if successful and (
            descriptor is None or not all(self.recovery.promoted_bitmap)
        ):
            raise StoredNodeLossStateError(
                "successful Complete lacks sealed bytes or every promotion"
            )
        if (
            not successful
            and self.graph.state is StoredNodeLossGraphState.COMMITTED
        ):
            raise StoredNodeLossStateError(
                "a pre-Complete publication cannot own a committed graph"
            )
        if successful and self.graph.state not in (
            StoredNodeLossGraphState.PREPARED,
            StoredNodeLossGraphState.COMMITTED,
        ):
            raise StoredNodeLossStateError(
                "a completed publication requires a prepared or committed "
                "graph"
            )

    @classmethod
    def from_intent(
        cls, node_death: protocol.NodeDeathRecord,
        recovery: StoredPublicationRecoveryRecord,
        graph: StoredNodeLossGraphWitness,
    ) -> "StoredNodeLossRequest":
        """Build recovery proof directly from the durable pre-effect intent."""

        if not isinstance(recovery, StoredPublicationRecoveryRecord):
            raise TypeError(
                "recovery must be a StoredPublicationRecoveryRecord"
            )
        intent = recovery.intent
        if intent is None:
            raise StoredNodeLossStateError(
                "Node-loss recovery requires an intent-backed record"
            )
        return cls(
            node_death, NodeIncarnationWitness.from_intent(intent),
            recovery, graph,
        )

    @property
    def publication_id(self) -> StoredPublicationID:
        return self.recovery.publication_id

    @property
    def header(self) -> StoredPublicationHeader:
        return self.node_incarnation.header

    @property
    def mode(self) -> StoredNodeLossMode:
        terminal = self.recovery.lease_terminal
        if (
            terminal is not None
            and terminal.status is protocol.TaskReplyStatus.SUCCEEDED
        ):
            return StoredNodeLossMode.POSTCOMPLETE_LOST_RESULT
        return StoredNodeLossMode.PRECOMPLETE_ROLLBACK

    @property
    def envelope(self) -> Optional[StoredPublicationEnvelope]:
        if self.mode is StoredNodeLossMode.PRECOMPLETE_ROLLBACK:
            return None
        descriptor = self.recovery.sealed_descriptor
        assert descriptor is not None
        return StoredPublicationEnvelope(
            self.publication_id, self.recovery.manifest.manifest_digest,
            descriptor, self.graph.manifest, self.header,
        )


@dataclass(frozen=True)
class RetireStoredOwnerEffect:
    retirement_id: str
    envelope: StoredPublicationEnvelope
    node_death: protocol.NodeDeathRecord

    def __post_init__(self) -> None:
        if not isinstance(self.retirement_id, str) or not self.retirement_id:
            raise ValueError("retirement_id must be a non-empty string")
        if not isinstance(self.envelope, StoredPublicationEnvelope):
            raise TypeError("envelope must be a StoredPublicationEnvelope")
        if not isinstance(self.node_death, protocol.NodeDeathRecord):
            raise TypeError("node_death must be a NodeDeathRecord")
        if self.envelope.header.node_id != self.node_death.node_id:
            raise ValueError("owner retirement death names another Node")

    @property
    def publication_id(self) -> StoredPublicationID:
        return self.envelope.publication_id


@dataclass(frozen=True)
class StoredOwnerRetirementReceipt:
    effect: RetireStoredOwnerEffect
    disposition: StoredOwnerRetirementDisposition

    def __post_init__(self) -> None:
        if not isinstance(self.effect, RetireStoredOwnerEffect):
            raise TypeError("effect must be a RetireStoredOwnerEffect")
        if not isinstance(
            self.disposition, StoredOwnerRetirementDisposition
        ):
            raise TypeError("owner retirement disposition is invalid")


@dataclass(frozen=True)
class AbortStoredGraphEffect:
    publication_id: StoredPublicationID
    manifest: ContainedGraphManifest

    def __post_init__(self) -> None:
        _validate_publication_graph(self.publication_id, self.manifest)


@dataclass(frozen=True)
class ReleaseStoredHoldEffect:
    publication_id: StoredPublicationID
    ordinal: int
    transfer_index: int
    kind: StoredNodeLossHoldKind
    contained_object_id: ObjectID
    contained_owner_worker_id: WorkerID
    contained_owner_address: Tuple[str, int]
    hold: ContainedReferenceHold

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        _require_non_negative_int(self.ordinal, "ordinal")
        _require_non_negative_int(self.transfer_index, "transfer_index")
        if not isinstance(self.kind, StoredNodeLossHoldKind):
            raise TypeError("kind must be a StoredNodeLossHoldKind")
        if not isinstance(self.contained_object_id, ObjectID):
            raise TypeError("contained_object_id must be an ObjectID")
        if not isinstance(self.contained_owner_worker_id, WorkerID):
            raise TypeError(
                "contained_owner_worker_id must be a WorkerID"
            )
        _validate_address(self.contained_owner_address)
        if not isinstance(self.hold, ContainedReferenceHold):
            raise TypeError("hold must be a ContainedReferenceHold")
        if self.hold.container_object_id != self.publication_id.outer_object_id:
            raise ValueError("hold belongs to another outer object")


@dataclass(frozen=True)
class StoredHoldResolutionReceipt:
    effect: ReleaseStoredHoldEffect
    disposition: StoredHoldResolutionDisposition
    owner_death: Optional[protocol.WorkerDeathRecord] = None

    def __post_init__(self) -> None:
        if not isinstance(self.effect, ReleaseStoredHoldEffect):
            raise TypeError("effect must be a ReleaseStoredHoldEffect")
        if not isinstance(self.disposition, StoredHoldResolutionDisposition):
            raise TypeError("hold resolution disposition is invalid")
        if self.disposition is StoredHoldResolutionDisposition.OWNER_DEAD:
            if (
                not isinstance(self.owner_death, protocol.WorkerDeathRecord)
                or self.owner_death.worker_id
                != self.effect.contained_owner_worker_id
            ):
                raise ValueError(
                    "OWNER_DEAD requires the exact child-owner death proof"
                )
        elif self.owner_death is not None:
            raise ValueError(
                "a released hold receipt cannot contain an owner death proof"
            )


@dataclass(frozen=True)
class ReleaseStoredGraphEffect:
    publication_id: StoredPublicationID
    manifest: ContainedGraphManifest
    container_object_id: ObjectID

    def __post_init__(self) -> None:
        _validate_publication_graph(self.publication_id, self.manifest)
        if self.container_object_id != self.publication_id.outer_object_id:
            raise ValueError("graph release targets another outer object")
        if any(
            edge.container_object_id != self.container_object_id
            for edge in self.manifest.ordered_edges
        ):
            raise ValueError(
                "graph release container must own every manifest edge"
            )


@dataclass(frozen=True)
class ProveStoredReplicaGoneEffect:
    publication_id: StoredPublicationID
    descriptor: Optional[protocol.ResultDescriptor]
    node_death: protocol.NodeDeathRecord
    node_incarnation: NodeIncarnationWitness

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        if self.descriptor is not None and not isinstance(
            self.descriptor, protocol.ResultDescriptor
        ):
            raise TypeError("descriptor must be a ResultDescriptor or None")
        if not isinstance(self.node_death, protocol.NodeDeathRecord):
            raise TypeError("node_death must be a NodeDeathRecord")
        if not isinstance(self.node_incarnation, NodeIncarnationWitness):
            raise TypeError(
                "node_incarnation must be a NodeIncarnationWitness"
            )
        self.node_incarnation.validate_death(self.node_death)
        if (
            self.node_incarnation.header.publication_id != self.publication_id
            or (
                self.descriptor is not None
                and self.descriptor.node_id != self.node_death.node_id
            )
        ):
            raise ValueError(
                "replica proof does not match publication and dead Node"
            )


@dataclass(frozen=True)
class StoredReplicaResolutionReceipt:
    effect: ProveStoredReplicaGoneEffect
    disposition: StoredReplicaResolutionDisposition

    def __post_init__(self) -> None:
        if not isinstance(self.effect, ProveStoredReplicaGoneEffect):
            raise TypeError(
                "effect must be a ProveStoredReplicaGoneEffect"
            )
        if not isinstance(
            self.disposition, StoredReplicaResolutionDisposition
        ):
            raise TypeError("replica resolution disposition is invalid")
        expected = (
            StoredReplicaResolutionDisposition.NO_REPLICA_REPORTED
            if self.effect.descriptor is None
            else StoredReplicaResolutionDisposition.DEAD_NODE_REPLICA_GONE
        )
        if self.disposition is not expected:
            raise ValueError(
                "replica resolution contradicts the recovered descriptor"
            )


StoredNodeLossEffect = Union[
    RetireStoredOwnerEffect, AbortStoredGraphEffect,
    ReleaseStoredHoldEffect, ReleaseStoredGraphEffect,
    ProveStoredReplicaGoneEffect,
]

StoredNodeLossEffectReceipt = Union[
    StoredOwnerRetirementReceipt, ContainedGraphManifestReceipt,
    StoredHoldResolutionReceipt, StoredReplicaResolutionReceipt,
]


@dataclass(frozen=True)
class StoredNodeLossResolution:
    """Permanent terminal receipt for one exact takeover request."""

    request: StoredNodeLossRequest
    terminal: StoredNodeLossTerminal
    effect_receipts: Tuple[
        Tuple[StoredNodeLossEffect, StoredNodeLossEffectReceipt], ...
    ]
    superseding_owner_death: Optional[protocol.WorkerDeathRecord] = None

    def __post_init__(self) -> None:
        if not isinstance(self.request, StoredNodeLossRequest):
            raise TypeError("request must be a StoredNodeLossRequest")
        if not isinstance(self.terminal, StoredNodeLossTerminal):
            raise TypeError("terminal must be a StoredNodeLossTerminal")
        if self.terminal is StoredNodeLossTerminal.SUPERSEDED:
            death = self.superseding_owner_death
            if (
                not isinstance(death, protocol.WorkerDeathRecord)
                or death.worker_id != self.request.header.outer_owner_worker_id
                or death.reason is protocol.WorkerDeathReason.EXPECTED
            ):
                raise ValueError(
                    "SUPERSEDED requires the exact outer-owner death proof"
                )
        else:
            expected = (
                StoredNodeLossTerminal.ROLLED_BACK
                if self.request.mode is StoredNodeLossMode.PRECOMPLETE_ROLLBACK
                else StoredNodeLossTerminal.LOST_RESULT_RETIRED
            )
            if self.terminal is not expected:
                raise ValueError(
                    "terminal contradicts the Complete boundary"
                )
            if self.superseding_owner_death is not None:
                raise ValueError(
                    "ordinary Node-loss terminal cannot carry owner death"
                )
        receipts = tuple(self.effect_receipts)
        if any(
            not isinstance(pair, tuple) or len(pair) != 2
            for pair in receipts
        ):
            raise TypeError("effect_receipts must contain effect/receipt pairs")
        object.__setattr__(self, "effect_receipts", receipts)

    @property
    def publication_id(self) -> StoredPublicationID:
        return self.request.publication_id


@dataclass(frozen=True)
class StoredNodeLossSnapshot:
    request: StoredNodeLossRequest
    mode: StoredNodeLossMode
    phase: StoredNodeLossPhase
    next_effect: Optional[StoredNodeLossEffect]
    acknowledged_effects: Tuple[StoredNodeLossEffect, ...]
    resolution: Optional[StoredNodeLossResolution]


@dataclass
class _SagaRecord:
    request: StoredNodeLossRequest
    owner_effect: Optional[RetireStoredOwnerEffect]
    graph_abort_effect: Optional[AbortStoredGraphEffect]
    hold_effects: Tuple[ReleaseStoredHoldEffect, ...]
    graph_release_effect: Optional[ReleaseStoredGraphEffect]
    replica_effect: ProveStoredReplicaGoneEffect
    effects: Tuple[StoredNodeLossEffect, ...]
    receipts: dict[StoredNodeLossEffect, StoredNodeLossEffectReceipt] = field(
        default_factory=dict
    )
    resolution: Optional[StoredNodeLossResolution] = None


class StoredNodeLossRecoveryCoordinator:
    """Replay-safe pure saga reducer owned by GCS."""

    def __init__(self) -> None:
        self._records: dict[StoredPublicationID, _SagaRecord] = {}
        self._admission_closed = False
        self._lock = RLock()

    def begin(self, request: StoredNodeLossRequest) -> StoredNodeLossSnapshot:
        """Freeze one exact Node-loss decision, or replay it."""

        if not isinstance(request, StoredNodeLossRequest):
            raise TypeError("request must be a StoredNodeLossRequest")
        with self._lock:
            previous = self._records.get(request.publication_id)
            if previous is not None:
                if previous.request != request:
                    raise StoredNodeLossConflictError(
                        "publication Node-loss evidence changed on replay"
                    )
                return self._snapshot(previous)
            if self._admission_closed:
                raise StoredNodeLossStateError(
                    "stored Node-loss recovery admission is closed"
                )
            record = self._build_record(request)
            self._records[request.publication_id] = record
            return self._snapshot(record)

    def next_effect(
        self, publication_id: StoredPublicationID
    ) -> Optional[StoredNodeLossEffect]:
        """Return the first unacknowledged exact effect."""

        with self._lock:
            record = self._record(publication_id)
            return self._next_effect(record)

    def acknowledge(
        self, effect: StoredNodeLossEffect,
        receipt: StoredNodeLossEffectReceipt,
    ) -> StoredNodeLossSnapshot:
        """Commit one exact ACK; exact response-loss replay is harmless."""

        if not isinstance(
            effect,
            (RetireStoredOwnerEffect, AbortStoredGraphEffect,
             ReleaseStoredHoldEffect, ReleaseStoredGraphEffect,
             ProveStoredReplicaGoneEffect),
        ):
            raise TypeError("effect is not a stored Node-loss effect")
        publication_id = effect.publication_id
        with self._lock:
            record = self._record(publication_id)
            previous = record.receipts.get(effect)
            if previous is not None:
                if previous != receipt:
                    raise StoredNodeLossConflictError(
                        "effect acknowledgement changed on replay"
                    )
                return self._snapshot(record)
            expected = self._next_effect(record)
            if expected != effect:
                raise StoredNodeLossStateError(
                    "effect is not the saga's next exact obligation"
                )
            self._validate_receipt(effect, receipt)
            record.receipts[effect] = receipt
            if effect is record.replica_effect:
                terminal = (
                    StoredNodeLossTerminal.ROLLED_BACK
                    if record.request.mode
                    is StoredNodeLossMode.PRECOMPLETE_ROLLBACK
                    else StoredNodeLossTerminal.LOST_RESULT_RETIRED
                )
                record.resolution = StoredNodeLossResolution(
                    record.request, terminal, tuple(
                        (item, record.receipts[item]) for item in record.effects
                    ),
                )
            return self._snapshot(record)

    def snapshot(
        self, publication_id: StoredPublicationID
    ) -> StoredNodeLossSnapshot:
        with self._lock:
            return self._snapshot(self._record(publication_id))

    def resolution(
        self, publication_id: StoredPublicationID
    ) -> Optional[StoredNodeLossResolution]:
        with self._lock:
            return self._record(publication_id).resolution

    def supersede(
        self, publication_id: StoredPublicationID,
        owner_death: protocol.WorkerDeathRecord,
    ) -> StoredNodeLossResolution:
        """Terminalize an active saga at the owner-death authority switch.

        Already acknowledged effects are retained in the tombstone.  No new
        effect is performed here; owner-death cleanup may import those exact
        receipts before it starts driving its remaining obligations.
        """

        if not isinstance(owner_death, protocol.WorkerDeathRecord):
            raise TypeError("owner_death must be a WorkerDeathRecord")
        with self._lock:
            record = self._record(publication_id)
            previous = record.resolution
            if previous is not None:
                if (
                    previous.terminal is StoredNodeLossTerminal.SUPERSEDED
                    and previous.superseding_owner_death == owner_death
                ):
                    return previous
                raise StoredNodeLossStateError(
                    "completed Node-loss saga cannot be superseded"
                )
            resolution = StoredNodeLossResolution(
                record.request, StoredNodeLossTerminal.SUPERSEDED,
                tuple(
                    (effect, record.receipts[effect])
                    for effect in record.effects
                    if effect in record.receipts
                ),
                owner_death,
            )
            record.resolution = resolution
            return resolution

    def close_admission(self) -> None:
        """Fence unseen sagas while allowing exact cleanup replays."""

        with self._lock:
            self._admission_closed = True

    def has_active_operations(self) -> bool:
        with self._lock:
            return any(
                record.resolution is None for record in self._records.values()
            )

    @staticmethod
    def _build_record(request: StoredNodeLossRequest) -> _SagaRecord:
        owner_effect = None
        if request.mode is StoredNodeLossMode.POSTCOMPLETE_LOST_RESULT:
            envelope = request.envelope
            assert envelope is not None
            owner_effect = RetireStoredOwnerEffect(
                _retirement_id(request), envelope, request.node_death
            )

        graph_abort_effect = None
        if request.graph.state in (
            StoredNodeLossGraphState.NOT_FOUND,
            StoredNodeLossGraphState.PREPARED,
        ) and request.graph.manifest.ordered_edges:
            # ABORT on NOT_FOUND intentionally writes the graph tombstone.  It
            # fences a PREPARE that left the dead Node before the death barrier
            # but reaches GCS after takeover begins.
            graph_abort_effect = AbortStoredGraphEffect(
                request.publication_id, request.graph.manifest
            )

        holds = _hold_effects(request)
        graph_release_effect = None
        if (
            request.graph.state is StoredNodeLossGraphState.COMMITTED
            and not request.graph.container_released(
                request.publication_id.outer_object_id
            )
            and request.graph.manifest.ordered_edges
        ):
            graph_release_effect = ReleaseStoredGraphEffect(
                request.publication_id, request.graph.manifest,
                request.publication_id.outer_object_id,
            )

        replica_effect = ProveStoredReplicaGoneEffect(
            request.publication_id, request.recovery.sealed_descriptor,
            request.node_death, request.node_incarnation,
        )
        effects = tuple(
            item for item in (owner_effect, graph_abort_effect)
            if item is not None
        ) + holds + tuple(
            item for item in (graph_release_effect, replica_effect)
            if item is not None
        )
        return _SagaRecord(
            request, owner_effect, graph_abort_effect, holds,
            graph_release_effect, replica_effect, effects,
        )

    @staticmethod
    def _next_effect(record: _SagaRecord) -> Optional[StoredNodeLossEffect]:
        if record.resolution is not None:
            return None
        for effect in record.effects:
            if effect not in record.receipts:
                return effect
        raise AssertionError(
            "all effects acknowledged without a terminal resolution"
        )

    @staticmethod
    def _validate_receipt(
        effect: StoredNodeLossEffect, receipt: StoredNodeLossEffectReceipt
    ) -> None:
        if isinstance(effect, RetireStoredOwnerEffect):
            if (
                not isinstance(receipt, StoredOwnerRetirementReceipt)
                or receipt.effect != effect
            ):
                raise StoredNodeLossStateError(
                    "owner retirement did not echo the exact CAS"
                )
            return
        if isinstance(effect, AbortStoredGraphEffect):
            if (
                not isinstance(receipt, ContainedGraphManifestReceipt)
                or receipt.manifest != effect.manifest
                or receipt.state is not ContainedGraphTransactionState.ABORTED
                or receipt.disposition not in (
                    ContainedGraphManifestDisposition.APPLIED,
                    ContainedGraphManifestDisposition.ALREADY_ABORTED,
                )
                or receipt.released_edges
            ):
                raise StoredNodeLossStateError(
                    "graph abort did not acknowledge the exact manifest"
                )
            return
        if isinstance(effect, ReleaseStoredHoldEffect):
            if (
                not isinstance(receipt, StoredHoldResolutionReceipt)
                or receipt.effect != effect
            ):
                raise StoredNodeLossStateError(
                    "hold resolution did not echo the exact hold"
                )
            return
        if isinstance(effect, ReleaseStoredGraphEffect):
            if (
                not isinstance(receipt, ContainedGraphManifestReceipt)
                or receipt.manifest != effect.manifest
                or receipt.state
                is not ContainedGraphTransactionState.COMMITTED
                or receipt.disposition not in (
                    ContainedGraphManifestDisposition.RELEASED,
                    ContainedGraphManifestDisposition.ALREADY_RELEASED,
                )
                or receipt.released_edges
                != effect.manifest.ordered_edges
            ):
                raise StoredNodeLossStateError(
                    "graph release did not acknowledge every exact edge"
                )
            return
        if (
            not isinstance(effect, ProveStoredReplicaGoneEffect)
            or not isinstance(receipt, StoredReplicaResolutionReceipt)
            or receipt.effect != effect
        ):
            raise StoredNodeLossStateError(
                "replica resolution did not echo the exact death proof"
            )

    def _record(self, publication_id: StoredPublicationID) -> _SagaRecord:
        if not isinstance(publication_id, StoredPublicationID):
            raise TypeError("publication_id must be a StoredPublicationID")
        try:
            return self._records[publication_id]
        except KeyError as exc:
            raise StoredNodeLossStateError(
                "stored Node-loss recovery has not begun"
            ) from exc

    @classmethod
    def _snapshot(cls, record: _SagaRecord) -> StoredNodeLossSnapshot:
        next_effect = cls._next_effect(record)
        if record.resolution is not None:
            phase = StoredNodeLossPhase.RESOLVED
        elif next_effect is record.owner_effect:
            phase = StoredNodeLossPhase.OWNER_RETIREMENT
        elif next_effect is record.graph_abort_effect:
            phase = StoredNodeLossPhase.GRAPH_ABORT
        elif isinstance(next_effect, ReleaseStoredHoldEffect):
            phase = StoredNodeLossPhase.HOLDS
        elif next_effect is record.graph_release_effect:
            phase = StoredNodeLossPhase.GRAPH_RELEASE
        else:
            assert next_effect is record.replica_effect
            phase = StoredNodeLossPhase.REPLICA_PROOF
        return StoredNodeLossSnapshot(
            record.request, record.request.mode, phase, next_effect,
            tuple(
                effect for effect in record.effects
                if effect in record.receipts
            ),
            record.resolution,
        )


def _hold_effects(
    request: StoredNodeLossRequest,
) -> Tuple[ReleaseStoredHoldEffect, ...]:
    values: list[tuple[int, StoredNodeLossHoldKind, PreparedContainedTransfer,
                       ContainedReferenceHold]] = []
    transfers = request.recovery.manifest.transfers
    # Final first, then possibly-live provisional identities.  A false GCS bit
    # is ambiguous: promotion may not have happened, or it happened and its
    # monotonic report was lost.  Tombstoning both identities is the only safe
    # compensation.
    for index in reversed(range(len(transfers))):
        transfer = transfers[index]
        values.append((
            index, StoredNodeLossHoldKind.FINAL, transfer,
            transfer.final_hold,
        ))
    for index in reversed(range(len(transfers))):
        if request.recovery.promoted_bitmap[index]:
            continue
        transfer = transfers[index]
        values.append((
            index, StoredNodeLossHoldKind.PROVISIONAL, transfer,
            transfer.provisional_hold,
        ))
    return tuple(
        ReleaseStoredHoldEffect(
            request.publication_id, ordinal, index, kind,
            transfer.contained_object_id,
            transfer.contained_owner_worker_id,
            transfer.contained_owner_address, hold,
        )
        for ordinal, (index, kind, transfer, hold) in enumerate(values)
    )


def _retirement_id(request: StoredNodeLossRequest) -> str:
    publication = request.publication_id
    death = request.node_death
    digest = hashlib.sha256()
    digest.update(b"miniray-stored-node-loss-retirement-v1\0")
    digest.update(bytes(publication.lease_id))
    digest.update(bytes(publication.task_id))
    digest.update(publication.attempt_id.attempt_number.to_bytes(8, "big"))
    digest.update(bytes(publication.outer_object_id.task_id))
    digest.update(publication.outer_object_id.return_index.to_bytes(8, "big"))
    digest.update(bytes(request.header.outer_owner_worker_id))
    digest.update(bytes(death.node_id))
    digest.update(death.node_pid.to_bytes(8, "big", signed=False))
    digest.update(death.registration_epoch.to_bytes(8, "big", signed=False))
    detection = death.detection_id.encode("utf-8")
    digest.update(len(detection).to_bytes(8, "big"))
    digest.update(detection)
    return "stored-node-loss:" + digest.hexdigest()


def _validate_publication_graph(
    publication_id: StoredPublicationID, manifest: ContainedGraphManifest
) -> None:
    if not isinstance(publication_id, StoredPublicationID):
        raise TypeError("publication_id must be a StoredPublicationID")
    if not isinstance(manifest, ContainedGraphManifest):
        raise TypeError("manifest must be a ContainedGraphManifest")
    if (
        manifest.publication_id != publication_id
        or manifest.transaction_id != publication_id.graph_transaction_id
    ):
        raise ValueError("graph manifest belongs to another publication")


def _validate_address(value: object) -> None:
    if (
        not isinstance(value, tuple)
        or len(value) != 2
        or not isinstance(value[0], str)
        or not value[0]
        or isinstance(value[1], bool)
        or not isinstance(value[1], int)
        or not 1 <= value[1] <= 65535
    ):
        raise ValueError(
            "contained_owner_address must be a bound (host, port) tuple"
        )


def _require_positive_int(value: object, label: str) -> None:
    if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
        raise ValueError(f"{label} must be a positive integer")


def _require_non_negative_int(value: object, label: str) -> None:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError(f"{label} must be a non-negative integer")
