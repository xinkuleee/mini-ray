"""Metadata-only GCS recovery authority for contained INLINE publications.

The Node reports the immutable manifest before effects, but result bytes stay
at the Node and outer owner.  Local Complete does not wait for its GCS report.
Consequently PREPARED without a terminal report means completion is *unknown*,
not that execution can be rolled back.  After Node death the outer owner must
choose KEEP (it retained the completed envelope) or DROP (fence late delivery
and clean up the lost result).  That choice is monotonic and contains no bytes.

Frozen work remains an immutable effect identity.  The later owner decision
lives in the current snapshot so retries cannot rewrite the original evidence.
This module performs no RPC and never reconstructs a payload from metadata.
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from enum import Enum
from threading import RLock
from typing import TYPE_CHECKING, Optional, Tuple

from .ids import NodeID, WorkerID
from .inline_contained_publication import (
    InlinePublicationManifest, InlinePublicationID,
)
from .protocol import (
    NodeDeathRecord, TaskReplyStatus, WorkerDeathReason, WorkerDeathRecord,
)

if TYPE_CHECKING:
    from .publication_owner_death import PublicationOwnerDeathResolution


class InlineRecoveryError(RuntimeError):
    """Base class for pure INLINE recovery failures."""


class InlineRecoveryConflictError(InlineRecoveryError):
    """A stable identity was replayed with different immutable facts."""


class InlineRecoveryStateError(InlineRecoveryError):
    """A requested transition contradicts the monotonic recovery state."""


class UnknownInlineRecoveryError(InlineRecoveryStateError, LookupError):
    """No complete INLINE intent is registered for the publication."""


def _positive_int(value: object, label: str) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
        raise ValueError("{} must be a positive integer".format(label))
    return value


@dataclass(frozen=True, order=True)
class InlinePublicationNodeIncarnation:
    """Exact publishing Node process registered with GCS."""

    node_id: NodeID
    node_pid: int
    registration_epoch: int

    def __post_init__(self) -> None:
        if not isinstance(self.node_id, NodeID):
            raise TypeError("node_id must be a NodeID")
        _positive_int(self.node_pid, "node_pid")
        _positive_int(self.registration_epoch, "registration_epoch")

    @classmethod
    def from_death(
        cls, death: NodeDeathRecord,
    ) -> "InlinePublicationNodeIncarnation":
        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        return cls(
            death.node_id, death.node_pid, death.registration_epoch
        )


@dataclass(frozen=True)
class InlinePublicationIntent:
    """Byte-free pre-effect identity retained by GCS."""

    manifest: InlinePublicationManifest
    node_pid: int
    registration_epoch: int

    def __post_init__(self) -> None:
        if not isinstance(self.manifest, InlinePublicationManifest):
            raise TypeError("manifest must be an InlinePublicationManifest")
        _positive_int(self.node_pid, "node_pid")
        _positive_int(self.registration_epoch, "registration_epoch")

    @property
    def publication_id(self) -> InlinePublicationID:
        return self.manifest.publication_id

    @property
    def node_incarnation(self) -> InlinePublicationNodeIncarnation:
        return InlinePublicationNodeIncarnation(
            self.manifest.publishing_node_id,
            self.node_pid, self.registration_epoch,
        )


@dataclass(frozen=True)
class InlinePublicationTerminal:
    """Durable successful Complete boundary for one exact publication."""

    publication_id: InlinePublicationID
    status: TaskReplyStatus

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, InlinePublicationID):
            raise TypeError("publication_id must be an InlinePublicationID")
        if self.status is not TaskReplyStatus.SUCCEEDED:
            raise ValueError(
                "INLINE publication terminal must be SUCCEEDED"
            )


class InlineRecoveryStage(str, Enum):
    INTENT = "INTENT"
    PREPARED = "PREPARED"
    TERMINAL = "TERMINAL"
    DECIDED = "DECIDED"
    ABORTED = "ABORTED"
    ADOPTED = "ADOPTED"
    RETIRED = "RETIRED"
    RESOLVED = "RESOLVED"


class InlineRecoveryDisposition(str, Enum):
    APPLIED = "APPLIED"
    ALREADY_RECORDED = "ALREADY_RECORDED"


class InlineRecoveryAction(str, Enum):
    PRECOMPLETE_ROLLBACK = "PRECOMPLETE_ROLLBACK"
    COMPLETION_UNKNOWN = "COMPLETION_UNKNOWN"
    POSTCOMPLETE_RESOLVE = "POSTCOMPLETE_RESOLVE"


class InlineRecoveryOwnerDecision(str, Enum):
    """Owner-local payload custody projected as a byte-free GCS fence."""

    KEEP = "KEEP"
    DROP = "DROP"


class InlineRecoveryResolution(str, Enum):
    ABORTED = "ABORTED"
    ROLLED_BACK = "ROLLED_BACK"
    ADOPTED = "ADOPTED"
    RETIRED = "RETIRED"
    LOST = "LOST"
    SUPERSEDED = "SUPERSEDED"


class InlineRecoveryState(str, Enum):
    """Compact diagnostic projection; immutable facts remain authoritative."""

    INTENT = "INTENT"
    PREPARED = "PREPARED"
    TERMINAL = "TERMINAL"
    FROZEN = "FROZEN"
    RESOLVED = "RESOLVED"


@dataclass(frozen=True)
class InlinePublicationRecoverySnapshot:
    """Immutable query result and frozen recovery identity."""

    intent: InlinePublicationIntent
    prepared: bool = False
    terminal: Optional[InlinePublicationTerminal] = None
    frozen_death: Optional[NodeDeathRecord] = None
    resolution: Optional[InlineRecoveryResolution] = None
    superseding_owner_death: Optional[WorkerDeathRecord] = None
    owner_decision: Optional[InlineRecoveryOwnerDecision] = None

    def __post_init__(self) -> None:
        if not isinstance(self.intent, InlinePublicationIntent):
            raise TypeError("intent must be an InlinePublicationIntent")
        if type(self.prepared) is not bool:
            raise TypeError("prepared must be a bool")
        if self.owner_decision is not None:
            if not isinstance(self.owner_decision, InlineRecoveryOwnerDecision):
                raise TypeError(
                    "owner_decision must be an InlineRecoveryOwnerDecision or None"
                )
            if not self.prepared or self.frozen_death is None:
                raise InlineRecoveryStateError(
                    "owner decision requires a prepared, death-frozen publication"
                )
        if self.terminal is not None:
            if not isinstance(self.terminal, InlinePublicationTerminal):
                raise TypeError(
                    "terminal must be an InlinePublicationTerminal or None"
                )
            if self.terminal.publication_id != self.publication_id:
                raise InlineRecoveryConflictError(
                    "terminal belongs to another INLINE publication"
                )
            if not self.prepared:
                raise InlineRecoveryStateError(
                    "successful terminal requires prepared publication"
                )
        if self.frozen_death is not None:
            if not isinstance(self.frozen_death, NodeDeathRecord):
                raise TypeError("frozen_death must be a NodeDeathRecord or None")
            if (
                InlinePublicationNodeIncarnation.from_death(
                    self.frozen_death
                ) != self.node_incarnation
            ):
                raise InlineRecoveryConflictError(
                    "death proof names another publishing Node incarnation"
                )
        if self.resolution is not None:
            if not isinstance(self.resolution, InlineRecoveryResolution):
                raise TypeError(
                    "resolution must be an InlineRecoveryResolution or None"
                )
            if self.frozen_death is not None:
                if (
                    self.resolution
                    is not InlineRecoveryResolution.SUPERSEDED
                    and self.resolution not in self.expected_resolutions
                ):
                    raise InlineRecoveryConflictError(
                        "resolution contradicts the frozen recovery action"
                    )
            elif self.resolution in (
                InlineRecoveryResolution.ROLLED_BACK,
                InlineRecoveryResolution.LOST,
            ):
                raise InlineRecoveryStateError(
                    "Node-loss cleanup requires a frozen Node death"
                )
            elif (
                self.resolution is InlineRecoveryResolution.ABORTED
                and self.terminal is not None
            ):
                raise InlineRecoveryConflictError(
                    "a successful INLINE publication cannot be aborted"
                )
            elif (
                self.resolution is InlineRecoveryResolution.ADOPTED
                and self.terminal is None
            ):
                raise InlineRecoveryConflictError(
                    "INLINE adoption requires the successful terminal"
                )
            elif (
                self.resolution is InlineRecoveryResolution.RETIRED
                and self.terminal is None
            ):
                raise InlineRecoveryConflictError(
                    "INLINE stale retirement requires the successful terminal"
                )
        if self.resolution is InlineRecoveryResolution.SUPERSEDED:
            death = self.superseding_owner_death
            if (
                not isinstance(death, WorkerDeathRecord)
                or death.worker_id != self.manifest.outer_owner_worker_id
                or death.reason is WorkerDeathReason.EXPECTED
                or self.frozen_death is None
            ):
                raise InlineRecoveryConflictError(
                    "SUPERSEDED requires the exact outer-owner death proof "
                    "and frozen Node-loss work"
                )
        if (
            self.resolution is not InlineRecoveryResolution.SUPERSEDED
            and self.superseding_owner_death is not None
        ):
            raise InlineRecoveryConflictError(
                "only SUPERSEDED may carry an outer-owner death proof"
            )
        if (
            self.resolution is InlineRecoveryResolution.LOST
            and self.owner_decision is not InlineRecoveryOwnerDecision.DROP
        ):
            raise InlineRecoveryStateError(
                "lost-result cleanup requires the exact owner DROP decision"
            )
        if (
            self.owner_decision is InlineRecoveryOwnerDecision.DROP
            and self.resolution in (
                InlineRecoveryResolution.ADOPTED,
                InlineRecoveryResolution.RETIRED,
            )
        ):
            raise InlineRecoveryConflictError(
                "DROP cannot be resolved through owner adoption or retirement"
            )
        if (
            self.owner_decision is not None
            and self.resolution is InlineRecoveryResolution.ABORTED
        ):
            raise InlineRecoveryConflictError(
                "owner-decided Node-loss work cannot become a live abort tombstone"
            )
        if (
            self.terminal is None
            and self.resolution in (
                InlineRecoveryResolution.ADOPTED, InlineRecoveryResolution.RETIRED
            )
            and self.owner_decision is not InlineRecoveryOwnerDecision.KEEP
        ):
            raise InlineRecoveryStateError(
                "unknown-Complete adoption requires the owner's KEEP witness"
            )

    @property
    def publication_id(self) -> InlinePublicationID:
        return self.intent.publication_id

    @property
    def manifest(self) -> InlinePublicationManifest:
        return self.intent.manifest

    @property
    def node_incarnation(self) -> InlinePublicationNodeIncarnation:
        return self.intent.node_incarnation

    @property
    def recovery_action(self) -> InlineRecoveryAction:
        if self.terminal is not None:
            return InlineRecoveryAction.POSTCOMPLETE_RESOLVE
        if self.prepared:
            return InlineRecoveryAction.COMPLETION_UNKNOWN
        return InlineRecoveryAction.PRECOMPLETE_ROLLBACK

    @property
    def expected_resolution(self) -> InlineRecoveryResolution:
        if self.recovery_action is InlineRecoveryAction.PRECOMPLETE_ROLLBACK:
            return InlineRecoveryResolution.ROLLED_BACK
        if self.owner_decision is InlineRecoveryOwnerDecision.DROP:
            return InlineRecoveryResolution.LOST
        return InlineRecoveryResolution.ADOPTED

    @property
    def expected_resolutions(self) -> Tuple[InlineRecoveryResolution, ...]:
        """Legal terminal outcomes when a Node death freezes this record.

        ``ABORTED`` is a pre-existing live-Node rollback tombstone, not the
        result of death recovery.  It is included so the exact death-bound
        query can expose that no INLINE result survived.
        """

        if self.recovery_action is InlineRecoveryAction.PRECOMPLETE_ROLLBACK:
            return (
                InlineRecoveryResolution.ROLLED_BACK,
                InlineRecoveryResolution.ABORTED,
            )
        return (
            InlineRecoveryResolution.ADOPTED,
            InlineRecoveryResolution.RETIRED,
            InlineRecoveryResolution.LOST,
        ) + (
            (InlineRecoveryResolution.ABORTED,)
            if self.terminal is None else ()
        )

    @property
    def frozen(self) -> bool:
        return self.frozen_death is not None

    @property
    def resolved(self) -> bool:
        return self.resolution is not None

    @property
    def state(self) -> InlineRecoveryState:
        if self.resolved:
            return InlineRecoveryState.RESOLVED
        if self.frozen:
            return InlineRecoveryState.FROZEN
        if self.terminal is not None:
            return InlineRecoveryState.TERMINAL
        if self.prepared:
            return InlineRecoveryState.PREPARED
        return InlineRecoveryState.INTENT


@dataclass(frozen=True)
class InlinePublicationRecoveryAck:
    """Exact acknowledgement of one monotonic registry transition."""

    stage: InlineRecoveryStage
    disposition: InlineRecoveryDisposition
    snapshot: InlinePublicationRecoverySnapshot

    def __post_init__(self) -> None:
        if not isinstance(self.stage, InlineRecoveryStage):
            raise TypeError("stage must be an InlineRecoveryStage")
        if not isinstance(self.disposition, InlineRecoveryDisposition):
            raise TypeError(
                "disposition must be an InlineRecoveryDisposition"
            )
        if not isinstance(self.snapshot, InlinePublicationRecoverySnapshot):
            raise TypeError(
                "snapshot must be an InlinePublicationRecoverySnapshot"
            )

    @property
    def record(self) -> InlinePublicationRecoverySnapshot:
        """Compatibility spelling for report/reply adapters."""

        return self.snapshot


@dataclass(frozen=True)
class InlinePublicationRecoveryWork:
    """One exact immutable item frozen by a publishing Node death."""

    death: NodeDeathRecord
    snapshot: InlinePublicationRecoverySnapshot
    action: InlineRecoveryAction

    def __post_init__(self) -> None:
        if not isinstance(self.death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        if not isinstance(self.snapshot, InlinePublicationRecoverySnapshot):
            raise TypeError(
                "snapshot must be an InlinePublicationRecoverySnapshot"
            )
        if not isinstance(self.action, InlineRecoveryAction):
            raise TypeError("action must be an InlineRecoveryAction")
        if self.snapshot.frozen_death != self.death:
            raise InlineRecoveryConflictError(
                "work item does not carry its exact frozen death proof"
            )
        if self.snapshot.recovery_action is not self.action:
            raise InlineRecoveryConflictError(
                "work action contradicts its frozen snapshot"
            )

    @property
    def publication_id(self) -> InlinePublicationID:
        return self.snapshot.publication_id

    @property
    def manifest(self) -> InlinePublicationManifest:
        return self.snapshot.manifest


class InlinePublicationRecoveryAuthority:
    """Thread-safe, side-effect-free GCS registry for INLINE recovery."""

    def __init__(self) -> None:
        self._records: dict[
            InlinePublicationID, InlinePublicationRecoverySnapshot
        ] = {}
        self._by_incarnation: dict[
            InlinePublicationNodeIncarnation, set[InlinePublicationID]
        ] = {}
        self._by_outer_owner: dict[
            WorkerID, set[InlinePublicationID]
        ] = {}
        self._deaths: dict[
            InlinePublicationNodeIncarnation, NodeDeathRecord
        ] = {}
        self._frozen_worksets: dict[
            InlinePublicationNodeIncarnation,
            Tuple[InlinePublicationRecoveryWork, ...],
        ] = {}
        self._owner_deaths: dict[WorkerID, WorkerDeathRecord] = {}
        self._frozen_owner_worksets: dict[
            WorkerID, Tuple[InlinePublicationRecoverySnapshot, ...]
        ] = {}
        self._owner_death_resolutions: dict[
            InlinePublicationID, PublicationOwnerDeathResolution
        ] = {}
        self._admission_closed = False
        self._lock = RLock()

    def report_intent(
        self, intent: InlinePublicationIntent,
    ) -> InlinePublicationRecoveryAck:
        """Register the complete byte-free manifest before effects."""

        if not isinstance(intent, InlinePublicationIntent):
            raise TypeError("intent must be an InlinePublicationIntent")
        with self._lock:
            publication_id = intent.publication_id
            previous = self._records.get(publication_id)
            if previous is not None:
                if previous.intent != intent:
                    raise InlineRecoveryConflictError(
                        "INLINE recovery intent changed on replay"
                    )
                return self._ack(
                    InlineRecoveryStage.INTENT, False, previous
                )
            if self._admission_closed:
                raise InlineRecoveryStateError(
                    "INLINE recovery admission is closed"
                )
            if intent.node_incarnation in self._deaths:
                raise InlineRecoveryStateError(
                    "publishing Node incarnation is already death-frozen"
                )
            owner_worker_id = (
                intent.manifest.outer_owner_worker_id
            )
            if owner_worker_id in self._owner_deaths:
                raise InlineRecoveryStateError(
                    "outer owner is already death-frozen"
                )
            snapshot = InlinePublicationRecoverySnapshot(intent)
            self._records[publication_id] = snapshot
            self._by_incarnation.setdefault(
                intent.node_incarnation, set()
            ).add(publication_id)
            self._by_outer_owner.setdefault(
                owner_worker_id, set()
            ).add(publication_id)
            return self._ack(InlineRecoveryStage.INTENT, True, snapshot)

    def report_prepared(
        self, publication_id: InlinePublicationID,
    ) -> InlinePublicationRecoveryAck:
        """Record graph PREPARE plus every final child hold ACK."""

        with self._lock:
            snapshot = self._record(publication_id)
            if snapshot.prepared:
                return self._ack(
                    InlineRecoveryStage.PREPARED, False, snapshot
                )
            self._require_mutable(snapshot)
            snapshot = replace(snapshot, prepared=True)
            self._records[publication_id] = snapshot
            return self._ack(InlineRecoveryStage.PREPARED, True, snapshot)

    def report_terminal(
        self, terminal: InlinePublicationTerminal,
    ) -> InlinePublicationRecoveryAck:
        """Record successful Complete only after preparation is durable."""

        if not isinstance(terminal, InlinePublicationTerminal):
            raise TypeError(
                "terminal must be an InlinePublicationTerminal"
            )
        with self._lock:
            snapshot = self._record(terminal.publication_id)
            if snapshot.terminal is not None:
                if snapshot.terminal != terminal:
                    raise InlineRecoveryConflictError(
                        "INLINE recovery terminal changed on replay"
                    )
                return self._ack(
                    InlineRecoveryStage.TERMINAL, False, snapshot
                )
            self._require_mutable(snapshot)
            if not snapshot.prepared:
                raise InlineRecoveryStateError(
                    "successful terminal requires prepared publication"
                )
            snapshot = replace(snapshot, terminal=terminal)
            self._records[terminal.publication_id] = snapshot
            return self._ack(InlineRecoveryStage.TERMINAL, True, snapshot)

    def snapshot(
        self, publication_id: InlinePublicationID,
    ) -> InlinePublicationRecoverySnapshot:
        with self._lock:
            return self._record(publication_id)

    def publication_ids(self) -> Tuple[InlinePublicationID, ...]:
        with self._lock:
            return tuple(sorted(self._records))

    def records_for_node_incarnation(
        self, incarnation: InlinePublicationNodeIncarnation,
    ) -> Tuple[InlinePublicationRecoverySnapshot, ...]:
        if not isinstance(incarnation, InlinePublicationNodeIncarnation):
            raise TypeError(
                "incarnation must be an InlinePublicationNodeIncarnation"
            )
        with self._lock:
            frozen = self._frozen_worksets.get(incarnation)
            if frozen is not None:
                return tuple(work.snapshot for work in frozen)
            return tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_incarnation.get(incarnation, ())
                )
            )

    def records_for_outer_owner(
        self, owner_worker_id: WorkerID,
    ) -> Tuple[InlinePublicationRecoverySnapshot, ...]:
        """Return the current records indexed by one exact outer owner.

        The index is installed in the same critical section as the durable
        intent record.  A caller can therefore never observe an admitted
        publication in ``publication_ids()`` without also observing it here.
        """

        if not isinstance(owner_worker_id, WorkerID):
            raise TypeError("owner_worker_id must be a WorkerID")
        with self._lock:
            frozen = self._frozen_owner_worksets.get(owner_worker_id)
            if frozen is not None:
                return frozen
            return tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_outer_owner.get(owner_worker_id, ())
                )
            )

    def freeze_owner_death(
        self, death: WorkerDeathRecord,
    ) -> Tuple[InlinePublicationRecoverySnapshot, ...]:
        """Freeze the complete point-in-time workset for an outer owner.

        Owner death is orthogonal to publishing-Node recovery, so this does
        not overload ``snapshot.frozen_death`` or mutate publication state.
        The tuple contains every record admitted for the owner at the linear
        ization point, including resolved tombstones.  Orderly Worker exit is
        not an owner-loss proof and creates neither a workset nor an
        admission fence.
        """

        self._validate_owner_death(death)
        owner_worker_id = death.worker_id
        with self._lock:
            previous = self._owner_deaths.get(owner_worker_id)
            if previous is not None:
                if previous != death:
                    raise InlineRecoveryConflictError(
                        "outer-owner death proof changed on replay"
                    )
                return self._frozen_owner_worksets[owner_worker_id]
            workset = tuple(
                self._records[publication_id]
                for publication_id in sorted(
                    self._by_outer_owner.get(owner_worker_id, ())
                )
            )
            self._owner_deaths[owner_worker_id] = death
            self._frozen_owner_worksets[owner_worker_id] = workset
            return workset

    def frozen_records_for_owner_death(
        self, death: WorkerDeathRecord,
    ) -> Tuple[InlinePublicationRecoverySnapshot, ...]:
        """Return only the immutable workset for this exact death proof."""

        self._validate_owner_death(death)
        owner_worker_id = death.worker_id
        with self._lock:
            previous = self._owner_deaths.get(owner_worker_id)
            if previous is None:
                raise InlineRecoveryStateError(
                    "outer owner is not death-frozen"
                )
            if previous != death:
                raise InlineRecoveryConflictError(
                    "outer-owner death proof changed on query"
                )
            return self._frozen_owner_worksets[owner_worker_id]

    def acknowledge_owner_death_cleaned(
        self, resolution: PublicationOwnerDeathResolution,
    ) -> PublicationOwnerDeathResolution:
        """Retain an independent outer-owner lifetime tombstone.

        Publication recovery and owner lifetime are orthogonal: this method
        deliberately does not rewrite ``snapshot.resolution``.  It only
        accepts the exact INLINE manifest and history captured in the frozen
        owner workset, after the owner-death saga has externally completed.
        """

        from .publication_owner_death import (
            InlinePublicationOwnerDeathKey,
            InlinePublicationOwnerDeathRequest,
            PublicationOwnerDeathResolution, PublicationRecoveryHistory,
        )

        if not isinstance(resolution, PublicationOwnerDeathResolution):
            raise TypeError(
                "resolution must be a PublicationOwnerDeathResolution"
            )
        request = resolution.request
        if (
            not isinstance(request, InlinePublicationOwnerDeathRequest)
            or not isinstance(request.key, InlinePublicationOwnerDeathKey)
        ):
            raise TypeError(
                "resolution must carry an INLINE owner-death key"
            )
        publication_id = request.publication_id
        with self._lock:
            self._record(publication_id)
            owner_worker_id = request.owner_worker_id
            frozen_death = self._owner_deaths.get(owner_worker_id)
            if frozen_death is None:
                raise InlineRecoveryStateError(
                    "outer owner is not death-frozen"
                )
            if frozen_death != request.owner_death:
                raise InlineRecoveryConflictError(
                    "owner-death resolution changed the frozen death proof"
                )
            canonical = next((
                snapshot
                for snapshot in self._frozen_owner_worksets[owner_worker_id]
                if snapshot.publication_id == publication_id
            ), None)
            if canonical is None:
                raise InlineRecoveryConflictError(
                    "owner-death resolution is outside the frozen workset"
                )
            if request.manifest != canonical.manifest:
                raise InlineRecoveryConflictError(
                    "owner-death resolution changed the frozen INLINE manifest"
                )
            expected_history = (
                PublicationRecoveryHistory.ADOPTED
                if canonical.resolution is InlineRecoveryResolution.ADOPTED
                else PublicationRecoveryHistory.NOT_COMMITTED
                if not canonical.prepared
                else PublicationRecoveryHistory.COMMIT_UNKNOWN
            )
            if request.publication_history is not expected_history:
                raise InlineRecoveryConflictError(
                    "owner-death resolution changed publication history"
                )
            previous = self._owner_death_resolutions.get(publication_id)
            if previous is not None:
                if previous != resolution:
                    raise InlineRecoveryConflictError(
                        "owner-death resolution changed on replay"
                    )
                return previous
            self._owner_death_resolutions[publication_id] = resolution
            return resolution

    def owner_death_resolution(
        self, publication_id: InlinePublicationID,
    ) -> Optional[PublicationOwnerDeathResolution]:
        """Return the independent owner-lifetime terminal, if present."""

        with self._lock:
            self._record(publication_id)
            return self._owner_death_resolutions.get(publication_id)

    def freeze_node_death(
        self, death: NodeDeathRecord,
    ) -> Tuple[InlinePublicationRecoveryWork, ...]:
        """Freeze one exact incarnation and its deterministic workset."""

        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        from .protocol import NodeDeathReason
        if death.reason is not NodeDeathReason.PROCESS_EXIT:
            return ()
        incarnation = InlinePublicationNodeIncarnation.from_death(death)
        with self._lock:
            previous_death = self._deaths.get(incarnation)
            if previous_death is not None:
                if previous_death != death:
                    raise InlineRecoveryConflictError(
                        "publishing Node death proof changed on replay"
                    )
                return self._frozen_worksets[incarnation]
            work = []
            for publication_id in sorted(
                self._by_incarnation.get(incarnation, ())
            ):
                snapshot = self._records[publication_id]
                if snapshot.resolved:
                    # A normal terminal acknowledgement may have committed
                    # just before its reply was lost.  Freeze every exact
                    # terminal tombstone: ADOPTED/RETIRED converge the
                    # completed-result path, while ABORTED proves the INLINE
                    # candidate was fully rolled back before Node death.  The
                    # latter carries no new cleanup effect but must remain
                    # queryable so Core may safely fall back to STORED/retry.
                    if snapshot.resolution not in (
                        InlineRecoveryResolution.ABORTED,
                        InlineRecoveryResolution.ADOPTED,
                        InlineRecoveryResolution.RETIRED,
                    ):
                        continue
                frozen = replace(snapshot, frozen_death=death)
                self._records[publication_id] = frozen
                work.append(InlinePublicationRecoveryWork(
                    death, frozen, frozen.recovery_action
                ))
            result = tuple(work)
            self._deaths[incarnation] = death
            self._frozen_worksets[incarnation] = result
            return result

    def frozen_workset(
        self, death: NodeDeathRecord,
    ) -> Tuple[InlinePublicationRecoveryWork, ...]:
        """Return only the work frozen by this exact death proof."""

        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        incarnation = InlinePublicationNodeIncarnation.from_death(death)
        with self._lock:
            previous = self._deaths.get(incarnation)
            if previous is None:
                raise InlineRecoveryStateError(
                    "publishing Node incarnation is not death-frozen"
                )
            if previous != death:
                raise InlineRecoveryConflictError(
                    "publishing Node death proof changed on query"
                )
            return self._frozen_worksets[incarnation]

    def work_for_node_death(
        self, publication_id: InlinePublicationID, death: NodeDeathRecord,
    ) -> Optional[InlinePublicationRecoveryWork]:
        """Find one item in the immutable workset for an exact query."""

        workset = self.frozen_workset(death)
        return next((
            work for work in workset
            if work.publication_id == publication_id
        ), None)

    def snapshot_for_node_death(
        self, publication_id: InlinePublicationID, death: NodeDeathRecord,
    ) -> Optional[InlinePublicationRecoverySnapshot]:
        """Return one exact death-bound snapshot, including tombstones.

        A frozen work item is an immutable effect identity.  Queries should
        nevertheless return the authority's current snapshot after an effect
        resolves, while normal ABORTED/ADOPTED/RETIRED acknowledgements that
        won before death must remain discoverable even though they need no new
        work.  This projection centralizes both cases under the authority lock.
        """

        if not isinstance(publication_id, InlinePublicationID):
            raise TypeError(
                "publication_id must be an InlinePublicationID"
            )
        if not isinstance(death, NodeDeathRecord):
            raise TypeError("death must be a NodeDeathRecord")
        incarnation = InlinePublicationNodeIncarnation.from_death(death)
        with self._lock:
            previous = self._deaths.get(incarnation)
            if previous is None:
                raise InlineRecoveryStateError(
                    "publishing Node incarnation is not death-frozen"
                )
            if previous != death:
                raise InlineRecoveryConflictError(
                    "publishing Node death proof changed on query"
                )
            if not any(
                work.publication_id == publication_id
                for work in self._frozen_worksets[incarnation]
            ):
                return None
            return self._records[publication_id]

    def acknowledge_rollback(
        self, work: InlinePublicationRecoveryWork,
    ) -> InlinePublicationRecoveryAck:
        """Acknowledge externally completed child/graph cleanup."""

        return self._acknowledge_resolution(
            work, InlineRecoveryAction.PRECOMPLETE_ROLLBACK,
            InlineRecoveryResolution.ROLLED_BACK,
        )

    def acknowledge_adoption(
        self, work: InlinePublicationRecoveryWork,
    ) -> InlinePublicationRecoveryAck:
        """Acknowledge a KEEP owner's atomically published local payload."""

        return self._acknowledge_resolution(
            work, (InlineRecoveryAction.COMPLETION_UNKNOWN,
                   InlineRecoveryAction.POSTCOMPLETE_RESOLVE),
            InlineRecoveryResolution.ADOPTED,
            owner_decision=InlineRecoveryOwnerDecision.KEEP,
        )

    def acknowledge_retirement(
        self, work: InlinePublicationRecoveryWork,
    ) -> InlinePublicationRecoveryAck:
        """Resolve frozen post-Complete work after stale-result cleanup."""

        return self._acknowledge_resolution(
            work, (InlineRecoveryAction.COMPLETION_UNKNOWN,
                   InlineRecoveryAction.POSTCOMPLETE_RESOLVE),
            InlineRecoveryResolution.RETIRED,
            owner_decision=InlineRecoveryOwnerDecision.KEEP,
        )

    def acknowledge_lost(
        self, work: InlinePublicationRecoveryWork,
    ) -> InlinePublicationRecoveryAck:
        """Resolve DROP only after child holds and graph are cleaned up."""

        return self._acknowledge_resolution(
            work, (InlineRecoveryAction.COMPLETION_UNKNOWN,
                   InlineRecoveryAction.POSTCOMPLETE_RESOLVE),
            InlineRecoveryResolution.LOST,
            owner_decision=InlineRecoveryOwnerDecision.DROP,
        )

    def decide_owner_resolution(
        self, work: InlinePublicationRecoveryWork, owner_worker_id: WorkerID,
        decision: InlineRecoveryOwnerDecision,
    ) -> InlinePublicationRecoveryAck:
        """Choose KEEP or DROP once, without putting payload bytes in GCS.

        The owner makes this decision under its local receipt/delivery lock.
        GCS composes this metadata fence with graph mutation; this reducer
        checks the exact frozen work and rejects every conflicting replay.
        """

        if not isinstance(owner_worker_id, WorkerID):
            raise TypeError("owner_worker_id must be a WorkerID")
        if not isinstance(decision, InlineRecoveryOwnerDecision):
            raise TypeError("decision must be an InlineRecoveryOwnerDecision")
        if not isinstance(work, InlinePublicationRecoveryWork):
            raise TypeError("work must be an InlinePublicationRecoveryWork")
        if work.action is InlineRecoveryAction.PRECOMPLETE_ROLLBACK:
            raise InlineRecoveryStateError(
                "proven pre-Complete work does not require an owner decision"
            )
        with self._lock:
            snapshot = self._require_exact_work(work)
            if snapshot.manifest.outer_owner_worker_id != owner_worker_id:
                raise InlineRecoveryConflictError(
                    "owner decision names another INLINE publication owner"
                )
            if snapshot.owner_decision is not None:
                if snapshot.owner_decision is not decision:
                    raise InlineRecoveryConflictError(
                        "INLINE owner decision changed on replay"
                    )
                return self._ack(InlineRecoveryStage.DECIDED, False, snapshot)
            if snapshot.resolved:
                raise InlineRecoveryStateError(
                    "resolved INLINE publication cannot acquire an owner decision"
                )
            self._require_owner_mutable(snapshot)
            snapshot = replace(snapshot, owner_decision=decision)
            self._records[work.publication_id] = snapshot
            return self._ack(InlineRecoveryStage.DECIDED, True, snapshot)

    def supersede_owner_death(
        self, publication_id: InlinePublicationID, death: WorkerDeathRecord,
    ) -> InlinePublicationRecoveryAck:
        """Terminalize unresolved Node-loss work at owner-death freeze."""

        self._validate_owner_death(death)
        with self._lock:
            snapshot = self._record(publication_id)
            if (
                snapshot.manifest.outer_owner_worker_id
                != death.worker_id
            ):
                raise InlineRecoveryConflictError(
                    "owner death names another INLINE publication owner"
                )
            if snapshot.resolution is not None:
                if (
                    snapshot.resolution
                    is InlineRecoveryResolution.SUPERSEDED
                    and snapshot.superseding_owner_death == death
                ):
                    return self._ack(
                        InlineRecoveryStage.RESOLVED, False, snapshot
                    )
                raise InlineRecoveryStateError(
                    "completed INLINE recovery cannot be superseded"
                )
            if not snapshot.frozen:
                raise InlineRecoveryStateError(
                    "INLINE publication has no active Node-loss saga"
                )
            snapshot = replace(
                snapshot, resolution=InlineRecoveryResolution.SUPERSEDED,
                superseding_owner_death=death,
            )
            self._records[publication_id] = snapshot
            return self._ack(InlineRecoveryStage.RESOLVED, True, snapshot)

    def acknowledge_aborted(
        self, publication_id: InlinePublicationID, manifest_digest: str,
    ) -> InlinePublicationRecoveryAck:
        """Retire a live-Node publication after full cleanup and graph ABORT.

        This is an acknowledgement, not permission to perform cleanup.  The
        caller must first release/tombstone every child slot and obtain the
        exact graph ABORT.  The immutable manifest digest prevents a delayed
        acknowledgement from retiring a different publication incarnation.
        """

        with self._lock:
            snapshot = self._record(publication_id)
            self._require_manifest_digest(snapshot, manifest_digest)
            if snapshot.resolution is InlineRecoveryResolution.ABORTED:
                # The live Node may lose this ACK immediately before its own
                # death is committed.  Once death freezes the same terminal
                # tombstone, an exact retry remains a harmless replay.
                return self._ack(
                    InlineRecoveryStage.ABORTED, False, snapshot
                )
            if snapshot.frozen:
                raise InlineRecoveryStateError(
                    "death-frozen rollback requires its exact work item"
                )
            if snapshot.terminal is not None:
                raise InlineRecoveryStateError(
                    "a successful INLINE publication cannot be aborted"
                )
            if snapshot.resolution is not None:
                raise InlineRecoveryConflictError(
                    "INLINE recovery was resolved through another action"
                )
            self._require_owner_mutable(snapshot)
            snapshot = replace(
                snapshot, resolution=InlineRecoveryResolution.ABORTED
            )
            self._records[publication_id] = snapshot
            return self._ack(InlineRecoveryStage.ABORTED, True, snapshot)

    def acknowledge_adopted(
        self, publication_id: InlinePublicationID, manifest_digest: str,
        owner_worker_id: WorkerID,
    ) -> InlinePublicationRecoveryAck:
        """Retire normal success after graph commit plus atomic owner publish."""

        if not isinstance(owner_worker_id, WorkerID):
            raise TypeError("owner_worker_id must be a WorkerID")
        with self._lock:
            snapshot = self._record(publication_id)
            self._require_manifest_digest(snapshot, manifest_digest)
            if (
                snapshot.manifest.outer_owner_worker_id
                != owner_worker_id
            ):
                raise InlineRecoveryConflictError(
                    "adoption acknowledgement names another outer owner"
                )
            if snapshot.frozen:
                raise InlineRecoveryStateError(
                    "death-frozen adoption requires its exact work item"
                )
            if not snapshot.prepared:
                raise InlineRecoveryStateError(
                    "INLINE adoption requires a prepared publication"
                )
            if snapshot.resolution is not None:
                if snapshot.resolution is not InlineRecoveryResolution.ADOPTED:
                    raise InlineRecoveryConflictError(
                        "INLINE recovery was resolved through another action"
                    )
                return self._ack(
                    InlineRecoveryStage.ADOPTED, False, snapshot
                )
            self._require_owner_mutable(snapshot)
            # The exact owner has received a successful Complete envelope and
            # committed it atomically.  That is a completion witness even when
            # Node's asynchronous terminal report has not arrived yet.
            snapshot = replace(
                snapshot, resolution=InlineRecoveryResolution.ADOPTED,
                terminal=(snapshot.terminal or InlinePublicationTerminal(
                    publication_id, TaskReplyStatus.SUCCEEDED
                )),
            )
            self._records[publication_id] = snapshot
            return self._ack(InlineRecoveryStage.ADOPTED, True, snapshot)

    def acknowledge_retired(
        self, publication_id: InlinePublicationID, manifest_digest: str,
        owner_worker_id: WorkerID,
    ) -> InlinePublicationRecoveryAck:
        """Retire a terminal result fenced stale before owner publication."""

        if not isinstance(owner_worker_id, WorkerID):
            raise TypeError("owner_worker_id must be a WorkerID")
        with self._lock:
            snapshot = self._record(publication_id)
            self._require_manifest_digest(snapshot, manifest_digest)
            if snapshot.manifest.outer_owner_worker_id != owner_worker_id:
                raise InlineRecoveryConflictError(
                    "retirement acknowledgement names another outer owner"
                )
            if snapshot.frozen:
                raise InlineRecoveryStateError(
                    "death-frozen retirement requires its exact work item"
                )
            if not snapshot.prepared:
                raise InlineRecoveryStateError(
                    "INLINE stale retirement requires a prepared publication"
                )
            if snapshot.resolution is not None:
                if snapshot.resolution is not InlineRecoveryResolution.RETIRED:
                    raise InlineRecoveryConflictError(
                        "INLINE recovery was resolved through another action"
                    )
                return self._ack(
                    InlineRecoveryStage.RETIRED, False, snapshot
                )
            self._require_owner_mutable(snapshot)
            snapshot = replace(
                snapshot, resolution=InlineRecoveryResolution.RETIRED,
                terminal=(snapshot.terminal or InlinePublicationTerminal(
                    publication_id, TaskReplyStatus.SUCCEEDED
                )),
            )
            self._records[publication_id] = snapshot
            return self._ack(InlineRecoveryStage.RETIRED, True, snapshot)

    # The report spellings fit GCS request handlers; the acknowledge spellings
    # make the required external-before-registry ordering explicit.
    report_aborted = acknowledge_aborted
    report_adopted = acknowledge_adopted
    report_retired = acknowledge_retired

    def close_admission(self) -> bool:
        """Fence new intents while preserving exact terminal replays."""

        with self._lock:
            changed = not self._admission_closed
            self._admission_closed = True
            return changed

    @property
    def admission_closed(self) -> bool:
        with self._lock:
            return self._admission_closed

    def has_active_operations(self) -> bool:
        with self._lock:
            return any(
                not snapshot.resolved
                and publication_id not in self._owner_death_resolutions
                for publication_id, snapshot in self._records.items()
            )

    def has_active_obligations(self) -> bool:
        """Compatibility alias for other shutdown-aware authorities."""

        return self.has_active_operations()

    def unresolved_snapshots(
        self,
    ) -> Tuple[InlinePublicationRecoverySnapshot, ...]:
        with self._lock:
            return tuple(
                self._records[publication_id]
                for publication_id in sorted(self._records)
                if not self._records[publication_id].resolved
            )

    def _acknowledge_resolution(
        self, work: InlinePublicationRecoveryWork,
        action: InlineRecoveryAction | tuple[InlineRecoveryAction, ...],
        resolution: InlineRecoveryResolution,
        *, owner_decision: Optional[InlineRecoveryOwnerDecision] = None,
    ) -> InlinePublicationRecoveryAck:
        if not isinstance(work, InlinePublicationRecoveryWork):
            raise TypeError("work must be an InlinePublicationRecoveryWork")
        allowed_actions = (
            (action,) if isinstance(action, InlineRecoveryAction) else action
        )
        if work.action not in allowed_actions:
            raise InlineRecoveryStateError(
                "recovery work requires {}".format(work.action.value)
            )
        with self._lock:
            snapshot = self._require_exact_work(work)
            if snapshot.resolution is not None:
                if snapshot.resolution is not resolution:
                    raise InlineRecoveryConflictError(
                        "INLINE recovery was resolved through another action"
                    )
                return self._ack(
                    InlineRecoveryStage.RESOLVED, False, snapshot
                )
            if snapshot.owner_decision is not owner_decision:
                raise InlineRecoveryStateError(
                    "resolution requires the exact owner decision"
                )
            self._require_owner_mutable(snapshot)
            snapshot = replace(snapshot, resolution=resolution)
            self._records[work.publication_id] = snapshot
            return self._ack(InlineRecoveryStage.RESOLVED, True, snapshot)

    def _require_exact_work(
        self, work: InlinePublicationRecoveryWork,
    ) -> InlinePublicationRecoverySnapshot:
        """Validate frozen evidence without confusing later owner decisions."""

        snapshot = self._record(work.publication_id)
        incarnation = InlinePublicationNodeIncarnation.from_death(work.death)
        canonical = next((
            candidate
            for candidate in self._frozen_worksets.get(incarnation, ())
            if candidate.publication_id == work.publication_id
        ), None)
        if canonical != work or (
            snapshot.frozen_death != work.death
            or work.snapshot.intent != snapshot.intent
            or work.snapshot.prepared != snapshot.prepared
            or work.snapshot.terminal != snapshot.terminal
        ):
            raise InlineRecoveryConflictError(
                "operation does not match the exact frozen work item"
            )
        return snapshot

    def _record(
        self, publication_id: InlinePublicationID,
    ) -> InlinePublicationRecoverySnapshot:
        if not isinstance(publication_id, InlinePublicationID):
            raise TypeError("publication_id must be an InlinePublicationID")
        try:
            return self._records[publication_id]
        except KeyError as exc:
            raise UnknownInlineRecoveryError(
                "INLINE recovery intent was not reported"
            ) from exc

    def _require_mutable(
        self, snapshot: InlinePublicationRecoverySnapshot,
    ) -> None:
        if snapshot.frozen:
            raise InlineRecoveryStateError(
                "publishing Node incarnation is already death-frozen"
            )
        if snapshot.resolved:
            raise InlineRecoveryStateError(
                "INLINE recovery is already resolved"
            )
        self._require_owner_mutable(snapshot)

    def _require_owner_mutable(
        self, snapshot: InlinePublicationRecoverySnapshot,
    ) -> None:
        owner_worker_id = (
            snapshot.manifest.outer_owner_worker_id
        )
        if owner_worker_id in self._owner_deaths:
            raise InlineRecoveryStateError(
                "outer owner Worker is already death-frozen"
            )

    @staticmethod
    def _require_manifest_digest(
        snapshot: InlinePublicationRecoverySnapshot, digest: object,
    ) -> None:
        if (
            not isinstance(digest, str)
            or digest.lower() != snapshot.manifest.manifest_digest
        ):
            raise InlineRecoveryConflictError(
                "retirement acknowledgement changed manifest identity"
            )

    @staticmethod
    def _validate_owner_death(death: WorkerDeathRecord) -> None:
        if not isinstance(death, WorkerDeathRecord):
            raise TypeError("death must be a WorkerDeathRecord")
        if death.reason is WorkerDeathReason.EXPECTED:
            raise InlineRecoveryStateError(
                "EXPECTED Worker exit is not an owner-death proof"
            )

    @staticmethod
    def _ack(
        stage: InlineRecoveryStage, changed: bool,
        snapshot: InlinePublicationRecoverySnapshot,
    ) -> InlinePublicationRecoveryAck:
        return InlinePublicationRecoveryAck(
            stage,
            (InlineRecoveryDisposition.APPLIED if changed else
             InlineRecoveryDisposition.ALREADY_RECORDED),
            snapshot,
        )
