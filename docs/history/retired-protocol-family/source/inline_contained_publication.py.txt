"""Pure identity and lifecycle model for INLINE results containing ObjectRefs.

INLINE publication shares the job-scoped contained-reference DAG with stored
publication, but it is not a stored object and deliberately has no replica,
seal, custody-promotion, Node-journal, or owner-adoption state.  This module
therefore defines its own identity domain and the smallest useful reducer:

``PREPARED -> PINS_INSTALLED -> COMMITTING -> COMMITTED -> COLLECTING -> RELEASED``

A failure before ``COMMITTED`` instead follows ``ABORTING -> ABORTED``.  The
bitmaps record acknowledgements only; callers own the actual child-owner and
GCS RPCs.  Every reducer method validates the complete immutable identity so a
lost acknowledgement is replayed, never inferred.
"""

from __future__ import annotations

import hashlib
from dataclasses import dataclass, field
from enum import Enum
from threading import RLock
from typing import Optional, Tuple

from .contained_cycle import (
    ContainedGraphManifest,
    ContainedGraphManifestDisposition,
    ContainedGraphManifestReceipt,
    ContainedGraphTransactionState,
)
from .contained_edges import ContainedReferenceEdge
from .ids import AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID
from .protocol import ResultDescriptor, ResultStorage
from .stored_publication import (
    BorrowedContainedSource, OwnedContainedSource, PreparedContainedTransfer,
    prepared_contained_transfer_fingerprint,
)


_GRAPH_ID_DOMAIN = b"miniray-inline-contained-graph-v1\0"
_MANIFEST_DOMAIN = b"miniray-inline-contained-manifest-v1\0"


class InlinePublicationError(RuntimeError):
    """Base class for pure INLINE contained-publication failures."""


class InlinePublicationConflictError(InlinePublicationError):
    """One stable identity was rebound to different immutable content."""


class InlinePublicationStateError(InlinePublicationError):
    """A transition contradicts the publication's monotonic lifecycle."""


class UnknownInlinePublicationError(InlinePublicationError, LookupError):
    """The reducer has never prepared the requested publication."""


def _framed(digest: object, value: bytes) -> None:
    digest.update(len(value).to_bytes(8, "big"))
    digest.update(value)


def _uint(value: int, label: str) -> bytes:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError(f"{label} must be a non-negative integer")
    return value.to_bytes(8, "big")


def _digest(value: object, label: str) -> str:
    if not isinstance(value, str) or len(value) != 64:
        raise ValueError(f"{label} must be a SHA-256 hex digest")
    normalized = value.lower()
    try:
        bytes.fromhex(normalized)
    except ValueError as exc:
        raise ValueError(f"{label} must be a SHA-256 hex digest") from exc
    return normalized


def _object_identity(value: ObjectID) -> bytes:
    if not isinstance(value, ObjectID):
        raise TypeError("object identity requires an ObjectID")
    return bytes(value.task_id) + _uint(value.return_index, "return_index")


@dataclass(frozen=True, order=True)
class InlinePublicationID:
    """Stable identity of one INLINE result within one lease attempt.

    The fields intentionally match the logical execution coordinates used by
    stored publication, but the Python type and hash domain are distinct.
    """

    lease_id: LeaseID
    task_id: TaskID
    attempt_id: AttemptID
    outer_object_id: ObjectID

    def __post_init__(self) -> None:
        if not isinstance(self.lease_id, LeaseID):
            raise TypeError("lease_id must be a LeaseID")
        if not isinstance(self.task_id, TaskID):
            raise TypeError("task_id must be a TaskID")
        if not isinstance(self.attempt_id, AttemptID):
            raise TypeError("attempt_id must be an AttemptID")
        if not isinstance(self.outer_object_id, ObjectID):
            raise TypeError("outer_object_id must be an ObjectID")
        if self.attempt_id.task_id != self.task_id:
            raise ValueError("attempt_id belongs to another task")
        if self.outer_object_id.task_id != self.task_id:
            raise ValueError("outer_object_id belongs to another task")

    @property
    def graph_transaction_id(self) -> str:
        """Deterministic GCS DAG identity in an INLINE-only domain."""

        digest = hashlib.sha256()
        digest.update(_GRAPH_ID_DOMAIN)
        _framed(digest, bytes(self.lease_id))
        _framed(digest, bytes(self.task_id))
        _framed(digest, _uint(
            self.attempt_id.attempt_number, "attempt_number"
        ))
        _framed(digest, _object_identity(self.outer_object_id))
        return "inline-contained-graph:" + digest.hexdigest()


def _manifest_digest(
    publication_id: InlinePublicationID,
    job_id: JobID,
    executor_worker_id: WorkerID,
    outer_owner_worker_id: WorkerID,
    publishing_node_id: NodeID,
    size_bytes: int,
    checksum: str,
    ordered_edges: Tuple[ContainedReferenceEdge, ...],
    transfers: Tuple[PreparedContainedTransfer, ...],
) -> str:
    digest = hashlib.sha256()
    digest.update(_MANIFEST_DOMAIN)
    _framed(digest, bytes(publication_id.lease_id))
    _framed(digest, bytes(publication_id.task_id))
    _framed(digest, _uint(
        publication_id.attempt_id.attempt_number, "attempt_number"
    ))
    _framed(digest, _object_identity(publication_id.outer_object_id))
    _framed(digest, bytes(job_id))
    _framed(digest, bytes(executor_worker_id))
    _framed(digest, bytes(outer_owner_worker_id))
    _framed(digest, bytes(publishing_node_id))
    _framed(digest, _uint(size_bytes, "size_bytes"))
    _framed(digest, bytes.fromhex(checksum))
    if len(ordered_edges) != len(transfers):
        raise ValueError(
            "INLINE edges and transfer credentials must have equal length"
        )
    _framed(digest, _uint(len(transfers), "transfer count"))
    for index, (edge, transfer) in enumerate(zip(ordered_edges, transfers)):
        _framed(digest, _uint(index, "edge index"))
        _framed(digest, _object_identity(edge.container_object_id))
        _framed(digest, _object_identity(edge.contained_object_id))
        _framed(digest, bytes(edge.contained_owner_worker_id))
        host, port = edge.contained_owner_address
        _framed(digest, host.encode("utf-8"))
        _framed(digest, _uint(port, "contained owner port"))
        _framed(digest, edge.transfer_token.encode("utf-8"))
        _framed(digest, prepared_contained_transfer_fingerprint(transfer))
    return digest.hexdigest()


def _owned_transfers_from_edges(
    publication_id: InlinePublicationID, executor_worker_id: WorkerID,
    outer_owner_worker_id: WorkerID,
    ordered_edges: Tuple[ContainedReferenceEdge, ...],
) -> Tuple[PreparedContainedTransfer, ...]:
    """Compatibility constructor for the original owned-child API."""

    values = []
    for edge in ordered_edges:
        if edge.contained_owner_worker_id != executor_worker_id:
            raise ValueError(
                "foreign INLINE children require explicit source credentials"
            )
        provisional = edge.incoming_hold(executor_worker_id)
        final = edge.incoming_hold(outer_owner_worker_id)
        values.append(PreparedContainedTransfer(
            edge.contained_object_id, edge.contained_owner_worker_id,
            edge.contained_owner_address,
            OwnedContainedSource(edge.contained_owner_worker_id),
            provisional, final,
        ))
    return tuple(values)


@dataclass(frozen=True)
class InlinePublicationManifest:
    """Frozen serialization boundary used by child pins and the GCS DAG."""

    publication_id: InlinePublicationID
    job_id: JobID
    executor_worker_id: WorkerID
    outer_owner_worker_id: WorkerID
    publishing_node_id: NodeID
    size_bytes: int
    checksum: str
    ordered_edges: Tuple[ContainedReferenceEdge, ...]
    transfers: Tuple[PreparedContainedTransfer, ...]
    manifest_digest: str

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, InlinePublicationID):
            raise TypeError("publication_id must be an InlinePublicationID")
        if not isinstance(self.job_id, JobID):
            raise TypeError("job_id must be a JobID")
        if not isinstance(self.executor_worker_id, WorkerID):
            raise TypeError("executor_worker_id must be a WorkerID")
        if not isinstance(self.outer_owner_worker_id, WorkerID):
            raise TypeError("outer_owner_worker_id must be a WorkerID")
        if not isinstance(self.publishing_node_id, NodeID):
            raise TypeError("publishing_node_id must be a NodeID")
        _uint(self.size_bytes, "size_bytes")
        checksum = _digest(self.checksum, "checksum")
        edges = tuple(self.ordered_edges)
        if not edges:
            raise ValueError(
                "an INLINE contained publication requires at least one edge"
            )
        if any(not isinstance(edge, ContainedReferenceEdge) for edge in edges):
            raise TypeError(
                "ordered_edges must contain ContainedReferenceEdge values"
            )
        if len(edges) != len(set(edges)):
            raise ValueError("ordered_edges must be unique")
        transfers = tuple(self.transfers)
        if any(
            not isinstance(value, PreparedContainedTransfer)
            for value in transfers
        ):
            raise TypeError(
                "transfers must contain PreparedContainedTransfer values"
            )
        if len(transfers) != len(edges):
            raise ValueError(
                "INLINE transfer credentials must match every edge"
            )
        if any(
            transfer.edge != edge
            for transfer, edge in zip(transfers, edges)
        ):
            raise ValueError(
                "INLINE transfer credentials must match their edge slots"
            )
        outer = self.publication_id.outer_object_id
        if any(edge.container_object_id != outer for edge in edges):
            raise ValueError("every edge must name the INLINE outer object")
        for transfer in transfers:
            if (
                transfer.provisional_hold.container_object_id != outer
                or transfer.final_hold.container_object_id != outer
                or transfer.provisional_hold.container_owner_worker_id
                != self.executor_worker_id
                or transfer.final_hold.container_owner_worker_id
                != self.outer_owner_worker_id
            ):
                raise ValueError(
                    "INLINE transfer custody must match its executor and "
                    "outer owner"
                )
            if isinstance(transfer.source, OwnedContainedSource):
                if transfer.source.owner_worker_id != self.executor_worker_id:
                    raise ValueError(
                        "only the child-owning executor may use an owned "
                        "INLINE source"
                    )
            elif transfer.source.borrower_worker_id != self.executor_worker_id:
                raise ValueError(
                    "borrowed INLINE source must belong to the executor"
                )
        supplied = _digest(self.manifest_digest, "manifest_digest")
        expected = _manifest_digest(
            self.publication_id, self.job_id, self.executor_worker_id,
            self.outer_owner_worker_id, self.publishing_node_id,
            self.size_bytes, checksum, edges, transfers,
        )
        if supplied != expected:
            raise InlinePublicationConflictError(
                "manifest_digest does not match the immutable INLINE manifest"
            )
        object.__setattr__(self, "checksum", checksum)
        object.__setattr__(self, "ordered_edges", edges)
        object.__setattr__(self, "transfers", transfers)
        object.__setattr__(self, "manifest_digest", supplied)

    @classmethod
    def create(
        cls,
        publication_id: InlinePublicationID,
        job_id: JobID,
        executor_worker_id: WorkerID,
        outer_owner_worker_id: WorkerID,
        publishing_node_id: NodeID,
        size_bytes: int,
        checksum: str,
        ordered_edges: Tuple[ContainedReferenceEdge, ...],
        transfers: Optional[Tuple[PreparedContainedTransfer, ...]] = None,
    ) -> "InlinePublicationManifest":
        edges = tuple(ordered_edges)
        normalized_checksum = _digest(checksum, "checksum")
        normalized_transfers = (
            _owned_transfers_from_edges(
                publication_id, executor_worker_id, outer_owner_worker_id, edges
            )
            if transfers is None else tuple(transfers)
        )
        digest = _manifest_digest(
            publication_id, job_id, executor_worker_id,
            outer_owner_worker_id, publishing_node_id, size_bytes,
            normalized_checksum, edges, normalized_transfers,
        )
        return cls(
            publication_id, job_id, executor_worker_id,
            outer_owner_worker_id, publishing_node_id, size_bytes,
            normalized_checksum, edges, normalized_transfers, digest,
        )

    def to_graph_manifest(self) -> ContainedGraphManifest:
        """Project the exact INLINE identity into the shared GCS DAG."""

        return ContainedGraphManifest(
            transaction_id=self.publication_id.graph_transaction_id,
            publication_id=self.publication_id,
            outer_owner_worker_id=self.outer_owner_worker_id,
            manifest_digest=self.manifest_digest,
            ordered_edges=self.ordered_edges,
        )


@dataclass(frozen=True)
class InlinePublicationEnvelope:
    """Immutable owner hand-off: INLINE bytes plus their exact DAG manifest."""

    manifest: InlinePublicationManifest
    descriptor: ResultDescriptor

    def __post_init__(self) -> None:
        if not isinstance(self.manifest, InlinePublicationManifest):
            raise TypeError("manifest must be an InlinePublicationManifest")
        if not isinstance(self.descriptor, ResultDescriptor):
            raise TypeError("descriptor must be a ResultDescriptor")
        manifest = self.manifest
        descriptor = self.descriptor
        if (
            descriptor.storage is not ResultStorage.INLINE
            or descriptor.object_id != manifest.publication_id.outer_object_id
            or descriptor.owner_worker_id != manifest.outer_owner_worker_id
            or descriptor.node_id != manifest.publishing_node_id
            or descriptor.size_bytes != manifest.size_bytes
            or descriptor.checksum != manifest.checksum
            or not isinstance(descriptor.inline_data, bytes)
        ):
            raise InlinePublicationConflictError(
                "INLINE descriptor does not match its publication manifest"
            )

    @classmethod
    def create(
        cls,
        publication_id: InlinePublicationID,
        job_id: JobID,
        executor_worker_id: WorkerID,
        outer_owner_worker_id: WorkerID,
        publishing_node_id: NodeID,
        inline_data: bytes,
        ordered_edges: Tuple[ContainedReferenceEdge, ...],
        transfers: Optional[Tuple[PreparedContainedTransfer, ...]] = None,
    ) -> "InlinePublicationEnvelope":
        if not isinstance(inline_data, bytes):
            raise TypeError("inline_data must be bytes")
        checksum = hashlib.sha256(inline_data).hexdigest()
        manifest = InlinePublicationManifest.create(
            publication_id, job_id, executor_worker_id,
            outer_owner_worker_id, publishing_node_id, len(inline_data),
            checksum, tuple(ordered_edges), transfers,
        )
        descriptor = ResultDescriptor(
            publication_id.outer_object_id, ResultStorage.INLINE,
            len(inline_data), outer_owner_worker_id, publishing_node_id,
            checksum, inline_data,
        )
        return cls(manifest, descriptor)

    @classmethod
    def from_graph_manifest(
        cls,
        graph: ContainedGraphManifest,
        *,
        job_id: JobID,
        executor_worker_id: WorkerID,
        publishing_node_id: NodeID,
        descriptor: ResultDescriptor,
        transfers: Optional[Tuple[PreparedContainedTransfer, ...]] = None,
    ) -> "InlinePublicationEnvelope":
        """Rebuild an envelope while revalidating an exact GCS projection."""

        if not isinstance(graph, ContainedGraphManifest):
            raise TypeError("graph must be a ContainedGraphManifest")
        if not isinstance(graph.publication_id, InlinePublicationID):
            raise TypeError(
                "graph publication_id must be an InlinePublicationID"
            )
        if not isinstance(descriptor, ResultDescriptor):
            raise TypeError("descriptor must be a ResultDescriptor")
        normalized_transfers = (
            _owned_transfers_from_edges(
                graph.publication_id, executor_worker_id,
                graph.outer_owner_worker_id, graph.ordered_edges,
            )
            if transfers is None else tuple(transfers)
        )
        manifest = InlinePublicationManifest(
            graph.publication_id, job_id, executor_worker_id,
            graph.outer_owner_worker_id, publishing_node_id,
            descriptor.size_bytes, descriptor.checksum, graph.ordered_edges,
            normalized_transfers, graph.manifest_digest,
        )
        envelope = cls(manifest, descriptor)
        if envelope.to_graph_manifest() != graph:
            raise InlinePublicationConflictError(
                "GCS graph does not exactly match the INLINE envelope"
            )
        return envelope

    @property
    def publication_id(self) -> InlinePublicationID:
        return self.manifest.publication_id

    @property
    def ordered_edges(self) -> Tuple[ContainedReferenceEdge, ...]:
        return self.manifest.ordered_edges

    def to_graph_manifest(self) -> ContainedGraphManifest:
        return self.manifest.to_graph_manifest()

    def edge_effect(self, index: int) -> "InlineEdgeEffect":
        _uint(index, "edge index")
        try:
            edge = self.ordered_edges[index]
        except IndexError as exc:
            raise IndexError("edge index is outside the manifest") from exc
        return InlineEdgeEffect(
            self.publication_id, index, edge, self.manifest.transfers[index]
        )


@dataclass(frozen=True)
class InlineEdgeEffect:
    """One index-bound child hold effect from an immutable manifest."""

    publication_id: InlinePublicationID
    index: int
    edge: ContainedReferenceEdge
    transfer: PreparedContainedTransfer

    def __post_init__(self) -> None:
        if not isinstance(self.publication_id, InlinePublicationID):
            raise TypeError("publication_id must be an InlinePublicationID")
        _uint(self.index, "edge index")
        if not isinstance(self.edge, ContainedReferenceEdge):
            raise TypeError("edge must be a ContainedReferenceEdge")
        if not isinstance(self.transfer, PreparedContainedTransfer):
            raise TypeError(
                "transfer must be a PreparedContainedTransfer"
            )
        if self.transfer.edge != self.edge:
            raise ValueError(
                "INLINE effect transfer must match its edge"
            )


class InlinePublicationState(str, Enum):
    PREPARED = "PREPARED"
    PINS_INSTALLED = "PINS_INSTALLED"
    COMMITTING = "COMMITTING"
    COMMITTED = "COMMITTED"
    ABORTING = "ABORTING"
    ABORTED = "ABORTED"
    COLLECTING = "COLLECTING"
    RELEASED = "RELEASED"


@dataclass(frozen=True)
class InlinePublicationSnapshot:
    envelope: InlinePublicationEnvelope
    state: InlinePublicationState
    pin_bitmap: Tuple[bool, ...]
    abort_id: Optional[str]
    compensation_bitmap: Tuple[bool, ...]
    collection_id: Optional[str]
    collection_bitmap: Tuple[bool, ...]
    graph_released: bool

    @property
    def publication_id(self) -> InlinePublicationID:
        return self.envelope.publication_id

    @property
    def ready_to_commit(self) -> bool:
        return (
            self.state is InlinePublicationState.PINS_INSTALLED
            and all(self.pin_bitmap)
        )

    @property
    def pending_compensations(self) -> Tuple[InlineEdgeEffect, ...]:
        return tuple(
            self.envelope.edge_effect(index)
            for index in reversed(range(len(self.compensation_bitmap)))
            if not self.compensation_bitmap[index]
        ) if self.state is InlinePublicationState.ABORTING else ()

    @property
    def pending_collection_releases(self) -> Tuple[InlineEdgeEffect, ...]:
        return tuple(
            self.envelope.edge_effect(index)
            for index, acknowledged in enumerate(self.collection_bitmap)
            if not acknowledged
        ) if self.state is InlinePublicationState.COLLECTING else ()


@dataclass
class _InlinePublicationRecord:
    envelope: InlinePublicationEnvelope
    state: InlinePublicationState = InlinePublicationState.PREPARED
    pins: list[bool] = field(default_factory=list)
    abort_id: Optional[str] = None
    compensations: list[bool] = field(default_factory=list)
    collection_id: Optional[str] = None
    collection_releases: list[bool] = field(default_factory=list)
    graph_released: bool = False


class InlineContainedPublicationReducer:
    """Thread-safe, side-effect-free INLINE publication state machine.

    Runtime adapters should call methods only after receiving the corresponding
    exact child/GCS acknowledgement.  The reducer intentionally exposes the
    next legal compensation and collection effects so ordering is executable
    documentation rather than a comment.
    """

    def __init__(self) -> None:
        self._records: dict[InlinePublicationID, _InlinePublicationRecord] = {}
        self._lock = RLock()

    def prepare(
        self, envelope: InlinePublicationEnvelope,
        graph_receipt: ContainedGraphManifestReceipt,
    ) -> bool:
        """Record one exact GCS-prepared manifest before any child pin."""

        if not isinstance(envelope, InlinePublicationEnvelope):
            raise TypeError("envelope must be an InlinePublicationEnvelope")
        with self._lock:
            previous = self._records.get(envelope.publication_id)
            if previous is not None:
                self._require_envelope(previous, envelope)
                self._require_graph_receipt(
                    envelope, graph_receipt,
                    state=ContainedGraphTransactionState.PREPARED,
                    dispositions=(
                        ContainedGraphManifestDisposition.APPLIED,
                        ContainedGraphManifestDisposition.ALREADY_PREPARED,
                    ),
                )
                if previous.state is InlinePublicationState.ABORTED:
                    raise InlinePublicationStateError(
                        "an aborted INLINE publication cannot prepare"
                    )
                return False
            self._require_graph_receipt(
                envelope, graph_receipt,
                state=ContainedGraphTransactionState.PREPARED,
                dispositions=(
                    ContainedGraphManifestDisposition.APPLIED,
                    ContainedGraphManifestDisposition.ALREADY_PREPARED,
                ),
            )
            size = len(envelope.ordered_edges)
            self._records[envelope.publication_id] = _InlinePublicationRecord(
                envelope=envelope, pins=[False] * size,
                compensations=[False] * size,
                collection_releases=[False] * size,
            )
            return True

    def mark_pin_installed(self, effect: InlineEdgeEffect) -> bool:
        """Acknowledge one exact final hold; installation order is free."""

        with self._lock:
            record = self._record(effect.publication_id)
            self._require_effect(record, effect)
            if record.pins[effect.index]:
                return False
            if record.state is not InlinePublicationState.PREPARED:
                raise InlinePublicationStateError(
                    "child pins can be installed only while PREPARED"
                )
            record.pins[effect.index] = True
            if all(record.pins):
                record.state = InlinePublicationState.PINS_INSTALLED
            return True

    def begin_commit(self, envelope: InlinePublicationEnvelope) -> bool:
        """Persist the roll-forward boundary before sending GCS COMMIT.

        Once this intent exists, an unknown COMMIT outcome may already be
        authoritative at GCS.  The caller must replay/query that same commit; it
        may no longer compensate with ABORT.
        """

        if not isinstance(envelope, InlinePublicationEnvelope):
            raise TypeError("envelope must be an InlinePublicationEnvelope")
        with self._lock:
            record = self._record(envelope.publication_id)
            self._require_envelope(record, envelope)
            if record.state is InlinePublicationState.COMMITTING:
                return False
            if record.state in (
                InlinePublicationState.COMMITTED,
                InlinePublicationState.COLLECTING,
                InlinePublicationState.RELEASED,
            ):
                return False
            if record.state is not InlinePublicationState.PINS_INSTALLED:
                raise InlinePublicationStateError(
                    "GCS commit cannot begin before every pin ACK"
                )
            record.state = InlinePublicationState.COMMITTING
            return True

    def commit(
        self, envelope: InlinePublicationEnvelope,
        graph_receipt: ContainedGraphManifestReceipt,
    ) -> bool:
        """Record the exact GCS graph commit after its intent was frozen."""

        if not isinstance(envelope, InlinePublicationEnvelope):
            raise TypeError("envelope must be an InlinePublicationEnvelope")
        self._require_graph_receipt(
            envelope, graph_receipt,
            state=ContainedGraphTransactionState.COMMITTED,
            dispositions=(
                ContainedGraphManifestDisposition.APPLIED,
                ContainedGraphManifestDisposition.ALREADY_COMMITTED,
            ),
        )
        with self._lock:
            record = self._record(envelope.publication_id)
            self._require_envelope(record, envelope)
            if record.state in (
                InlinePublicationState.COMMITTED,
                InlinePublicationState.COLLECTING,
                InlinePublicationState.RELEASED,
            ):
                return False
            if record.state is not InlinePublicationState.COMMITTING:
                raise InlinePublicationStateError(
                    "INLINE publication has no matching GCS commit intent"
                )
            record.state = InlinePublicationState.COMMITTED
            return True

    def begin_abort(
        self, publication_id: InlinePublicationID, abort_id: str
    ) -> bool:
        """Freeze forward progress and expose reverse-order compensation."""

        self._require_operation_id(abort_id, "abort_id")
        with self._lock:
            record = self._record(publication_id)
            if record.abort_id is not None and record.abort_id != abort_id:
                raise InlinePublicationConflictError(
                    "INLINE publication abort identity changed"
                )
            if record.state in (
                InlinePublicationState.ABORTING, InlinePublicationState.ABORTED
            ):
                return False
            if record.state not in (
                InlinePublicationState.PREPARED,
                InlinePublicationState.PINS_INSTALLED,
            ):
                raise InlinePublicationStateError(
                    "a committed INLINE publication cannot abort"
                )
            record.abort_id = abort_id
            record.state = InlinePublicationState.ABORTING
            return True

    def next_compensation(
        self, publication_id: InlinePublicationID
    ) -> Optional[InlineEdgeEffect]:
        """Return the next reverse-order release-before-add compensator."""

        with self._lock:
            record = self._record(publication_id)
            if record.state is InlinePublicationState.ABORTED:
                return None
            if record.state is not InlinePublicationState.ABORTING:
                raise InlinePublicationStateError(
                    "compensation requires an ABORTING publication"
                )
            index = self._last_unacknowledged(record.compensations)
            return (
                None if index is None else record.envelope.edge_effect(index)
            )

    def acknowledge_compensation(self, effect: InlineEdgeEffect) -> bool:
        """Acknowledge exactly the next reverse-order child release."""

        with self._lock:
            record = self._record(effect.publication_id)
            self._require_effect(record, effect)
            if record.compensations[effect.index]:
                return False
            if record.state is not InlinePublicationState.ABORTING:
                raise InlinePublicationStateError(
                    "compensation ACK requires an ABORTING publication"
                )
            expected = self._last_unacknowledged(record.compensations)
            if effect.index != expected:
                raise InlinePublicationStateError(
                    "INLINE compensation must follow reverse manifest order"
                )
            record.compensations[effect.index] = True
            return True

    def finish_abort(
        self, publication_id: InlinePublicationID, abort_id: str,
        graph_receipt: ContainedGraphManifestReceipt,
    ) -> bool:
        """Record GCS ABORT only after every child compensator ACK."""

        self._require_operation_id(abort_id, "abort_id")
        with self._lock:
            record = self._record(publication_id)
            self._require_graph_receipt(
                record.envelope, graph_receipt,
                state=ContainedGraphTransactionState.ABORTED,
                dispositions=(
                    ContainedGraphManifestDisposition.APPLIED,
                    ContainedGraphManifestDisposition.ALREADY_ABORTED,
                ),
            )
            if record.abort_id != abort_id:
                raise InlinePublicationConflictError(
                    "INLINE publication abort identity changed"
                )
            if record.state is InlinePublicationState.ABORTED:
                return False
            if record.state is not InlinePublicationState.ABORTING:
                raise InlinePublicationStateError(
                    "finish_abort requires an ABORTING publication"
                )
            if not all(record.compensations):
                raise InlinePublicationStateError(
                    "GCS graph cannot abort before every compensation ACK"
                )
            record.state = InlinePublicationState.ABORTED
            return True

    def begin_collection(
        self, publication_id: InlinePublicationID, collection_id: str
    ) -> bool:
        """Freeze a committed outer for ordered reverse-lifetime work."""

        self._require_operation_id(collection_id, "collection_id")
        with self._lock:
            record = self._record(publication_id)
            if (
                record.collection_id is not None
                and record.collection_id != collection_id
            ):
                raise InlinePublicationConflictError(
                    "INLINE collection identity changed"
                )
            if record.state in (
                InlinePublicationState.COLLECTING,
                InlinePublicationState.RELEASED,
            ):
                return False
            if record.state is not InlinePublicationState.COMMITTED:
                raise InlinePublicationStateError(
                    "only a committed INLINE publication can be collected"
                )
            record.collection_id = collection_id
            record.state = InlinePublicationState.COLLECTING
            return True

    def next_collection_release(
        self, publication_id: InlinePublicationID
    ) -> Optional[InlineEdgeEffect]:
        """Return the next manifest-order final-hold release."""

        with self._lock:
            record = self._record(publication_id)
            if record.state is InlinePublicationState.RELEASED:
                return None
            if record.state is not InlinePublicationState.COLLECTING:
                raise InlinePublicationStateError(
                    "child release requires a COLLECTING publication"
                )
            index = self._first_unacknowledged(record.collection_releases)
            return (
                None if index is None else record.envelope.edge_effect(index)
            )

    def acknowledge_collection_release(
        self, collection_id: str, effect: InlineEdgeEffect
    ) -> bool:
        """Acknowledge exactly the next manifest-order final-hold release."""

        self._require_operation_id(collection_id, "collection_id")
        with self._lock:
            record = self._record(effect.publication_id)
            self._require_collection(record, collection_id)
            self._require_effect(record, effect)
            if record.collection_releases[effect.index]:
                return False
            if record.state is not InlinePublicationState.COLLECTING:
                raise InlinePublicationStateError(
                    "collection ACK requires a COLLECTING publication"
                )
            expected = self._first_unacknowledged(
                record.collection_releases
            )
            if effect.index != expected:
                raise InlinePublicationStateError(
                    "INLINE collection must follow manifest order"
                )
            record.collection_releases[effect.index] = True
            return True

    def mark_graph_released(
        self, publication_id: InlinePublicationID, collection_id: str,
        graph_receipt: ContainedGraphManifestReceipt,
    ) -> bool:
        """Record exact GCS RELEASE after every child release ACK."""

        self._require_operation_id(collection_id, "collection_id")
        with self._lock:
            record = self._record(publication_id)
            self._require_collection(record, collection_id)
            self._require_graph_receipt(
                record.envelope, graph_receipt,
                state=ContainedGraphTransactionState.COMMITTED,
                dispositions=(
                    ContainedGraphManifestDisposition.RELEASED,
                    ContainedGraphManifestDisposition.ALREADY_RELEASED,
                ),
                released_edges=record.envelope.ordered_edges,
            )
            if record.graph_released:
                return False
            if record.state is not InlinePublicationState.COLLECTING:
                raise InlinePublicationStateError(
                    "graph release requires a COLLECTING publication"
                )
            if not all(record.collection_releases):
                raise InlinePublicationStateError(
                    "GCS graph cannot release before every child release ACK"
                )
            record.graph_released = True
            return True

    def finish_collection(
        self, publication_id: InlinePublicationID, collection_id: str
    ) -> bool:
        """Record owner metadata collection after the GCS reverse barrier."""

        self._require_operation_id(collection_id, "collection_id")
        with self._lock:
            record = self._record(publication_id)
            self._require_collection(record, collection_id)
            if record.state is InlinePublicationState.RELEASED:
                return False
            if record.state is not InlinePublicationState.COLLECTING:
                raise InlinePublicationStateError(
                    "finish_collection requires a COLLECTING publication"
                )
            if not record.graph_released:
                raise InlinePublicationStateError(
                    "owner metadata cannot collect before GCS graph release"
                )
            record.state = InlinePublicationState.RELEASED
            return True

    def snapshot(
        self, publication_id: InlinePublicationID
    ) -> InlinePublicationSnapshot:
        with self._lock:
            record = self._record(publication_id)
            return InlinePublicationSnapshot(
                record.envelope, record.state, tuple(record.pins),
                record.abort_id, tuple(record.compensations),
                record.collection_id, tuple(record.collection_releases),
                record.graph_released,
            )

    def publication_ids(self) -> Tuple[InlinePublicationID, ...]:
        with self._lock:
            return tuple(sorted(self._records))

    def _record(
        self, publication_id: InlinePublicationID
    ) -> _InlinePublicationRecord:
        if not isinstance(publication_id, InlinePublicationID):
            raise TypeError("publication_id must be an InlinePublicationID")
        try:
            return self._records[publication_id]
        except KeyError as exc:
            raise UnknownInlinePublicationError(
                "INLINE publication has not been prepared"
            ) from exc

    @staticmethod
    def _require_envelope(
        record: _InlinePublicationRecord, envelope: InlinePublicationEnvelope
    ) -> None:
        if record.envelope != envelope:
            raise InlinePublicationConflictError(
                "INLINE publication identity names another envelope"
            )

    @staticmethod
    def _require_effect(
        record: _InlinePublicationRecord, effect: InlineEdgeEffect
    ) -> None:
        if not isinstance(effect, InlineEdgeEffect):
            raise TypeError("effect must be an InlineEdgeEffect")
        edges = record.envelope.ordered_edges
        transfers = record.envelope.manifest.transfers
        if (
            effect.index >= len(edges)
            or edges[effect.index] != effect.edge
            or transfers[effect.index] != effect.transfer
        ):
            raise InlinePublicationConflictError(
                "INLINE edge effect changed its manifest slot"
            )

    @staticmethod
    def _require_graph_receipt(
        envelope: InlinePublicationEnvelope,
        receipt: ContainedGraphManifestReceipt,
        *,
        state: ContainedGraphTransactionState,
        dispositions: Tuple[ContainedGraphManifestDisposition, ...],
        released_edges: Tuple[ContainedReferenceEdge, ...] = (),
    ) -> None:
        if not isinstance(receipt, ContainedGraphManifestReceipt):
            raise TypeError("graph_receipt must be a ContainedGraphManifestReceipt")
        if (
            receipt.manifest != envelope.to_graph_manifest()
            or receipt.state is not state
            or receipt.disposition not in dispositions
            or receipt.released_edges != released_edges
        ):
            raise InlinePublicationConflictError(
                "GCS graph receipt does not match the INLINE transition"
            )

    @staticmethod
    def _require_collection(
        record: _InlinePublicationRecord, collection_id: str
    ) -> None:
        if record.collection_id != collection_id:
            raise InlinePublicationConflictError(
                "INLINE collection identity changed"
            )

    @staticmethod
    def _require_operation_id(value: object, label: str) -> None:
        if not isinstance(value, str) or not value:
            raise ValueError(f"{label} must be a non-empty string")

    @staticmethod
    def _first_unacknowledged(bitmap: list[bool]) -> Optional[int]:
        return next((index for index, value in enumerate(bitmap) if not value), None)

    @staticmethod
    def _last_unacknowledged(bitmap: list[bool]) -> Optional[int]:
        return next((
            index for index in reversed(range(len(bitmap))) if not bitmap[index]
        ), None)
