import hashlib
from dataclasses import replace

import pytest

from miniray import protocol
from miniray.contained_cycle import ContainedReferenceGraphAuthority
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.stored_publication import (

from miniray.stored_publication_owner import (

    StoredOwnerEffectDisposition, StoredOwnerOutcome, StoredOwnerWakeReceipt,
    StoredPublicationClaimReceipt, StoredPublicationCollectionAdapter,
    StoredPublicationCollectionReceipt, StoredPublicationOwnerAdapter,
    StoredPublicationOwnerConflictError, StoredPublicationOwnerDiedError,
    StoredPublicationOwnerStateError, StoredPublicationPromotionReceipt,
)
from miniray.task_outputs import TaskExecutionKey, TaskOutputManifest


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self, edge_count: int = 2) -> None:
        self.job = JobID.random()
        self.task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), 0
        )
        self.attempt = AttemptID(self.task, 0)
        self.execution = TaskExecutionKey(
            TaskOutputManifest.for_task(self.task, 1), self.attempt
        )
        self.outer = self.execution.output_ids[0]
        self.owner = WorkerID.random()
        self.executor = WorkerID.random()
        self.node = NodeID.random()
        self.publication = StoredPublicationID(
            LeaseID.random(), self.task, self.attempt, self.outer
        )
        self.header = StoredPublicationHeader(
            self.publication, self.job, self.executor, self.owner, self.node
        )
        self.edges = tuple(
            ContainedReferenceEdge(
                self.outer,
                ObjectID.for_task(TaskID.derive(
                    self.job, TaskID.for_driver(self.job), index + 10
                )),
                WorkerID.random(), ("127.0.0.1", 33000 + index),
                f"pin-{index}",
            )
            for index in range(edge_count)
        )
        self.digest = hashlib.sha256(b"manifest").hexdigest()
        self.graph = self._graph()
        checksum = hashlib.sha256(b"payload").hexdigest()
        self.descriptor = protocol.ResultDescriptor(
            self.outer, protocol.ResultStorage.OBJECT_STORE, 7, self.owner,
            self.node, checksum,
        )
        self.envelope = StoredPublicationEnvelope(
            self.publication, self.digest, self.descriptor, self.graph,
            self.header,
        )

    def _graph(self):
        from miniray.contained_cycle import ContainedGraphManifest

        return ContainedGraphManifest(
            self.publication.graph_transaction_id, self.publication,
            self.owner, self.digest, self.edges,
        )


class _Effects:
    def __init__(self, fixture: _Fixture) -> None:
        self.f = fixture
        self.graph = ContainedReferenceGraphAuthority()
        self.graph.prepare_manifest(fixture.graph)
        self.events: list[str] = []
        self.fail_once: set[str] = set()
        self.claim = None
        self.promotions = None
        self.owner_commit = None
        self.wakes = {}
        self.adoption = None
        self.death_proof = None
        self.death_commit = None
        self.released = set()
        self.collection = None

    def _fail(self, name: str) -> None:
        if name in self.fail_once:
            self.fail_once.remove(name)
            raise TimeoutError(f"{name} ACK lost")

    def claim_publication(self, request):
        self.events.append("claim")
        self._fail("claim")
        if self.claim is not None and self.claim != request:
            raise AssertionError("claim changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if self.claim is None else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.claim = request
        return StoredPublicationClaimReceipt(request, disposition)

    def replay_promotions(self, request):
        self.events.append("promotions")
        self._fail("promotions")
        if self.promotions is not None and self.promotions != request:
            raise AssertionError("promotion request changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if self.promotions is None
            else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.promotions = request
        return StoredPublicationPromotionReceipt(request, disposition)

    def commit_graph(self, request):
        self.events.append("graph")
        self._fail("graph")
        return protocol.StoredPublicationGraphReply(
            request, self.graph.commit_manifest(request.manifest)
        )

    def commit_owner(self, request):
        self.events.append("owner")
        self._fail("owner")
        if self.owner_commit is not None and self.owner_commit != request:
            raise AssertionError("owner commit changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if self.owner_commit is None
            else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.owner_commit = request
        return StoredOwnerCommitReceipt(request, disposition)

    def wake(self, request):
        self.events.append("wake-" + request.outcome.value)
        self._fail("wake")
        previous = self.wakes.get(request.outcome)
        if previous is not None and previous != request:
            raise AssertionError("wake changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if previous is None else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.wakes[request.outcome] = request
        return StoredOwnerWakeReceipt(request, disposition)

    def ack_adopted(self, adoption):
        self.events.append("ack")
        self._fail("ack")
        if self.adoption is not None and self.adoption != adoption:
            raise AssertionError("adoption changed")
        self.adoption = adoption
        return adoption

    def owner_death(self, _owner):
        return self.death_proof

    def commit_owner_death(self, request):
        self.events.append("owner-death")
        self._fail("owner-death")
        if self.death_commit is not None and self.death_commit != request:
            raise AssertionError("death commit changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if self.death_commit is None
            else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.death_commit = request
        return StoredOwnerDeathCommitReceipt(request, disposition)

    def release_child(self, request):
        self.events.append("release-" + request.transfer_token)
        self._fail("release-" + request.transfer_token)
        key = (request.object_id, request.owner_worker_id, request.hold)
        changed = key not in self.released
        self.released.add(key)
        return protocol.ReleaseContainedReferenceReply(
            request.object_id, request.owner_worker_id, request.hold, True,
            changed,
        )

    def release_graph(self, manifest, object_id):
        self.events.append("release-graph")
        self._fail("release-graph")
        return self.graph.release_manifest_container(manifest, object_id)

    def complete_collection(self, request):
        self.events.append("complete-collection")
        self._fail("complete-collection")
        if self.collection is not None and self.collection != request:
            raise AssertionError("collection changed")
        disposition = (
            StoredOwnerEffectDisposition.APPLIED
            if self.collection is None
            else StoredOwnerEffectDisposition.ALREADY_APPLIED
        )
        self.collection = request
        return StoredPublicationCollectionReceipt(request, disposition)

    def adapter(self):
        collection = StoredPublicationCollectionAdapter(
            release_child=self.release_child, release_graph=self.release_graph,
            complete_owner=self.complete_collection,
        )
        return StoredPublicationOwnerAdapter(
            self.f.owner, claim_publication=self.claim_publication,
            replay_promotions=self.replay_promotions,
            commit_graph=self.commit_graph, commit_owner=self.commit_owner,
            wake_owner=self.wake, ack_adopted=self.ack_adopted,
            owner_death_lookup=self.owner_death,
            commit_owner_death=self.commit_owner_death, collection=collection,
        )


def test_adoption_orders_authorities_before_wake_and_node_ack() -> None:
    f = _Fixture()
    effects = _Effects(f)
    adapter = effects.adapter()

    adoption = adapter.adopt(f.execution, f.envelope)

    assert effects.events == [
        "claim", "promotions", "graph", "owner",
        "wake-READY", "ack",
    ]
    assert adoption.claim == effects.claim.claim
    assert effects.owner_commit.envelope == f.envelope
    assert effects.owner_commit.envelope.edges == f.edges
    snapshot = adapter.snapshot(f.publication)
    assert snapshot.owner_committed and snapshot.graph_committed
    assert snapshot.ready_woken and snapshot.adopted_acked

    assert adapter.adopt(f.execution, f.envelope) == adoption
    assert effects.events == [
        "claim", "promotions", "graph", "owner",
        "wake-READY", "ack",
    ]


@pytest.mark.parametrize(
    "failure,expected_prefix",
    [
        ("claim", []),
        ("promotions", ["claim"]),
        ("graph", ["claim", "promotions"]),
        ("owner", ["claim", "promotions", "graph"]),
        ("wake", ["claim", "promotions", "graph", "owner"]),
        ("ack", ["claim", "promotions", "graph", "owner",
                  "wake-READY"]),
    ],
)
def test_lost_ack_replays_only_the_pending_exact_effect(
    failure, expected_prefix
) -> None:
    f = _Fixture()
    effects = _Effects(f)
    adapter = effects.adapter()
    effects.fail_once.add(failure)

    with pytest.raises(TimeoutError, match=failure):
        adapter.adopt(f.execution, f.envelope)
    snapshot = adapter.snapshot(f.publication)
    assert not snapshot.adopted_acked
    if failure in ("wake", "ack"):
        assert snapshot.owner_committed and snapshot.graph_committed

    adapter.adopt(f.execution, f.envelope)
    # Every prefix effect completed before the fault runs once.  The failing
    # effect runs twice; later effects run once and READY never appears early.
    for event in expected_prefix:
        assert effects.events.count(event) == 1
    assert effects.events.count(failure if failure != "wake" else "wake-READY") == 2
    assert effects.events.index("wake-READY") > effects.events.index("owner")


def test_conflicting_envelope_replay_is_rejected_without_new_effect() -> None:
    f = _Fixture()
    effects = _Effects(f)
    adapter = effects.adapter()
    adapter.adopt(f.execution, f.envelope)
    before = tuple(effects.events)
    changed = replace(
        f.envelope,
        descriptor=replace(f.descriptor, size_bytes=f.descriptor.size_bytes + 1),
    )

    with pytest.raises(StoredPublicationOwnerConflictError, match="changed"):
        adapter.adopt(f.execution, changed)
    assert tuple(effects.events) == before


def test_owner_death_never_allows_ready_or_another_owner_takeover() -> None:
    f = _Fixture()
    effects = _Effects(f)
    effects.death_proof = "death:owner"
    adapter = effects.adapter()

    with pytest.raises(StoredPublicationOwnerDiedError):
        adapter.adopt(f.execution, f.envelope)
    assert effects.events == []

    other = StoredPublicationOwnerAdapter(
        WorkerID.random(), claim_publication=effects.claim_publication,
        replay_promotions=effects.replay_promotions,
        commit_graph=effects.commit_graph, commit_owner=effects.commit_owner,
        wake_owner=effects.wake, ack_adopted=effects.ack_adopted,
    )
    with pytest.raises(StoredPublicationOwnerConflictError, match="another"):
        other.adopt(f.execution, f.envelope)


def test_owner_death_cleanup_releases_children_before_graph_and_terminal_wake() -> None:
    f = _Fixture(2)
    effects = _Effects(f)
    effects.death_proof = "death:owner"
    adapter = effects.adapter()

    adoption = adapter.cleanup_owner_death(f.execution, f.envelope)

    assert effects.events == [
        "claim", "promotions", "graph", "owner-death",
        "release-pin-0", "release-pin-1", "release-graph",
        "complete-collection", "wake-OWNER_DIED", "ack",
    ]
    assert StoredOwnerOutcome.READY not in effects.wakes
    assert adoption.claim.claimant_worker_id == f.owner
    assert adapter.snapshot(f.publication).owner_death_committed


def test_collection_lost_child_ack_keeps_graph_until_retry() -> None:
    f = _Fixture(2)
    effects = _Effects(f)
    effects.death_proof = "death:owner"
    adapter = effects.adapter()
    effects.fail_once.add("release-pin-1")

    with pytest.raises(TimeoutError, match="pin-1"):
        adapter.cleanup_owner_death(f.execution, f.envelope)
    assert "release-graph" not in effects.events
    assert "complete-collection" not in effects.events
    assert "wake-OWNER_DIED" not in effects.events

    adapter.cleanup_owner_death(f.execution, f.envelope)
    assert effects.events.count("release-pin-0") == 1
    assert effects.events.count("release-pin-1") == 2
    assert effects.events.index("release-graph") > effects.events.index(
        "release-pin-1"
    )


def test_malformed_receipt_never_advances_or_wakes() -> None:
    f = _Fixture()
    effects = _Effects(f)
    adapter = effects.adapter()
    adapter._commit_owner = lambda request: StoredOwnerCommitReceipt(
        replace(request, adoption=replace(
            request.adoption, owner_commit_id="wrong"
        )),
        StoredOwnerEffectDisposition.APPLIED,
    )

    with pytest.raises(StoredPublicationOwnerStateError, match="owner commit"):
        adapter.adopt(f.execution, f.envelope)
    snapshot = adapter.snapshot(f.publication)
    assert snapshot.graph_committed and not snapshot.owner_committed
    assert not snapshot.ready_woken and not snapshot.adopted_acked
