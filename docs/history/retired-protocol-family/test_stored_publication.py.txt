class _Fixture:
    def __init__(self, transfer_count: int = 2) -> None:
        self.job_id = JobID.random()
        self.task_id = TaskID.derive(
            self.job_id, TaskID.for_driver(self.job_id), 0
        )
        self.attempt_id = AttemptID(self.task_id, 0)
        self.outer_object_id = ObjectID.for_task(self.task_id)
        self.executor = WorkerID.random()
        self.outer_owner = WorkerID.random()
        self.node_id = NodeID.random()
        self.publication_id = StoredPublicationID(
            LeaseID.random(), self.task_id, self.attempt_id,
            self.outer_object_id,
        )
        self.header = StoredPublicationHeader(
            self.publication_id, self.job_id, self.executor,
            self.outer_owner, self.node_id,
        )
        self.payload = b"stored outer with contained ObjectRefs"
        self.transfers = tuple(
            self._transfer(index) for index in range(transfer_count)
        )
        self.manifest = StoredPublicationManifest(
            self.publication_id, self.transfers, len(self.payload),
            hashlib.sha256(self.payload).hexdigest(),
            hashlib.sha256(
                b"ordered-contained-manifest"
                + bytes([transfer_count])
            ).hexdigest(),
        )
        self.graph = self.manifest.graph_manifest(self.outer_owner)
        self.descriptor = ResultDescriptor(
            self.outer_object_id, ResultStorage.OBJECT_STORE,
            len(self.payload), self.outer_owner, self.node_id,

            self.publication_id, TaskReplyStatus.SUCCEEDED
        )
        self.intent = StoredPublicationIntent(
            self.header, self.manifest, 4321, 9
        )

    def _transfer(self, index: int) -> PreparedContainedTransfer:
        child_task = TaskID.derive(
            self.job_id, TaskID.for_driver(self.job_id), index + 10
        )
        child = ObjectID.for_task(child_task)
        token = f"publication-pin:{index}"
        return PreparedContainedTransfer(
            contained_object_id=child,
            contained_owner_worker_id=self.executor,
            contained_owner_address=("127.0.0.1", 26000 + index),
            source=OwnedContainedSource(self.executor),
            provisional_hold=ContainedReferenceHold(
                self.outer_object_id, self.executor, token
            ),
            final_hold=ContainedReferenceHold(
                self.outer_object_id, self.outer_owner, token
            ),
        )


def _prepared(fixture: _Fixture) -> StoredPublicationJournal:
    journal = StoredPublicationJournal()
    assert journal.open(fixture.header)
    for index, transfer in enumerate(fixture.transfers):
        assert journal.append_pin_intent(
            fixture.publication_id, index, transfer
        )
        assert journal.mark_pin_prepared(
            fixture.publication_id, index, transfer
        )
    assert journal.freeze_discovery(
        fixture.publication_id, fixture.manifest
    )
    return journal


def _owner_death(
    owner_worker_id: WorkerID, *,
    reason: protocol.WorkerDeathReason = (
        protocol.WorkerDeathReason.PROCESS_EXIT
    ),
    detection_id: str = "stored-owner-death",
) -> protocol.WorkerDeathRecord:
    return protocol.WorkerDeathRecord(
        detection_id,
        protocol.WorkerIncarnation(
            NodeID.random(), 7654, 3, owner_worker_id, 8765
        ),
        17, -9, reason,
    )


def _stored_owner_resolution(
    fixture: _Fixture, record: object, death: protocol.WorkerDeathRecord, *,
    history: PublicationRecoveryHistory = (
        PublicationRecoveryHistory.NOT_COMMITTED
    ),
    graph_state: PublicationOwnerDeathGraphState = (
        PublicationOwnerDeathGraphState.NOT_FOUND
    ),
) -> PublicationOwnerDeathResolution:
    active_edges = (
        fixture.graph.ordered_edges
        if graph_state is PublicationOwnerDeathGraphState.COMMITTED
        else ()
    )
    request = StoredPublicationOwnerDeathRequest.from_recovery(
        record, death, history,
        PublicationOwnerDeathGraphWitness(
            fixture.graph, graph_state, active_edges,
        ),
    )
    return PublicationOwnerDeathResolution(
        request, PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED, (),
    )


def _sealed(fixture: _Fixture) -> StoredPublicationJournal:
    journal = _prepared(fixture)
    assert journal.begin_graph_prepare(
        fixture.publication_id, fixture.graph
    )
    assert journal.mark_graph_prepared(
        fixture.publication_id, fixture.graph
    )
    assert journal.begin_replica_seal(
        fixture.publication_id, fixture.descriptor
    )
    assert journal.mark_replica_sealed(
        fixture.publication_id, fixture.descriptor
    )
    return journal


def _promoted(fixture: _Fixture) -> StoredPublicationJournal:
    journal = _sealed(fixture)
    for index, transfer in enumerate(fixture.transfers):
        assert journal.begin_pin_promotion(
            fixture.publication_id, index, transfer
        )
        assert journal.mark_pin_promoted(
            fixture.publication_id, index, transfer
        )
    return journal

def test_pin_intent_must_precede_prepare_and_replay_is_exact() -> None:
    fixture = _Fixture(1)
    journal = StoredPublicationJournal()
    journal.open(fixture.header)
    transfer = fixture.transfers[0]

    with pytest.raises(StoredPublicationStateError, match="index"):
        journal.mark_pin_prepared(fixture.publication_id, 0, transfer)
    assert journal.append_pin_intent(fixture.publication_id, 0, transfer)
    assert not journal.append_pin_intent(fixture.publication_id, 0, transfer)
    conflicting = replace(
        transfer,
        provisional_hold=replace(
            transfer.provisional_hold, transfer_token="changed"
        ),
        final_hold=replace(transfer.final_hold, transfer_token="changed"),
    )
    with pytest.raises(StoredPublicationConflictError, match="index"):
        journal.append_pin_intent(fixture.publication_id, 0, conflicting)

    assert journal.mark_pin_prepared(fixture.publication_id, 0, transfer)
    assert not journal.mark_pin_prepared(fixture.publication_id, 0, transfer)
    snapshot = journal.snapshot(fixture.publication_id)
    assert snapshot.intent_written == (True,)
    assert snapshot.prepared_bitmap == (True,)

def test_freeze_is_an_all_prepare_barrier_and_closes_discovery() -> None:
    fixture = _Fixture(2)
    journal = StoredPublicationJournal()
    journal.open(fixture.header)
    for index, transfer in enumerate(fixture.transfers):
        journal.append_pin_intent(fixture.publication_id, index, transfer)
    journal.mark_pin_prepared(
        fixture.publication_id, 0, fixture.transfers[0]
    )

    with pytest.raises(StoredPublicationStateError, match="every pin"):
        journal.freeze_discovery(fixture.publication_id, fixture.manifest)
    journal.mark_pin_prepared(
        fixture.publication_id, 1, fixture.transfers[1]
    )
    assert journal.freeze_discovery(fixture.publication_id, fixture.manifest)
    assert not journal.freeze_discovery(fixture.publication_id, fixture.manifest)
    with pytest.raises(StoredPublicationConflictError, match="frozen"):
        journal.freeze_discovery(
            fixture.publication_id,
            replace(fixture.manifest, manifest_digest="ab" * 32),
        )
    with pytest.raises(StoredPublicationStateError, match="PINS_PREPARED"):
        journal.append_pin_intent(
            fixture.publication_id, 2, fixture._transfer(2)
        )

def test_partial_promotion_cannot_complete_succeeded_lease() -> None:
    fixture = _Fixture(2)
    journal = _sealed(fixture)
    first, second = fixture.transfers

    with pytest.raises(StoredPublicationStateError, match="intent"):
        journal.mark_pin_promoted(fixture.publication_id, 0, first)
    journal.begin_pin_promotion(fixture.publication_id, 0, first)
    journal.mark_pin_promoted(fixture.publication_id, 0, first)
    snapshot = journal.snapshot(fixture.publication_id)
    assert snapshot.promoted_bitmap == (True, False)
    assert not snapshot.ready_to_complete
    with pytest.raises(StoredPublicationStateError, match="every final"):
        journal.complete_lease(fixture.publication_id, fixture.terminal)

    journal.begin_pin_promotion(fixture.publication_id, 1, second)
    journal.mark_pin_promoted(fixture.publication_id, 1, second)
    assert journal.snapshot(fixture.publication_id).ready_to_complete
    envelope = journal.complete_lease(
        fixture.publication_id, fixture.terminal
    )
    assert envelope.descriptor == fixture.descriptor
    assert envelope.edges == tuple(item.edge for item in fixture.transfers)

def test_executor_crash_before_complete_freezes_exact_reverse_rollback() -> None:
    fixture = _Fixture(2)
    journal = _sealed(fixture)
    journal.begin_pin_promotion(
        fixture.publication_id, 0, fixture.transfers[0]
    )
    journal.mark_pin_promoted(
        fixture.publication_id, 0, fixture.transfers[0]
    )

    obligations = journal.begin_rollback(
        fixture.publication_id, "executor died before Complete"
    )

    assert [value.kind for value in obligations] == [
        StoredRollbackKind.GRAPH_MANIFEST,
        StoredRollbackKind.SEALED_REPLICA,
        StoredRollbackKind.FINAL_HOLD,
        StoredRollbackKind.FINAL_HOLD,
        StoredRollbackKind.PROVISIONAL_HOLD,
        StoredRollbackKind.PROVISIONAL_HOLD,
    ]
    assert [
        value.identity.transfer_index
        for value in obligations
        if value.kind is StoredRollbackKind.FINAL_HOLD
    ] == [1, 0]
    assert [
        value.identity.transfer_index
        for value in obligations
        if value.kind is StoredRollbackKind.PROVISIONAL_HOLD
    ] == [1, 0]
    # Both final and provisional identities are released regardless of the
    # observed promotion bitmap; release-before-add tombstones close ACK races.
    assert obligations[2].identity.hold == fixture.transfers[1].final_hold
    assert obligations[4].identity.hold == fixture.transfers[1].provisional_hold


def test_compensation_ack_loss_retains_identity_and_enforces_order() -> None:
    fixture = _Fixture(1)
    journal = _sealed(fixture)
    obligations = journal.begin_rollback(
        fixture.publication_id, "prepare path failed"
    )

    with pytest.raises(StoredPublicationStateError, match="order"):
        journal.mark_compensation_acked(
            fixture.publication_id, 1, obligations[1]
        )
    assert journal.mark_compensation_acked(
        fixture.publication_id, 0, obligations[0]

        fixture.publication_id, 0, obligations[0]
    )

    assert after_lost_ack.pending_compensations == obligations[1:]
    for obligation in obligations[1:]:
        assert journal.mark_compensation_acked(
            fixture.publication_id, obligation.ordinal, obligation
        )
    terminal = journal.snapshot(fixture.publication_id)
    assert terminal.state is StoredPublicationState.ROLLED_BACK
    assert terminal.pending_compensations == ()

def test_claim_and_adoption_conflicts_are_non_mutating() -> None:

        journal.claim(fixture.publication_id, wrong)

    assert journal.claim(fixture.publication_id, claim)
    before = journal.snapshot(fixture.publication_id)

        journal.claim(

    journal.mark_adopted(
        fixture.publication_id, StoredPublicationAdoption(claim, "commit:a")
    )
    with pytest.raises(StoredPublicationConflictError, match="receipt"):
        journal.mark_adopted(
            fixture.publication_id,
            StoredPublicationAdoption(claim, "commit:b"),
        )

def test_gcs_manifest_api_retains_full_identity_for_ack_recovery() -> None:

    assert released.disposition is ContainedGraphManifestDisposition.RELEASED
    assert released.released_edges == fixture.graph.ordered_edges
    assert authority.release_manifest_container(
        fixture.graph, fixture.outer_object_id
    ).disposition is ContainedGraphManifestDisposition.ALREADY_RELEASED
    replay = authority.release_manifest_container(
        fixture.graph, fixture.outer_object_id
    )
    assert replay.released_edges == fixture.graph.ordered_edges


def test_manifest_release_is_exact_per_container_and_legacy_safe() -> None:
    fixture = _Fixture(2)
    other_container = ObjectID.for_task(
        TaskID.derive(fixture.job_id, TaskID.for_driver(fixture.job_id), 90)
    )
    second = replace(
        fixture.graph.ordered_edges[1],
        container_object_id=other_container,
    )
    graph = replace(
        fixture.graph, ordered_edges=(fixture.graph.ordered_edges[0], second)
    )
    authority = ContainedReferenceGraphAuthority()
    authority.prepare_manifest(graph)
    authority.commit_manifest(graph)

    # ObjectID-only legacy operations cannot downgrade a full manifest.
    transaction = ContainedGraphTransaction(
        graph.transaction_id, graph.ordered_edges
    )
    for mutate in (authority.prepare, authority.commit, authority.abort,
                   authority.publish):
        with pytest.raises(
            ContainedGraphTransactionConflictError, match="edge-only"
        ):
            mutate(transaction)
    assert authority.release_container(fixture.outer_object_id) == ()

    first = authority.release_manifest_container(
        graph, fixture.outer_object_id
    )
    first_replay = authority.release_manifest_container(
        graph, fixture.outer_object_id
    )
    assert first.released_edges == first_replay.released_edges == (
        graph.ordered_edges[0],
    )
    assert authority.has_active_obligations()
    second_release = authority.release_manifest_container(
        graph, other_container
    )
    assert second_release.released_edges == (second,)
    assert not authority.has_active_obligations()
    snapshot = authority.snapshot()
    assert snapshot.manifests[0].released_container_ids == tuple(
        sorted((fixture.outer_object_id, other_container))
    )

def test_owner_death_and_intent_admission_linearize_without_orphan_record() -> None:
    fixture = _Fixture(0)
    authority = StoredPublicationRecoveryAuthority()
    death = _owner_death(fixture.outer_owner)
    barrier = Barrier(2)

    def report() -> object:
        barrier.wait()
        return authority.report_intent(fixture.intent)

    def freeze() -> tuple[object, ...]:
        barrier.wait()
        return authority.freeze_owner_death(death)

    with ThreadPoolExecutor(max_workers=2) as pool:
        report_future = pool.submit(report)
        freeze_future = pool.submit(freeze)
        frozen = freeze_future.result()
        try:
            acknowledgement = report_future.result()
        except StoredPublicationStateError:
            acknowledgement = None

    if acknowledgement is None:
        assert frozen == ()
        with pytest.raises(StoredPublicationStateError, match="not reported"):
            authority.snapshot(fixture.publication_id)
    else:
        assert frozen == (acknowledgement.record,)
        exact_replay = authority.report_intent(fixture.intent)
        assert exact_replay.disposition is (
            StoredPublicationRecoveryDisposition.ALREADY_RECORDED
        )
    assert authority.records_for_outer_owner(fixture.outer_owner) is frozen
    assert authority.frozen_records_for_owner_death(death) is frozen
