"""Pure contracts for stored outer contained-reference publication."""

from __future__ import annotations

import hashlib

import pytest

from miniray.ids import AttemptID, JobID, NodeID, ObjectID, TaskID, WorkerID
from miniray.stored_contained import (
    CommitStoredEdgesReply,
    ContainedReferenceIdentity,
    PrepareContainedPinReply,
    RollbackKind,
    RollbackReply,
    SealStoredOuterReply,
    StoredContainedState,
    StoredContainedStateError,
    StoredContainedTransaction,
    StoredContainedTransactionError,
    StoredOuterIdentity,
    StoredReplicaIdentity,
)


pytestmark = pytest.mark.unit


def _outer() -> tuple[StoredOuterIdentity, StoredReplicaIdentity]:
    job_id = JobID.random()
    task_id = TaskID.derive(job_id, TaskID.for_driver(job_id), 0)
    object_id = ObjectID.for_task(task_id)
    outer = StoredOuterIdentity(
        object_id, AttemptID(task_id, 0), WorkerID.random()
    )
    payload = b"stored outer payload"
    return outer, StoredReplicaIdentity(
        outer, NodeID.random(), len(payload), hashlib.sha256(payload).hexdigest()
    )


def _source(
    index: int, *, borrower_token: str | None = None
) -> ContainedReferenceIdentity:
    job_id = JobID.random()
    parent = TaskID.for_driver(job_id)
    object_id = ObjectID.for_task(TaskID.derive(job_id, parent, index))
    return ContainedReferenceIdentity(
        object_id, WorkerID.random(), ("127.0.0.1", 25000 + index),
        borrower_token,
    )


class _Callbacks:
    def __init__(self) -> None:
        self.events: list[tuple[str, object]] = []
        self.rollback_failures = 0

    def prepare(self, request):
        self.events.append(("prepare", request))
        return PrepareContainedPinReply(request, True)

    def seal(self, replica):
        self.events.append(("seal", replica))
        return SealStoredOuterReply(replica, True)

    def commit(self, manifest):
        self.events.append(("commit", manifest))
        return CommitStoredEdgesReply(manifest, True)

    def rollback(self, obligation):
        self.events.append(("rollback", obligation))
        if self.rollback_failures:
            self.rollback_failures -= 1
            raise TimeoutError("rollback ACK lost")
        return RollbackReply(obligation, True)


def _transaction(callbacks: _Callbacks, tokens: list[str]):
    outer, replica = _outer()
    token_iterator = iter(tokens)
    transaction = StoredContainedTransaction(
        outer,
        prepare_pin=callbacks.prepare,
        seal_outer=callbacks.seal,
        commit_edges=callbacks.commit,
        rollback=callbacks.rollback,
        token_factory=lambda: next(token_iterator),
    )
    return transaction, replica


def test_local_and_foreign_sources_share_one_prepare_seal_commit_protocol() -> None:
    callbacks = _Callbacks()
    transaction, replica = _transaction(callbacks, ["local", "foreign"])
    local = _source(1)
    foreign = _source(2, borrower_token="active-borrower")

    local_pin = transaction.prepare_reference(local)
    foreign_pin = transaction.prepare_reference(foreign)
    assert local_pin.transfer_token == "local"
    assert foreign_pin.transfer_token == "foreign"
    assert local_pin.exported_reference[:3] == (
        local.object_id, local.owner_worker_id, local.owner_address
    )
    assert foreign_pin.exported_reference[:3] == (
        foreign.object_id, foreign.owner_worker_id, foreign.owner_address
    )

    assert transaction.finish_pin_preparation() == (local_pin, foreign_pin)
    transaction.seal(replica)
    manifest = transaction.commit_edge_manifest()

    assert transaction.state is StoredContainedState.COMMITTED
    assert transaction.pending_rollback == ()
    assert tuple(edge.contained_object_id for edge in manifest.edges) == (
        local.object_id, foreign.object_id
    )
    assert tuple(edge.transfer_token for edge in manifest.edges) == (
        "local", "foreign"
    )
    assert [name for name, _value in callbacks.events] == [
        "prepare", "prepare", "seal", "commit"
    ]
    assert callbacks.events[0][1].source.borrower_token is None
    assert callbacks.events[1][1].source.borrower_token == "active-borrower"


def test_seal_is_fenced_until_all_pin_discovery_is_closed() -> None:
    callbacks = _Callbacks()
    transaction, replica = _transaction(callbacks, ["pin"])
    transaction.prepare_reference(_source(1))

    with pytest.raises(StoredContainedStateError, match="PINS_PREPARED"):
        transaction.seal(replica)
    assert [name for name, _value in callbacks.events] == ["prepare"]

    transaction.finish_pin_preparation()
    transaction.seal(replica)
    with pytest.raises(StoredContainedStateError, match="PREPARING_PINS"):
        transaction.prepare_reference(_source(2))


def test_later_pin_prepare_failure_rolls_back_every_identity_in_reverse() -> None:
    callbacks = _Callbacks()
    first = True

    def prepare(request):
        nonlocal first
        callbacks.events.append(("prepare", request))
        if first:
            first = False
            return PrepareContainedPinReply(request, True)
        raise TimeoutError("prepare ACK lost")

    transaction, _replica = _transaction(callbacks, ["one", "two"])
    transaction._prepare_pin = prepare
    transaction.prepare_reference(_source(1))

    with pytest.raises(StoredContainedTransactionError, match="prepare"):
        transaction.prepare_reference(_source(2))

    assert transaction.state is StoredContainedState.ROLLED_BACK
    rolled_back = [
        value.identity.transfer_token
        for name, value in callbacks.events
        if name == "rollback"
    ]
    # The ambiguous second prepare is compensated before the acknowledged one.
    assert rolled_back == ["two", "one"]
    assert transaction.pending_rollback == ()


@pytest.mark.parametrize("failure_phase", ("seal", "edge"))
def test_seal_or_edge_failure_compensates_in_exact_reverse_order(
    failure_phase: str,
) -> None:
    callbacks = _Callbacks()
    transaction, replica = _transaction(callbacks, ["a", "b"])
    transaction.prepare_reference(_source(1))
    transaction.prepare_reference(_source(2, borrower_token="borrow"))
    transaction.finish_pin_preparation()

    if failure_phase == "seal":
        def reject_seal(value):
            callbacks.events.append(("seal", value))
            return SealStoredOuterReply(value, False, "seal rejected")

        transaction._seal_outer = reject_seal
        invoke = lambda: transaction.seal(replica)
    else:
        transaction.seal(replica)

        def lose_edge_ack(value):
            callbacks.events.append(("commit", value))
            raise TimeoutError("edge ACK lost")

        transaction._commit_edges = lose_edge_ack
        invoke = transaction.commit_edge_manifest

    with pytest.raises(StoredContainedTransactionError):
        invoke()

    kinds = [
        value.kind
        for name, value in callbacks.events
        if name == "rollback"
    ]
    expected = (
        [
            RollbackKind.SEALED_OUTER,
            RollbackKind.CONTAINED_PIN,
            RollbackKind.CONTAINED_PIN,
        ]
        if failure_phase == "seal"
        else [
            RollbackKind.EDGE_MANIFEST,
            RollbackKind.SEALED_OUTER,
            RollbackKind.CONTAINED_PIN,
            RollbackKind.CONTAINED_PIN,
        ]
    )
    assert kinds == expected
    assert transaction.state is StoredContainedState.ROLLED_BACK
    assert transaction.pending_rollback == ()


def test_lost_rollback_ack_retains_exact_obligation_for_retry() -> None:
    callbacks = _Callbacks()
    callbacks.rollback_failures = 1
    transaction, replica = _transaction(callbacks, ["one", "two"])
    transaction.prepare_reference(_source(1))
    transaction.prepare_reference(_source(2))
    transaction.finish_pin_preparation()
    transaction.seal(replica)

    def fail_commit(manifest):
        callbacks.events.append(("commit", manifest))
        raise TimeoutError("commit ACK lost")

    transaction._commit_edges = fail_commit
    with pytest.raises(StoredContainedTransactionError):
        transaction.commit_edge_manifest()

    assert transaction.state is StoredContainedState.ROLLBACK_PENDING
    pending = transaction.pending_rollback
    assert len(pending) == 4
    assert pending[0].kind is RollbackKind.EDGE_MANIFEST
    frozen = pending[0]

    assert transaction.retry_rollback()
    assert transaction.state is StoredContainedState.ROLLED_BACK
    assert transaction.pending_rollback == ()
    edge_attempts = [
        value for name, value in callbacks.events
        if name == "rollback" and value.kind is RollbackKind.EDGE_MANIFEST
    ]
    assert edge_attempts == [frozen, frozen]


def test_wrong_rollback_ack_cannot_discharge_pending_identity() -> None:
    callbacks = _Callbacks()
    transaction, _replica = _transaction(callbacks, ["pin"])
    pin = transaction.prepare_reference(_source(1))
    expected = transaction.pending_rollback[0]

    def wrong_ack(_obligation):
        other = expected.__class__(
            expected.kind,
            pin.__class__(
                pin.object_id, pin.owner_worker_id, pin.owner_address, "other"
            ),
        )
        return RollbackReply(other, True)

    transaction._rollback = wrong_ack
    assert not transaction.rollback()
    assert transaction.state is StoredContainedState.ROLLBACK_PENDING
    assert transaction.pending_rollback == (expected,)

    transaction._rollback = callbacks.rollback
    assert transaction.retry_rollback()
    assert transaction.pending_rollback == ()


def test_successful_edge_commit_transfers_cleanup_ownership_to_manifest() -> None:
    callbacks = _Callbacks()
    transaction, replica = _transaction(callbacks, ["pin"])
    transaction.prepare_reference(_source(1))
    transaction.finish_pin_preparation()
    transaction.seal(replica)
    manifest = transaction.commit_edge_manifest()

    with pytest.raises(StoredContainedStateError, match="committed"):
        transaction.rollback()
    assert transaction.manifest is manifest
    assert transaction.pending_rollback == ()
    assert not any(name == "rollback" for name, _value in callbacks.events)
