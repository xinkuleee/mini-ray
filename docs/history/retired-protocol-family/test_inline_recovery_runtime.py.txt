def _freeze(
    fixture: _Fixture, *, completed: bool = False, prepared: bool = False,
) -> tuple[InlinePublicationRecoveryAuthority, object]:
    authority = InlinePublicationRecoveryAuthority()
    authority.report_intent(fixture.intent)
    if prepared or completed:
        authority.report_prepared(fixture.publication_id)
    if completed:
        authority.report_terminal(InlinePublicationTerminal(
            fixture.publication_id, protocol.TaskReplyStatus.SUCCEEDED
        ))
    return authority, authority.freeze_node_death(fixture.death)[0]


def _release_receipt(
    effect: ReleaseInlineRecoveryHoldEffect, *, released: bool = True,
) -> InlineRecoveryHoldReceipt:
    reply = protocol.ReleaseContainedReferenceReply(
        effect.request.object_id, effect.request.owner_worker_id,
        effect.request.hold, True, released,
    )
    return InlineRecoveryHoldReceipt(
        effect,
        (InlineRecoveryHoldDisposition.RELEASED if released else
         InlineRecoveryHoldDisposition.ALREADY_RELEASED),
        release_reply=reply,
    )


def _abort_reply(
    effect: AbortInlineRecoveryGraphEffect, *, replay: bool = False,
) -> protocol.ContainedGraphReply:
    receipt = ContainedGraphManifestReceipt(
        effect.request.manifest, ContainedGraphTransactionState.ABORTED,
        (ContainedGraphManifestDisposition.ALREADY_ABORTED if replay else
         ContainedGraphManifestDisposition.APPLIED),
    )
    return protocol.ContainedGraphReply(effect.request, receipt=receipt)


def _graph_cleanup_for(work, graph) -> InlineRecoveryGraphEffect:
    """Single-thread stand-in for GCS's composition-locked graph choice."""

    manifest = work.manifest.to_graph_manifest()
    current = next((
        snapshot for snapshot in graph.snapshot().manifests
        if snapshot.manifest.transaction_id == manifest.transaction_id
    ), None)
    assert current is None or current.manifest == manifest
    if current is not None and current.state is ContainedGraphTransactionState.COMMITTED:
        return ReleaseInlineRecoveryGraphEffect(
            work, protocol.ReleaseContainedGraphContainer(
                manifest, work.publication_id.outer_object_id
            ),
        )
    return AbortInlineRecoveryGraphEffect(
        work, protocol.AbortContainedGraph(manifest)
    )


def _release_graph_reply(
    effect: ReleaseInlineRecoveryGraphEffect, *, replay: bool = False,
) -> protocol.ContainedGraphReply:
    return protocol.ContainedGraphReply(
        effect.request, receipt=ContainedGraphManifestReceipt(
            effect.request.manifest, ContainedGraphTransactionState.COMMITTED,
            (ContainedGraphManifestDisposition.ALREADY_RELEASED if replay else
             ContainedGraphManifestDisposition.RELEASED),
            effect.work.manifest.ordered_edges,
        ),
    )


def test_drive_releases_every_slot_reverse_aborts_then_resolves() -> None:
    fixture = _Fixture()
    authority, work = _freeze(fixture)
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.envelope.to_graph_manifest())
    events: list[tuple[str, object]] = []

    def release(effect: ReleaseInlineRecoveryHoldEffect):
        assert authority.snapshot(
            fixture.publication_id
        ).resolution is None
        events.append(("release", effect.index))
        return _release_receipt(effect, released=effect.index != 1)

    def abort(effect: AbortInlineRecoveryGraphEffect):
        assert [value for name, value in events if name == "release"] == [
            2, 1, 0
        ]
        assert authority.snapshot(
            fixture.publication_id
        ).resolution is None
        events.append(("abort", effect.request))
        return protocol.ContainedGraphReply(
            effect.request, receipt=graph.abort_manifest(
                effect.request.manifest
            ),
        )

    runtime = InlineNodeLossRuntime(
        authority, release_child=release, abort_graph=abort
    )
    result = runtime.drive(work)

    assert [name for name, _value in events] == [
        "release", "release", "release", "abort"
    ]
    assert result.release_bitmap == (True, True, True)
    assert result.graph_aborted and result.authority_acknowledged
    assert result.resolution_ack is not None
    assert result.resolution_ack.snapshot.resolution is (
        InlineRecoveryResolution.ROLLED_BACK
    )
    assert not runtime.has_active_operations()
    assert not runtime.has_active_obligations()
    before = tuple(events)
    assert runtime.drive(work) == result
    assert tuple(events) == before


def test_manual_effect_order_and_authority_barriers_are_explicit() -> None:
    fixture = _Fixture(2)
    authority, work = _freeze(fixture)
    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: _release_receipt(effect),
        abort_graph=lambda effect: _abort_reply(effect),
    )
    # The frozen GCS item is a shutdown obligation before first local pull.
    assert runtime.has_active_operations()

    second = runtime.next_effect(work)
    assert isinstance(second, ReleaseInlineRecoveryHoldEffect)
    assert second.index == 1
    assert runtime.has_active_operations()
    assert runtime.has_active_obligations()
    with pytest.raises(InlineNodeLossRuntimeStateError, match="authority"):
        runtime.acknowledge_authority(work)

    snapshot = runtime.acknowledge(second, _release_receipt(second))
    first = snapshot.next_effect
    assert isinstance(first, ReleaseInlineRecoveryHoldEffect)
    assert first.index == 0
    snapshot = runtime.acknowledge(first, _release_receipt(first, released=False))
    graph_effect = snapshot.next_effect
    assert isinstance(graph_effect, AbortInlineRecoveryGraphEffect)
    snapshot = runtime.acknowledge(graph_effect, _abort_reply(graph_effect))
    assert snapshot.graph_aborted and not snapshot.authority_acknowledged
    assert authority.snapshot(fixture.publication_id).resolution is None

    snapshot = runtime.acknowledge_authority(work)
    assert snapshot.complete
    assert authority.snapshot(fixture.publication_id).resolution is (
        InlineRecoveryResolution.ROLLED_BACK
    )


def test_lost_release_ack_replays_same_exact_effect_only() -> None:
    fixture = _Fixture(2)
    authority, work = _freeze(fixture)
    attempts: list[ReleaseInlineRecoveryHoldEffect] = []

    def release(effect: ReleaseInlineRecoveryHoldEffect):
        attempts.append(effect)
        if len(attempts) == 1:
            raise ConnectionError("release ACK lost")
        return _release_receipt(effect, released=len(attempts) != 2)

    runtime = InlineNodeLossRuntime(
        authority, release_child=release,
        abort_graph=lambda effect: _abort_reply(effect),
    )
    with pytest.raises(ConnectionError, match="ACK lost"):
        runtime.drive(work)
    assert runtime.snapshot(work).release_bitmap == (False, False)
    assert runtime.has_active_operations()

    done = runtime.drive(work)
    assert attempts[0] == attempts[1]
    assert [effect.index for effect in attempts] == [1, 1, 0]
    assert done.complete


def test_malformed_release_ack_does_not_advance_and_exact_replay_recovers() -> None:
    fixture = _Fixture(2)
    authority, work = _freeze(fixture)
    attempts: list[ReleaseInlineRecoveryHoldEffect] = []

    def release(effect: ReleaseInlineRecoveryHoldEffect):
        attempts.append(effect)
        if len(attempts) == 1:
            other = InlineNodeLossRuntime._hold_effect(work, 0)
            return _release_receipt(other)
        return _release_receipt(effect)

    runtime = InlineNodeLossRuntime(
        authority, release_child=release,
        abort_graph=lambda effect: _abort_reply(effect),
    )
    with pytest.raises(
        InlineNodeLossRuntimeConflictError, match="exact effect"
    ):
        runtime.drive(work)
    pending = runtime.snapshot(work)
    assert pending.release_bitmap == (False, False)
    assert isinstance(pending.next_effect, ReleaseInlineRecoveryHoldEffect)
    assert pending.next_effect == attempts[0]

    assert runtime.drive(work).complete
    assert attempts[0] == attempts[1]
    assert [effect.index for effect in attempts] == [1, 1, 0]


def test_owner_death_is_exact_terminal_proof_and_expected_exit_is_not() -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture)
    effect = InlineNodeLossRuntime._hold_effect(work, 0)
    receipt = InlineRecoveryHoldReceipt(
        effect, InlineRecoveryHoldDisposition.OWNER_DEAD,
        owner_death=fixture.worker_death(),
    )
    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda _effect: receipt,
        abort_graph=lambda graph_effect: _abort_reply(graph_effect),
    )
    assert runtime.drive(work).complete

    with pytest.raises(ValueError, match="non-expected"):
        InlineRecoveryHoldReceipt(

            owner_death=fixture.worker_death(
                reason=protocol.WorkerDeathReason.EXPECTED
            ),
        )
    with pytest.raises(ValueError, match="exact non-expected"):
        InlineRecoveryHoldReceipt(
            effect, InlineRecoveryHoldDisposition.OWNER_DEAD,
            owner_death=fixture.worker_death(WorkerID.random()),
        )


def test_lost_graph_ack_replays_exact_abort_before_authority_ack() -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture)
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.envelope.to_graph_manifest())
    calls: list[AbortInlineRecoveryGraphEffect] = []

    def abort(effect: AbortInlineRecoveryGraphEffect):
        calls.append(effect)
        receipt = graph.abort_manifest(effect.request.manifest)
        if len(calls) == 1:
            raise ConnectionError("ABORT ACK lost")
        return protocol.ContainedGraphReply(effect.request, receipt=receipt)

    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: _release_receipt(effect),
        abort_graph=abort,
    )
    with pytest.raises(ConnectionError, match="ACK lost"):
        runtime.drive(work)
    pending = runtime.snapshot(work)
    assert pending.release_bitmap == (True,)
    assert not pending.graph_aborted
    assert authority.snapshot(fixture.publication_id).resolution is None

    done = runtime.drive(work)
    assert calls[0] == calls[1]
    assert done.complete
    assert calls[1].request == calls[0].request

    assert authority.snapshot(fixture.publication_id).resolution is None

    assert runtime.drive(work).complete

def test_malformed_authority_ack_replays_only_exact_resolution() -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture)
    events: list[str] = []
    original = authority.acknowledge_rollback
    acknowledgements = 0

    def malformed_once(candidate):
        nonlocal acknowledgements
        acknowledgements += 1
        ack = original(candidate)
        if acknowledgements == 1:
            return replace(ack, snapshot=candidate.snapshot)
        return ack

    authority.acknowledge_rollback = (  # type: ignore[method-assign]
        malformed_once
    )
    runtime = InlineNodeLossRuntime(
        authority,
        release_child=lambda effect: (
            events.append("release") or _release_receipt(effect)
        ),
        abort_graph=lambda effect: (
            events.append("abort") or _abort_reply(effect)
        ),
    )
    with pytest.raises(
        InlineNodeLossRuntimeConflictError, match="exact rollback"
    ):
        runtime.drive(work)
    pending = runtime.snapshot(work)
    assert pending.graph_aborted and not pending.authority_acknowledged

    done = runtime.drive(work)
    assert done.complete
    assert events == ["release", "abort"]
    assert acknowledgements == 2


def test_authority_ack_loss_replays_only_resolution_not_prior_effects() -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture)
    events: list[str] = []
    original = authority.acknowledge_rollback
    acknowledgements = 0

    def lose_once(candidate):
        nonlocal acknowledgements
        acknowledgements += 1
        ack = original(candidate)
        if acknowledgements == 1:
            raise ConnectionError("authority ACK lost")
        return ack

    authority.acknowledge_rollback = lose_once  # type: ignore[method-assign]
    runtime = InlineNodeLossRuntime(
        authority,
        release_child=lambda effect: (
            events.append("release") or _release_receipt(effect)
        ),
        abort_graph=lambda effect: (
            events.append("abort") or _abort_reply(effect)
        ),
    )
    with pytest.raises(ConnectionError, match="ACK lost"):
        runtime.drive(work)
    pending = runtime.snapshot(work)
    assert pending.graph_aborted and not pending.authority_acknowledged
    assert authority.snapshot(fixture.publication_id).resolution is (
        InlineRecoveryResolution.ROLLED_BACK
    )

    done = runtime.drive(work)
    assert events == ["release", "abort"]
    assert acknowledgements == 2
    assert done.complete
    assert done.resolution_ack is not None
    assert done.resolution_ack.disposition is (
        InlineRecoveryDisposition.ALREADY_RECORDED
    )

@pytest.mark.parametrize(
    "completed", [False, True], ids=["completion_unknown", "postcomplete"]
)
@pytest.mark.parametrize("graph_state", ["absent", "prepared", "aborted", "committed"])
def test_drop_reuses_reverse_cleanup_and_selects_exact_graph_terminal(
    completed: bool, graph_state: str,
) -> None:
    fixture = _Fixture(3)
    authority, work = _freeze(fixture, prepared=True, completed=completed)
    graph = ContainedReferenceGraphAuthority()
    manifest = fixture.manifest.to_graph_manifest()
    if graph_state != "absent":
        graph.prepare_manifest(manifest)
    if graph_state == "aborted":
        graph.abort_manifest(manifest)
    if graph_state == "committed":
        graph.commit_manifest(manifest)

    events: list[tuple[str, object]] = []

    def choose(candidate):
        assert candidate == work
        assert [value for name, value in events if name == "child"] == [2, 1, 0]
        assert authority.snapshot(fixture.publication_id).owner_decision is (
            InlineRecoveryOwnerDecision.DROP
        )
        events.append(("choose", candidate))
        return _graph_cleanup_for(candidate, graph)

    def abort(effect):
        events.append(("abort", effect))
        return protocol.ContainedGraphReply(
            effect.request, receipt=graph.abort_manifest(effect.request.manifest)
        )

    def release_graph(effect):
        events.append(("release_graph", effect))
        return protocol.ContainedGraphReply(
            effect.request, receipt=graph.release_manifest_container(
                effect.request.manifest, effect.request.container_object_id
            ),
        )

    runtime = InlineNodeLossRuntime(
        authority,
        release_child=lambda effect: (
            events.append(("child", effect.index))
            or _release_receipt(effect, released=effect.index != 1)
        ),
        abort_graph=abort, choose_graph_cleanup=choose,
        release_graph=release_graph,
    )
    assert not runtime.has_active_operations()
    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    assert runtime.has_active_operations()
    assert work.snapshot.owner_decision is None

    result = runtime.drive(work)
    assert result.complete and result.graph_cleaned
    assert result.graph_aborted is (graph_state != "committed")
    assert result.graph_released is (graph_state == "committed")
    assert result.release_bitmap == (True, True, True)
    assert result.resolution_ack is not None
    assert result.resolution_ack.snapshot == replace(
        work.snapshot, owner_decision=InlineRecoveryOwnerDecision.DROP,
        resolution=InlineRecoveryResolution.LOST,
    )
    assert [name for name, _ in events] == [
        "child", "child", "child", "choose",
        "release_graph" if graph_state == "committed" else "abort",
    ]
    assert not graph.has_active_obligations()
    assert not runtime.has_active_operations()
    before = tuple(events)
    assert runtime.drive(work) == result
    assert tuple(events) == before


@pytest.mark.parametrize("committed", [False, True], ids=["abort", "release"])
def test_drop_graph_ack_loss_replays_selected_effect_without_rechoosing(
    committed: bool,
) -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture, prepared=True)
    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.manifest.to_graph_manifest())
    if committed:
        graph.commit_manifest(fixture.manifest.to_graph_manifest())
    selected: list[object] = []
    calls: list[InlineRecoveryGraphEffect] = []
    child_calls: list[object] = []

    def choose(candidate):
        selected.append(candidate)
        return _graph_cleanup_for(candidate, graph)

    def clean(effect):
        calls.append(effect)
        receipt = (
            graph.release_manifest_container(
                effect.request.manifest, effect.request.container_object_id
            )
            if isinstance(effect, ReleaseInlineRecoveryGraphEffect)
            else graph.abort_manifest(effect.request.manifest)
        )
        if len(calls) == 1:
            raise ConnectionError("graph cleanup ACK lost")
        return protocol.ContainedGraphReply(effect.request, receipt=receipt)

    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: (
            child_calls.append(effect) or _release_receipt(effect)
        ),
        abort_graph=clean, choose_graph_cleanup=choose, release_graph=clean,
    )
    with pytest.raises(ConnectionError, match="ACK lost"):
        runtime.drive(work)
    waiting = runtime.snapshot(work)
    assert waiting.next_effect == calls[0]
    assert not waiting.graph_cleaned
    assert authority.snapshot(fixture.publication_id).resolution is None
    assert runtime.drive(work).complete
    assert calls[0] == calls[1]
    assert selected == [work]
    assert len(child_calls) == 1


@pytest.mark.parametrize("failure", ["lost_ack", "malformed_ack"])
def test_drop_authority_ack_replay_never_reexecutes_cleanup(failure: str) -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture, completed=True)
    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    events: list[str] = []
    original = authority.acknowledge_lost
    calls = 0

    def lose_once(candidate):
        nonlocal calls
        calls += 1
        result = original(candidate)
        if calls == 1:
            if failure == "lost_ack":
                raise ConnectionError("LOST ACK lost")
            return replace(
                result, snapshot=replace(result.snapshot, resolution=None)
            )
        return result

    authority.acknowledge_lost = lose_once  # type: ignore[method-assign]
    runtime = InlineNodeLossRuntime(
        authority,
        release_child=lambda effect: (events.append("child") or _release_receipt(effect)),
        abort_graph=lambda effect: (events.append("abort") or _abort_reply(effect)),
        choose_graph_cleanup=lambda candidate: AbortInlineRecoveryGraphEffect(
            candidate, protocol.AbortContainedGraph(candidate.manifest.to_graph_manifest())
        ),
        release_graph=lambda effect: pytest.fail("uncommitted graph was released"),
    )
    expected_exception = (
        ConnectionError if failure == "lost_ack" else InlineNodeLossRuntimeConflictError
    )
    with pytest.raises(expected_exception):
        runtime.drive(work)
    waiting = runtime.snapshot(work)
    assert waiting.graph_cleaned and not waiting.authority_acknowledged
    assert authority.snapshot(fixture.publication_id).resolution is InlineRecoveryResolution.LOST
    done = runtime.drive(work)
    assert done.complete
    assert done.resolution_ack.disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert calls == 2
    assert events == ["child", "abort"]

def test_release_graph_rejects_partial_edge_receipt_and_replays_exact_effect() -> None:
    fixture = _Fixture(2)

    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    calls: list[ReleaseInlineRecoveryGraphEffect] = []

    def release_graph(effect):
        calls.append(effect)
        if len(calls) == 1:
            # Simulate a malformed wire object that skipped dataclass validation.
            valid = _release_graph_reply(effect)
            malformed = object.__new__(protocol.ContainedGraphReply)
            object.__setattr__(malformed, "request", effect.request)
            object.__setattr__(malformed, "error_kind", None)
            object.__setattr__(malformed, "error", None)
            object.__setattr__(malformed, "receipt", replace(
                valid.receipt, released_edges=effect.work.manifest.ordered_edges[:1]
            ))
            return malformed
        return _release_graph_reply(effect, replay=True)

    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: _release_receipt(effect),
        abort_graph=lambda effect: pytest.fail("committed graph was aborted"),

                candidate.manifest.to_graph_manifest(), candidate.publication_id.outer_object_id

    assert authority.snapshot(fixture.publication_id).resolution is None
    assert runtime.drive(work).complete
    assert calls[0] == calls[1]


def test_drop_graph_release_manual_ack_order_and_replay_are_exact() -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture, prepared=True)
    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    graph_effect = ReleaseInlineRecoveryGraphEffect(
        work, protocol.ReleaseContainedGraphContainer(
            work.manifest.to_graph_manifest(), fixture.outer
        ),
    )
    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: _release_receipt(effect),
        abort_graph=lambda effect: _abort_reply(effect),
        choose_graph_cleanup=lambda candidate: graph_effect,
        release_graph=lambda effect: _release_graph_reply(effect),
    )
    with pytest.raises(InlineNodeLossRuntimeStateError, match="next exact effect"):
        runtime.acknowledge(graph_effect, _release_graph_reply(graph_effect))
    child = runtime.next_effect(work)
    runtime.acknowledge(child, _release_receipt(child))
    with pytest.raises(InlineNodeLossRuntimeStateError, match="before all holds"):
        runtime.acknowledge_authority(work)
    receipt = _release_graph_reply(graph_effect)
    acknowledged = runtime.acknowledge(graph_effect, receipt)
    assert acknowledged.graph_released and not acknowledged.graph_aborted
    assert runtime.acknowledge(graph_effect, receipt) == acknowledged
    with pytest.raises(InlineNodeLossRuntimeStateError, match="next exact effect"):
        runtime.acknowledge(graph_effect, _release_graph_reply(graph_effect, replay=True))
    wrong_effect = AbortInlineRecoveryGraphEffect(
        work, protocol.AbortContainedGraph(work.manifest.to_graph_manifest())
    )
    with pytest.raises(InlineNodeLossRuntimeStateError, match="next exact effect"):
        runtime.acknowledge(wrong_effect, receipt)
    assert runtime.acknowledge_authority(work).complete

@pytest.mark.parametrize("when", ["before_first_effect", "after_child_ack", "after_choice"])
def test_owner_death_supersession_fences_new_or_retained_cleanup(when: str) -> None:
    fixture = _Fixture(1)
    authority, work = _freeze(fixture, prepared=True)
    authority.decide_owner_resolution(
        work, fixture.owner, InlineRecoveryOwnerDecision.DROP
    )
    events: list[str] = []

    def supersede():
        authority.supersede_owner_death(
            fixture.publication_id, fixture.worker_death(fixture.owner)
        )

    def choose(candidate):
        events.append("choose")
        effect = AbortInlineRecoveryGraphEffect(
            candidate, protocol.AbortContainedGraph(candidate.manifest.to_graph_manifest())
        )
        if when == "after_child_ack":
            supersede()
        return effect

    runtime = InlineNodeLossRuntime(
        authority, release_child=lambda effect: (events.append("child") or _release_receipt(effect)),
        abort_graph=lambda effect: (events.append("abort") or _abort_reply(effect)),
        choose_graph_cleanup=choose,
        release_graph=lambda effect: events.append("release_graph"),
    )
    if when == "before_first_effect":
        supersede()
    elif when == "after_choice":
        child = runtime.next_effect(work)
        progress = runtime.acknowledge(child, _release_receipt(child))
        assert isinstance(progress.next_effect, AbortInlineRecoveryGraphEffect)
        supersede()
    with pytest.raises(InlineNodeLossRuntimeStateError, match="superseded"):
        runtime.drive(work)
    assert "abort" not in events and "release_graph" not in events
    assert not runtime.has_active_operations()
