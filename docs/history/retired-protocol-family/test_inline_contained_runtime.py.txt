"""Executable ordering contracts for contained INLINE publication."""

from __future__ import annotations

import pytest

from miniray import protocol
from miniray.contained_cycle import (
    ContainedGraphManifestDisposition,
    ContainedGraphManifestReceipt,
    ContainedGraphTransactionState,
    ContainedReferenceGraphAuthority,
)
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.inline_contained_publication import (
    InlineContainedPublicationReducer, InlinePublicationEnvelope,
    InlinePublicationID, InlinePublicationState, InlinePublicationStateError,
    UnknownInlinePublicationError,
)
from miniray.inline_contained_runtime import (
    InlineChildHoldInstallReply, InlineContainedPublicationCoordinator,
    InlineContainedRuntimeCleanupError,
    InlineContainedRuntimeProtocolError, InlineOwnerCommitCoordinator,
    InlineOwnerCommitReply,
)


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self, edge_count: int = 3) -> None:
        self.job = JobID.random()
        driver = TaskID.for_driver(self.job)
        self.task = TaskID.derive(self.job, driver, 1)
        self.attempt = AttemptID(self.task, 0)
        self.outer = ObjectID.for_task(self.task)
        self.executor = WorkerID.random()
        self.owner = WorkerID.random()
        self.node = NodeID.random()
        self.publication_id = InlinePublicationID(
            LeaseID.random(), self.task, self.attempt, self.outer
        )
        self.edges = tuple(
            ContainedReferenceEdge(
                self.outer,
                ObjectID.for_task(TaskID.derive(
                    self.job, driver, index + 10
                )),
                self.executor, ("127.0.0.1", 38000 + index),
                "inline-runtime:{}".format(index),
            )
            for index in range(edge_count)
        )
        self.envelope = InlinePublicationEnvelope.create(
            self.publication_id, self.job, self.executor, self.owner,
            self.node, b"inline result with contained refs", self.edges,
        )


class _Harness:
    def __init__(self, fixture: _Fixture) -> None:
        self.fixture = fixture
        self.authority = ContainedReferenceGraphAuthority()
        self.events: list[tuple[str, object]] = []
        self.active: set[object] = set()
        self.tombstones: set[object] = set()
        self.install_attempts: list[object] = []
        self.commit_calls = 0
        self.fail_install: int | None = None
        self.raise_commit_once = False

    def graph(self, request: object) -> protocol.ContainedGraphReply:
        self.events.append((type(request).__name__, request))
        if isinstance(request, protocol.PrepareContainedGraph):
            receipt = self.authority.prepare_manifest(request.manifest)
        elif isinstance(request, protocol.CommitContainedGraph):
            self.commit_calls += 1
            receipt = self.authority.commit_manifest(request.manifest)
            if self.raise_commit_once:
                self.raise_commit_once = False
                raise RuntimeError("commit ACK lost")
        elif isinstance(request, protocol.AbortContainedGraph):
            receipt = self.authority.abort_manifest(request.manifest)
        else:
            raise AssertionError(request)
        return protocol.ContainedGraphReply(request, receipt=receipt)

    def install(self, effect: object) -> InlineChildHoldInstallReply:
        self.events.append(("install", effect))
        self.install_attempts.append(effect)
        if effect.index == self.fail_install:
            # Model mutate-then-fail.  The adapter must tombstone this slot too.
            if effect not in self.tombstones:
                self.active.add(effect)
            raise RuntimeError("install ACK lost")
        if effect in self.tombstones:
            raise AssertionError("late install crossed its release tombstone")
        changed = effect not in self.active
        self.active.add(effect)
        return InlineChildHoldInstallReply(effect, True, changed)

    def release(
        self, request: protocol.ReleaseContainedReference,
    ) -> protocol.ReleaseContainedReferenceReply:
        index = next(
            index for index, edge in enumerate(self.fixture.edges)
            if (
                edge.contained_object_id == request.object_id
                and edge.contained_owner_worker_id == request.owner_worker_id
                and edge.transfer_token == request.hold.transfer_token
            )
        )
        effect = self.fixture.envelope.edge_effect(index)
        self.events.append(("release", effect))
        released = effect in self.active
        self.active.discard(effect)
        self.tombstones.add(effect)
        return protocol.ReleaseContainedReferenceReply(
            request.object_id, request.owner_worker_id, request.hold,
            True, released,
        )

    def coordinator(self) -> InlineContainedPublicationCoordinator:
        return InlineContainedPublicationCoordinator(
            InlineContainedPublicationReducer(),
            prepare_graph=self.graph, install_child=self.install,
            release_child=self.release, commit_graph=self.graph,
            abort_graph=self.graph,
        )


def _state(
    harness: _Harness, transaction_id: str,
) -> ContainedGraphTransactionState:
    return dict(harness.authority.snapshot().transactions)[transaction_id]


def test_prepare_precedes_pins_and_exact_replay_performs_no_effect() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    coordinator = harness.coordinator()

    prepared = coordinator.prepare(fixture.envelope)
    assert prepared.pin_bitmap == (True, True, True)
    assert [name for name, _ in harness.events] == [
        "PrepareContainedGraph", "install", "install", "install"
    ]
    assert [effect.index for effect in harness.install_attempts] == [0, 1, 2]

    assert coordinator.prepare(fixture.envelope) == prepared
    assert [name for name, _ in harness.events] == [
        "PrepareContainedGraph", "install", "install", "install"
    ]


def test_partial_pin_failure_tombstones_every_slot_reverse_then_aborts() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    harness.fail_install = 1
    coordinator = harness.coordinator()

    with pytest.raises(RuntimeError, match="install ACK lost"):
        coordinator.prepare(fixture.envelope)

    releases = [
        effect.index for name, effect in harness.events if name == "release"
    ]
    assert releases == [2, 1, 0]
    assert harness.active == set()
    assert harness.tombstones == {
        fixture.envelope.edge_effect(index) for index in range(3)
    }
    snapshot = coordinator.reducer.snapshot(fixture.publication_id)
    assert snapshot.state is InlinePublicationState.ABORTED
    assert snapshot.pin_bitmap == (True, False, False)
    assert snapshot.compensation_bitmap == (True, True, True)
    assert _state(
        harness, fixture.publication_id.graph_transaction_id
    ) is ContainedGraphTransactionState.ABORTED

    # Terminal replay neither releases nor aborts a second time.
    before = tuple(harness.events)
    report = coordinator.abort(fixture.envelope)
    assert report.reducer_tracked and report.graph_aborted
    assert tuple(harness.events) == before


def test_ambiguous_prepare_is_fenced_by_full_tombstones_and_exact_abort() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    coordinator = harness.coordinator()

    def prepare_then_lose(
        request: protocol.PrepareContainedGraph,
    ) -> protocol.ContainedGraphReply:
        harness.graph(request)
        raise RuntimeError("prepare ACK lost")

    coordinator._prepare_graph = prepare_then_lose
    with pytest.raises(RuntimeError, match="prepare ACK lost"):
        coordinator.prepare(fixture.envelope)

    with pytest.raises(UnknownInlinePublicationError):
        coordinator.reducer.snapshot(fixture.publication_id)
    assert [
        effect.index for name, effect in harness.events if name == "release"
    ] == [2, 1, 0]
    assert _state(
        harness, fixture.publication_id.graph_transaction_id
    ) is ContainedGraphTransactionState.ABORTED
    before = tuple(harness.events)
    report = coordinator.abort(fixture.envelope)
    assert not report.reducer_tracked
    assert tuple(harness.events) == before


def test_rollback_resumes_only_missing_release_after_cleanup_failure() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    harness.fail_install = 0
    real_release = harness.release
    failed_once = False

    def flaky_release(
        request: protocol.ReleaseContainedReference,
    ) -> protocol.ReleaseContainedReferenceReply:
        nonlocal failed_once
        index = next(
            index for index, edge in enumerate(fixture.edges)
            if edge.contained_object_id == request.object_id
        )
        if index == 1 and not failed_once:
            failed_once = True
            raise RuntimeError("release unavailable")
        return real_release(request)

    coordinator = InlineContainedPublicationCoordinator(
        InlineContainedPublicationReducer(),
        prepare_graph=harness.graph, install_child=harness.install,
        release_child=flaky_release, commit_graph=harness.graph,
        abort_graph=harness.graph,
    )
    with pytest.raises(InlineContainedRuntimeCleanupError) as raised:
        coordinator.prepare(fixture.envelope)
    assert isinstance(raised.value.primary_error, RuntimeError)
    assert isinstance(raised.value.cleanup_error, RuntimeError)
    assert coordinator.reducer.snapshot(
        fixture.publication_id
    ).compensation_bitmap == (False, False, True)

    report = coordinator.abort(fixture.envelope)
    assert report.compensation_bitmap == (True, True, True)
    # Slot 2 acknowledged before interruption and was not replayed.
    assert [
        effect.index for name, effect in harness.events if name == "release"
    ] == [2, 1, 0]


def test_ambiguous_commit_is_roll_forward_only_and_replays_exact_commit() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    coordinator = harness.coordinator()
    prepared = coordinator.prepare(fixture.envelope)
    harness.raise_commit_once = True

    with pytest.raises(RuntimeError, match="commit ACK lost"):
        coordinator.commit(prepared)
    assert coordinator.reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.COMMITTING
    assert not [event for event in harness.events if event[0] == "release"]
    with pytest.raises(InlinePublicationStateError, match="cannot abort"):
        coordinator.abort(fixture.envelope)

    committed = coordinator.commit(prepared)
    assert committed.state is InlinePublicationState.COMMITTED
    assert harness.commit_calls == 2
    commit_requests = [
        request for name, request in harness.events
        if name == "CommitContainedGraph"
    ]
    assert len(commit_requests) == 2
    assert commit_requests[0] == commit_requests[1]
    assert coordinator.commit(prepared) == committed
    assert harness.commit_calls == 2


def test_malformed_install_ack_is_not_recorded_and_all_slots_rollback() -> None:
    fixture = _Fixture(2)
    harness = _Harness(fixture)
    coordinator = harness.coordinator()

    def wrong_install(effect: object) -> InlineChildHoldInstallReply:
        harness.events.append(("install", effect))
        return InlineChildHoldInstallReply(
            fixture.envelope.edge_effect(1 - effect.index), True, True
        )

    coordinator._install_child = wrong_install
    with pytest.raises(
        InlineContainedRuntimeProtocolError, match="changed effect identity"
    ):
        coordinator.prepare(fixture.envelope)
    snapshot = coordinator.reducer.snapshot(fixture.publication_id)
    assert snapshot.pin_bitmap == (False, False)
    assert snapshot.compensation_bitmap == (True, True)


def test_owner_commit_happens_after_graph_and_is_cached_after_exact_ack() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    publication = harness.coordinator()
    prepared = publication.prepare(fixture.envelope)
    owner_calls = []

    def commit_owner(request: object) -> InlineOwnerCommitReply:
        owner_calls.append(request)
        assert publication.reducer.snapshot(
            fixture.publication_id
        ).state is InlinePublicationState.COMMITTED
        return InlineOwnerCommitReply(request, True, len(owner_calls) == 1)

    owner = InlineOwnerCommitCoordinator(
        publication, commit_owner=commit_owner
    )
    receipt = owner.commit(prepared)
    assert receipt.request.envelope == fixture.envelope
    assert [name for name, _ in harness.events][-1] == "CommitContainedGraph"
    assert len(owner_calls) == 1

    # A known exact owner ACK is a local terminal proof: no repeated effects.
    assert owner.commit(prepared) == receipt
    assert len(owner_calls) == 1
    assert harness.commit_calls == 1


def test_owner_ack_loss_replays_owner_only_never_graph_or_abort() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    publication = harness.coordinator()
    prepared = publication.prepare(fixture.envelope)
    owner_calls = []

    def commit_owner(request: object) -> InlineOwnerCommitReply:
        owner_calls.append(request)
        if len(owner_calls) == 1:
            raise RuntimeError("owner ACK lost")
        return InlineOwnerCommitReply(request, True, False)

    owner = InlineOwnerCommitCoordinator(
        publication, commit_owner=commit_owner
    )
    with pytest.raises(RuntimeError, match="owner ACK lost"):
        owner.commit(prepared)
    assert publication.reducer.snapshot(
        fixture.publication_id
    ).state is InlinePublicationState.COMMITTED
    assert harness.commit_calls == 1

    receipt = owner.commit(prepared)
    assert not receipt.changed
    assert owner_calls[0] == owner_calls[1]
    assert harness.commit_calls == 1
    assert not [event for event in harness.events if event[0] == "release"]


def test_graph_reply_identity_drift_fails_before_reducer_transition() -> None:
    fixture = _Fixture(1)
    harness = _Harness(fixture)
    coordinator = harness.coordinator()
    other = _Fixture(1)

    def wrong_prepare(
        request: protocol.PrepareContainedGraph,
    ) -> protocol.ContainedGraphReply:
        other_request = protocol.PrepareContainedGraph(
            other.envelope.to_graph_manifest()
        )
        receipt = ContainedGraphManifestReceipt(
            other_request.manifest, ContainedGraphTransactionState.PREPARED,
            ContainedGraphManifestDisposition.APPLIED,
        )
        return protocol.ContainedGraphReply(other_request, receipt=receipt)

    coordinator._prepare_graph = wrong_prepare
    with pytest.raises(
        InlineContainedRuntimeProtocolError, match="changed request identity"
    ):
        coordinator.prepare(fixture.envelope)
    with pytest.raises(UnknownInlinePublicationError):
        coordinator.reducer.snapshot(fixture.publication_id)
    assert harness.tombstones == {fixture.envelope.edge_effect(0)}


def test_one_coordinator_cannot_be_rebound_to_another_envelope() -> None:
    fixture = _Fixture(1)
    harness = _Harness(fixture)
    coordinator = harness.coordinator()
    coordinator.prepare(fixture.envelope)
    before = tuple(harness.events)

    with pytest.raises(InlineContainedRuntimeProtocolError, match="rebound"):
        coordinator.prepare(_Fixture(1).envelope)
    assert tuple(harness.events) == before
