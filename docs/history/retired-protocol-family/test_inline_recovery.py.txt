# Pure reducers are marked individually; real threads/sockets are bounded
# loopback smoke tests and must not inherit the default unit marker.


def _race_two(first, second):
    """Start exactly two real racers with bounded barriers and teardown.

    The runner owns the process-level deadline; daemon racers cannot keep a
    failed pytest child alive after the bounded join reports a stuck lock.
    """

    barrier = Barrier(3, timeout=1.0)
    results = [None, None]
    failures = []

    def run(index, operation):
        try:
            barrier.wait(timeout=1.0)
            results[index] = operation()
        except BaseException as exc:
            failures.append(exc)

    threads = tuple(
        Thread(target=run, args=(index, operation), daemon=True)
        for index, operation in enumerate((first, second))
    )
    started = []
    try:
        deadline = time.monotonic() + 2.0
        for thread in threads:
            thread.start()
            started.append(thread)
        barrier.wait(timeout=max(0.0, deadline - time.monotonic()))
        for thread in started:
            thread.join(max(0.0, deadline - time.monotonic()))
        assert not any(thread.is_alive() for thread in started)
        if failures:
            raise failures[0]
        return tuple(results)
    finally:
        barrier.abort()
        cleanup_deadline = time.monotonic() + 1.0
        for thread in threads:
            if thread.ident is not None:
                thread.join(max(0.0, cleanup_deadline - time.monotonic()))
        assert not any(thread.is_alive() for thread in threads)


class _Fixture:
    def __init__(
        self, *, node: NodeID | None = None, node_pid: int = 4101,
        registration_epoch: int = 7, seed: int = 1,
        owner: WorkerID | None = None,
    ) -> None:
        self.job = JobID.random()
        driver = TaskID.for_driver(self.job)
        self.task = TaskID.derive(self.job, driver, seed)
        self.attempt = AttemptID(self.task, 0)
        self.outer = ObjectID.for_task(self.task)
        self.executor = WorkerID.random()
        self.owner = owner or WorkerID.random()
        self.node = node or NodeID.random()
        self.publication_id = InlinePublicationID(
            LeaseID.random(), self.task, self.attempt, self.outer
        )
        child_task = TaskID.derive(self.job, driver, seed + 100)
        self.edge = ContainedReferenceEdge(
            self.outer, ObjectID.for_task(child_task), self.executor,
            ("127.0.0.1", 39000 + seed),
            "inline-recovery:{}".format(seed),
        )
        self.envelope = InlinePublicationEnvelope.create(
            self.publication_id, self.job, self.executor, self.owner,
            self.node, b"small inline envelope", (self.edge,),
        )
        self.intent = InlinePublicationIntent(
            self.envelope.manifest, node_pid, registration_epoch
        )
        self.terminal = InlinePublicationTerminal(
            self.publication_id, TaskReplyStatus.SUCCEEDED
        )
        self.death = NodeDeathRecord(
            "inline-death:{}".format(seed), self.node, node_pid,
            registration_epoch, seed + 20, -9, NodeDeathReason.PROCESS_EXIT,
            "publishing Node exited",
        )


def _owner_death(
    owner_worker_id: WorkerID, *,
    reason: WorkerDeathReason = WorkerDeathReason.PROCESS_EXIT,
    detection_id: str = "inline-owner-death",
) -> WorkerDeathRecord:
    return WorkerDeathRecord(
        detection_id,
        WorkerIncarnation(
            NodeID.random(), 7654, 3, owner_worker_id, 8765
        ),
        17, -9, reason,
    )


def _prepared(
    authority: InlinePublicationRecoveryAuthority, fixture: _Fixture,
) -> None:
    authority.report_intent(fixture.intent)
    authority.report_prepared(fixture.publication_id)


def _completed(
    authority: InlinePublicationRecoveryAuthority, fixture: _Fixture,
) -> None:
    _prepared(authority, fixture)
    authority.report_terminal(fixture.terminal)


def _owner_resolution(
    fixture: _Fixture, death: WorkerDeathRecord, *,
    history: PublicationRecoveryHistory = (
        PublicationRecoveryHistory.NOT_COMMITTED
    ),
    graph_state: PublicationOwnerDeathGraphState = (
        PublicationOwnerDeathGraphState.NOT_FOUND
    ),
) -> PublicationOwnerDeathResolution:
    active_edges = (
        fixture.envelope.ordered_edges
        if graph_state is PublicationOwnerDeathGraphState.COMMITTED
        else ()
    )
    request = InlinePublicationOwnerDeathRequest(
        InlinePublicationOwnerDeathKey(fixture.publication_id),
        fixture.envelope.manifest, death, history,
        PublicationOwnerDeathGraphWitness(
            fixture.envelope.to_graph_manifest(), graph_state, active_edges,
        ),
    )
    return PublicationOwnerDeathResolution(
        request, PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED, (),
    )

@pytest.mark.unit
def test_aborted_ack_loss_then_node_death_is_exact_terminal_tombstone() -> None:
    fixture = _Fixture()
    authority = InlinePublicationRecoveryAuthority()
    authority.report_intent(fixture.intent)
    digest = fixture.envelope.manifest.manifest_digest
    first = authority.acknowledge_aborted(
        fixture.publication_id, digest
    )

    workset = authority.freeze_node_death(fixture.death)

    assert len(workset) == 1
    work = workset[0]
    assert work.action is InlineRecoveryAction.PRECOMPLETE_ROLLBACK
    assert work.snapshot.resolution is InlineRecoveryResolution.ABORTED
    current = authority.snapshot(fixture.publication_id)
    assert current.frozen_death == fixture.death
    assert current.resolution is InlineRecoveryResolution.ABORTED
    assert authority.snapshot_for_node_death(
        fixture.publication_id, fixture.death
    ) == current
    assert authority.snapshot_for_node_death(
        _Fixture(node=fixture.node, seed=99).publication_id, fixture.death
    ) is None
    # The Node's lost normal ACK can be retried after the death freeze without
    # manufacturing another rollback or changing the tombstone identity.
    replay = authority.acknowledge_aborted(
        fixture.publication_id, digest.upper()
    )
    assert first.stage is replay.stage is InlineRecoveryStage.ABORTED
    assert replay.disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert replay.snapshot == current
    assert authority.freeze_node_death(fixture.death) == workset
    assert not authority.has_active_operations()


@pytest.mark.unit
def test_node_death_freezes_exact_incarnation_and_pre_post_actions() -> None:
    node = NodeID.random()
    pre = _Fixture(node=node, seed=1)
    post = _Fixture(node=node, seed=2)
    authority = InlinePublicationRecoveryAuthority()
    authority.report_intent(pre.intent)
    _completed(authority, post)
    death = replace(pre.death, detection_id="shared-death")

    workset = authority.freeze_node_death(death)
    replay = authority.freeze_node_death(death)

    assert replay == workset
    assert tuple(work.publication_id for work in workset) == tuple(sorted(
        (pre.publication_id, post.publication_id)
    ))
    by_id = {work.publication_id: work for work in workset}
    assert by_id[pre.publication_id].action is (
        InlineRecoveryAction.PRECOMPLETE_ROLLBACK
    )
    assert by_id[post.publication_id].action is (
        InlineRecoveryAction.POSTCOMPLETE_RESOLVE
    )
    assert by_id[post.publication_id].manifest == post.envelope.manifest
    assert authority.frozen_workset(death) == workset
    assert authority.work_for_node_death(pre.publication_id, death) == (
        by_id[pre.publication_id]
    )
    assert authority.snapshot(pre.publication_id).state is (
        InlineRecoveryState.FROZEN
    )

    active = _Fixture(owner=owner, seed=11)

        prepared.publication_id
    ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert authority.report_terminal(
        terminal.terminal
    ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert authority.acknowledge_adopted(
        adopted.publication_id,
        adopted.envelope.manifest.manifest_digest, owner,
    ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert authority.acknowledge_retired(
        retired.publication_id,
        retired.envelope.manifest.manifest_digest, owner,
    ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert authority.acknowledge_aborted(
        aborted.publication_id,
        aborted.envelope.manifest.manifest_digest,
    ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED

    with pytest.raises(InlineRecoveryStateError, match="owner Worker"):
        authority.report_prepared(active.publication_id)
    with pytest.raises(InlineRecoveryStateError, match="owner Worker"):
        authority.report_terminal(prepared.terminal)
    with pytest.raises(InlineRecoveryStateError, match="owner Worker"):
        authority.acknowledge_adopted(
            terminal.publication_id,
            terminal.envelope.manifest.manifest_digest, owner,
        )

    late = _Fixture(owner=owner, seed=18)
    with pytest.raises(InlineRecoveryStateError, match="owner.*death-frozen"):
        authority.report_intent(late.intent)
    assert authority.records_for_outer_owner(owner) is frozen
    conflicting = replace(death, detection_id="changed-owner-proof")
    with pytest.raises(InlineRecoveryConflictError, match="death proof"):
        authority.freeze_owner_death(conflicting)
    with pytest.raises(InlineRecoveryConflictError, match="death proof"):
        authority.frozen_records_for_owner_death(conflicting)

@pytest.mark.loopback_smoke
def test_owner_death_and_intent_admission_linearize_atomically() -> None:
    fixture = _Fixture(seed=23)
    authority = InlinePublicationRecoveryAuthority()
    death = _owner_death(fixture.owner)
    def report() -> object:
        try:
            return authority.report_intent(fixture.intent)
        except InlineRecoveryStateError:
            return None

    def freeze() -> tuple[object, ...]:
        return authority.freeze_owner_death(death)

    acknowledgement, frozen = _race_two(report, freeze)

    if acknowledgement is None:
        assert frozen == ()
        with pytest.raises(UnknownInlineRecoveryError):
            authority.snapshot(fixture.publication_id)
    else:
        assert frozen == (acknowledgement.snapshot,)
        assert authority.report_intent(
            fixture.intent
        ).disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert authority.records_for_outer_owner(fixture.owner) is frozen
    assert authority.frozen_records_for_owner_death(death) is frozen

@pytest.mark.parametrize("reported_terminal", (False, True))
@pytest.mark.parametrize("decision", tuple(InlineRecoveryOwnerDecision))
@pytest.mark.unit
def test_frozen_owner_choice_is_monotonic_and_preserves_original_evidence(
    reported_terminal: bool, decision: InlineRecoveryOwnerDecision,
) -> None:
    fixture = _Fixture()
    authority = InlinePublicationRecoveryAuthority()
    _prepared(authority, fixture)
    if reported_terminal:
        authority.report_terminal(fixture.terminal)
    work = authority.freeze_node_death(fixture.death)[0]

    first = authority.decide_owner_resolution(work, fixture.owner, decision)
    replay = authority.decide_owner_resolution(work, fixture.owner, decision)

    assert first.stage is InlineRecoveryStage.DECIDED
    assert first.snapshot.owner_decision is decision
    assert replay.disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert work.snapshot.owner_decision is None
    assert authority.frozen_workset(fixture.death) == (work,)
    assert first.snapshot.terminal == work.snapshot.terminal
    opposite = (
        InlineRecoveryOwnerDecision.DROP
        if decision is InlineRecoveryOwnerDecision.KEEP
        else InlineRecoveryOwnerDecision.KEEP
    )
    with pytest.raises(InlineRecoveryConflictError, match="decision changed"):
        authority.decide_owner_resolution(work, fixture.owner, opposite)
    with pytest.raises(InlineRecoveryConflictError, match="another.*owner"):
        authority.decide_owner_resolution(work, WorkerID.random(), decision)

    if decision is InlineRecoveryOwnerDecision.KEEP:
        with pytest.raises(InlineRecoveryStateError, match="owner decision"):
            authority.acknowledge_lost(work)
        ack = authority.acknowledge_adoption(work)
        assert ack.snapshot.resolution is InlineRecoveryResolution.ADOPTED
    else:
        with pytest.raises(InlineRecoveryStateError, match="owner decision"):
            authority.acknowledge_adoption(work)
        ack = authority.acknowledge_lost(work)
        assert ack.snapshot.resolution is InlineRecoveryResolution.LOST
    assert ack.snapshot.terminal == work.snapshot.terminal
    assert authority.frozen_workset(fixture.death) == (work,)
    assert not authority.has_active_operations()
