pytestmark = pytest.mark.unit


_RESULT_PAYLOAD = b"result bytes must remain outside GCS recovery metadata"
_OPAQUE_IDS = (JobID, TaskID, LeaseID, NodeID, WorkerID)


def _assert_metadata_only(value: object, path: str = "message") -> None:
    """Only opaque ID leaves may contain bytes, never result descriptors."""

    assert not isinstance(
        value, (InlinePublicationEnvelope, protocol.ResultDescriptor)
    ), path
    if isinstance(value, _OPAQUE_IDS):
        assert tuple(field.name for field in fields(value)) == ("value",)
        assert isinstance(value.value, bytes) and len(value.value) == 16
    elif is_dataclass(value):
        for field in fields(value):
            assert field.name not in {
                "envelope", "descriptor", "data", "inline_data", "payload",
            }, f"{path}.{field.name}"
            _assert_metadata_only(
                getattr(value, field.name), f"{path}.{field.name}"
            )
    elif isinstance(value, (tuple, list, set, frozenset)):
        for index, item in enumerate(value):
            _assert_metadata_only(item, f"{path}[{index}]")
    elif isinstance(value, dict):
        for key, item in value.items():
            _assert_metadata_only(key, f"{path}.key")
            _assert_metadata_only(item, f"{path}[{key!r}]")
    else:
        assert value is None or isinstance(value, (str, int, bool)), path


def _assert_metadata_round_trip(value: object) -> None:
    _assert_metadata_only(value)
    encoded = pickle.dumps(value)
    assert _RESULT_PAYLOAD not in encoded
    restored = pickle.loads(encoded)
    assert restored == value
    _assert_metadata_only(restored)


class _Fixture:
    def __init__(self, *, source: str = "owned") -> None:
        self.job = JobID.random()
        self.task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), 51
        )
        self.attempt = AttemptID(self.task, 1)
        self.lease = LeaseID.random()
        self.outer = ObjectID.for_task(self.task)
        self.executor = WorkerID.random()
        self.owner = WorkerID.random()
        self.node = NodeID.random()
        self.publication_id = InlinePublicationID(
            self.lease, self.task, self.attempt, self.outer
        )
        child = ObjectID.for_task(TaskID.derive(
            self.job, TaskID.for_driver(self.job), 52
        ))
        child_owner = self.executor if source == "owned" else WorkerID.random()
        edge = ContainedReferenceEdge(
            self.outer, child, child_owner, ("127.0.0.1", 39501),
            "inline-recovery:child",
        )
        transfers = None
        if source != "owned":
            if source == "task":
                original_source = protocol.TaskHoldSource(
                    protocol.TaskReferenceHold(
                        protocol.TaskReferenceHoldKind.RETAINED, self.executor,
                        self.task, self.attempt,
                    )
                )
            else:
                assert source == "contained"
                original_outer = ObjectID.for_task(TaskID.derive(
                    self.job, TaskID.for_driver(self.job), 53
                ))
                original_source = protocol.ContainedTransferSource(
                    ContainedReferenceHold(
                        original_outer, self.executor, "original-inline-source"
                    )
                )
            transfers = (PreparedContainedTransfer(
                child, child_owner, edge.contained_owner_address,
                BorrowedContainedSource(
                    self.executor, "borrowed-inline-recovery", original_source
                ),
                edge.incoming_hold(self.executor), edge.incoming_hold(self.owner),
            ),)
        self.envelope = InlinePublicationEnvelope.create(
            self.publication_id, self.job, self.executor, self.owner,
            self.node, _RESULT_PAYLOAD, (edge,), transfers,
        )
        self.manifest = self.envelope.manifest
        self.intent = InlinePublicationIntent(self.manifest, 43210, 7)
        self.terminal = InlinePublicationTerminal(
            self.publication_id, protocol.TaskReplyStatus.SUCCEEDED
        )
        self.death = protocol.NodeDeathRecord(
            "inline-recovery-death", self.node, 43210, 7, 9, -9,
            protocol.NodeDeathReason.PROCESS_EXIT, "test process exit",
        )

    def authority(self, *, prepared: bool = True, terminal: bool = False):
        assert prepared or not terminal
        authority = InlinePublicationRecoveryAuthority()
        intent_ack = authority.report_intent(self.intent)
        prepared_ack = (
            authority.report_prepared(self.publication_id) if prepared else None
        )
        terminal_ack = (
            authority.report_terminal(self.terminal) if terminal else None
        )
        return authority, intent_ack, prepared_ack, terminal_ack


def test_live_node_reports_round_trip_with_exact_acknowledgements() -> None:
    f = _Fixture()
    authority, intent_ack, prepared_ack, terminal_ack = f.authority(
        terminal=True
    )
    requests = (
        protocol.ReportInlinePublicationIntent(f.intent),
        protocol.ReportInlinePublicationPrepared(f.publication_id),
        protocol.ReportInlinePublicationTerminal(f.terminal),
    )
    replies = (
        protocol.InlinePublicationRecoveryReply(requests[0], intent_ack),
        protocol.InlinePublicationRecoveryReply(requests[1], prepared_ack),
        protocol.InlinePublicationRecoveryReply(requests[2], terminal_ack),
    )

    _assert_metadata_round_trip((requests, replies))
    assert replies[0].ack.snapshot.manifest == f.manifest
    assert authority.snapshot(f.publication_id).terminal == f.terminal
    assert not hasattr(f.intent, "envelope")
    assert not hasattr(authority.snapshot(f.publication_id), "envelope")


@pytest.mark.parametrize("source", ("owned", "task", "contained"))
@pytest.mark.parametrize("phase", ("intent", "prepared", "terminal"))
def test_intent_snapshot_and_frozen_work_retain_metadata_not_result_bytes(
    source: str, phase: str,
) -> None:
    f = _Fixture(source=source)
    authority, *_ = f.authority(
        prepared=phase != "intent", terminal=phase == "terminal"
    )
    work = authority.freeze_node_death(f.death)[0]

    assert f.envelope.descriptor.inline_data == _RESULT_PAYLOAD
    assert work.manifest == f.manifest
    assert not hasattr(work, "envelope")
    _assert_metadata_round_trip((
        protocol.ReportInlinePublicationIntent(f.intent),
        f.intent, authority.snapshot(f.publication_id), work,
    ))
    with pytest.raises(TypeError, match="manifest"):
        InlinePublicationIntent(f.envelope, 43210, 7)


def test_normal_abort_adoption_and_retirement_acks_are_exact() -> None:
    aborted = _Fixture()
    abort_authority, *_ = aborted.authority()
    abort_request = protocol.AcknowledgeInlinePublicationAborted(
        aborted.publication_id, aborted.envelope.manifest.manifest_digest
    )
    abort_reply = protocol.InlinePublicationRecoveryReply(
        abort_request, abort_authority.acknowledge_aborted(
            aborted.publication_id, aborted.envelope.manifest.manifest_digest
        ),
    )
    assert abort_reply.ack.snapshot.resolution is InlineRecoveryResolution.ABORTED

    adopted = _Fixture()
    adopt_authority, *_ = adopted.authority(terminal=True)
    adopt_request = protocol.AcknowledgeInlinePublicationAdopted(
        adopted.publication_id, adopted.envelope.manifest.manifest_digest,
        adopted.owner,
    )
    adopt_reply = protocol.InlinePublicationRecoveryReply(
        adopt_request, adopt_authority.acknowledge_adopted(
            adopted.publication_id,
            adopted.envelope.manifest.manifest_digest, adopted.owner,
        ),
    )
    assert adopt_reply.ack.snapshot.resolution is InlineRecoveryResolution.ADOPTED
    retired = _Fixture()
    retire_authority, *_ = retired.authority(terminal=True)
    retire_request = protocol.AcknowledgeInlinePublicationRetired(
        retired.publication_id, retired.envelope.manifest.manifest_digest,
        retired.owner,
    )
    retire_reply = protocol.InlinePublicationRecoveryReply(
        retire_request, retire_authority.acknowledge_retired(
            retired.publication_id,
            retired.envelope.manifest.manifest_digest, retired.owner,
        ),
    )
    assert retire_reply.ack.snapshot.resolution is InlineRecoveryResolution.RETIRED
    _assert_metadata_round_trip((abort_reply, adopt_reply, retire_reply))


def test_normal_abort_ack_replay_remains_valid_after_exact_node_death() -> None:
    fixture = _Fixture()
    authority, *_ = fixture.authority()
    digest = fixture.envelope.manifest.manifest_digest
    authority.acknowledge_aborted(fixture.publication_id, digest)
    work = authority.freeze_node_death(fixture.death)[0]
    request = protocol.AcknowledgeInlinePublicationAborted(
        fixture.publication_id, digest
    )

    reply = protocol.InlinePublicationRecoveryReply(
        request, authority.acknowledge_aborted(
            fixture.publication_id, digest
        ),
    )

    assert work.snapshot.resolution is InlineRecoveryResolution.ABORTED
    assert reply.ack.snapshot.frozen_death == fixture.death
    _assert_metadata_round_trip(reply)
    pull = protocol.GetInlineNodeLoss(
        fixture.publication_id, fixture.owner, fixture.death
    )
    resolved = protocol.GetInlineNodeLossReply(
        pull, protocol.InlineNodeLossDisposition.RESOLVED,
        snapshot=reply.ack.snapshot,
    )
    _assert_metadata_round_trip(resolved)


@pytest.mark.parametrize("terminal", (False, True))
def test_node_loss_exposes_only_owner_bound_unknown_or_postcomplete_work(
    terminal: bool,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=terminal)
    work = authority.freeze_node_death(f.death)[0]
    assert work.action is (
        InlineRecoveryAction.POSTCOMPLETE_RESOLVE if terminal
        else InlineRecoveryAction.COMPLETION_UNKNOWN
    )
    request = protocol.GetInlineNodeLoss(
        f.publication_id, f.owner, f.death
    )
    reply = protocol.GetInlineNodeLossReply(
        request, protocol.InlineNodeLossDisposition.WORK, work=work
    )
    _assert_metadata_round_trip(reply)

    with pytest.raises(ProtocolError, match="owner identity"):
        protocol.GetInlineNodeLossReply(
            protocol.GetInlineNodeLoss(
                f.publication_id, WorkerID.random(), f.death
            ),
            protocol.InlineNodeLossDisposition.WORK, work=work,
        )
    with pytest.raises(ProtocolError, match="action"):
        before = _Fixture()
        pre_authority, *_ = before.authority(prepared=False)
        rollback = pre_authority.freeze_node_death(before.death)[0]
        protocol.GetInlineNodeLossReply(
            protocol.GetInlineNodeLoss(
                before.publication_id, before.owner, before.death
            ),
            protocol.InlineNodeLossDisposition.WORK, work=rollback,
        )
    for forged in (
        replace(request, publication_id=_Fixture().publication_id),
        replace(request, node_death=replace(f.death, death_epoch=10)),
    ):
        with pytest.raises(ProtocolError, match="publication, death"):
            protocol.GetInlineNodeLossReply(
                forged, protocol.InlineNodeLossDisposition.WORK, work=work
            )


def test_node_loss_ack_and_internal_progress_require_matching_resolution() -> None:
    post = _Fixture()
    authority, *_ = post.authority(terminal=True)
    work = authority.freeze_node_death(post.death)[0]
    with pytest.raises(ProtocolError, match="contradicts"):
        protocol.AcknowledgeInlineNodeLoss(
            work, post.owner, InlineRecoveryResolution.ROLLED_BACK
        )
    authority.decide_owner_resolution(
        work, post.owner, InlineRecoveryOwnerDecision.KEEP
    )
    request = protocol.AcknowledgeInlineNodeLoss(
        work, post.owner, InlineRecoveryResolution.ADOPTED
    )
    reply = protocol.AcknowledgeInlineNodeLossReply(
        request, authority.acknowledge_adoption(work)
    )
    _assert_metadata_round_trip(reply)

    retired = _Fixture()
    retire_authority, *_ = retired.authority(terminal=True)
    retire_work = retire_authority.freeze_node_death(retired.death)[0]
    retire_authority.decide_owner_resolution(
        retire_work, retired.owner, InlineRecoveryOwnerDecision.KEEP
    )
    retire_request = protocol.AcknowledgeInlineNodeLoss(
        retire_work, retired.owner, InlineRecoveryResolution.RETIRED
    )
    retire_reply = protocol.AcknowledgeInlineNodeLossReply(
        retire_request, retire_authority.acknowledge_retirement(retire_work)
    )
    resolved = protocol.GetInlineNodeLossReply(
        protocol.GetInlineNodeLoss(
            retired.publication_id, retired.owner, retired.death
        ),
        protocol.InlineNodeLossDisposition.RESOLVED,
        snapshot=retire_reply.ack.snapshot,
    )
    _assert_metadata_round_trip((retire_reply, resolved))

    pre = _Fixture()
    pre_authority, *_ = pre.authority(prepared=False)
    rollback = pre_authority.freeze_node_death(pre.death)[0]
    progress = protocol.ProgressInlineNodeLoss(
        pre.publication_id, pre.death
    )
    progressed = protocol.ProgressInlineNodeLossReply(
        progress, pre_authority.acknowledge_rollback(rollback)
    )
    assert progressed.ack.snapshot.resolution is (
        InlineRecoveryResolution.ROLLED_BACK
    )
    _assert_metadata_round_trip(progressed)


@pytest.mark.parametrize("terminal", (False, True))
@pytest.mark.parametrize("decision", tuple(InlineRecoveryOwnerDecision))
def test_owner_decision_acks_preserve_exact_frozen_metadata(
    terminal: bool, decision: InlineRecoveryOwnerDecision,
) -> None:
    f = _Fixture(source="task")
    authority, *_ = f.authority(terminal=terminal)
    work = authority.freeze_node_death(f.death)[0]
    request = protocol.DecideInlineNodeLoss(work, f.owner, decision)
    ack = authority.decide_owner_resolution(work, f.owner, decision)
    replay = authority.decide_owner_resolution(work, f.owner, decision)

    assert ack.stage is InlineRecoveryStage.DECIDED
    assert ack.disposition is InlineRecoveryDisposition.APPLIED
    assert replay.disposition is InlineRecoveryDisposition.ALREADY_RECORDED
    assert replay.snapshot == ack.snapshot
    assert ack.snapshot.owner_decision is decision
    assert ack.snapshot.resolution is None
    assert ack.snapshot.intent == work.snapshot.intent
    assert ack.snapshot.terminal == work.snapshot.terminal
    assert work.snapshot.owner_decision is None
    assert authority.frozen_workset(f.death) == (work,)
    _assert_metadata_round_trip((
        request, protocol.DecideInlineNodeLossReply(request, ack),
        protocol.DecideInlineNodeLossReply(request, replay),
    ))


def test_owner_decision_request_requires_exact_owner_enum_and_action() -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=True)
    work = authority.freeze_node_death(f.death)[0]
    for wrong_owner in (WorkerID.random(), "not-a-worker", None):
        with pytest.raises(ProtocolError, match="another owner"):
            protocol.DecideInlineNodeLoss(
                work, wrong_owner, InlineRecoveryOwnerDecision.KEEP
            )
    for wrong_decision in ("KEEP", "DROP", None, InlineRecoveryResolution.LOST):
        with pytest.raises(ProtocolError, match="KEEP or DROP"):
            protocol.DecideInlineNodeLoss(work, f.owner, wrong_decision)

    pre = _Fixture()
    pre_authority, *_ = pre.authority(prepared=False)
    rollback = pre_authority.freeze_node_death(pre.death)[0]
    for decision in InlineRecoveryOwnerDecision:
        with pytest.raises(ProtocolError, match="pre-Complete rollback"):
            protocol.DecideInlineNodeLoss(rollback, pre.owner, decision)


@pytest.mark.parametrize("changed", (
    "stage", "owner_decision", "manifest", "frozen_death", "terminal",
))
def test_owner_decision_reply_rejects_changed_choice_or_frozen_evidence(
    changed: str,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority()
    work = authority.freeze_node_death(f.death)[0]
    request = protocol.DecideInlineNodeLoss(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    ack = authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    reply = protocol.DecideInlineNodeLossReply(request, ack)
    if changed == "stage":
        object.__setattr__(ack, "stage", InlineRecoveryStage.RESOLVED)
    elif changed == "owner_decision":
        object.__setattr__(
            ack.snapshot, "owner_decision", InlineRecoveryOwnerDecision.DROP
        )
    elif changed == "manifest":
        # A self-consistent but different digest is not the frozen intent.
        other_envelope = InlinePublicationEnvelope.create(
            f.publication_id, f.job, f.executor, f.owner, f.node,
            b"a different completed result", f.manifest.ordered_edges,
        )
        object.__setattr__(
            ack.snapshot, "intent",
            replace(f.intent, manifest=other_envelope.manifest),
        )
    elif changed == "frozen_death":
        object.__setattr__(
            ack.snapshot, "frozen_death", replace(f.death, death_epoch=10)
        )
    else:
        # KEEP acknowledges local bytes; it may not invent a GCS terminal.
        object.__setattr__(ack.snapshot, "terminal", f.terminal)

    with pytest.raises(ProtocolError, match="frozen work or choice"):
        protocol.DecideInlineNodeLossReply(request, ack)
    with pytest.raises(ProtocolError, match="frozen work or choice"):
        pickle.loads(pickle.dumps(reply))


@pytest.mark.parametrize("changed", ("owner_worker_id", "decision"))
def test_rejected_owner_decision_reply_revalidates_request(
    changed: str,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority()
    work = authority.freeze_node_death(f.death)[0]
    request = protocol.DecideInlineNodeLoss(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    rejected = protocol.DecideInlineNodeLossReply(
        request, error_kind=protocol.StoredPublicationRPCErrorKind.INVALID_STATE,
        error="publication not ready",
    )
    _assert_metadata_round_trip(rejected)
    object.__setattr__(request, changed, "invalid")

    with pytest.raises(ProtocolError):
        protocol.DecideInlineNodeLossReply(
            request,
            error_kind=protocol.StoredPublicationRPCErrorKind.INVALID_REQUEST,
            error="invalid request",
        )
    with pytest.raises(ProtocolError):
        pickle.loads(pickle.dumps(rejected))


@pytest.mark.parametrize("terminal", (False, True))
@pytest.mark.parametrize("resolution, acknowledge", (
    (InlineRecoveryResolution.ADOPTED, "acknowledge_adoption"),
    (InlineRecoveryResolution.RETIRED, "acknowledge_retirement"),
))
def test_owner_ack_requires_keep_then_can_adopt_or_retire(
    terminal: bool, resolution: InlineRecoveryResolution, acknowledge: str,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=terminal)
    work = authority.freeze_node_death(f.death)[0]
    finish = getattr(authority, acknowledge)
    with pytest.raises(InlineRecoveryStateError, match="exact owner decision"):
        finish(work)
    authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    request = protocol.AcknowledgeInlineNodeLoss(work, f.owner, resolution)
    ack = finish(work)
    assert ack.snapshot.owner_decision is InlineRecoveryOwnerDecision.KEEP
    assert ack.snapshot.resolution is resolution
    assert ack.snapshot.terminal == work.snapshot.terminal
    if not terminal:
        # Owner-local custody proves adoption, not an invented GCS report.
        assert ack.snapshot.terminal is None
    _assert_metadata_round_trip((
        protocol.AcknowledgeInlineNodeLossReply(request, ack),
        protocol.AcknowledgeInlineNodeLossReply(request, finish(work)),
    ))


@pytest.mark.parametrize("changed", ("result_metadata", "borrowed_source"))
def test_terminal_ack_rejects_valid_manifest_rebound_under_same_identity(
    changed: str,
) -> None:
    f = _Fixture(source="task")
    authority, *_ = f.authority(terminal=True)
    work = authority.freeze_node_death(f.death)[0]
    authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    ack = authority.acknowledge_adoption(work)
    request = protocol.AcknowledgeInlineNodeLoss(
        work, f.owner, InlineRecoveryResolution.ADOPTED
    )
    reply = protocol.AcknowledgeInlineNodeLossReply(request, ack)
    transfers = f.manifest.transfers
    payload = _RESULT_PAYLOAD
    if changed == "result_metadata":
        payload = b"another result with another valid size and checksum"
    else:
        transfer = transfers[0]
        transfers = (replace(
            transfer, source=replace(
                transfer.source, borrower_token="other-valid-borrower-token"
            ),
        ),)
    rebound = InlinePublicationEnvelope.create(
        f.publication_id, f.job, f.executor, f.owner, f.node, payload,
        f.manifest.ordered_edges, transfers,
    )
    forged_snapshot = replace(
        ack.snapshot, intent=replace(f.intent, manifest=rebound.manifest)
    )
    assert forged_snapshot.publication_id == work.publication_id
    assert forged_snapshot.frozen_death == work.death
    assert forged_snapshot.manifest.manifest_digest != f.manifest.manifest_digest
    # Each manifest is valid by itself; only the exact frozen-work comparison
    # can reject this otherwise self-consistent replacement.
    _assert_metadata_round_trip(
        protocol.ReportInlinePublicationIntent(forged_snapshot.intent)
    )
    object.__setattr__(ack, "snapshot", forged_snapshot)

    with pytest.raises(ProtocolError, match="terminal identity"):
        protocol.AcknowledgeInlineNodeLossReply(request, ack)
    with pytest.raises(ProtocolError, match="terminal identity"):
        pickle.loads(pickle.dumps(reply))


@pytest.mark.parametrize("terminal", (False, True))
@pytest.mark.parametrize("changed", ("prepared", "terminal"))
def test_terminal_ack_rejects_prepared_or_terminal_changed_after_freeze(
    terminal: bool, changed: str,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=terminal)
    work = authority.freeze_node_death(f.death)[0]
    authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    ack = authority.acknowledge_adoption(work)
    request = protocol.AcknowledgeInlineNodeLoss(
        work, f.owner, InlineRecoveryResolution.ADOPTED
    )
    reply = protocol.AcknowledgeInlineNodeLossReply(request, ack)
    if changed == "prepared":
        object.__setattr__(ack.snapshot, "prepared", False)
    else:
        # Adding a report to UNKNOWN and erasing one from POSTCOMPLETE are
        # equally invalid: neither may rewrite the frozen evidence.
        object.__setattr__(ack.snapshot, "terminal", None if terminal else f.terminal)
    assert ack.snapshot.publication_id == work.publication_id
    assert ack.snapshot.frozen_death == work.death
    assert ack.snapshot.resolution is request.resolution

    with pytest.raises(ProtocolError):
        protocol.AcknowledgeInlineNodeLossReply(request, ack)
    with pytest.raises(ProtocolError):
        pickle.loads(pickle.dumps(reply))


@pytest.mark.parametrize("decision", (None, InlineRecoveryOwnerDecision.DROP))
def test_terminal_ack_rejects_missing_or_changed_keep_decision(
    decision: InlineRecoveryOwnerDecision | None,
) -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=True)
    work = authority.freeze_node_death(f.death)[0]
    authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    ack = authority.acknowledge_adoption(work)
    request = protocol.AcknowledgeInlineNodeLoss(
        work, f.owner, InlineRecoveryResolution.ADOPTED
    )
    reply = protocol.AcknowledgeInlineNodeLossReply(request, ack)
    if decision is None:
        # This snapshot is valid for a pre-death normal-adoption tombstone,
        # but cannot acknowledge new frozen work without KEEP.
        object.__setattr__(
            ack, "snapshot", replace(ack.snapshot, owner_decision=None)
        )
    else:
        object.__setattr__(ack.snapshot, "owner_decision", decision)
    assert ack.snapshot.intent == work.snapshot.intent
    assert ack.snapshot.prepared == work.snapshot.prepared
    assert ack.snapshot.terminal == work.snapshot.terminal
    assert ack.snapshot.frozen_death == work.death
    assert ack.snapshot.resolution is request.resolution

    with pytest.raises(ProtocolError):
        protocol.AcknowledgeInlineNodeLossReply(request, ack)
    with pytest.raises(ProtocolError):
        pickle.loads(pickle.dumps(reply))


@pytest.mark.parametrize("terminal", (False, True))
def test_drop_resolves_to_lost_through_ack_and_internal_progress(
    terminal: bool,
) -> None:
    f = _Fixture(source="contained")
    authority, *_ = f.authority(terminal=terminal)
    work = authority.freeze_node_death(f.death)[0]
    progress = protocol.ProgressInlineNodeLoss(f.publication_id, f.death)
    with pytest.raises(InlineRecoveryStateError, match="exact owner decision"):
        authority.acknowledge_lost(work)
    decision = protocol.DecideInlineNodeLoss(
        work, f.owner, InlineRecoveryOwnerDecision.DROP
    )
    decision_ack = authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.DROP
    )
    # The metadata choice itself is not an acknowledgement of child cleanup.
    assert decision_ack.snapshot.resolution is None
    with pytest.raises(ProtocolError, match="exact cleanup"):
        protocol.ProgressInlineNodeLossReply(progress, decision_ack)
    for finish in (authority.acknowledge_adoption, authority.acknowledge_retirement):
        with pytest.raises(InlineRecoveryStateError, match="exact owner decision"):
            finish(work)
    ack = authority.acknowledge_lost(work)
    assert ack.snapshot.resolution is InlineRecoveryResolution.LOST
    assert ack.snapshot.owner_decision is InlineRecoveryOwnerDecision.DROP
    ack_request = protocol.AcknowledgeInlineNodeLoss(
        work, f.owner, InlineRecoveryResolution.LOST
    )
    pull = protocol.GetInlineNodeLoss(f.publication_id, f.owner, f.death)
    _assert_metadata_round_trip((
        protocol.DecideInlineNodeLossReply(decision, decision_ack),
        protocol.AcknowledgeInlineNodeLossReply(ack_request, ack),
        protocol.ProgressInlineNodeLossReply(progress, ack),
        protocol.GetInlineNodeLossReply(
            pull, protocol.InlineNodeLossDisposition.RESOLVED, snapshot=ack.snapshot
        ),
    ))
    with pytest.raises(ProtocolError, match="exact identity"):
        protocol.GetInlineNodeLossReply(
            replace(pull, owner_worker_id=WorkerID.random()),
            protocol.InlineNodeLossDisposition.RESOLVED, snapshot=ack.snapshot,
        )


def test_progress_and_terminal_acks_reject_other_resolutions_or_identities() -> None:
    f = _Fixture()
    authority, *_ = f.authority()
    work = authority.freeze_node_death(f.death)[0]
    authority.decide_owner_resolution(
        work, f.owner, InlineRecoveryOwnerDecision.KEEP
    )
    adopted = authority.acknowledge_adoption(work)
    progress = protocol.ProgressInlineNodeLoss(f.publication_id, f.death)
    with pytest.raises(ProtocolError, match="exact cleanup"):
        protocol.ProgressInlineNodeLossReply(progress, adopted)
    with pytest.raises(ProtocolError, match="terminal identity"):
        protocol.AcknowledgeInlineNodeLossReply(
            protocol.AcknowledgeInlineNodeLoss(
                work, f.owner, InlineRecoveryResolution.RETIRED
            ),
            adopted,
        )
    with pytest.raises(ProtocolError, match="another owner"):
        protocol.AcknowledgeInlineNodeLoss(
            work, WorkerID.random(), InlineRecoveryResolution.ADOPTED
        )

    pre = _Fixture()
    pre_authority, *_ = pre.authority(prepared=False)
    rollback = pre_authority.freeze_node_death(pre.death)[0]
    rolled_back = pre_authority.acknowledge_rollback(rollback)
    for resolution in (InlineRecoveryResolution.ADOPTED, InlineRecoveryResolution.LOST):
        with pytest.raises(ProtocolError, match="contradicts"):
            protocol.AcknowledgeInlineNodeLoss(rollback, pre.owner, resolution)
    for forged in (
        protocol.ProgressInlineNodeLoss(f.publication_id, pre.death),
        protocol.ProgressInlineNodeLoss(
            pre.publication_id, replace(pre.death, death_epoch=10)
        ),
    ):
        with pytest.raises(ProtocolError, match="exact cleanup"):
            protocol.ProgressInlineNodeLossReply(forged, rolled_back)

@pytest.mark.parametrize("boundary", ("intent", "work", "ack"))
@pytest.mark.parametrize("source_kind", ("task", "contained"))
@pytest.mark.parametrize("corruption", ("borrower_token", "source_hold", "hold_type"))
def test_recovery_wire_deeply_revalidates_borrowed_source_credentials(
    boundary: str, source_kind: str, corruption: str,
) -> None:
    f = _Fixture(source=source_kind)
    authority, *_ = f.authority(terminal=True)
    work = authority.freeze_node_death(f.death)[0]
    manifest = f.manifest
    if boundary == "intent":
        message = protocol.ReportInlinePublicationIntent(f.intent)
    else:
        decision = protocol.DecideInlineNodeLoss(
            work, f.owner, InlineRecoveryOwnerDecision.KEEP
        )
        if boundary == "work":
            message = decision
        else:
            # Corrupt only the reply ACK, so request validation cannot hide
            # a missing recursive validation of acknowledged credentials.
            ack = pickle.loads(pickle.dumps(authority.decide_owner_resolution(
                work, f.owner, InlineRecoveryOwnerDecision.KEEP
            )))
            message = protocol.DecideInlineNodeLossReply(decision, ack)
            manifest = ack.snapshot.manifest
    source = manifest.transfers[0].source
    assert isinstance(source, BorrowedContainedSource)
    if corruption == "borrower_token":
        object.__setattr__(source, "borrower_token", "rebound-borrower-token")
    elif corruption == "source_hold":
        # Structurally valid source changes are also bound by the digest.
        if source_kind == "task":
            object.__setattr__(
                source.original_source.hold, "origin_attempt_id", f.attempt.next()
            )
        else:
            object.__setattr__(
                source.original_source.hold, "transfer_token", "rebound-source"
            )
    elif source_kind == "task":
        object.__setattr__(source.original_source.hold, "kind", "RETAINED")
    else:
        object.__setattr__(
            source.original_source.hold, "container_owner_worker_id", "not-a-worker"
        )

    with pytest.raises(ProtocolError, match="invalid"):
        type(message)(*(getattr(message, field.name) for field in fields(message)))
    with pytest.raises(ProtocolError, match="invalid"):
        pickle.loads(pickle.dumps(message))


def test_node_loss_reply_shapes_do_not_mix_work_snapshot_or_errors() -> None:
    f = _Fixture()
    authority, *_ = f.authority(terminal=True)
    work = authority.freeze_node_death(f.death)[0]
    request = protocol.GetInlineNodeLoss(
        f.publication_id, f.owner, f.death
    )
    with pytest.raises(ProtocolError, match="cannot expose state"):
        protocol.GetInlineNodeLossReply(
            request, protocol.InlineNodeLossDisposition.PENDING, work=work
        )
    with pytest.raises(ProtocolError, match="only its work"):
        protocol.GetInlineNodeLossReply(
            request, protocol.InlineNodeLossDisposition.WORK,
            work=work, snapshot=work.snapshot,
        )
    with pytest.raises(ProtocolError, match="cannot expose state"):
        protocol.GetInlineNodeLossReply(
            request, protocol.InlineNodeLossDisposition.REJECTED, work=work,
            error_kind=protocol.StoredPublicationRPCErrorKind.INVALID_STATE,
            error="rejected",
        )
