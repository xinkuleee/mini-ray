"""Pure dominance tests for publication owner-death cleanup."""

from __future__ import annotations

from dataclasses import replace

import pytest

from miniray import protocol
from miniray.contained_cycle import ContainedGraphManifest
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.inline_contained_publication import (
    InlinePublicationEnvelope, InlinePublicationID,
)
from miniray.publication_owner_death import (
    InlinePublicationOwnerDeathKey, InlinePublicationOwnerDeathRequest,
    PublicationOwnerDeathGraphState, PublicationOwnerDeathGraphWitness,
    PublicationOwnerDeathResolution, PublicationOwnerDeathTerminal,
    PublicationRecoveryHistory, StoredPublicationOwnerDeathKey,
)
from miniray.publication_owner_death_arbitration import (
    PublicationArbitrationDecision, PublicationArbitrationOperation,
    PublicationNodeLossSupersessionState,
    PublicationOwnerDeathArbitrationConflictError,
    PublicationOwnerDeathArbitrationPhase,
    PublicationOwnerDeathArbitrationStateError,
    PublicationOwnerDeathArbitrator,
    UnknownPublicationOwnerDeathArbitrationError,
)
from miniray.stored_publication import StoredPublicationID


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self) -> None:
        job = JobID(bytes.fromhex("11" * 16))
        driver = TaskID.for_driver(job)
        task = TaskID.derive(job, driver, 1)
        attempt = AttemptID(task, 0)
        outer = ObjectID.for_task(task)
        child_task = TaskID.derive(job, driver, 2)
        child = ObjectID.for_task(child_task)
        lease = LeaseID(bytes.fromhex("22" * 16))
        executor = WorkerID(bytes.fromhex("33" * 16))
        self.owner = WorkerID(bytes.fromhex("44" * 16))
        publisher = NodeID(bytes.fromhex("55" * 16))
        edge = ContainedReferenceEdge(
            outer, child, executor, ("127.0.0.1", 24001),
            "owner-arbitration-child",
        )
        self.inline_id = InlinePublicationID(
            lease, task, attempt, outer
        )
        self.stored_id = StoredPublicationID(
            lease, task, attempt, outer
        )
        self.inline_key = InlinePublicationOwnerDeathKey(self.inline_id)
        self.stored_key = StoredPublicationOwnerDeathKey(self.stored_id)
        envelope = InlinePublicationEnvelope.create(
            self.inline_id, job, executor, self.owner, publisher,
            b"inline result", (edge,),
        )
        graph = envelope.to_graph_manifest()
        assert isinstance(graph, ContainedGraphManifest)
        self.death = self.owner_death("owner-death", 7)
        self.request = InlinePublicationOwnerDeathRequest(
            self.inline_key, envelope.manifest, self.death,
            PublicationRecoveryHistory.NOT_COMMITTED,
            PublicationOwnerDeathGraphWitness(
                graph, PublicationOwnerDeathGraphState.NOT_FOUND
            ),
        )

    def owner_death(
        self, detection_id: str, death_epoch: int,
        *, reason: protocol.WorkerDeathReason = (
            protocol.WorkerDeathReason.PROCESS_EXIT
        ),
    ) -> protocol.WorkerDeathRecord:
        return protocol.WorkerDeathRecord(
            detection_id,
            protocol.WorkerIncarnation(
                NodeID(bytes.fromhex("66" * 16)), 4100, 3,
                self.owner, 5100,
            ),
            death_epoch, -9, reason,
        )

    def resolution(
        self,
        request: InlinePublicationOwnerDeathRequest | None = None,
    ) -> PublicationOwnerDeathResolution:
        return PublicationOwnerDeathResolution(
            self.request if request is None else request,
            PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED, (),
        )


def test_owner_death_freeze_dominates_every_other_writer() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()

    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.NORMAL_FORWARD
    ) is PublicationArbitrationDecision.ALLOWED
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.NODE_LOSS
    ) is PublicationArbitrationDecision.ALLOWED
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.OWNER_DEATH
    ) is PublicationArbitrationDecision.ALLOWED

    snapshot = arbitration.freeze(fixture.inline_key, fixture.death)

    assert snapshot.phase is PublicationOwnerDeathArbitrationPhase.FROZEN
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.NORMAL_FORWARD
    ) is PublicationArbitrationDecision.SUPERSEDED
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.NODE_LOSS
    ) is PublicationArbitrationDecision.SUPERSEDED
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.OWNER_DEATH
    ) is PublicationArbitrationDecision.OWNER_ONLY
    assert arbitration.has_active_operations()
    assert arbitration.has_active_obligations()


def test_freeze_is_exact_replay_and_conflicts_on_changed_evidence() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    original = arbitration.freeze(
        fixture.inline_key, fixture.death, ("node-loss", 1)
    )

    assert arbitration.freeze(
        fixture.inline_key, fixture.death, ("node-loss", 1)
    ) is original
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="death proof changed",
    ):
        arbitration.freeze(
            fixture.inline_key,
            fixture.owner_death("changed", 8),
            ("node-loss", 1),
        )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="join identity changed",
    ):
        arbitration.freeze(
            fixture.inline_key, fixture.death, ("node-loss", 2)
        )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="join identity changed",
    ):
        arbitration.freeze(fixture.inline_key, fixture.death)


def test_typed_inline_and_stored_keys_never_alias() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    arbitration.freeze(fixture.inline_key, fixture.death)

    assert fixture.inline_key.publication_id.outer_object_id == (
        fixture.stored_key.publication_id.outer_object_id
    )
    assert arbitration.decision(
        fixture.stored_key, PublicationArbitrationOperation.NORMAL_FORWARD
    ) is PublicationArbitrationDecision.ALLOWED
    stored = arbitration.freeze(fixture.stored_key, fixture.death)
    assert arbitration.publication_keys() == (
        fixture.inline_key, fixture.stored_key,
    )
    assert stored.key is fixture.stored_key


def test_cleanup_claim_is_exact_and_must_match_the_frozen_death() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    arbitration.freeze(fixture.inline_key, fixture.death)

    claimed = arbitration.claim_owner_cleanup(
        fixture.inline_key, fixture.request
    )
    assert claimed.phase is PublicationOwnerDeathArbitrationPhase.CLAIMED
    assert arbitration.claim_owner_cleanup(
        fixture.inline_key, fixture.request
    ) is claimed

    changed = replace(
        fixture.request,
        publication_history=PublicationRecoveryHistory.COMMIT_UNKNOWN,
    )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="claimant changed",
    ):
        arbitration.claim_owner_cleanup(fixture.inline_key, changed)

    other = PublicationOwnerDeathArbitrator()
    other.freeze(
        fixture.inline_key, fixture.owner_death("newer", 9)
    )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="frozen death proof",
    ):
        other.claim_owner_cleanup(fixture.inline_key, fixture.request)


def test_active_node_loss_is_joined_and_receipts_import_without_effects() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    identity = ("stored-node-loss", fixture.stored_id)
    arbitration.freeze(fixture.stored_key, fixture.death, identity)

    imported = arbitration.import_node_loss_receipt(
        fixture.stored_key, identity, ("graph-abort", 1),
        ("already-aborted", 1),
    )
    assert imported.node_loss is not None
    assert imported.node_loss.state is (
        PublicationNodeLossSupersessionState.JOINED_SUPERSEDED
    )
    assert len(imported.node_loss.imported_receipts) == 1
    assert arbitration.import_node_loss_receipt(
        fixture.stored_key, identity, ("graph-abort", 1),
        ("already-aborted", 1),
    ) is imported
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="effect receipt changed",
    ):
        arbitration.import_node_loss_receipt(
            fixture.stored_key, identity, ("graph-abort", 1),
            ("different-receipt", 1),
        )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="joined saga",
    ):
        arbitration.import_node_loss_receipt(
            fixture.stored_key, ("other-saga", 1),
            ("hold", 1), ("released", 1),
        )


def test_joined_node_loss_needs_exact_terminal_before_owner_resolution() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    identity = ("inline-node-loss", fixture.inline_id)
    arbitration.freeze(fixture.inline_key, fixture.death, identity)
    arbitration.claim_owner_cleanup(fixture.inline_key, fixture.request)
    resolution = fixture.resolution()

    with pytest.raises(
        PublicationOwnerDeathArbitrationStateError,
        match="must be terminalized",
    ):
        arbitration.resolve(fixture.inline_key, resolution)

    terminal = ("SUPERSEDED", identity)
    terminalized = arbitration.terminalize_node_loss(
        fixture.inline_key, identity, terminal
    )
    assert terminalized.node_loss is not None
    assert terminalized.node_loss.state is (
        PublicationNodeLossSupersessionState.TERMINALIZED
    )
    assert arbitration.terminalize_node_loss(
        fixture.inline_key, identity, terminal
    ) is terminalized
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="terminal receipt changed",
    ):
        arbitration.terminalize_node_loss(
            fixture.inline_key, identity, ("SUPERSEDED", "changed")
        )

    resolved = arbitration.resolve(fixture.inline_key, resolution)
    assert resolved.phase is PublicationOwnerDeathArbitrationPhase.RESOLVED
    assert resolved.resolution is resolution
    assert arbitration.resolve(fixture.inline_key, resolution) is resolved
    assert not arbitration.has_active_operations()


def test_resolution_requires_claim_and_retains_dominant_tombstone() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    frozen = arbitration.freeze(fixture.inline_key, fixture.death)
    resolution = fixture.resolution()

    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="exact cleanup claim",
    ):
        arbitration.resolve(fixture.inline_key, resolution)

    arbitration.claim_owner_cleanup(fixture.inline_key, fixture.request)
    tombstone = arbitration.resolve(fixture.inline_key, resolution)
    assert arbitration.snapshot(fixture.inline_key) is tombstone
    assert arbitration.freeze(fixture.inline_key, fixture.death) is tombstone
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.NORMAL_FORWARD
    ) is PublicationArbitrationDecision.SUPERSEDED
    assert arbitration.decision(
        fixture.inline_key, PublicationArbitrationOperation.OWNER_DEATH
    ) is PublicationArbitrationDecision.OWNER_ONLY
    assert frozen is not tombstone

    changed_request = replace(
        fixture.request,
        publication_history=PublicationRecoveryHistory.COMMIT_UNKNOWN,
    )
    with pytest.raises(
        PublicationOwnerDeathArbitrationConflictError,
        match="resolution changed",
    ):
        arbitration.resolve(
            fixture.inline_key, fixture.resolution(changed_request)
        )


def test_no_join_rejects_node_loss_import_and_terminalization() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    arbitration.freeze(fixture.inline_key, fixture.death)

    with pytest.raises(
        PublicationOwnerDeathArbitrationStateError,
        match="no Node-loss saga",
    ):
        arbitration.import_node_loss_receipt(
            fixture.inline_key, "node-loss", "effect", "receipt"
        )
    with pytest.raises(
        PublicationOwnerDeathArbitrationStateError,
        match="no Node-loss saga",
    ):
        arbitration.terminalize_node_loss(
            fixture.inline_key, "node-loss", "SUPERSEDED"
        )


def test_shutdown_fences_new_freezes_but_preserves_exact_replay() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()
    original = arbitration.freeze(fixture.inline_key, fixture.death)

    assert arbitration.close_admission()
    assert not arbitration.close_admission()
    assert arbitration.admission_closed
    assert arbitration.freeze(fixture.inline_key, fixture.death) is original
    with pytest.raises(
        PublicationOwnerDeathArbitrationStateError,
        match="admission is closed",
    ):
        arbitration.freeze(fixture.stored_key, fixture.death)


def test_invalid_death_unknown_key_and_unstable_identity_are_rejected() -> None:
    fixture = _Fixture()
    arbitration = PublicationOwnerDeathArbitrator()

    with pytest.raises(
        PublicationOwnerDeathArbitrationStateError,
        match="EXPECTED",
    ):
        arbitration.freeze(
            fixture.inline_key,
            fixture.owner_death(
                "expected", 10, reason=protocol.WorkerDeathReason.EXPECTED
            ),
        )
    with pytest.raises(UnknownPublicationOwnerDeathArbitrationError):
        arbitration.snapshot(fixture.inline_key)
    with pytest.raises(TypeError, match="hashable"):
        arbitration.freeze(fixture.inline_key, fixture.death, [])
