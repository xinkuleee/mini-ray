class _Harness:
    def __init__(self, fixture: _Fixture) -> None:
        self.fixture = fixture
        self.events = []
        self.fence_statuses = {}
        self.drop_statuses = {}
        self.node_deaths = {}
        self.child_death = None

    def runtime(self) -> PublicationOwnerDeathRuntime:
        return PublicationOwnerDeathRuntime(
            PublicationOwnerDeathCoordinator(),
            install_node_fence=self.install_fence,
            release_child=self.release_child,
            abort_graph=self.abort_graph,
            release_graph=self.release_graph,
            drop_replica=self.drop_replica,
            finalize_publisher=self.finalize,
            node_death_lookup=lambda node_id: self.node_deaths.get(node_id),
        )

    def install_fence(self, request):
        self.events.append(("fence", request))
        observations = tuple(
            protocol.OwnerDeathReplicaObservation(
                descriptor, self.fence_statuses.get(
                    descriptor.node_id, protocol.OwnerDeathReplicaStatus.PRESENT
                ),
                1 if self.fence_statuses.get(descriptor.node_id)
                is protocol.OwnerDeathReplicaStatus.PINNED else 0,
            )
            for descriptor in request.expected_replicas
        )
        return protocol.InstallOwnerDeathFenceReply(
            request, protocol.OwnerDeathFenceDisposition.FENCED, observations
        )

    def release_child(self, _effect, request):
        self.events.append(("child", request))
        if self.child_death is not None:
            return self.child_death
        return protocol.ReleaseContainedReferenceReply(
            request.object_id, request.owner_worker_id, request.hold,
            True, True,
        )

    def abort_graph(self, request):
        self.events.append(("abort", request))
        return protocol.ContainedGraphReply(
            request, ContainedGraphManifestReceipt(
                request.manifest, ContainedGraphTransactionState.ABORTED,
                ContainedGraphManifestDisposition.APPLIED,
            ),
        )

    def release_graph(self, request):
        self.events.append(("release_graph", request))
        return protocol.ContainedGraphReply(
            request, ContainedGraphManifestReceipt(
                request.manifest, ContainedGraphTransactionState.COMMITTED,
                ContainedGraphManifestDisposition.RELEASED,
                request.manifest.ordered_edges,
            ),
        )

    def drop_replica(self, request):
        self.events.append(("drop", request))
        status = self.drop_statuses.get(
            request.node_id, protocol.DropObjectReplicaStatus.DROPPED
        )
        return protocol.DropObjectReplicaReply(
            request.object_id, request.producer_attempt_id,
            request.owner_worker_id, request.node_id, request.checksum, status,
            None if status in (
                protocol.DropObjectReplicaStatus.DROPPED,
                protocol.DropObjectReplicaStatus.ALREADY_DROPPED,
            ) else "retryable drop",
        )

    def finalize(self, effect):
        self.events.append(("finalize", effect))
        return protocol.FinalizePublicationOwnerDeathReply(
            effect,
            protocol.PublicationOwnerDeathFinalizeDisposition
            .OWNER_DEATH_CLEANED,
            terminal=PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED,
        )


def test_drive_defaults_to_one_effect_and_explicit_none_converges_inline() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    runtime = harness.runtime()
    request = fixture.inline_request()
    started = runtime.begin(request)

    one = runtime.drive(request.key)
    assert len(one.acknowledged_effects) == 1
    assert one.phase is PublicationOwnerDeathPhase.HOLDS
    assert [name for name, _ in harness.events] == ["fence"]

    done = runtime.drive(request.key, max_effects=None)
    assert done.phase is PublicationOwnerDeathPhase.RESOLVED
    assert [name for name, _ in harness.events] == [
        "fence", "child", "abort", "finalize"
    ]
    assert not runtime.has_active_operations()
    assert runtime.drive_one(request.key) == done
    assert runtime.begin(request) == done
    assert started.request == done.request


def test_fence_observations_drive_absent_and_pinned_replica_policy() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    harness.fence_statuses[fixture.replica_node] = (
        protocol.OwnerDeathReplicaStatus.ABSENT
    )
    harness.fence_statuses[fixture.publisher] = (
        protocol.OwnerDeathReplicaStatus.PINNED
    )
    harness.drop_statuses[fixture.publisher] = (
        protocol.DropObjectReplicaStatus.PINNED
    )
    runtime = harness.runtime()
    request = fixture.stored_request()
    runtime.begin(request)

    stalled = runtime.drive(request.key, max_effects=None)
    assert stalled.phase is PublicationOwnerDeathPhase.REPLICAS
    pending = stalled.next_effect
    assert pending is not None
    assert pending.descriptor.node_id == fixture.publisher
    assert [name for name, _ in harness.events].count("drop") == 1

    # PINNED at the fence remains a candidate and PINNED at drop is retryable.
    harness.drop_statuses[fixture.publisher] = (
        protocol.DropObjectReplicaStatus.DROPPED
    )
    done = runtime.drive(request.key, max_effects=None)
    assert done.phase is PublicationOwnerDeathPhase.RESOLVED
    drops = [request for name, request in harness.events if name == "drop"]
    assert [request.node_id for request in drops] == [
        fixture.publisher, fixture.publisher
    ]
    # ABSENT_AT_FENCE discharges the other descriptor without deletion RPC.
    assert all(request.node_id != fixture.replica_node for request in drops)


def test_dead_nodes_replace_fence_drop_and_finalize_rpcs() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    harness.node_deaths[fixture.publisher] = fixture.node_death(
        fixture.publisher
    )
    harness.node_deaths[fixture.replica_node] = fixture.node_death(
        fixture.replica_node
    )
    runtime = harness.runtime()
    request = fixture.stored_request()
    runtime.begin(request)

    done = runtime.drive(request.key, max_effects=None)

    assert done.phase is PublicationOwnerDeathPhase.RESOLVED
    names = [name for name, _ in harness.events]
    assert "fence" not in names
    assert "drop" not in names
    assert "finalize" not in names
    assert names == ["child", "release_graph"]


def test_child_owner_death_proof_replaces_release_rpc_ack() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    harness.child_death = fixture.worker_death(fixture.executor)
    runtime = harness.runtime()
    request = fixture.inline_request()
    runtime.begin(request)

    done = runtime.drive(request.key, max_effects=None)

    assert done.phase is PublicationOwnerDeathPhase.RESOLVED
    resolution = done.resolution
    assert resolution is not None
    hold_receipts = tuple(
        receipt for _effect, receipt in resolution.effect_receipts
        if hasattr(receipt, "contained_owner_death")
    )
    assert len(hold_receipts) == 1
    assert hold_receipts[0].contained_owner_death == harness.child_death


def test_lost_ack_replays_exact_effect_and_malformed_reply_never_advances() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    runtime = harness.runtime()
    request = fixture.inline_request()
    runtime.begin(request)
    attempts = []

    def lost_once(fence_request):
        attempts.append(fence_request)
        if len(attempts) == 1:
            raise ConnectionError("fence ACK lost")
        return protocol.InstallOwnerDeathFenceReply(
            fence_request, protocol.OwnerDeathFenceDisposition.FENCED
        )

    runtime._install_node_fence = lost_once
    with pytest.raises(ConnectionError, match="ACK lost"):
        runtime.drive_one(request.key)
    assert runtime.snapshot(request.key).acknowledged_effects == ()
    runtime.drive_one(request.key)
    assert attempts[0] == attempts[1]

    effect = runtime.next_effect(request.key)
    assert effect is not None
    runtime._release_child = lambda _effect, _request: object()
    with pytest.raises(
        PublicationOwnerDeathRuntimeStateError, match="neither an ACK"
    ):
        runtime.drive_one(request.key)
    assert runtime.next_effect(request.key) == effect


def test_conflicting_fence_or_drop_identity_is_never_acknowledged() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    runtime = harness.runtime()
    request = fixture.stored_request()
    runtime.begin(request)

    def wrong_fence(fence_request):
        rebound = protocol.InstallOwnerDeathFence(
            fence_request.request_id + "-wrong", fence_request.owner_death,
            fence_request.node_id, fence_request.expected_replicas,
        )
        return protocol.InstallOwnerDeathFenceReply(
            rebound, protocol.OwnerDeathFenceDisposition.FENCED,
            tuple(
                protocol.OwnerDeathReplicaObservation(
                    descriptor, protocol.OwnerDeathReplicaStatus.PRESENT
                )
                for descriptor in rebound.expected_replicas
            ),
        )

    runtime._install_node_fence = wrong_fence
    with pytest.raises(
        PublicationOwnerDeathRuntimeConflictError, match="exact request"
    ):
        runtime.drive_one(request.key)
    assert runtime.snapshot(request.key).acknowledged_effects == ()


def test_finalize_receipt_must_echo_exact_effect() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    runtime = harness.runtime()
    request = fixture.inline_request()
    runtime.begin(request)
    while not isinstance(
        runtime.next_effect(request.key), FinalizePublicationOwnerDeathEffect
    ):
        runtime.drive_one(request.key)
    effect = runtime.next_effect(request.key)
    assert isinstance(effect, FinalizePublicationOwnerDeathEffect)
    wire = protocol.FinalizePublicationOwnerDeath.from_effect(effect)
    wrong = replace(wire, request_id=wire.request_id + "-wrong")
    runtime._finalize_publisher = lambda _request: (
        protocol.FinalizePublicationOwnerDeathReply(
            wrong,
            protocol.PublicationOwnerDeathFinalizeDisposition
            .OWNER_DEATH_CLEANED,
            terminal=PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED,
        )
    )

    with pytest.raises(
        PublicationOwnerDeathRuntimeConflictError, match="exact request"
    ):
        runtime.drive_one(request.key)
    assert runtime.next_effect(request.key) == effect


def test_constructor_and_drive_bounds_are_strict() -> None:
    fixture = _Fixture()
    harness = _Harness(fixture)
    runtime = harness.runtime()
    request = fixture.inline_request()
    runtime.begin(request)

    for invalid in (0, -1, True, "all"):
        with pytest.raises(ValueError, match="positive integer"):
            runtime.drive(request.key, max_effects=invalid)
    with pytest.raises(TypeError, match="coordinator"):
        PublicationOwnerDeathRuntime(
            object(), install_node_fence=harness.install_fence,
            release_child=harness.release_child, abort_graph=harness.abort_graph,
            release_graph=harness.release_graph,
            drop_replica=harness.drop_replica,
            finalize_publisher=harness.finalize,
            node_death_lookup=lambda _node: None,
        )
