"""Authority-bearing wire contracts for contained INLINE publication."""

from __future__ import annotations

import pickle
from typing import get_args
from dataclasses import replace

import pytest

from miniray import protocol
from miniray.contained_edges import ContainedReferenceEdge
from miniray.errors import ProtocolError
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.inline_contained_publication import (
    InlinePublicationEnvelope, InlinePublicationID,
)
from miniray.inline_contained_node import (
    InlineChildEffectRequest as NodeInlineChildEffectRequest,
)
from miniray.stored_publication import StoredPublicationID
from miniray.stored_publication import (
    BorrowedContainedSource, PreparedContainedTransfer,
)


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self) -> None:
        self.job = JobID.random()
        self.task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), 17
        )
        self.attempt = AttemptID(self.task, 2)
        self.lease = LeaseID.random()
        self.outer = ObjectID.for_task(self.task)
        self.executor = WorkerID.random()
        self.owner = WorkerID.random()
        self.node = NodeID.random()
        self.publication_id = InlinePublicationID(
            self.lease, self.task, self.attempt, self.outer
        )
        child = ObjectID.for_task(TaskID.derive(
            self.job, TaskID.for_driver(self.job), 18
        ))
        self.edge = ContainedReferenceEdge(
            self.outer, child, self.executor, ("127.0.0.1", 39101),
            "inline-protocol:0",
        )
        self.payload = b"inline outer containing one ObjectRef"
        self.envelope = InlinePublicationEnvelope.create(
            self.publication_id, self.job, self.executor, self.owner,
            self.node, self.payload, (self.edge,),
        )


def test_inline_worker_node_steps_are_exact_byte_free_and_domain_separated(
) -> None:
    fixture = _Fixture()
    opened = protocol.OpenInlineContainedPublication(fixture.envelope)
    prepared = protocol.PrepareInlineContainedPublication(
        fixture.publication_id, fixture.envelope.manifest.manifest_digest
    )
    replies = (
        protocol.InlinePublicationStepReply(
            opened.request_identity, True, changed=True
        ),
        protocol.InlinePublicationStepReply(
            prepared.request_identity, True, changed=False
        ),
    )

    assert pickle.loads(pickle.dumps((opened, prepared, replies))) == (
        opened, prepared, replies
    )
    assert fixture.payload not in pickle.dumps(replies)
    assert opened.request_identity.step is protocol.InlinePublicationStep.OPEN
    assert prepared.request_identity.step is protocol.InlinePublicationStep.PREPARE

    stored = StoredPublicationID(
        fixture.lease, fixture.task, fixture.attempt, fixture.outer
    )
    with pytest.raises(ProtocolError, match="InlinePublicationID"):
        protocol.PrepareInlineContainedPublication(
            stored, fixture.envelope.manifest.manifest_digest
        )


def test_inline_child_install_echoes_complete_owner_bound_effect() -> None:
    fixture = _Fixture()
    effect = fixture.envelope.edge_effect(0)
    child = protocol.InlineChildEffectRequest(
        effect, fixture.owner, fixture.executor
    )
    request = protocol.InstallInlineContainedPin(child)
    reply = protocol.InstallInlineContainedPinReply(
        request, True, changed=True
    )

    assert child.hold == fixture.edge.incoming_hold(fixture.owner)
    # The standalone Node reducer owns an internal effect value; protocol owns
    # the transport boundary and additionally binds the child authority.  Keep
    # the conversion explicit instead of relying on isomorphic duck typing.
    standalone = NodeInlineChildEffectRequest(effect, fixture.owner)
    assert child.effect == standalone.effect
    assert child.container_owner_worker_id == (
        standalone.container_owner_worker_id
    )
    assert child.authority_worker_id == fixture.executor
    assert pickle.loads(pickle.dumps(reply)) == reply
    with pytest.raises(ProtocolError, match="contained object owner"):
        protocol.InlineChildEffectRequest(
            effect, fixture.owner, WorkerID.random()
        )
    wrong_outer = ObjectID.for_task(TaskID.random())
    with pytest.raises(ProtocolError, match="another outer"):
        protocol.InlineChildEffectRequest(
            replace(
                effect,
                edge=replace(
                    fixture.edge, container_object_id=wrong_outer
                ),
                transfer=replace(
                    effect.transfer,
                    provisional_hold=replace(
                        effect.transfer.provisional_hold,
                        container_object_id=wrong_outer,
                    ),
                    final_hold=replace(
                        effect.transfer.final_hold,
                        container_object_id=wrong_outer,
                    ),
                ),
            ),
            fixture.owner, fixture.executor,
        )


def test_borrowed_inline_source_crosses_open_and_child_rpc_exactly() -> None:
    fixture = _Fixture()
    foreign_owner = WorkerID.random()
    edge = replace(
        fixture.edge, contained_owner_worker_id=foreign_owner,
        contained_owner_address=("127.0.0.1", 39109),
    )
    source_hold = protocol.TaskReferenceHold(
        protocol.TaskReferenceHoldKind.RETAINED, fixture.executor,
        fixture.task, fixture.attempt,
    )
    transfer = PreparedContainedTransfer(
        edge.contained_object_id, foreign_owner, edge.contained_owner_address,
        BorrowedContainedSource(
            fixture.executor, "borrowed-inline-wire",
            protocol.TaskHoldSource(source_hold),
        ),
        edge.incoming_hold(fixture.executor),
        edge.incoming_hold(fixture.owner),
    )
    envelope = InlinePublicationEnvelope.create(
        fixture.publication_id, fixture.job, fixture.executor, fixture.owner,
        fixture.node, fixture.payload, (edge,), (transfer,),
    )
    opened = protocol.OpenInlineContainedPublication(envelope)
    child = protocol.InstallInlineContainedPin(protocol.InlineChildEffectRequest(
        envelope.edge_effect(0), fixture.owner, foreign_owner,
    ))

    restored_open, restored_child = pickle.loads(
        pickle.dumps((opened, child))
    )
    assert restored_open == opened
    assert restored_child == child
    assert restored_child.request.effect.transfer.source == transfer.source


def test_inline_envelope_crosses_all_terminal_replies_exactly() -> None:
    fixture = _Fixture()
    completion = protocol.CompleteWorkerLeaseReply(
        fixture.lease, fixture.task, fixture.attempt, fixture.executor,
        protocol.TaskReplyStatus.SUCCEEDED,
        protocol.LeaseExecutionState.COMPLETED, True, True,
        inline_publication=fixture.envelope,
    )
    task = protocol.TaskReply(
        fixture.task, fixture.attempt, fixture.executor,
        protocol.TaskReplyStatus.SUCCEEDED, (fixture.envelope.descriptor,),
        inline_publication=fixture.envelope,
    )
    outcome = protocol.GetWorkerLeaseOutcomeReply(
        fixture.lease, fixture.task, fixture.attempt, fixture.executor,
        fixture.owner, (fixture.outer,), fixture.node, True, False,
        state=protocol.LeaseExecutionState.COMPLETED,
        completion_status=protocol.TaskReplyStatus.SUCCEEDED,
        inline_publication=fixture.envelope,
    )

    assert pickle.loads(pickle.dumps((completion, task, outcome))) == (
        completion, task, outcome
    )
    assert task.contained_edges == () and task.stored_publication is None
    assert outcome.descriptors == outcome.orphan_descriptors == ()


def test_inline_terminal_authority_is_mutually_exclusive_and_exact() -> None:
    fixture = _Fixture()
    task_values = dict(
        task_id=fixture.task, attempt_id=fixture.attempt,
        worker_id=fixture.executor, status=protocol.TaskReplyStatus.SUCCEEDED,
        results=(fixture.envelope.descriptor,),
        inline_publication=fixture.envelope,
    )
    with pytest.raises(ProtocolError, match="mutually exclusive"):
        protocol.TaskReply(
            **dict(task_values, contained_edges=(fixture.edge,))
        )
    # A valid stored envelope is unnecessary here: the protocol must reject
    # competing authority fields before consulting either nested value.
    with pytest.raises(ProtocolError, match="mutually exclusive"):
        protocol.TaskReply(
            **dict(task_values, stored_publication=object())
        )
    with pytest.raises(ProtocolError, match="exactly its single"):
        protocol.TaskReply(**dict(task_values, results=()))
    with pytest.raises(ProtocolError, match="lease, task, attempt"):
        protocol.CompleteWorkerLeaseReply(
            LeaseID.random(), fixture.task, fixture.attempt, fixture.executor,
            protocol.TaskReplyStatus.SUCCEEDED,
            protocol.LeaseExecutionState.COMPLETED, True, True,
            inline_publication=fixture.envelope,
        )
    with pytest.raises(ProtocolError, match="both stored and inline"):
        protocol.CompleteWorkerLeaseReply(
            fixture.lease, fixture.task, fixture.attempt, fixture.executor,
            protocol.TaskReplyStatus.SUCCEEDED,
            protocol.LeaseExecutionState.COMPLETED, True, True,
            stored_publication=object(), inline_publication=fixture.envelope,
        )
    with pytest.raises(ProtocolError, match="both stored and inline"):
        protocol.GetWorkerLeaseOutcomeReply(
            fixture.lease, fixture.task, fixture.attempt, fixture.executor,
            fixture.owner, (fixture.outer,), fixture.node, True, False,
            state=protocol.LeaseExecutionState.COMPLETED,
            completion_status=protocol.TaskReplyStatus.SUCCEEDED,
            stored_publication=object(), inline_publication=fixture.envelope,
        )
    with pytest.raises(ProtocolError, match="replica descriptors"):
        descriptor = protocol.ObjectStoreDescriptor(
            fixture.outer, fixture.owner, fixture.attempt, fixture.node,
            len(fixture.payload), fixture.envelope.descriptor.checksum,
        )
        protocol.GetWorkerLeaseOutcomeReply(
            fixture.lease, fixture.task, fixture.attempt, fixture.executor,
            fixture.owner, (fixture.outer,), fixture.node, True, False,
            state=protocol.LeaseExecutionState.COMPLETED,
            completion_status=protocol.TaskReplyStatus.SUCCEEDED,
            descriptors=(descriptor,), inline_publication=fixture.envelope,
        )


def test_terminal_pickle_rebuild_revalidates_nested_inline_envelope() -> None:
    fixture = _Fixture()
    task = protocol.TaskReply(
        fixture.task, fixture.attempt, fixture.executor,
        protocol.TaskReplyStatus.SUCCEEDED, (fixture.envelope.descriptor,),
        inline_publication=fixture.envelope,
    )
    object.__setattr__(
        fixture.envelope.manifest, "executor_worker_id", WorkerID.random()
    )

    with pytest.raises(ProtocolError, match="inline publication is invalid"):
        pickle.loads(pickle.dumps(task))


def test_protocol_union_contains_inline_authority_values() -> None:
    messages = set(get_args(protocol.ProtocolMessage))
    assert protocol.InlinePublicationStepIdentity in messages
    assert protocol.InlineChildEffectRequest in messages
