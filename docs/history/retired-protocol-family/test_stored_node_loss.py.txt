pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(
        self, *, completed: bool, graph_state: StoredNodeLossGraphState,
        promoted: tuple[bool, ...] = (True, True), sealed: bool = True,
    ) -> None:
        self.job = JobID(bytes.fromhex("11" * 16))
        driver = TaskID.for_driver(self.job)
        self.task = TaskID.derive(self.job, driver, 1)
        self.attempt = AttemptID(self.task, 0)
        self.outer = ObjectID.for_task(self.task)
        self.lease = LeaseID(bytes.fromhex("22" * 16))
        self.executor = WorkerID(bytes.fromhex("33" * 16))
        self.owner = WorkerID(bytes.fromhex("44" * 16))
        self.node = NodeID(bytes.fromhex("55" * 16))
        self.publication = StoredPublicationID(
            self.lease, self.task, self.attempt, self.outer
        )
        self.header = StoredPublicationHeader(
            self.publication, self.job, self.executor, self.owner, self.node
        )
        self.transfers = tuple(self._transfer(index) for index in range(2))
        payload = b"lost stored result"
        checksum = hashlib.sha256(payload).hexdigest()
        self.manifest = StoredPublicationManifest(
            self.publication, self.transfers, len(payload), checksum,
            hashlib.sha256(b"frozen manifest").hexdigest(),
        )
        self.graph_manifest = self.manifest.graph_manifest(self.owner)
        self.descriptor = protocol.ResultDescriptor(
            self.outer, protocol.ResultStorage.OBJECT_STORE, len(payload),
            self.owner, self.node, checksum,
        )
        terminal = (
            StoredPublicationLeaseTerminal(
                self.publication, protocol.TaskReplyStatus.SUCCEEDED
            )
            if completed else None
        )
        self.recovery = StoredPublicationRecoveryRecord(
            self.manifest, self.descriptor if sealed else None, promoted,
            terminal, StoredPublicationIntent(
                self.header, self.manifest, 1234, 7
            ),
        )
        self.death = protocol.NodeDeathRecord(
            "node-death-1", self.node, 1234, 7, 11, -9,
            protocol.NodeDeathReason.PROCESS_EXIT, "publishing Node exited",
        )
        self.incarnation = NodeIncarnationWitness(self.header, 1234, 7)
        active = (
            self.graph_manifest.ordered_edges
            if graph_state is StoredNodeLossGraphState.COMMITTED else ()
        )
        self.graph = StoredNodeLossGraphWitness(
            self.graph_manifest, graph_state, active
        )
        self.request = StoredNodeLossRequest(
            self.death, self.incarnation, self.recovery, self.graph
        )

    def _transfer(self, index: int) -> PreparedContainedTransfer:
        child_task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), index + 10
        )
        token = "stored-child-{}".format(index)
        return PreparedContainedTransfer(
            ObjectID.for_task(child_task), self.executor,
            ("127.0.0.1", 25000 + index),
            OwnedContainedSource(self.executor),
            ContainedReferenceHold(self.outer, self.executor, token),
            ContainedReferenceHold(self.outer, self.owner, token),
        )


def _ack(
    coordinator: StoredNodeLossRecoveryCoordinator, effect: object, *,
    retirement: StoredOwnerRetirementDisposition = (
        StoredOwnerRetirementDisposition.PENDING_MARKED_LOST
    ),
    hold: StoredHoldResolutionDisposition = (
        StoredHoldResolutionDisposition.RELEASED
    ),
):
    if isinstance(effect, RetireStoredOwnerEffect):
        receipt = StoredOwnerRetirementReceipt(effect, retirement)
    elif isinstance(effect, AbortStoredGraphEffect):
        receipt = ContainedGraphManifestReceipt(
            effect.manifest, ContainedGraphTransactionState.ABORTED,
            ContainedGraphManifestDisposition.APPLIED,
        )
    elif isinstance(effect, ReleaseStoredHoldEffect):
        receipt = StoredHoldResolutionReceipt(effect, hold)
    elif isinstance(effect, ReleaseStoredGraphEffect):
        receipt = ContainedGraphManifestReceipt(
            effect.manifest, ContainedGraphTransactionState.COMMITTED,
            ContainedGraphManifestDisposition.RELEASED,
            effect.manifest.ordered_edges,
        )
    elif isinstance(effect, ProveStoredReplicaGoneEffect):
        receipt = StoredReplicaResolutionReceipt(
            effect,
            (StoredReplicaResolutionDisposition.NO_REPLICA_REPORTED
             if effect.descriptor is None else
             StoredReplicaResolutionDisposition.DEAD_NODE_REPLICA_GONE),
        )
    else:
        raise AssertionError(type(effect).__name__)
    return coordinator.acknowledge(effect, receipt), receipt


def _drive(
    coordinator: StoredNodeLossRecoveryCoordinator,
    publication: StoredPublicationID,
) -> tuple[object, ...]:
    effects = []
    while True:
        effect = coordinator.next_effect(publication)
        if effect is None:
            return tuple(effects)
        effects.append(effect)
        _ack(coordinator, effect)


def test_precomplete_prepared_aborts_before_both_ambiguous_hold_identities() -> None:
    fixture = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.PREPARED,
        promoted=(True, False),
    )
    coordinator = StoredNodeLossRecoveryCoordinator()

    started = coordinator.begin(fixture.request)
    effects = _drive(coordinator, fixture.publication)

    assert started.mode is StoredNodeLossMode.PRECOMPLETE_ROLLBACK
    assert started.phase is StoredNodeLossPhase.GRAPH_ABORT
    assert isinstance(effects[0], AbortStoredGraphEffect)
    holds = tuple(
        effect for effect in effects
        if isinstance(effect, ReleaseStoredHoldEffect)
    )
    assert tuple((effect.transfer_index, effect.kind) for effect in holds) == (
        (1, StoredNodeLossHoldKind.FINAL),
        (0, StoredNodeLossHoldKind.FINAL),
        (1, StoredNodeLossHoldKind.PROVISIONAL),
    )
    assert isinstance(effects[-1], ProveStoredReplicaGoneEffect)
    resolution = coordinator.resolution(fixture.publication)
    assert resolution is not None
    assert resolution.terminal is StoredNodeLossTerminal.ROLLED_BACK
    assert coordinator.snapshot(fixture.publication).phase is (
        StoredNodeLossPhase.RESOLVED
    )


@pytest.mark.parametrize(
    "disposition",
    tuple(StoredOwnerRetirementDisposition),
)
def test_postcomplete_committed_owner_cas_precedes_holds_and_graph_release(
    disposition: StoredOwnerRetirementDisposition,
) -> None:
    fixture = _Fixture(
        completed=True, graph_state=StoredNodeLossGraphState.COMMITTED
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    started = coordinator.begin(fixture.request)

    owner = coordinator.next_effect(fixture.publication)
    assert started.phase is StoredNodeLossPhase.OWNER_RETIREMENT
    assert isinstance(owner, RetireStoredOwnerEffect)
    _ack(coordinator, owner, retirement=disposition)
    effects = _drive(coordinator, fixture.publication)

    assert all(
        isinstance(effect, ReleaseStoredHoldEffect) for effect in effects[:2]
    )
    assert all(
        effect.kind is StoredNodeLossHoldKind.FINAL
        for effect in effects[:2]
    )
    assert isinstance(effects[2], ReleaseStoredGraphEffect)
    assert isinstance(effects[3], ProveStoredReplicaGoneEffect)
    resolution = coordinator.resolution(fixture.publication)
    assert resolution is not None
    assert resolution.terminal is StoredNodeLossTerminal.LOST_RESULT_RETIRED


def test_postcomplete_prepared_owner_cas_precedes_abort_and_holds() -> None:
    fixture = _Fixture(
        completed=True, graph_state=StoredNodeLossGraphState.PREPARED
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    coordinator.begin(fixture.request)

    owner = coordinator.next_effect(fixture.publication)
    assert isinstance(owner, RetireStoredOwnerEffect)
    _ack(coordinator, owner)
    graph = coordinator.next_effect(fixture.publication)
    assert isinstance(graph, AbortStoredGraphEffect)
    _ack(coordinator, graph)
    assert isinstance(
        coordinator.next_effect(fixture.publication), ReleaseStoredHoldEffect
    )


def test_effect_order_and_ack_identity_cannot_be_skipped_or_rebound() -> None:
    fixture = _Fixture(
        completed=True, graph_state=StoredNodeLossGraphState.COMMITTED
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    coordinator.begin(fixture.request)
    owner = coordinator.next_effect(fixture.publication)
    assert isinstance(owner, RetireStoredOwnerEffect)
    later = ReleaseStoredGraphEffect(
        fixture.publication, fixture.graph_manifest, fixture.outer
    )
    with pytest.raises(StoredNodeLossStateError, match="next exact"):
        _ack(coordinator, later)

    snapshot, receipt = _ack(coordinator, owner)
    assert snapshot.phase is StoredNodeLossPhase.HOLDS
    replay = coordinator.acknowledge(owner, receipt)
    assert replay == snapshot
    changed = StoredOwnerRetirementReceipt(
        owner, StoredOwnerRetirementDisposition.FENCED
    )
    with pytest.raises(StoredNodeLossConflictError, match="changed"):
        coordinator.acknowledge(owner, changed)


def test_child_owner_death_can_discharge_only_its_exact_hold() -> None:
    fixture = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.ABORTED,
        promoted=(True, True),
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    coordinator.begin(fixture.request)
    effect = coordinator.next_effect(fixture.publication)
    assert isinstance(effect, ReleaseStoredHoldEffect)
    other = protocol.WorkerDeathRecord(
        "other-child",
        protocol.WorkerIncarnation(
            NodeID.random(), 2000, 1, WorkerID.random(), 3000
        ),
        1, -9, protocol.WorkerDeathReason.PROCESS_EXIT,
    )
    with pytest.raises(ValueError, match="exact child-owner"):
        StoredHoldResolutionReceipt(
            effect, StoredHoldResolutionDisposition.OWNER_DEAD, other
        )

    death = replace(
        other,
        detection_id="matching-child",
        incarnation=replace(
            other.incarnation, worker_id=effect.contained_owner_worker_id
        ),
    )
    receipt = StoredHoldResolutionReceipt(
        effect, StoredHoldResolutionDisposition.OWNER_DEAD, death
    )
    snapshot = coordinator.acknowledge(effect, receipt)
    assert snapshot.phase is StoredNodeLossPhase.HOLDS


def test_node_incarnation_graph_and_descriptor_evidence_are_exact() -> None:
    fixture = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.PREPARED
    )
    with pytest.raises(StoredNodeLossConflictError, match="incarnation"):
        StoredNodeLossRequest(
            replace(fixture.death, node_pid=9999), fixture.incarnation,
            fixture.recovery, fixture.graph,
        )
    with pytest.raises(StoredNodeLossConflictError, match="descriptor"):
        StoredNodeLossRequest(
            fixture.death, fixture.incarnation,
            replace(
                fixture.recovery,
                sealed_descriptor=replace(
                    fixture.descriptor, node_id=NodeID.random()
                ),
            ),
            fixture.graph,
        )
    with pytest.raises(StoredNodeLossConflictError, match="graph witness"):
        StoredNodeLossRequest(
            fixture.death, fixture.incarnation, fixture.recovery,
            replace(
                fixture.graph,
                manifest=replace(
                    fixture.graph_manifest, manifest_digest="ab" * 32
                ),
            ),
        )


def test_node_loss_request_constructs_witness_from_durable_intent() -> None:
    fixture = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.PREPARED
    )

    request = StoredNodeLossRequest.from_intent(
        fixture.death, fixture.recovery, fixture.graph
    )

    assert request == fixture.request
    legacy = replace(fixture.recovery, intent=None)
    with pytest.raises(StoredNodeLossStateError, match="intent-backed"):
        StoredNodeLossRequest.from_intent(
            fixture.death, legacy, fixture.graph
        )


def test_precomplete_committed_graph_is_rejected_without_saga_state() -> None:
    fixture = _Fixture(
        completed=True, graph_state=StoredNodeLossGraphState.COMMITTED
    )
    precomplete = replace(fixture.recovery, lease_terminal=None)
    # Construction itself is the reducer's no-mutation preflight.
    with pytest.raises(StoredNodeLossStateError, match="pre-Complete"):
        StoredNodeLossRequest(
            fixture.death, fixture.incarnation, precomplete, fixture.graph
        )


def test_no_replica_report_has_an_explicit_terminal_proof() -> None:
    fixture = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.NOT_FOUND,
        promoted=(False, False), sealed=False,
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    coordinator.begin(fixture.request)
    effects = _drive(coordinator, fixture.publication)
    replica = effects[-1]
    assert isinstance(replica, ProveStoredReplicaGoneEffect)
    assert replica.descriptor is None
    receipt = coordinator.resolution(
        fixture.publication
    ).effect_receipts[-1][1]  # type: ignore[union-attr]
    assert isinstance(receipt, StoredReplicaResolutionReceipt)
    assert receipt.disposition is (
        StoredReplicaResolutionDisposition.NO_REPLICA_REPORTED
    )


def test_terminal_and_inflight_records_gate_shutdown_and_replay() -> None:
    first = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.ABORTED
    )
    coordinator = StoredNodeLossRecoveryCoordinator()
    original = coordinator.begin(first.request)
    assert coordinator.has_active_operations()
    assert coordinator.begin(first.request) == original
    with pytest.raises(StoredNodeLossConflictError, match="evidence changed"):
        coordinator.begin(
            replace(
                first.request,
                node_death=replace(
                    first.death, detection_id="different-proof"
                ),
            )
        )

    _drive(coordinator, first.publication)
    assert not coordinator.has_active_operations()
    terminal = coordinator.begin(first.request)
    assert terminal.phase is StoredNodeLossPhase.RESOLVED
    coordinator.close_admission()
    assert coordinator.begin(first.request) == terminal

    unseen = _Fixture(
        completed=False, graph_state=StoredNodeLossGraphState.ABORTED
    )
    unseen_task = TaskID.derive(
        unseen.job, TaskID.for_driver(unseen.job), 99
    )
    # A minimal distinct valid request avoids relying on random identities.
    unseen_publication = replace(
        unseen.publication, task_id=unseen_task,
        attempt_id=AttemptID(unseen_task, 0),
        outer_object_id=ObjectID.for_task(unseen_task),
    )
    unseen_header = replace(
        unseen.header, publication_id=unseen_publication
    )
    unseen_transfers = tuple(
        replace(
            transfer,
            provisional_hold=replace(
                transfer.provisional_hold,
                container_object_id=unseen_publication.outer_object_id,
            ),
            final_hold=replace(
                transfer.final_hold,
                container_object_id=unseen_publication.outer_object_id,
            ),
        )
        for transfer in unseen.transfers
    )
    unseen_manifest = replace(
        unseen.manifest, publication_id=unseen_publication,
        transfers=unseen_transfers,
    )
    unseen_recovery = replace(
        unseen.recovery, manifest=unseen_manifest,
        sealed_descriptor=replace(
            unseen.descriptor, object_id=unseen_publication.outer_object_id
        ),
        intent=StoredPublicationIntent(
            unseen_header, unseen_manifest, 1234, 7
        ),
    )
    unseen_graph = StoredNodeLossGraphWitness(
        unseen_manifest.graph_manifest(unseen.owner),
        StoredNodeLossGraphState.ABORTED,
    )
    unseen_request = StoredNodeLossRequest(
        unseen.death, NodeIncarnationWitness(unseen_header, 1234, 7),
        unseen_recovery, unseen_graph,
    )
    with pytest.raises(StoredNodeLossStateError, match="admission is closed"):
        coordinator.begin(unseen_request)
