pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self) -> None:
        self.job = JobID.random()
        driver = TaskID.for_driver(self.job)
        self.task = TaskID.derive(self.job, driver, 1)
        self.attempt = AttemptID(self.task, 0)
        self.owner = WorkerID.random()
        self.executor = WorkerID.random()
        self.node = NodeID.random()
        self.spec = protocol.TaskSpec(
            self.job, self.task, self.attempt,
            protocol.FunctionKey(self.job, __name__, "producer", "v1"),
            (), 1, ResourceVector(), self.owner, max_retries=1,
        )
        self.execution = TaskExecutionKey.from_task_spec(self.spec)
        self.outer = self.execution.output_ids[0]
        self.edges = tuple(
            ContainedReferenceEdge(
                self.outer, ObjectID.for_task(TaskID.derive(
                    self.job, driver, index + 2
                )), self.executor, ("127.0.0.1", 35200),
                f"unreceived-child-{index}",
            )
            for index in range(2)
        )
        self.envelope = self.envelope_for(self.attempt)
        self.graph = self.envelope.to_graph_manifest()
        self.plan = InlineOwnerPublicationPlan(
            self.execution, self.envelope.descriptor, self.graph,
        )

    def envelope_for(self, attempt: AttemptID) -> InlinePublicationEnvelope:
        return InlinePublicationEnvelope.create(
            InlinePublicationID(LeaseID.random(), self.task, attempt, self.outer),
            self.job, self.executor, self.owner, self.node,
            b"unreceived inline payload", self.edges,
        )

    def table(self, *, local_token: str | None = None) -> ObjectOwnerTable:
        table = ObjectOwnerTable()
        table.register_task_outputs(self.spec, local_tokens=(local_token,))
        return table


def test_retirement_is_metadata_only_atomic_and_preserves_lineage() -> None:
    fixture = _Fixture()
    table = fixture.table(local_token="outer-handle")
    lineage = LineageReferenceEdge(
        fixture.outer, fixture.edges[0].contained_object_id, "producer-input"
    )
    table.add_outgoing_lineage_edge(fixture.outer, lineage)
    before = table.snapshot(fixture.outer)

    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)

    retired = table.snapshot(fixture.outer)
    assert retired == replace(before, state=ObjectState.LOST)
    assert retired.inline_data is None
    assert retired.canonical_stored_result is None
    assert retired.inline_publication is None
    assert retired.stored_publication is None
    assert retired.locations == frozenset()
    assert retired.outgoing_contained_edges == frozenset()
    assert retired.outgoing_lineage_edges == frozenset((lineage,))
    assert retired.producer_task_spec is fixture.spec
    assert table._unreceived_inline_retirements == {
        fixture.execution: fixture.graph
    }
    assert table._inline_publication_receipts == {}
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert table.snapshot(fixture.outer) == retired


def test_accepted_inline_bytes_win_without_installing_drop_tombstone() -> None:
    fixture = _Fixture()
    table = fixture.table()
    table.commit_inline_owner_publication(fixture.plan)
    before = table.snapshot(fixture.outer)

    assert not table.retire_unreceived_inline_result(
        fixture.execution, fixture.graph
    )
    assert table.snapshot(fixture.outer) == before
    assert table._unreceived_inline_retirements == {}
    assert table.commit_inline_owner_publication(fixture.plan).disposition is (
        InlineOwnerPublicationDisposition.ALREADY_APPLIED
    )


def test_retired_attempt_is_fenced_but_new_attempt_reconstructs() -> None:
    fixture = _Fixture()
    table = fixture.table()
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    retired = table.snapshot(fixture.outer)
    assert table.validate_inline_owner_publication(fixture.plan) is (
        InlineOwnerPublicationDisposition.FENCED
    )
    assert not table.commit_inline_owner_publication(fixture.plan).committed
    assert table.snapshot(fixture.outer) == retired

    next_attempt = fixture.attempt.next()
    assert table.advance_task_outputs(fixture.execution, next_attempt)
    pending = table.snapshot(fixture.outer)
    assert pending.producer_task_spec is fixture.spec
    assert pending.state is ObjectState.PENDING
    assert not table.retire_unreceived_inline_result(
        fixture.execution, fixture.graph
    )
    assert not table.commit_inline_owner_publication(fixture.plan).committed
    assert table.snapshot(fixture.outer) == pending
    with pytest.raises(ValueError, match="execution and graph"):
        InlineOwnerPublicationPlan(
            fixture.execution.for_attempt(next_attempt),
            fixture.envelope.descriptor, fixture.graph,
        )

    next_envelope = fixture.envelope_for(next_attempt)
    next_plan = InlineOwnerPublicationPlan(
        fixture.execution.for_attempt(next_attempt), next_envelope.descriptor,
        next_envelope.to_graph_manifest(),
    )
    assert table.commit_inline_owner_publication(next_plan).committed
    assert table.snapshot(fixture.outer).producer_task_spec is fixture.spec


def test_unreceived_inline_uses_existing_lineage_reconstruction_coordinator() -> None:
    fixture = _Fixture()
    table = fixture.table(local_token="live-reconstruction-request")
    recovery = RecoveryManager()
    recovery.register_task(fixture.spec, max_retries=fixture.spec.max_retries)
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    recovery.record_task_success(fixture.task, fixture.attempt)
    coordinator = ReconstructionCoordinator(recovery, table)

    graph_plan = coordinator.preflight_graph(fixture.outer)
    assert len(graph_plan.steps) == 1
    assert graph_plan.steps[0].task_spec == fixture.spec
    outcome = coordinator.request(fixture.outer)

    assert outcome.disposition is ReconstructionDisposition.START
    assert outcome.plan is not None
    assert outcome.plan.attempt_id == fixture.attempt.next()
    assert outcome.plan.output_ids == (fixture.outer,)
    assert outcome.plan.task_spec == replace(
        fixture.spec, attempt_id=fixture.attempt.next()
    )
    assert table.snapshot(fixture.outer).producer_task_spec is fixture.spec
    assert table.snapshot(fixture.outer).state is ObjectState.PENDING
    assert not table.commit_inline_owner_publication(fixture.plan).committed


@pytest.mark.parametrize("changed_field", ("manifest_digest", "ordered_edges"))
def test_retirement_replay_binds_the_exact_graph(changed_field: str) -> None:
    fixture = _Fixture()
    table = fixture.table()
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    before = table.snapshot(fixture.outer)
    value = (
        "ab" * 32 if changed_field == "manifest_digest"
        else tuple(reversed(fixture.graph.ordered_edges))
    )
    changed = replace(fixture.graph, **{changed_field: value})

    with pytest.raises(InlineOwnerPublicationConflictError, match="rebound"):
        table.retire_unreceived_inline_result(fixture.execution, changed)
    with pytest.raises(InlineOwnerPublicationConflictError, match="rebound"):
        table.commit_inline_owner_publication(replace(
            fixture.plan, graph_manifest=changed
        ))
    assert table.snapshot(fixture.outer) == before
    assert table._unreceived_inline_retirements[fixture.execution] == fixture.graph


@pytest.mark.parametrize("case", (
    "owner", "task", "attempt", "transaction", "empty", "foreign-edge",
))
def test_retirement_rejects_foreign_identity_without_mutation(case: str) -> None:
    fixture = _Fixture()
    table = fixture.table()
    before = table.snapshot(fixture.outer)
    graph = fixture.graph
    if case == "owner":
        graph = replace(graph, outer_owner_worker_id=WorkerID.random())
    elif case == "task":
        other = _Fixture()
        graph = other.graph
    elif case == "attempt":
        graph = fixture.envelope_for(fixture.attempt.next()).to_graph_manifest()
    elif case == "transaction":
        graph = replace(graph, transaction_id="another-graph-transaction")
    elif case == "empty":
        graph = replace(graph, ordered_edges=())
    else:
        graph = replace(graph, ordered_edges=(replace(
            fixture.edges[0], container_object_id=fixture.edges[0].contained_object_id
        ),))

    with pytest.raises(InlineOwnerPublicationConflictError):
        table.retire_unreceived_inline_result(fixture.execution, graph)
    assert table.snapshot(fixture.outer) == before
    assert table._unreceived_inline_retirements == {}


def test_retirement_requires_one_return_typed_identity_and_lineage() -> None:
    fixture = _Fixture()
    table = fixture.table()
    with pytest.raises(TypeError, match="execution"):
        table.retire_unreceived_inline_result(fixture.attempt, fixture.graph)
    with pytest.raises(TypeError, match="ContainedGraphManifest"):
        table.retire_unreceived_inline_result(fixture.execution, fixture.envelope)
    with pytest.raises(TypeError, match="InlinePublicationID"):
        table.retire_unreceived_inline_result(
            fixture.execution, replace(fixture.graph, publication_id="opaque")
        )
    with pytest.raises(ValueError, match="one return"):
        table.retire_unreceived_inline_result(
            TaskExecutionKey.from_task_spec(replace(fixture.spec, num_returns=2)),
            fixture.graph,
        )
    assert table.snapshot(fixture.outer).state is ObjectState.PENDING
    assert table._unreceived_inline_retirements == {}

    no_lineage = ObjectOwnerTable()
    no_lineage.register(fixture.outer, current_attempt=fixture.attempt)
    with pytest.raises(InvalidObjectTransitionError, match="producer lineage"):
        no_lineage.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert no_lineage.snapshot(fixture.outer).state is ObjectState.PENDING


@pytest.mark.parametrize("field", (
    "inline_data", "error", "canonical_stored_result", "location_attempts",
    "outgoing_contained_edges", "inline_publication",
))
def test_partial_pending_metadata_cannot_be_erased_by_retirement(field: str) -> None:
    fixture = _Fixture()
    table = fixture.table()
    stored = replace(
        fixture.envelope.descriptor, storage=protocol.ResultStorage.OBJECT_STORE,
        inline_data=None,
    )
    corruptions = {
        "inline_data": b"already received",
        "error": RuntimeError("partial error"),
        "canonical_stored_result": stored,
        "location_attempts": {fixture.node: fixture.attempt},
        "outgoing_contained_edges": set(fixture.edges),
        "inline_publication": fixture.plan,
    }
    setattr(table._entries[fixture.outer], field, corruptions[field])
    before = table.snapshot(fixture.outer)

    with pytest.raises(InlineOwnerPublicationConflictError, match="partially"):
        table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert table.snapshot(fixture.outer) == before
    assert table._unreceived_inline_retirements == {}

def test_retired_inline_gc_is_metadata_and_lineage_only() -> None:
    fixture = _Fixture()
    table = fixture.table(local_token="handle")
    lineage = LineageReferenceEdge(
        fixture.outer, fixture.edges[0].contained_object_id, "lineage-after-loss"
    )
    table.add_outgoing_lineage_edge(fixture.outer, lineage)
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert table.begin_collection(fixture.outer) is None
    table.release_local_reference(fixture.outer, "handle")

    plan = table.begin_collection(fixture.outer, collection_id="lost-inline-gc")
    assert plan is not None
    assert plan.canonical_size_bytes is None and plan.canonical_checksum is None
    assert plan.locations == ()
    assert plan.contained_releases == ()
    assert plan.lineage_releases == (lineage,)
    assert plan.producer_task_spec is fixture.spec
    assert plan.producer_attempt_id == fixture.attempt
    with pytest.raises(ObjectCollectionInProgressError):
        table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert table.begin_collection(fixture.outer, collection_id="lost-inline-gc") == plan
    collection = table.complete_collection(plan)
    assert collection.collected
    assert collection.contained_releases == ()
    assert collection.lineage_releases == (lineage,)
    assert table.collection_state(fixture.outer) is ObjectCollectionState.COLLECTED
    assert not table.retire_unreceived_inline_result(
        fixture.execution, fixture.graph
    )
    assert table.commit_inline_owner_publication(fixture.plan).disposition is (
        InlineOwnerPublicationDisposition.FENCED
    )


def test_tombstone_does_not_make_contaminated_lost_metadata_collectible() -> None:
    fixture = _Fixture()

    table._entries[fixture.outer].outgoing_contained_edges.add(fixture.edges[0])

        table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    assert table.snapshot(fixture.outer) == before


def test_old_retirement_does_not_authorize_new_attempts_stored_gc() -> None:
    fixture = _Fixture()

        fixture.execution, fixture.graph
    )


@pytest.mark.parametrize("entrypoint", ("replica", "task-outputs", "publication"))
def test_retired_inline_attempt_cannot_reappear_through_stored_publication(
    entrypoint: str,
) -> None:
    fixture = _Fixture()
    table = fixture.table()
    assert table.retire_unreceived_inline_result(fixture.execution, fixture.graph)
    before = table.snapshot(fixture.outer)
    descriptor = replace(
        fixture.envelope.descriptor, storage=protocol.ResultStorage.OBJECT_STORE,
        inline_data=None,
    )

    def publish(execution: TaskExecutionKey) -> None:
        if entrypoint == "replica":
            table.publish_stored(
                fixture.outer, execution.attempt_id, fixture.node,
                descriptor=descriptor,
            )
        elif entrypoint == "task-outputs":
            table.publish_task_outputs(execution, (descriptor,))
        else:
            publication_id = StoredPublicationID(
                LeaseID.random(), fixture.task, execution.attempt_id, fixture.outer
            )
            manifest = StoredPublicationManifest(
                publication_id, fixture.envelope.manifest.transfers,
                descriptor.size_bytes, descriptor.checksum, "ab" * 32,
            )
            header = StoredPublicationHeader(
                publication_id, fixture.job, fixture.executor, fixture.owner,
                fixture.node,
            )
            envelope = StoredPublicationEnvelope(
                publication_id, manifest.manifest_digest, descriptor,
                manifest.graph_manifest(fixture.owner), header,
            )
            claim = StoredPublicationClaim(
                publication_id, "stored-adoption-claim", fixture.owner,
                manifest.manifest_digest,
            )
            plan = StoredOwnerPublicationPlan(
                execution, envelope, StoredPublicationAdoption(claim, "stored-commit")
            )
            table.commit_stored_owner_publication(plan)

    with pytest.raises(InvalidObjectTransitionError, match="retired inline execution"):
        publish(fixture.execution)
    assert table.snapshot(fixture.outer) == before
    assert table._stored_publication_receipts == {}
    assert table._stored_owner_commit_ids == {}
    assert table.advance_task_outputs(fixture.execution, fixture.attempt.next())
    publish(fixture.execution.for_attempt(fixture.attempt.next()))
    assert table.snapshot(fixture.outer).state is ObjectState.READY_STORED
    assert table.snapshot(fixture.outer).producer_task_spec is fixture.spec


@pytest.mark.parametrize("canonical", (False, True))
def test_other_lost_results_are_not_retirements_and_keep_stored_gc_rules(
    canonical: bool,
) -> None:
    fixture = _Fixture()
    table = fixture.table()
    descriptor = replace(
        fixture.envelope.descriptor, storage=protocol.ResultStorage.OBJECT_STORE,
        inline_data=None,
    )
    table.publish_stored(
        fixture.outer, fixture.attempt, fixture.node,
        descriptor=descriptor if canonical else None,
    )
    table.mark_lost(fixture.outer, fixture.attempt)
    before = table.snapshot(fixture.outer)

    assert not table.retire_unreceived_inline_result(
        fixture.execution, fixture.graph
    )
    assert table.snapshot(fixture.outer) == before
    assert table._unreceived_inline_retirements == {}
    with pytest.raises(InvalidObjectTransitionError, match="canonical metadata"):
        table.begin_collection(fixture.outer)
    if canonical:
        plan = table.begin_collection(
            fixture.outer, canonical_size_bytes=descriptor.size_bytes,
            canonical_checksum=descriptor.checksum,
        )
        assert plan is not None
        assert plan.canonical_size_bytes == descriptor.size_bytes
        assert table.complete_collection(plan).collected
