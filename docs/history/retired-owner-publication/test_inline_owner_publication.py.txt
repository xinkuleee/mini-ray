"""Owner-table atomicity for graph-associated INLINE outer objects."""

from __future__ import annotations

import hashlib
from dataclasses import fields, is_dataclass, replace
from enum import Enum

import pytest

from miniray import protocol
from miniray.contained_cycle import (
    ContainedGraphManifest, ContainedReferenceGraphAuthority,
)
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.inline_contained_publication import InlinePublicationID
from miniray.ownership import (
    InlineOwnerPublicationCollectionRequiredError,
    InlineOwnerPublicationConflictError, InlineOwnerPublicationDisposition,
    InlineOwnerPublicationPlan, ObjectCollectionState, ObjectOwnerTable,
    ObjectState,
)
from miniray.resources import ResourceVector
from miniray.task_outputs import TaskExecutionKey


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self) -> None:
        self.job = JobID.random()
        self.task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), 93
        )
        self.attempt = AttemptID(self.task, 0)
        self.owner = WorkerID.random()
        self.node = NodeID.random()
        self.spec = protocol.TaskSpec(
            self.job, self.task, self.attempt,
            protocol.FunctionKey(self.job, __name__, "producer", "v1"),
            (), 1, ResourceVector(), self.owner, max_retries=1,
        )
        self.execution = TaskExecutionKey.from_task_spec(self.spec)
        self.outer = self.execution.output_ids[0]
        self.publication_id = (
            "INLINE", self.task, self.attempt, self.outer,
        )
        raw_edges = tuple(
            ContainedReferenceEdge(
                self.outer,
                ObjectID.for_task(TaskID.derive(
                    self.job, TaskID.for_driver(self.job), index + 120
                )),
                WorkerID.random(), ("127.0.0.1", 35100 + index),
                f"inline-final-{index}",
            )
            for index in range(2)
        )
        self.edges = tuple(reversed(sorted(raw_edges)))
        payload = b"inline outer with contained refs"
        self.result = protocol.ResultDescriptor(
            self.outer, protocol.ResultStorage.INLINE, len(payload),
            self.owner, self.node, hashlib.sha256(payload).hexdigest(), payload,
        )
        self.graph = ContainedGraphManifest(
            "contained-graph:inline-owner", self.publication_id, self.owner,
            hashlib.sha256(b"inline-manifest").hexdigest(), self.edges,
        )
        self.plan = InlineOwnerPublicationPlan(
            self.execution, self.result, self.graph
        )

    def owner_table(
        self, *, local_token: str | None = None,
    ) -> ObjectOwnerTable:
        table = ObjectOwnerTable()
        table.register_task_outputs(
            self.spec, local_tokens=(local_token,)
        )
        return table


def test_inline_bytes_edges_and_graph_commit_atomically_and_replay() -> None:
    fixture = _Fixture()
    table = fixture.owner_table(local_token="outer-handle")

    assert table.validate_inline_owner_publication(fixture.plan) is (
        InlineOwnerPublicationDisposition.APPLIED
    )
    before = table.snapshot(fixture.outer)
    assert before.state is ObjectState.PENDING
    assert before.inline_data is None
    assert before.inline_publication is None
    assert not before.outgoing_contained_edges

    receipt = table.commit_inline_owner_publication(fixture.plan)
    assert receipt.plan == fixture.plan and receipt.committed
    assert receipt.disposition is InlineOwnerPublicationDisposition.APPLIED
    ready = table.snapshot(fixture.outer)
    assert ready.state is ObjectState.READY_INLINE
    assert ready.inline_data == fixture.result.inline_data
    assert ready.inline_publication == fixture.plan
    assert ready.outgoing_contained_edges == frozenset(fixture.edges)
    assert table.inline_owner_publication(fixture.outer) == fixture.plan

    replay = table.commit_inline_owner_publication(fixture.plan)
    assert replay.disposition is (
        InlineOwnerPublicationDisposition.ALREADY_APPLIED
    )
    assert table.snapshot(fixture.outer) == ready
    assert table.inline_owner_publication_receipt(
        fixture.outer
    ).plan == fixture.plan


def test_plan_rejects_empty_or_foreign_container_graphs() -> None:
    fixture = _Fixture()
    with pytest.raises(ValueError, match="requires outgoing"):
        InlineOwnerPublicationPlan(
            fixture.execution, fixture.result,
            replace(fixture.graph, ordered_edges=()),
        )

    foreign_outer = ObjectID.for_task(TaskID.derive(
        fixture.job, TaskID.for_driver(fixture.job), 999
    ))
    foreign_edge = replace(
        fixture.edges[0], container_object_id=foreign_outer
    )
    with pytest.raises(ValueError, match="sole inline output"):
        InlineOwnerPublicationPlan(
            fixture.execution, fixture.result,
            replace(fixture.graph, ordered_edges=(foreign_edge,)),
        )


def test_stale_and_conflicting_inline_commits_have_zero_mutation() -> None:
    fixture = _Fixture()
    next_spec = replace(fixture.spec, attempt_id=fixture.attempt.next())
    stale_table = ObjectOwnerTable()
    stale_table.register_task_outputs(next_spec)
    stale_before = stale_table.snapshot(fixture.outer)

    fenced = stale_table.commit_inline_owner_publication(fixture.plan)
    assert fenced.disposition is InlineOwnerPublicationDisposition.FENCED
    assert not fenced.committed
    assert stale_table.snapshot(fixture.outer) == stale_before

    table = fixture.owner_table()
    table.commit_inline_owner_publication(fixture.plan)
    committed = table.snapshot(fixture.outer)
    changed_payload = b"different inline bytes"
    changed_result = replace(
        fixture.result, size_bytes=len(changed_payload),
        checksum=hashlib.sha256(changed_payload).hexdigest(),
        inline_data=changed_payload,
    )
    conflict = replace(fixture.plan, result=changed_result)
    with pytest.raises(
        InlineOwnerPublicationConflictError, match="rebound"
    ):
        table.commit_inline_owner_publication(conflict)
    assert table.snapshot(fixture.outer) == committed

    partial = fixture.owner_table()
    assert partial.add_outgoing_contained_edge(
        fixture.outer, fixture.edges[0]
    )
    partial_before = partial.snapshot(fixture.outer)
    with pytest.raises(
        InlineOwnerPublicationConflictError, match="partially"
    ):
        partial.commit_inline_owner_publication(fixture.plan)
    assert partial.snapshot(fixture.outer) == partial_before


def test_inline_collection_requires_the_exact_graph_release_receipt() -> None:
    fixture = _Fixture()
    table = fixture.owner_table(local_token="outer-handle")
    table.commit_inline_owner_publication(fixture.plan)
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.graph)
    graph.commit_manifest(fixture.graph)

    assert table.begin_inline_publication_collection(
        fixture.outer, collection_id="inline-gc:1"
    ) is None
    assert table.release_local_reference(fixture.outer, "outer-handle")
    frozen = table.begin_inline_publication_collection(
        fixture.outer, collection_id="inline-gc:1"
    )
    assert frozen is not None
    assert frozen.publication == fixture.plan
    assert frozen.metadata_plan == table.begin_collection(
        fixture.outer, collection_id="inline-gc:1"
    )
    assert frozen.metadata_plan.locations == ()
    assert tuple(frozen.metadata_plan.contained_releases) == tuple(
        sorted(fixture.edges)
    )

    with pytest.raises(
        InlineOwnerPublicationCollectionRequiredError,
        match="graph release",
    ):
        table.complete_collection(frozen.metadata_plan)
    assert table.collection_state(fixture.outer) is (
        ObjectCollectionState.COLLECTING
    )

    release = graph.release_manifest_container(fixture.graph, fixture.outer)
    wrong = replace(
        release, manifest=replace(
            fixture.graph, transaction_id="contained-graph:wrong-inline"
        )
    )
    before_wrong = table.snapshot(fixture.outer)
    with pytest.raises(
        InlineOwnerPublicationConflictError, match="exact inline manifest"
    ):
        table.complete_inline_publication_collection(frozen, wrong)
    assert table.snapshot(fixture.outer) == before_wrong

    completed = table.complete_inline_publication_collection(frozen, release)
    assert completed.disposition is InlineOwnerPublicationDisposition.APPLIED
    assert completed.collection.contained_releases == tuple(
        sorted(fixture.edges)
    )
    assert table.collection_state(fixture.outer) is (
        ObjectCollectionState.COLLECTED
    )
    assert table.inline_owner_publication(fixture.outer) is None

    replay = table.complete_inline_publication_collection(frozen, release)
    assert replay.disposition is (
        InlineOwnerPublicationDisposition.ALREADY_APPLIED
    )
    assert table.inline_publication_collection_receipt(
        frozen
    ).collection == completed.collection
    assert table.begin_inline_publication_collection(
        fixture.outer, collection_id="inline-gc:1"
    ) is None


def test_legacy_inline_publication_keeps_legacy_collection_path() -> None:
    fixture = _Fixture()
    table = fixture.owner_table()
    assert table.publish_inline(
        fixture.outer, fixture.attempt, b"legacy-inline"
    )
    frozen = table.begin_collection(
        fixture.outer, collection_id="legacy-inline-gc"
    )
    assert frozen is not None
    assert table.complete_collection(frozen).collected


def test_terminal_inline_history_cannot_keep_result_or_lineage_payload_bytes() -> None:
    fixture = _Fixture()
    publication_id = InlinePublicationID(
        LeaseID.random(), fixture.task, fixture.attempt, fixture.outer
    )
    fixture.graph = replace(
        fixture.graph, publication_id=publication_id,
        transaction_id=publication_id.graph_transaction_id,
    )
    fixture.plan = replace(fixture.plan, graph_manifest=fixture.graph)
    argument_payload = b"argument-stream-no-longer-needed-after-gc"
    function_payload = b"serialized-function-no-longer-needed-after-gc"
    fixture.spec = replace(
        fixture.spec, args=(protocol.InlineArg(argument_payload),),
        function_definition=protocol.FunctionDefinition.from_payload(
            fixture.spec.function, function_payload
        ),
    )
    table = fixture.owner_table()
    table.commit_inline_owner_publication(fixture.plan)
    frozen = table.begin_inline_publication_collection(
        fixture.outer, collection_id="compact-inline-gc"
    )
    assert frozen is not None
    assert table.inline_publication_collection_receipt(frozen) is None
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.graph)
    graph.commit_manifest(fixture.graph)
    release = graph.release_manifest_container(fixture.graph, fixture.outer)

    completed = table.complete_inline_publication_collection(frozen, release)

    def check_metadata_tree(value: object) -> None:
        if isinstance(value, (JobID, TaskID, WorkerID, NodeID, LeaseID)):
            assert len(bytes(value)) == 16
            return
        assert not isinstance(value, (
            bytes, bytearray, memoryview, protocol.ResultDescriptor,
            protocol.TaskSpec, protocol.FunctionDefinition,
        ))
        if value is None or isinstance(value, (str, bool, int, Enum)):
            return
        if is_dataclass(value) and not isinstance(value, type):
            for item in fields(value):
                check_metadata_tree(getattr(value, item.name))
            return
        if isinstance(value, dict):
            for key, item in value.items():
                check_metadata_tree(key)
                check_metadata_tree(item)
            return
        if isinstance(value, (tuple, list, set, frozenset)):
            for item in value:
                check_metadata_tree(item)
            return
        raise AssertionError(f"unexpected history type: {type(value).__name__}")

    # The caller retains frozen/completed for replay, but the owner itself
    # must no longer own a path to result, argument, or function payloads.
    for name, value in vars(table).items():
        if name != "_lock":
            check_metadata_tree(value)
    assert not table.contains(fixture.outer)
    assert table.inline_publication_collection_receipt(frozen) == completed
    assert table.complete_inline_publication_collection(frozen, release) == replace(
        completed, disposition=InlineOwnerPublicationDisposition.ALREADY_APPLIED
    )
    rebuilt = replace(
        frozen, publication=replace(
            frozen.publication, result=replace(frozen.publication.result),
        ), metadata_plan=replace(
            frozen.metadata_plan, producer_task_spec=replace(fixture.spec),
        ),
    )
    assert rebuilt == frozen and rebuilt is not frozen
    assert table.inline_publication_collection_receipt(rebuilt) == completed
    assert table.begin_inline_publication_collection(
        fixture.outer, collection_id="compact-inline-gc"
    ) is None
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.begin_inline_publication_collection(
            fixture.outer, collection_id="different-collection"
        )

    # A graph-associated publication cannot resurrect a collected object.
    assert table.commit_inline_owner_publication(fixture.plan).disposition is (
        InlineOwnerPublicationDisposition.FENCED
    )
    changed_payload = b"changed-result-after-collection"
    changed_result = replace(
        fixture.result, inline_data=changed_payload, size_bytes=len(changed_payload),
        checksum=hashlib.sha256(changed_payload).hexdigest(),
    )
    with pytest.raises(InlineOwnerPublicationConflictError, match="rebound"):
        table.commit_inline_owner_publication(replace(
            fixture.plan, result=changed_result
        ))
    changed_publication = replace(
        frozen, publication=replace(frozen.publication, result=changed_result),
    )
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.complete_inline_publication_collection(changed_publication, release)
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.inline_publication_collection_receipt(changed_publication)

    # Exact replay hashes the entire frozen metadata identity, including
    # TaskSpec argument bytes, not merely the graph or collection_id.
    changed_metadata = replace(
        frozen.metadata_plan, producer_task_spec=replace(
            fixture.spec, args=(protocol.InlineArg(b"changed-argument"),)
        ),
    )
    changed_plan = replace(frozen, metadata_plan=changed_metadata)
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.complete_inline_publication_collection(changed_plan, release)
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.inline_publication_collection_receipt(changed_plan)
    changed_function = replace(
        frozen, metadata_plan=replace(
            frozen.metadata_plan, producer_task_spec=replace(
                fixture.spec,
                function_definition=protocol.FunctionDefinition.from_payload(
                    fixture.spec.function, b"changed-function"
                ),
            ),
        ),
    )
    with pytest.raises(InlineOwnerPublicationConflictError, match="identity changed"):
        table.inline_publication_collection_receipt(changed_function)
    for name, value in vars(table).items():
        if name != "_lock":
            check_metadata_tree(value)
