"""Owner-table atomicity for graph-associated stored outer objects."""

from __future__ import annotations

import hashlib
from dataclasses import replace

import pytest

from miniray import protocol
from miniray.contained_cycle import (
    ContainedGraphManifest, ContainedReferenceGraphAuthority,
)
from miniray.contained_edges import ContainedReferenceEdge
from miniray.ids import (
    AttemptID, JobID, LeaseID, NodeID, ObjectID, TaskID, WorkerID,
)
from miniray.ownership import (
    ObjectCollectionState, ObjectOwnerTable, ObjectState,
    StoredOwnerPublicationCollectionRequiredError,
    StoredOwnerPublicationConflictError, StoredOwnerPublicationDisposition,
    StoredOwnerPublicationPlan,
)
from miniray.resources import ResourceVector
from miniray.stored_publication import (
    StoredPublicationAdoption, StoredPublicationClaim,
    StoredPublicationEnvelope, StoredPublicationHeader, StoredPublicationID,
)
from miniray.task_outputs import TaskExecutionKey


pytestmark = pytest.mark.unit


class _Fixture:
    def __init__(self) -> None:
        self.job = JobID.random()
        self.task = TaskID.derive(
            self.job, TaskID.for_driver(self.job), 41
        )
        self.attempt = AttemptID(self.task, 0)
        self.owner = WorkerID.random()
        self.executor = WorkerID.random()
        self.node = NodeID.random()
        self.spec = protocol.TaskSpec(
            self.job, self.task, self.attempt,
            protocol.FunctionKey(self.job, __name__, "producer", "v1"),
            (), 1, ResourceVector(), self.owner, max_retries=1,
        )
        self.execution = TaskExecutionKey.from_task_spec(self.spec)
        self.outer = self.execution.output_ids[0]
        self.publication_id = StoredPublicationID(
            LeaseID.random(), self.task, self.attempt, self.outer
        )
        self.header = StoredPublicationHeader(
            self.publication_id, self.job, self.executor, self.owner, self.node
        )
        raw_edges = tuple(
            ContainedReferenceEdge(
                self.outer,
                ObjectID.for_task(TaskID.derive(
                    self.job, TaskID.for_driver(self.job), index + 70
                )),
                WorkerID.random(), ("127.0.0.1", 34700 + index),
                f"stored-final-{index}",
            )
            for index in range(2)
        )
        # Manifest order is semantically frozen and need not equal the local
        # collection plan's canonical sorted order.
        self.edges = tuple(reversed(sorted(raw_edges)))
        self.manifest_digest = hashlib.sha256(b"manifest").hexdigest()
        self.graph = ContainedGraphManifest(
            self.publication_id.graph_transaction_id, self.publication_id,
            self.owner, self.manifest_digest, self.edges,
        )
        payload = b"stored outer payload"
        self.descriptor = protocol.ResultDescriptor(
            self.outer, protocol.ResultStorage.OBJECT_STORE, len(payload),
            self.owner, self.node, hashlib.sha256(payload).hexdigest(),
        )
        self.envelope = StoredPublicationEnvelope(
            self.publication_id, self.manifest_digest, self.descriptor,
            self.graph, self.header,
        )
        self.claim = StoredPublicationClaim(
            self.publication_id, "claim:stored-owner", self.owner,
            self.manifest_digest,
        )
        self.adoption = StoredPublicationAdoption(
            self.claim, "owner-commit:stored-owner"
        )
        self.plan = StoredOwnerPublicationPlan(
            self.execution, self.envelope, self.adoption
        )

    def owner_table(self, *, local_token: str | None = None) -> ObjectOwnerTable:
        table = ObjectOwnerTable()
        table.register_task_outputs(
            self.spec, local_tokens=(local_token,)
        )
        return table


def test_descriptor_edges_and_adoption_commit_under_one_owner_transaction() -> None:
    fixture = _Fixture()
    table = fixture.owner_table(local_token="outer-handle")

    assert table.validate_stored_owner_publication(fixture.plan) is (
        StoredOwnerPublicationDisposition.APPLIED
    )
    pending = table.snapshot(fixture.outer)
    assert pending.state is ObjectState.PENDING
    assert pending.canonical_stored_result is None
    assert not pending.outgoing_contained_edges
    assert pending.stored_publication is None

    receipt = table.commit_stored_owner_publication(fixture.plan)
    assert receipt.plan == fixture.plan and receipt.committed
    assert receipt.disposition is StoredOwnerPublicationDisposition.APPLIED
    ready = table.snapshot(fixture.outer)
    assert ready.state is ObjectState.READY_STORED
    assert ready.canonical_stored_result == fixture.descriptor
    assert ready.locations == frozenset({fixture.node})
    assert ready.outgoing_contained_edges == frozenset(fixture.edges)
    assert ready.stored_publication == fixture.plan
    assert table.stored_owner_publication(fixture.outer) == fixture.plan

    replay = table.commit_stored_owner_publication(fixture.plan)
    assert replay.disposition is (
        StoredOwnerPublicationDisposition.ALREADY_APPLIED
    )
    assert table.stored_owner_publication_receipt(fixture.outer).plan == (
        fixture.plan
    )


def test_stale_attempt_is_fenced_and_conflicting_replay_changes_nothing() -> None:
    fixture = _Fixture()
    next_spec = replace(fixture.spec, attempt_id=fixture.attempt.next())
    stale_table = ObjectOwnerTable()
    stale_table.register_task_outputs(next_spec)
    before = stale_table.snapshot(fixture.outer)

    stale = stale_table.commit_stored_owner_publication(fixture.plan)
    assert stale.disposition is StoredOwnerPublicationDisposition.FENCED
    assert not stale.committed
    assert stale_table.snapshot(fixture.outer) == before

    table = fixture.owner_table()
    table.commit_stored_owner_publication(fixture.plan)
    committed = table.snapshot(fixture.outer)
    conflict = replace(
        fixture.plan,
        adoption=replace(
            fixture.adoption, owner_commit_id="owner-commit:changed"
        ),
    )
    with pytest.raises(
        StoredOwnerPublicationConflictError, match="rebound"
    ):
        table.commit_stored_owner_publication(conflict)
    assert table.snapshot(fixture.outer) == committed


def test_preexisting_partial_publication_is_rejected_without_more_mutation() -> None:
    fixture = _Fixture()
    table = fixture.owner_table()
    assert table.add_outgoing_contained_edge(
        fixture.outer, fixture.edges[0]
    )
    before = table.snapshot(fixture.outer)

    with pytest.raises(
        StoredOwnerPublicationConflictError, match="partially"
    ):
        table.commit_stored_owner_publication(fixture.plan)
    assert table.snapshot(fixture.outer) == before


def test_adopted_edge_manifest_is_immutable() -> None:
    fixture = _Fixture()
    table = fixture.owner_table()
    table.commit_stored_owner_publication(fixture.plan)

    assert not table.add_outgoing_contained_edge(
        fixture.outer, fixture.edges[0]
    )
    extra = replace(fixture.edges[0], transfer_token="extra")
    with pytest.raises(
        StoredOwnerPublicationConflictError, match="cannot change edges"
    ):
        table.add_outgoing_contained_edge(fixture.outer, extra)
    assert table.snapshot(fixture.outer).outgoing_contained_edges == (
        frozenset(fixture.edges)
    )


def test_collection_reuses_generic_freeze_but_requires_exact_graph_release() -> None:
    fixture = _Fixture()
    table = fixture.owner_table(local_token="outer-handle")
    table.commit_stored_owner_publication(fixture.plan)
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.graph)
    graph.commit_manifest(fixture.graph)

    assert table.begin_stored_publication_collection(
        fixture.outer, collection_id="stored-gc:1"
    ) is None
    assert table.release_local_reference(fixture.outer, "outer-handle")
    frozen = table.begin_stored_publication_collection(
        fixture.outer, collection_id="stored-gc:1"
    )
    assert frozen is not None
    assert frozen.publication == fixture.plan
    assert frozen.metadata_plan == table.begin_collection(
        fixture.outer, collection_id="stored-gc:1",
        canonical_size_bytes=fixture.descriptor.size_bytes,
        canonical_checksum=fixture.descriptor.checksum,
    )
    assert tuple(frozen.metadata_plan.contained_releases) == tuple(
        sorted(fixture.edges)
    )

    with pytest.raises(
        StoredOwnerPublicationCollectionRequiredError, match="graph release"
    ):
        table.complete_collection(frozen.metadata_plan)
    assert table.collection_state(fixture.outer) is (
        ObjectCollectionState.COLLECTING
    )

    # Simulate a lost first GCS reply: terminal graph replay still echoes the
    # original serialization order and is sufficient proof for owner commit.
    graph.release_manifest_container(fixture.graph, fixture.outer)
    graph_receipt = graph.release_manifest_container(
        fixture.graph, fixture.outer
    )
    assert tuple(graph_receipt.released_edges) == fixture.edges
    completed = table.complete_stored_publication_collection(
        frozen, graph_receipt
    )
    assert completed.disposition is StoredOwnerPublicationDisposition.APPLIED
    assert completed.collection.contained_releases == tuple(
        sorted(fixture.edges)
    )
    assert table.collection_state(fixture.outer) is (
        ObjectCollectionState.COLLECTED
    )
    assert table.stored_owner_publication(fixture.outer) is None

    replay = table.complete_stored_publication_collection(
        frozen, graph_receipt
    )
    assert replay.disposition is (
        StoredOwnerPublicationDisposition.ALREADY_APPLIED
    )
    assert table.stored_publication_collection_receipt(
        fixture.outer
    ).collection == completed.collection
    assert table.begin_stored_publication_collection(
        fixture.outer, collection_id="stored-gc:1"
    ) == frozen


def test_wrong_graph_release_cannot_delete_frozen_owner_metadata() -> None:
    fixture = _Fixture()
    table = fixture.owner_table()
    table.commit_stored_owner_publication(fixture.plan)
    frozen = table.begin_stored_publication_collection(
        fixture.outer, collection_id="stored-gc:wrong-proof"
    )
    assert frozen is not None
    graph = ContainedReferenceGraphAuthority()
    graph.prepare_manifest(fixture.graph)
    graph.commit_manifest(fixture.graph)
    release = graph.release_manifest_container(fixture.graph, fixture.outer)
    wrong = replace(
        release, manifest=replace(
            fixture.graph, transaction_id="contained-graph:wrong"
        )
    )

    with pytest.raises(
        StoredOwnerPublicationConflictError, match="exact stored manifest"
    ):
        table.complete_stored_publication_collection(frozen, wrong)
    assert table.collection_state(fixture.outer) is (
        ObjectCollectionState.COLLECTING
    )
    assert table.stored_owner_publication(fixture.outer) == fixture.plan
