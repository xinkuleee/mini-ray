class _Harness:
    def __init__(self, *, edge_count: int = 1) -> None:
        job = JobID.random()
        task = TaskID.derive(job, TaskID.for_driver(job), 31)
        attempt = AttemptID(task, 0)
        outer = ObjectID.for_task(task)
        lease = LeaseID.random()
        self.node_id = NodeID.random()
        self.executor = WorkerID.random()
        self.owner = WorkerID.random()
        publication_id = InlinePublicationID(lease, task, attempt, outer)
        edges = tuple(
            ContainedReferenceEdge(
                outer,
                ObjectID.for_task(TaskID.derive(
                    job, TaskID.for_driver(job), 40 + index
                )),
                self.executor, ("127.0.0.1", 33000 + index),
                "inline-node:{}".format(index),
            )
            for index in range(edge_count)
        )
        self.envelope = InlinePublicationEnvelope.create(
            publication_id, job, self.executor, self.owner, self.node_id,
            b"inline-node-server", edges,
        )
        request = protocol.RequestWorkerLease(
            lease, task, attempt, ResourceVector.of(CPU=1), NodeID.random(),
            self.owner, target_node_id=self.node_id, return_ids=(outer,),
        )
        grant = protocol.GrantWorkerLease(
            lease, task, attempt, self.node_id, self.executor,
            ("127.0.0.1", 33100), AllocationToken.random(),
        )
        self.record = _LeaseRecord(
            request, grant.allocation_token, grant,
            state=protocol.LeaseExecutionState.RUNNING,
        )
        self.node = object.__new__(NodeServer)
        self.node.node_id = self.node_id
        self.node._node_pid = 4321
        self.node._registration_epoch = 7
        self.node._leases = {lease: self.record}
        self.node._state_lock = threading.RLock()
        self.node._ledger = ResourceLedger(ResourceVector.of(CPU=1))
        self.node._ledger.allocate(request.resources, grant.allocation_token)
        self.node._cluster_nodes = ()
        self.node._worker_order = (self.executor,)
        self.node._shutdown_request_id = None
        self.node._inline_publication_journal = InlinePublicationNodeJournal()
        self.node._stored_publication_journal = None
        self.node._workers = {
            self.executor: _WorkerSlot(
                self.executor,
                process=SimpleNamespace(is_alive=lambda: True),
                active_lease_id=lease,
            )
        }
        self.graph = ContainedReferenceGraphAuthority()
        self.events: list[str] = []
        self.pins = set()
        self.commit_calls = 0
        self.recovery = InlinePublicationRecoveryAuthority()
        self.recovery_reports: list[object] = []
        self.timeline: list[str] = []

        def recovery_rpc(_handler, request):
            self.recovery_reports.append(request)
            if isinstance(request, protocol.ReportInlinePublicationIntent):
                self.timeline.append("intent")
                ack = self.recovery.report_intent(request.intent)
            elif isinstance(
                request, protocol.ReportInlinePublicationPrepared
            ):
                self.timeline.append("prepared")
                ack = self.recovery.report_prepared(
                    request.publication_id
                )
            elif isinstance(
                request, protocol.ReportInlinePublicationTerminal
            ):
                self.timeline.append("terminal")
                ack = self.recovery.report_terminal(request.terminal)
            elif isinstance(
                request, protocol.AcknowledgeInlinePublicationAborted
            ):
                self.timeline.append("aborted")
                ack = self.recovery.acknowledge_aborted(
                    request.publication_id, request.manifest_digest
                )
            else:
                raise AssertionError(
                    "Node sent a recovery transition it does not own"
                )
            return protocol.InlinePublicationRecoveryReply(
                request, ack=ack
            )

        self.recovery_rpc = recovery_rpc
        self.node._stored_gcs_rpc = recovery_rpc

        def prepare_graph(request):
            self.events.append("prepare-graph")
            self.timeline.append("prepare-graph")
            return protocol.ContainedGraphReply(
                request, self.graph.prepare_manifest(request.manifest)
            )

        def reject_commit(_request):
            self.commit_calls += 1
            raise AssertionError("Node must not commit INLINE graph")

        def abort_graph(request):
            self.events.append("abort-graph")
            self.timeline.append("abort-graph")
            return protocol.ContainedGraphReply(
                request, self.graph.abort_manifest(request.manifest)
            )

        def install(request):
            self.events.append("install:{}".format(request.effect.index))
            self.timeline.append("install:{}".format(request.effect.index))
            changed = request.effect not in self.pins
            self.pins.add(request.effect)
            return InlineChildEffectReply(request, True, changed)

        def release(request):
            self.events.append("release:{}".format(request.effect.index))
            self.timeline.append("release:{}".format(request.effect.index))
            changed = request.effect in self.pins
            self.pins.discard(request.effect)
            return InlineChildEffectReply(request, True, changed)

        self.node._inline_publications = InlinePublicationNodeRuntime(
            self.node._inline_publication_journal,
            prepare_graph=prepare_graph, commit_graph=reject_commit,
            abort_graph=abort_graph, install_child=install,
            release_child=release,
        )

    @property
    def publication_id(self):
        return self.envelope.publication_id

    def open(self):
        return self.node._handle_open_inline_contained_publication(
            protocol.OpenInlineContainedPublication(self.envelope)
        )

    def prepare(self):
        return self.node._handle_prepare_inline_contained_publication(
            protocol.PrepareInlineContainedPublication(
                self.publication_id, self.envelope.manifest.manifest_digest
            )
        )

    def complete(self, status=protocol.TaskReplyStatus.SUCCEEDED):
        return self.node._handle_complete_worker_lease(
            self.completion_request(status)
        )

    def completion_request(self, status=protocol.TaskReplyStatus.SUCCEEDED):
        return protocol.CompleteWorkerLease(
            self.record.request.lease_id, self.record.request.task_id,
            self.record.request.attempt_id, self.executor, status,
        )

    def outcome_request(self):
        return protocol.GetWorkerLeaseOutcome(
            self.record.request.lease_id, self.record.request.task_id,
            self.record.request.attempt_id, self.executor, self.owner,
            self.record.request.return_ids,
        )

    def lose_worker(self):
        self.node._workers[self.executor].process = SimpleNamespace(
            is_alive=lambda: False
        )
        with self.node._state_lock:
            return self.node._reclaim_active_lease_after_worker_exit_locked(
                self.executor
            )


@pytest.mark.unit
def test_open_prepare_bind_order_and_authority_exclusion() -> None:
    h = _Harness(edge_count=2)
    opened = h.open()
    assert opened.accepted and opened.changed
    assert h.record.inline_publication_id == h.publication_id
    assert h.record.stored_publication_id is None

    prepared = h.prepare()
    assert prepared.accepted and prepared.changed
    assert h.events == ["prepare-graph", "install:0", "install:1"]
    assert h.node._inline_publication_journal.snapshot(
        h.publication_id
    ).pin_bitmap == (True, True)
    assert h.timeline == [
        "intent", "prepare-graph", "install:0", "install:1",
        "prepared",
    ]
    recovery = h.recovery.snapshot(h.publication_id)
    assert recovery.intent.manifest == h.envelope.manifest
    assert not hasattr(recovery.intent, "envelope")
    assert recovery.prepared


@pytest.mark.unit
def test_open_ack_loss_replays_exact_intent_before_any_effect() -> None:
    h = _Harness()
    calls = 0

    def lose_first_reply(handler, request):
        nonlocal calls
        calls += 1
        reply = h.recovery_rpc(handler, request)
        if calls == 1:
            raise TimeoutError("intent ACK lost after apply")
        return reply

    h.node._stored_gcs_rpc = lose_first_reply
    with pytest.raises(TimeoutError, match="intent ACK lost"):
        h.open()
    assert not h.record.inline_publication_intent_acked
    assert h.events == []

    replay = h.open()

    assert replay.accepted and not replay.changed
    assert h.record.inline_publication_intent_acked
    assert calls == 2
    assert h.events == []

@pytest.mark.unit
def test_prepare_failure_is_trusted_only_after_explicit_rollback() -> None:
    h = _Harness(edge_count=2)
    h.open()
    calls = 0

    def fail_second(request):
        nonlocal calls
        calls += 1
        h.pins.add(request.effect)
        if request.effect.index == 1:
            raise ValueError("explicit pin rejection")
        return InlineChildEffectReply(request, True, True)

    h.node._inline_publications._install_child = fail_second
    reply = h.prepare()

    assert not reply.accepted
    assert "explicit pin rejection" in (reply.error or "")
    snapshot = h.node._inline_publication_journal.snapshot(h.publication_id)
    assert snapshot.rollback_finished
    assert not h.pins
    assert h.events[-3:] == ["release:1", "release:0", "abort-graph"]
    assert h.timeline[-4:] == [
        "release:1", "release:0", "abort-graph", "aborted",
    ]
    assert h.record.inline_publication_aborted_acked
    assert (
        h.recovery.snapshot(h.publication_id).resolution
        is InlineRecoveryResolution.ABORTED
    )
    failed = h.complete(protocol.TaskReplyStatus.SYSTEM_ERROR)
    assert failed.accepted and failed.released
    assert failed.inline_publication is None
    assert h.commit_calls == 0

@pytest.mark.unit
def test_worker_loss_resumes_an_ambiguous_explicit_abort() -> None:
    h = _Harness(edge_count=2)
    h.open(); h.prepare()
    original_abort = h.node._inline_publications._abort_graph
    calls = 0

    def lose_first_abort(request):
        nonlocal calls
        calls += 1
        if calls == 1:
            original_abort(request)
            raise TimeoutError("ABORT ACK lost")
        return original_abort(request)

    h.node._inline_publications._abort_graph = lose_first_abort
    with pytest.raises(TimeoutError, match="ABORT ACK lost"):
        h.node._inline_publications.abort(
            h.node._inline_binding_for_record(h.record, h.node_id),
            h.publication_id,
        )
    snapshot = h.node._inline_publication_journal.snapshot(h.publication_id)
    assert snapshot.explicit_abort_started and snapshot.rollback_started
    assert not snapshot.rollback_finished

    h.record.state = protocol.LeaseExecutionState.WORKER_LOST
    assert h.node._drive_terminal_inline_publications(h.executor)
    terminal = h.node._inline_publication_journal.snapshot(h.publication_id)
    assert terminal.rollback_finished
    assert terminal.worker_loss is None
    assert calls == 2

@pytest.mark.unit
def test_outcome_never_exposes_worker_lost_after_complete_boundary(
) -> None:
    h = _Harness()
    h.open(); h.prepare()

    def unavailable_terminal(_handler, request):
        if isinstance(request, protocol.ReportInlinePublicationTerminal):
            raise TimeoutError("GCS terminal report unavailable")
        return h.recovery_rpc(_handler, request)

    h.node._stored_gcs_rpc = unavailable_terminal
    complete = h.completion_request()
    h.record.inline_publication_complete_inflight = complete
    h.node._inline_publication_journal.record_complete(
        InlinePublicationCompleteBoundary(
            h.node._inline_binding_for_record(h.record, h.node_id),
            h.publication_id,

        "owner-terminal-race",
        protocol.WorkerIncarnation(NodeID.random(), 9101, 1, h.owner, 9102),
        1, 23, protocol.WorkerDeathReason.PROCESS_EXIT,
    )
    effect = FinalizePublicationOwnerDeathEffect(

    )

            protocol.FinalizePublicationOwnerDeath.from_effect(effect)

        return reply

    h.node._stored_gcs_rpc = finalize_after_apply

    assert h.node._drive_terminal_inline_publications()
    snapshot = h.node._inline_publication_journal.snapshot(h.publication_id)

    assert not h.record.inline_publication_terminal_acked
    assert h.record.inline_publication_complete_inflight is None
    assert h.node._inline_publications_clean_locked()
    assert h.node._drive_terminal_inline_publications()

@pytest.mark.loopback_smoke
def test_pending_terminal_rpc_does_not_lock_out_complete_or_outcome() -> None:
    """Real bounded concurrency gate; deliberately excluded from unit runs."""

    h = _Harness()
    h.open(); h.prepare()
    complete = h.completion_request()
    first = h.node._handle_complete_worker_lease(complete)
    assert first.accepted and first.released
    terminal_entered = threading.Event()
    allow_terminal_ack = threading.Event()
    original_rpc = h.recovery_rpc

    def block_terminal(handler, request):
        if isinstance(request, protocol.ReportInlinePublicationTerminal):
            terminal_entered.set()
            assert allow_terminal_ack.wait(1.0)
        return original_rpc(handler, request)

    h.node._stored_gcs_rpc = block_terminal
    complete_replies = []
    outcomes = []
    driver_results = []
    errors = []

    def drive():
        try:
            driver_results.append(
                h.node._drive_terminal_inline_publications(h.executor)
            )
        except BaseException as exc:
            errors.append(exc)

    def replay_and_query():
        try:
            complete_replies.append(
                h.node._handle_complete_worker_lease(complete)
            )
            outcomes.append(h.node._handle_get_worker_lease_outcome(
                h.outcome_request()
            ))
        except BaseException as exc:
            errors.append(exc)

    driver_thread = threading.Thread(target=drive, daemon=True)
    read_thread = threading.Thread(target=replay_and_query, daemon=True)
    driver_thread.start()
    try:
        assert terminal_entered.wait(1.0)
        assert not h.lose_worker()
        read_thread.start()
        read_thread.join(0.25)
        assert not read_thread.is_alive()
        assert driver_thread.is_alive()
        assert not h.record.inline_publication_terminal_acked
    finally:
        allow_terminal_ack.set()
        driver_thread.join(1.0)
        if read_thread.ident is not None:
            read_thread.join(1.0)

    assert not driver_thread.is_alive()
    assert not read_thread.is_alive()
    assert not errors
    assert complete_replies[0].accepted
    assert not complete_replies[0].released
    assert complete_replies[0].inline_publication == h.envelope
    assert outcomes[0].inline_publication == h.envelope
    assert outcomes[0].state is protocol.LeaseExecutionState.COMPLETED
    assert not outcomes[0].worker_alive
    assert driver_results == [True]
    assert h.record.state is protocol.LeaseExecutionState.COMPLETED
    assert h.record.completion == complete
    assert h.record.inline_publication_complete_inflight is None
    assert h.commit_calls == 0
