def _death(owner: WorkerID, *, identity: str = "owner-death"):
    return protocol.WorkerDeathRecord(
        identity,
        protocol.WorkerIncarnation(
            NodeID.random(), 8101, 3, owner, 8102
        ),
        9, -9, protocol.WorkerDeathReason.PROCESS_EXIT,
    )


def _lease_record(
    node: NodeServer, owner: WorkerID, executor: WorkerID,
    lease_id: LeaseID, task_id: TaskID, attempt_id: AttemptID,
    outer: ObjectID,
) -> _LeaseRecord:
    resources = ResourceVector({"CPU": 1})
    request = protocol.RequestWorkerLease(
        lease_id, task_id, attempt_id, resources, NodeID.random(),
        owner, target_node_id=node.node_id, return_ids=(outer,),
    )
    token = AllocationToken.random()
    node._ledger.allocate(resources, token)
    grant = protocol.GrantWorkerLease(
        lease_id, task_id, attempt_id, node.node_id, executor,
        ("127.0.0.1", 38101), token,
    )
    record = _LeaseRecord(
        request, token, grant,
        state=protocol.LeaseExecutionState.RUNNING,
    )
    node._workers[executor] = _WorkerSlot(
        executor, process=object(), address=grant.worker_address,
        active_lease_id=lease_id,
    )
    node._worker_order = tuple(node._workers)
    return record


def _node() -> NodeServer:
    node = object.__new__(NodeServer)
    node.node_id = NodeID.random()
    node._state_lock = threading.RLock()
    node._leases = {}
    total = ResourceVector({"CPU": 1})
    node._ledger = ResourceLedger(total)
    node._cluster_nodes = (NodeSnapshot(node.node_id, total, total),)
    node._workers = {}
    node._worker_order = ()
    node._legacy_worker_compat = False
    node._dependency_pin_cleanups = {}
    node._object_store = ObjectStore(1024)
    node._placement_group_digests = {}
    node._resource_report_version = 0
    node._owner_death_fences = {}
    node._owner_death_fence_outcomes = {}
    node._publication_owner_death_finalize_outcomes = {}
    node._object_localization_locks = {}
    node._stored_publication_locks = {}
    node._stored_publication_journal = StoredPublicationJournal()
    node._inline_publication_journal = InlinePublicationNodeJournal()
    return node


def _install_owner_fence(
    node: NodeServer, death: protocol.WorkerDeathRecord,
) -> None:
    reply = node._handle_install_owner_death_fence(
        protocol.InstallOwnerDeathFence(
            "fence:{}".format(death.detection_id), death, node.node_id
        )
    )
    assert reply.accepted


def _inline_fixture(node: NodeServer, owner: WorkerID):
    job = JobID.random()
    task = TaskID.derive(job, TaskID.for_driver(job), 1)
    attempt = AttemptID(task, 0)
    lease = LeaseID.random()
    outer = ObjectID.for_task(task)
    executor = WorkerID.random()
    publication_id = InlinePublicationID(lease, task, attempt, outer)
    child = ObjectID.for_task(
        TaskID.derive(job, TaskID.for_driver(job), 2)
    )
    edge = ContainedReferenceEdge(
        outer, child, executor, ("127.0.0.1", 38102),
        "owner-death-inline",
    )
    envelope = InlinePublicationEnvelope.create(
        publication_id, job, executor, owner, node.node_id,
        b"inline owner-death", (edge,),
    )
    record = _lease_record(node, owner, executor, lease, task, attempt, outer)
    record.inline_publication_id = publication_id
    node._leases[lease] = record
    node._inline_publication_journal.open(
        InlineExecutionBinding(
            lease, task, attempt, executor, node.node_id
        ),
        envelope,
    )
    return publication_id, record


def _stored_fixture(node: NodeServer, owner: WorkerID):
    job = JobID.random()
    task = TaskID.derive(job, TaskID.for_driver(job), 11)
    attempt = AttemptID(task, 0)
    lease = LeaseID.random()
    outer = ObjectID.for_task(task)
    executor = WorkerID.random()
    publication_id = StoredPublicationID(lease, task, attempt, outer)
    header = StoredPublicationHeader(
        publication_id, job, executor, owner, node.node_id
    )
    child = ObjectID.for_task(
        TaskID.derive(job, TaskID.for_driver(job), 12)
    )
    transfer = PreparedContainedTransfer(
        child, executor, ("127.0.0.1", 38103),
        OwnedContainedSource(executor),
        ContainedReferenceHold(outer, executor, "owner-death-stored"),
        ContainedReferenceHold(outer, owner, "owner-death-stored"),
    )
    payload = b"stored owner-death"
    manifest = StoredPublicationManifest(
        publication_id, (transfer,), len(payload),
        hashlib.sha256(payload).hexdigest(),
        hashlib.sha256(b"owner-death-manifest").hexdigest(),
    )
    record = _lease_record(node, owner, executor, lease, task, attempt, outer)
    record.stored_publication_id = publication_id
    node._leases[lease] = record
    node._stored_publication_journal.open_manifest(
        header, manifest, 4321, 7
    )
    return publication_id, record


def _request(
    request_id: str, key: object, death: protocol.WorkerDeathRecord,
    node_id: NodeID,
    history: PublicationRecoveryHistory = (
        PublicationRecoveryHistory.NOT_COMMITTED
    ),
) -> protocol.FinalizePublicationOwnerDeath:
    effect = FinalizePublicationOwnerDeathEffect(
        request_id, key, death, node_id, history
    )
    return protocol.FinalizePublicationOwnerDeath.from_effect(effect)


@pytest.mark.parametrize("tier", ["inline", "stored"])
def test_node_finalize_installs_independent_terminal_and_exact_replay(
    tier: str,
) -> None:
    node = _node()
    owner = WorkerID.random()
    death = _death(owner)
    _install_owner_fence(node, death)
    if tier == "inline":
        publication_id, record = _inline_fixture(node, owner)
        key = InlinePublicationOwnerDeathKey(publication_id)
    else:
        publication_id, record = _stored_fixture(node, owner)
        key = StoredPublicationOwnerDeathKey(publication_id)
    request = _request(
        "publisher-finalize:{}".format(tier), key, death, node.node_id
    )

    assert pickle.loads(pickle.dumps(request)) == request
    first = node._handle_finalize_publication_owner_death(request)
    replay = node._handle_finalize_publication_owner_death(request)

    assert first.accepted and first.changed
    assert first.terminal is PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED
    assert replay is first
    assert record.state is protocol.LeaseExecutionState.ABANDONED
    assert record.completion is None
    allocation = node._ledger.record(record.allocation_token)
    assert allocation is not None
    assert allocation.state is AllocationState.RELEASED
    assert node._workers[record.grant.worker_id].active_lease_id is None
    terminal = (
        node._inline_publication_journal.snapshot(publication_id)
        if tier == "inline" else
        node._stored_publication_journal.snapshot(publication_id)
    )
    assert terminal.owner_death_cleanup.publication_history is (
        PublicationRecoveryHistory.NOT_COMMITTED
    )
    if tier == "inline":
        assert terminal.phase is InlineNodePublicationPhase.OWNER_DEATH_CLEANED
        assert not terminal.adoption_exposed
        assert node._inline_publications_clean_locked()
    else:
        assert terminal.state is StoredPublicationState.OWNER_DEATH_CLEANED
        assert terminal.adoption is None
        assert node._stored_publications_clean_locked()
    assert record.inline_publication_id is None or tier == "inline"
    assert record.stored_publication_id is None or tier == "stored"


def test_node_finalize_rejects_request_id_owner_history_and_tier_conflicts() -> None:
    node = _node()
    owner = WorkerID.random()
    death = _death(owner)
    _install_owner_fence(node, death)
    publication_id, _ = _inline_fixture(node, owner)
    key = InlinePublicationOwnerDeathKey(publication_id)
    request = _request(
        "publisher-finalize-conflict", key, death, node.node_id
    )
    assert node._handle_finalize_publication_owner_death(request).accepted

    changed_history = _request(
        request.request_id, key, death, node.node_id,
        PublicationRecoveryHistory.COMMIT_UNKNOWN,
    )
    assert not node._handle_finalize_publication_owner_death(
        changed_history
    ).accepted

    wrong_owner = _death(WorkerID.random(), identity="wrong-owner")
    wrong_owner_request = _request(
        "wrong-owner-finalize", key, wrong_owner, node.node_id
    )
    assert not node._handle_finalize_publication_owner_death(
        wrong_owner_request
    ).accepted

    rebound_request_id = _request(
        "another-finalize-id", key, death, node.node_id
    )
    assert not node._handle_finalize_publication_owner_death(
        rebound_request_id
    ).accepted

    stored_key = StoredPublicationOwnerDeathKey(StoredPublicationID(
        publication_id.lease_id, publication_id.task_id,
        publication_id.attempt_id, publication_id.outer_object_id,
    ))
    wrong_tier = _request(
        "wrong-tier-finalize", stored_key, death, node.node_id
    )
    assert not node._handle_finalize_publication_owner_death(
        wrong_tier
    ).accepted


def test_finalize_wire_revalidates_corrupted_typed_key() -> None:
    node = _node()
    owner = WorkerID.random()
    death = _death(owner)
    publication_id, _ = _inline_fixture(node, owner)
    request = _request(
        "corrupt-finalize",
        InlinePublicationOwnerDeathKey(publication_id), death, node.node_id,
    )
    object.__setattr__(request.key, "publication_id", object())

    with pytest.raises(ProtocolError, match="identity is invalid"):
        pickle.loads(pickle.dumps(request))


def test_finalize_releases_dependency_pins_and_fences_late_inline_steps() -> None:
    node = _node()
    owner = WorkerID.random()
    publication_id, record = _inline_fixture(node, owner)
    dependency_job = JobID.random()
    dependency = ObjectID(
        TaskID.derive(
            dependency_job, TaskID.for_driver(dependency_job), 1
        ),
        0,
    )
    node._object_store.put(dependency, b"dependency")
    pin = ("lease-dependency", publication_id.lease_id, dependency)
    node._object_store.pin(dependency, pin)
    record.dependency_pins = ((dependency, pin),)
    record.inline_publication_complete_inflight = protocol.CompleteWorkerLease(
        record.request.lease_id, record.request.task_id,
        record.request.attempt_id, record.grant.worker_id,
        protocol.TaskReplyStatus.SUCCEEDED,
    )
    death = _death(owner)
    _install_owner_fence(node, death)

    request = _request(
        "publisher-finalize:inline-resource",
        InlinePublicationOwnerDeathKey(publication_id), death, node.node_id,
    )
    assert node._handle_finalize_publication_owner_death(request).accepted

    assert record.state is protocol.LeaseExecutionState.ABANDONED
    assert record.inline_publication_complete_inflight is None
    assert node._object_store.snapshot(dependency).pin_count == 0
    envelope = node._inline_publication_journal.snapshot(
        publication_id
    ).envelope
    opened = node._handle_open_inline_contained_publication(
        protocol.OpenInlineContainedPublication(envelope)
    )
    prepared = node._handle_prepare_inline_contained_publication(
        protocol.PrepareInlineContainedPublication(
            publication_id, envelope.manifest.manifest_digest
        )
    )
    completed = node._handle_complete_worker_lease_inner(
        protocol.CompleteWorkerLease(
            record.request.lease_id, record.request.task_id,
            record.request.attempt_id, record.grant.worker_id,
            protocol.TaskReplyStatus.SUCCEEDED,
        )
    )
    assert not opened.accepted and "fenced" in opened.error
    assert not prepared.accepted and "fenced" in prepared.error
    assert not completed.accepted and "fenced" in completed.error
    assert record.state is protocol.LeaseExecutionState.ABANDONED
    assert record.completion is None


def test_finalize_fences_late_stored_complete_and_claim() -> None:
    node = _node()
    owner = WorkerID.random()
    publication_id, record = _stored_fixture(node, owner)
    record.publication_complete_inflight = protocol.CompleteWorkerLease(
        record.request.lease_id, record.request.task_id,
        record.request.attempt_id, record.grant.worker_id,
        protocol.TaskReplyStatus.SUCCEEDED,
    )
    death = _death(owner)
    _install_owner_fence(node, death)
    request = _request(
        "publisher-finalize:stored-fence",
        StoredPublicationOwnerDeathKey(publication_id), death, node.node_id,
    )
    assert node._handle_finalize_publication_owner_death(request).accepted
    assert record.publication_complete_inflight is None

    complete = node._handle_complete_worker_lease_inner(
        protocol.CompleteWorkerLease(
            record.request.lease_id, record.request.task_id,
            record.request.attempt_id, record.grant.worker_id,
            protocol.TaskReplyStatus.SUCCEEDED,
        )
    )
    assert not complete.accepted and "fenced" in complete.error

    snapshot = node._stored_publication_journal.snapshot(publication_id)
    descriptor = protocol.ResultDescriptor(
        publication_id.outer_object_id, protocol.ResultStorage.OBJECT_STORE,
        snapshot.publication_intent.manifest.size_bytes, owner, node.node_id,
        snapshot.publication_intent.manifest.checksum,
    )
    envelope = StoredPublicationEnvelope(
        publication_id, snapshot.publication_intent.manifest.manifest_digest,
        descriptor,
        snapshot.publication_intent.manifest.graph_manifest(owner),
        snapshot.header,
    )
    claim = StoredPublicationClaim(
        publication_id, "late-claim", owner, envelope.manifest_digest
    )
    claim_request = StoredPublicationClaimRequest(
        TaskExecutionKey(
            TaskOutputManifest(publication_id.task_id,
                               (publication_id.outer_object_id,)),
            publication_id.attempt_id,
        ),
        envelope, claim,
    )
    assert "fenced" in node._stored_completed_claim_error_locked(
        claim_request
    )
