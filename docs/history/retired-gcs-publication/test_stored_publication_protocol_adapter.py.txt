def test_graph_adapter_echoes_identity_disposition_and_query() -> None:
    fixture = _Fixture()
    adapter = StoredPublicationControlAdapter()
    prepare = protocol.PrepareStoredPublicationGraph(fixture.graph)

    first = adapter.mutate_graph(prepare)
    replay = adapter.mutate_graph(prepare)
    query = protocol.GetStoredPublicationGraph(
        fixture.graph.transaction_id
    )
    found = adapter.get_graph(query)

    assert first.request is prepare and first.accepted
    assert first.receipt.disposition.value == "APPLIED"
    assert replay.request is prepare
    assert replay.receipt.disposition.value == "ALREADY_PREPARED"
    assert found.request is query
    assert found.disposition is protocol.StoredPublicationQueryDisposition.FOUND
    assert found.manifest == fixture.graph

    commit = protocol.CommitStoredPublicationGraph(fixture.graph)
    committed = adapter.mutate_graph(commit)
    commit_replay = adapter.mutate_graph(commit)
    assert committed.receipt.manifest == fixture.graph
    assert committed.receipt.disposition.value == "APPLIED"
    assert commit_replay.receipt.disposition.value == "ALREADY_COMMITTED"

    release = protocol.ReleaseStoredPublicationGraphContainer(
        fixture.graph, fixture.publication_id.outer_object_id
    )
    released = adapter.mutate_graph(release)
    release_replay = adapter.mutate_graph(release)
    assert released.request is release
    assert released.receipt.disposition.value == "RELEASED"
    assert released.receipt.released_edges == fixture.graph.ordered_edges
    assert release_replay.receipt.disposition.value == "ALREADY_RELEASED"
    assert release_replay.receipt.released_edges == fixture.graph.ordered_edges

def test_owner_to_node_adoption_messages_round_trip_and_echo_exact_requests(
) -> None:
    fixture = _Fixture()
    execution = TaskExecutionKey(
        TaskOutputManifest(
            fixture.publication_id.task_id,
            (fixture.publication_id.outer_object_id,),
        ),
        fixture.publication_id.attempt_id,
    )
    claim = StoredPublicationClaim(
        fixture.publication_id, "claim:wire", fixture.outer_owner,
        fixture.manifest.manifest_digest,
    )
    claim_request = StoredPublicationClaimRequest(
        execution, fixture.envelope, claim
    )
    claim_wire = protocol.ClaimStoredPublication(claim_request)
    promotion_wire = protocol.ReplayStoredPublicationPromotions(
        StoredPublicationPromotionReplay(claim_request)
    )
    adoption_wire = protocol.AcknowledgeStoredPublicationAdopted(
        StoredPublicationAdoption(claim, "owner-commit:wire")
    )
    messages = (
        claim_wire,
        protocol.ClaimStoredPublicationReply(
            claim_wire, True, changed=True
        ),
        promotion_wire,
        protocol.ReplayStoredPublicationPromotionsReply(
            promotion_wire, True, changed=False
        ),
        adoption_wire,
        protocol.AcknowledgeStoredPublicationAdoptedReply(
            adoption_wire, True, changed=True
        ),
    )

    assert pickle.loads(pickle.dumps(messages)) == messages
    with pytest.raises(ProtocolError, match="changed bool"):
        protocol.ClaimStoredPublicationReply(claim_wire, True)
    with pytest.raises(ProtocolError, match="cannot report a mutation"):
        protocol.AcknowledgeStoredPublicationAdoptedReply(
            adoption_wire, False, changed=False, error="rejected"
        )

def test_worker_to_node_publication_requests_reject_cross_bound_authority(
) -> None:
    fixture = _Fixture()
    other_task = TaskID.derive(
        fixture.job, TaskID.for_driver(fixture.job), 99
    )
    other_publication = StoredPublicationID(
        LeaseID.random(), other_task, AttemptID(other_task, 0),
        ObjectID.for_task(other_task),
    )

    with pytest.raises(ProtocolError, match="another publication"):
        protocol.OpenStoredPublication(
            fixture.header,
            replace(fixture.manifest, publication_id=other_publication),
        )
    with pytest.raises(ProtocolError, match="another publication"):
        protocol.FreezeStoredPublication(
            other_publication, fixture.manifest
        )
    with pytest.raises(ProtocolError, match="another publication"):
        protocol.SealStoredPublication(
            other_publication, fixture.payload, fixture.descriptor
        )

    transfer = fixture.manifest.transfers[0]
    changed = replace(
        transfer,
        provisional_hold=replace(
            transfer.provisional_hold,
            container_object_id=other_publication.outer_object_id,
        ),
        final_hold=replace(
            transfer.final_hold,
            container_object_id=other_publication.outer_object_id,
        ),
    )
    with pytest.raises(ProtocolError, match="another outer object"):
        protocol.PrepareStoredPublicationPin(
            fixture.publication_id, 0, changed
        )

        protocol.StoredPublicationStepReply(
            protocol.PrepareStoredPublicationPin(

            ).request_identity,
            True, changed=False,
        ),
        protocol.StoredPublicationStepReply(
            protocol.FreezeStoredPublication(

            True, changed=True,

        protocol.StoredPublicationStepReply(
            protocol.PrepareNodeStoredPublicationGraph(
                fixture.publication_id

            True, graph=fixture.graph,
        ),
        protocol.StoredPublicationStepReply(
            seal.request_identity, True, descriptor=fixture.descriptor

            protocol.PromoteStoredPublicationPin(
                fixture.publication_id, 0, fixture.manifest.transfers[0]

def test_stored_node_loss_pull_reply_has_unambiguous_state_shapes() -> None:
    fixture = _Fixture()
    pull = protocol.GetStoredNodeLoss(
        fixture.publication_id, fixture.outer_owner, fixture.death
    )

    absent = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.ABSENT
    )
    pending = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.PENDING
    )
    rejected = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.REJECTED,
        error_kind=protocol.StoredPublicationRPCErrorKind.INVALID_REQUEST,
        error="owner or death authority mismatch",
    )
    coordinator, owner_snapshot = fixture.node_loss_coordinator()
    owner = owner_snapshot.next_effect
    assert isinstance(owner, RetireStoredOwnerEffect)
    owner_reply = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.OWNER_EFFECT,
        owner_snapshot, owner,
    )

    receipt = StoredOwnerRetirementReceipt(
        owner, StoredOwnerRetirementDisposition.FENCED
    )
    non_owner = coordinator.acknowledge(owner, receipt)
    progress = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.NON_OWNER_PROGRESS,
        non_owner,
    )

    assert pickle.loads(pickle.dumps((
        absent, pending, rejected, owner_reply, progress,
    ))) == (absent, pending, rejected, owner_reply, progress)

    with pytest.raises(ProtocolError, match="cannot expose takeover state"):
        protocol.GetStoredNodeLossReply(
            pull, protocol.StoredNodeLossDisposition.ABSENT, owner_snapshot
        )
    with pytest.raises(ProtocolError, match="cannot expose state"):
        protocol.GetStoredNodeLossReply(
            pull, protocol.StoredNodeLossDisposition.REJECTED, owner_snapshot,
            error_kind=protocol.StoredPublicationRPCErrorKind.INVALID_REQUEST,
            error="rejected",
        )
    with pytest.raises(ProtocolError, match="next exact owner effect"):
        protocol.GetStoredNodeLossReply(
            pull, protocol.StoredNodeLossDisposition.OWNER_EFFECT,
            non_owner, owner,
        )


def test_stored_node_loss_resolved_reply_echoes_exact_resolution() -> None:
    fixture = _Fixture()
    pull = protocol.GetStoredNodeLoss(
        fixture.publication_id, fixture.outer_owner, fixture.death
    )
    coordinator, snapshot = fixture.node_loss_coordinator()
    while snapshot.next_effect is not None:
        effect = snapshot.next_effect
        if isinstance(effect, RetireStoredOwnerEffect):
            receipt = StoredOwnerRetirementReceipt(
                effect, StoredOwnerRetirementDisposition.FENCED
            )
        elif isinstance(effect, AbortStoredGraphEffect):
            receipt = ContainedGraphManifestReceipt(
                effect.manifest, ContainedGraphTransactionState.ABORTED,
                ContainedGraphManifestDisposition.APPLIED,
            )
        elif isinstance(effect, ReleaseStoredHoldEffect):
            receipt = StoredHoldResolutionReceipt(
                effect, StoredHoldResolutionDisposition.RELEASED
            )
        elif isinstance(effect, ProveStoredReplicaGoneEffect):
            receipt = StoredReplicaResolutionReceipt(
                effect,
                StoredReplicaResolutionDisposition.DEAD_NODE_REPLICA_GONE,
            )
        else:
            raise AssertionError(type(effect).__name__)
        snapshot = coordinator.acknowledge(effect, receipt)

    assert snapshot.resolution is not None
    reply = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.RESOLVED, snapshot,
        resolution=snapshot.resolution,
    )
    assert pickle.loads(pickle.dumps(reply)) == reply

    with pytest.raises(ProtocolError, match="exact terminal resolution"):
        protocol.GetStoredNodeLossReply(
            pull, protocol.StoredNodeLossDisposition.RESOLVED, snapshot
        )


def test_stored_owner_retirement_ack_and_progress_are_exact_metadata_only() -> None:
    fixture = _Fixture()
    coordinator, owner_snapshot = fixture.node_loss_coordinator()
    owner = owner_snapshot.next_effect
    assert isinstance(owner, RetireStoredOwnerEffect)
    receipt = StoredOwnerRetirementReceipt(
        owner, StoredOwnerRetirementDisposition.FENCED
    )
    ack = protocol.AcknowledgeStoredOwnerRetirement(
        fixture.publication_id, fixture.outer_owner, fixture.death,
        owner, receipt,
    )
    after_owner = coordinator.acknowledge(owner, receipt)
    ack_reply = protocol.AcknowledgeStoredOwnerRetirementReply(
        ack, after_owner
    )
    progress = protocol.ProgressStoredNodeLoss(
        fixture.publication_id, fixture.outer_owner, fixture.death
    )
    progress_reply = protocol.ProgressStoredNodeLossReply(
        progress, after_owner, False
    )

    wire = pickle.dumps((ack, ack_reply, progress, progress_reply))
    assert fixture.payload not in wire
    assert pickle.loads(wire) == (ack, ack_reply, progress, progress_reply)

    wrong_owner = WorkerID.random()
    with pytest.raises(ProtocolError, match="exact owner effect"):
        protocol.AcknowledgeStoredOwnerRetirement(
            fixture.publication_id, wrong_owner, fixture.death, owner, receipt
        )
    with pytest.raises(ProtocolError, match="requires progressed bool"):
        protocol.ProgressStoredNodeLossReply(progress, after_owner)

class _Server:
    def __init__(self, handlers, **_configuration) -> None:
        self.handlers = dict(handlers)
        self.address = ("127.0.0.1", 29199)
        self.is_running = False

    def start(self):
        self.is_running = True
        return self.address

    def stop(self):
        self.is_running = False


def test_gcs_default_composes_pure_authorities_and_advertises_handlers(
    monkeypatch,
) -> None:
    created = []

def test_gcs_stored_graph_shutdown_is_a_replayable_cleanliness_barrier(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    fixture = _Fixture()
    shutdown = protocol.Shutdown("stored-graph-shutdown", "test")
    prepare = protocol.PrepareStoredPublicationGraph(fixture.graph)

    assert service.prepare_stored_publication_graph(prepare).accepted
    pending = service.shutdown(shutdown)
    assert not pending.clean
    assert not service._stop_event.is_set()

    # Closing admission cannot strand an in-flight exact identity.
    replay = service.prepare_stored_publication_graph(prepare)
    assert replay.receipt.disposition.value == "ALREADY_PREPARED"
    committed = service.commit_stored_publication_graph(
        protocol.CommitStoredPublicationGraph(fixture.graph)
    )
    assert committed.accepted
    release = protocol.ReleaseStoredPublicationGraphContainer(
        fixture.graph, fixture.publication_id.outer_object_id
    )
    released = service.release_stored_publication_graph_container(release)
    assert released.accepted
    assert service.handle(release).receipt.disposition.value == (
        "ALREADY_RELEASED"
    )

    unseen = _Fixture()
    rejected = service.prepare_stored_publication_graph(
        protocol.PrepareStoredPublicationGraph(unseen.graph)
    )
    assert not rejected.accepted
    assert rejected.error_kind is (
        protocol.StoredPublicationRPCErrorKind.INVALID_STATE
    )

    clean = service.shutdown(shutdown)
    assert clean.clean
    assert service._stop_event.is_set()
