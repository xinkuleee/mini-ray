# Pure reducers are marked individually; real threads/sockets are bounded
# loopback smoke tests and must not inherit the default unit marker.


class _Server:
    def __init__(self, handlers, **_configuration) -> None:
        self.handlers = dict(handlers)
        self.address = ("127.0.0.1", 39999)
        self.is_running = False

    def start(self):
        self.is_running = True
        return self.address

    def stop(self):
        self.is_running = False


def _fixture(service: control.GCSLite, *, separate_owner: bool = False):
    job = JobID.random()
    driver = TaskID.for_driver(job)
    task = TaskID.derive(job, driver, 1)
    attempt = AttemptID(task, 0)
    outer = ObjectID.for_task(task)
    owner = WorkerID.random()
    executor = WorkerID.random()
    node = NodeID.random()
    registration = service.register_node(protocol.RegisterNode(
        node, 4101, ("127.0.0.1", 39101),
        ResourceVector({"CPU": 1}),
    ))
    owner_registration = registration
    owner_node = node
    if separate_owner:
        owner_node = NodeID.random()
        owner_registration = service.register_node(protocol.RegisterNode(
            owner_node, 4102, ("127.0.0.1", 39103),
            ResourceVector({"CPU": 1}),
        ))
    incarnation = protocol.WorkerIncarnation(
        owner_node, owner_registration.node_pid,
        owner_registration.registration_epoch,
        owner, 5101,
    )
    assert service.register_worker_incarnation(
        protocol.RegisterWorkerIncarnation(incarnation)
    ).accepted
    child = ObjectID.for_task(TaskID.derive(job, task, 2))
    edge = ContainedReferenceEdge(
        outer, child, executor, ("127.0.0.1", 39102), "owner-death"
    )
    publication = InlinePublicationID(
        LeaseID.random(), task, attempt, outer
    )
    envelope = InlinePublicationEnvelope.create(
        publication, job, executor, owner, node, b"owner-death", (edge,),
    )
    intent = InlinePublicationIntent(
        envelope.manifest, registration.node_pid, registration.registration_epoch
    )
    assert service.report_inline_publication_recovery(
        protocol.ReportInlinePublicationIntent(intent)
    ).ack is not None
    assert service.prepare_contained_graph(
        protocol.PrepareContainedGraph(envelope.to_graph_manifest())
    ).accepted
    return incarnation, envelope, edge


@pytest.mark.unit
def test_direct_worker_death_freezes_without_remote_effect_and_replays(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    rpc_calls = []

    def rpc(*args):
        rpc_calls.append(args)
        raise AssertionError("death report must not execute cleanup effects")

    service = control.GCSLite(stored_hold_rpc=rpc)
    incarnation, envelope, _edge = _fixture(service)
    request = protocol.ReportWorkerDeath(
        "direct-owner-exit", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    )

    first = service.report_worker_death(request)
    replay = service.report_worker_death(request)

    assert first.disposition is protocol.WorkerDeathDisposition.APPLIED
    assert replay.disposition is protocol.WorkerDeathDisposition.ALREADY_DEAD
    assert replay.death == first.death
    assert rpc_calls == []
    snapshots = service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    )
    assert len(snapshots) == 1
    assert snapshots[0].request.key.publication_id == envelope.publication_id
    assert snapshots[0].request.manifest == envelope.manifest
    assert snapshots[0].request.publication_history is (
        PublicationRecoveryHistory.NOT_COMMITTED
    )
    assert not hasattr(snapshots[0].request, "envelope")
    assert snapshots[0].phase is PublicationOwnerDeathPhase.NODE_FENCE
    arbitration = service.stored_publications.owner_death_arbitration.snapshot(
        InlinePublicationOwnerDeathKey(envelope.publication_id)
    )
    assert arbitration.phase is PublicationOwnerDeathArbitrationPhase.CLAIMED
    assert arbitration.owner_cleanup_request == snapshots[0].request
    assert not service.shutdown(
        protocol.Shutdown("owner-death-active", "test")
    ).clean
    service.stop()


@pytest.mark.unit
def test_node_loss_first_inline_is_superseded_before_owner_cleanup(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    incarnation, envelope, _edge = _fixture(service, separate_owner=True)
    service.report_inline_publication_recovery(
        protocol.ReportInlinePublicationPrepared(envelope.publication_id)
    )
    node_death = service.report_node_death(protocol.ReportNodeDeath(
        "publisher-first", envelope.manifest.publishing_node_id, 4101, 1, -9,
        protocol.NodeDeathReason.PROCESS_EXIT, "publisher exited",
    )).death
    assert node_death is not None
    frozen = service.stored_publications.inline_recovery.snapshot(
        envelope.publication_id
    )
    assert frozen.frozen and frozen.resolution is None
    assert frozen.recovery_action is InlineRecoveryAction.COMPLETION_UNKNOWN

    service.report_worker_death(protocol.ReportWorkerDeath(
        "owner-second", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    ))

    terminal = service.stored_publications.inline_recovery.snapshot(
        envelope.publication_id
    )
    assert terminal.resolution is InlineRecoveryResolution.SUPERSEDED
    assert terminal.superseding_owner_death is not None
    owner_work = service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    )[0]
    assert owner_work.request.publication_history is (
        PublicationRecoveryHistory.COMMIT_UNKNOWN
    )
    assert owner_work.request.manifest == envelope.manifest
    arbitration = (
        service.stored_publications.owner_death_arbitration.snapshot(
            InlinePublicationOwnerDeathKey(envelope.publication_id)
        )
    )
    assert arbitration.node_loss is not None
    assert arbitration.node_loss.terminal_receipt == terminal
    assert not service.stored_publications.inline_node_loss.has_active_operations()
    service.stop()


@pytest.mark.parametrize("choice", tuple(InlineRecoveryOwnerDecision))
@pytest.mark.unit
def test_unknown_inline_owner_death_preserves_choice_and_supersedes_effects(
    monkeypatch, choice,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    incarnation, envelope, _edge = _fixture(service, separate_owner=True)
    publication = envelope.publication_id
    assert service.report_inline_publication_recovery(
        protocol.ReportInlinePublicationPrepared(publication)
    ).ack is not None
    death = service.report_node_death(protocol.ReportNodeDeath(
        "publisher-before-choice", envelope.manifest.publishing_node_id,
        4101, 1, -9, protocol.NodeDeathReason.PROCESS_EXIT, "test",
    )).death
    assert death is not None
    work = service.stored_publications.inline_recovery.work_for_node_death(
        publication, death
    )
    decision = protocol.DecideInlineNodeLoss(work, incarnation.worker_id, choice)
    assert service.decide_inline_node_loss(decision).ack is not None

    assert service.report_worker_death(protocol.ReportWorkerDeath(
        "owner-after-choice", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    )).death is not None

    snapshot = service.stored_publications.inline_recovery.snapshot(publication)
    assert snapshot.resolution is InlineRecoveryResolution.SUPERSEDED
    assert snapshot.owner_decision is choice
    assert snapshot.terminal is None
    request = service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    )[0].request
    assert request.publication_history is PublicationRecoveryHistory.COMMIT_UNKNOWN
    assert request.manifest == envelope.manifest
    assert not hasattr(request, "envelope")
    assert service.decide_inline_node_loss(decision).ack is None
    assert service.progress_inline_node_loss(protocol.ProgressInlineNodeLoss(
        publication, death
    )).ack is None
    service.stop()

@pytest.mark.unit
def test_explicit_progress_drives_one_effect_and_global_drain_converges(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    graph_aborted = False
    calls = []

    def rpc(address, handler, request):
        nonlocal graph_aborted
        calls.append(handler)
        if handler == control.INSTALL_OWNER_DEATH_FENCE_HANDLER:
            return protocol.InstallOwnerDeathFenceReply(
                request, protocol.OwnerDeathFenceDisposition.FENCED
            )
        if handler == control.RELEASE_CONTAINED_REFERENCE_HANDLER:
            return protocol.ReleaseContainedReferenceReply(
                request.object_id, request.owner_worker_id, request.hold,
                True, True,
            )
        if handler == control.FINALIZE_PUBLICATION_OWNER_DEATH_HANDLER:
            from miniray.publication_owner_death import (
                PublicationOwnerDeathTerminal,
            )
            assert graph_aborted
            return protocol.FinalizePublicationOwnerDeathReply(
                request,
                protocol.PublicationOwnerDeathFinalizeDisposition
                .OWNER_DEATH_CLEANED,
                PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED,
            )
        raise AssertionError(handler)

    service = control.GCSLite(stored_hold_rpc=rpc)
    incarnation, envelope, _edge = _fixture(service)
    # The graph callback stays local to GCS; record when the pure authority is
    # reached without routing through the public frozen-mutation gate.
    original_abort = service.stored_publications.graph.abort_manifest

    def abort(manifest):
        nonlocal graph_aborted
        graph_aborted = True
        return original_abort(manifest)

    monkeypatch.setattr(
        service.stored_publications.graph, "abort_manifest", abort
    )
    death = service.report_worker_death(protocol.ReportWorkerDeath(
        "direct-owner-exit", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    ))
    assert death.death is not None

    progress = service.handlers[
        control.PROGRESS_PUBLICATION_OWNER_DEATH_HANDLER
    ](protocol.ProgressPublicationOwnerDeath(incarnation.worker_id))
    assert progress.progressed and not progress.clean
    # One owner-wide fence and one publication-specific fence advance through
    # independent bounded reducers in this explicit progress call.
    assert calls == [
        control.INSTALL_OWNER_DEATH_FENCE_HANDLER,
        control.INSTALL_OWNER_DEATH_FENCE_HANDLER,
    ]

    drain = protocol.DrainPublicationOwnerDeaths("drain-owner-death")
    latest = None
    for _ in range(4):
        latest = service.handle(drain)
        if latest.clean:
            break
    assert latest is not None and latest.clean
    assert calls == [
        control.INSTALL_OWNER_DEATH_FENCE_HANDLER,
        control.INSTALL_OWNER_DEATH_FENCE_HANDLER,
        control.RELEASE_CONTAINED_REFERENCE_HANDLER,
        control.FINALIZE_PUBLICATION_OWNER_DEATH_HANDLER,
    ]
    snapshot = service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    )[0]
    assert snapshot.phase is PublicationOwnerDeathPhase.RESOLVED
    arbitration = service.stored_publications.owner_death_arbitration.snapshot(
        InlinePublicationOwnerDeathKey(envelope.publication_id)
    )
    assert arbitration.phase is PublicationOwnerDeathArbitrationPhase.RESOLVED
    assert arbitration.resolution == snapshot.resolution
    assert service.stored_publications.inline_recovery.owner_death_resolution(
        envelope.publication_id
    ) == snapshot.resolution
    assert not service.stored_publications.has_active_operations()
    service.stop()


@pytest.mark.loopback_smoke
@pytest.mark.parametrize("failed_domain", ("fence", "publication"))
def test_live_background_converges_owner_wide_and_publication_sagas(
    monkeypatch, failed_domain,
) -> None:
    """Worker death needs no explicit progress RPC or shutdown driver."""

    monkeypatch.setattr(control, "TCPServer", _Server)
    graph_aborted = threading.Event()
    converged = threading.Event()
    calls = []

    def rpc(_address, handler, request):
        calls.append((handler, getattr(request, "scope", None)))
        if handler == control.INSTALL_OWNER_DEATH_FENCE_HANDLER:
            return protocol.InstallOwnerDeathFenceReply(
                request, protocol.OwnerDeathFenceDisposition.FENCED
            )
        if handler == control.RELEASE_CONTAINED_REFERENCE_HANDLER:
            return protocol.ReleaseContainedReferenceReply(
                request.object_id, request.owner_worker_id, request.hold,
                True, True,
            )
        if handler == control.FINALIZE_PUBLICATION_OWNER_DEATH_HANDLER:
            from miniray.publication_owner_death import (
                PublicationOwnerDeathTerminal,
            )

            assert graph_aborted.is_set()
            return protocol.FinalizePublicationOwnerDeathReply(
                request,
                protocol.PublicationOwnerDeathFinalizeDisposition
                .OWNER_DEATH_CLEANED,
                PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED,
            )
        raise AssertionError(handler)

    service = control.GCSLite(stored_hold_rpc=rpc)
    incarnation, envelope, _edge = _fixture(service)
    original_abort = service.stored_publications.graph.abort_manifest

    def abort(manifest):
        graph_aborted.set()
        return original_abort(manifest)

    monkeypatch.setattr(
        service.stored_publications.graph, "abort_manifest", abort
    )
    death = service.report_worker_death(protocol.ReportWorkerDeath(
        "background-owner-exit", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    ))
    assert death.death is not None

    # Each exact parameter case injects just one failed progress operation.
    # Together they prove both domains are failure-isolated without putting
    # two faults in one bounded live-thread test.
    original_fence_progress = service._drive_owner_death_fence_once
    original_publication_progress = (
        service.stored_publications.drive_owner_deaths_background_round
    )
    progress_events = []
    injected_failures = []
    fence_round = 0
    publication_round = 0

    def fence_progress():
        nonlocal fence_round
        fence_round += 1
        progress_events.append(("fence", fence_round))
        if failed_domain == "fence" and fence_round == 1:
            injected_failures.append("fence")
            raise RuntimeError("injected owner-wide progress failure")
        return original_fence_progress()

    def publication_progress():
        nonlocal publication_round
        publication_round += 1
        progress_events.append(("publication", publication_round))
        if failed_domain == "publication" and publication_round == 1:
            injected_failures.append("publication")
            raise RuntimeError("injected publication progress failure")
        result = original_publication_progress()
        if (
            not service.owner_death_fences.has_active_operations()
            and service.stored_publications.active_owner_death_count() == 0
        ):
            converged.set()
        return result

    monkeypatch.setattr(
        service, "_drive_owner_death_fence_once", fence_progress
    )
    monkeypatch.setattr(
        service.stored_publications,
        "drive_owner_deaths_background_round",
        publication_progress,
    )

    try:
        service.start()
        # The progress callback sets this only after both real authorities
        # converge.  The test never substitutes a polling sleep for an ACK.
        assert converged.wait(3.0)
        assert not service.owner_death_fences.has_active_operations()
        assert service.stored_publications.active_owner_death_count() == 0
        assert service._owner_death_progress_thread is not None
        assert service._owner_death_progress_thread.is_alive()

        snapshot = service.stored_publications.owner_death_snapshots(
            incarnation.worker_id
        )[0]
        assert snapshot.phase is PublicationOwnerDeathPhase.RESOLVED
        assert snapshot.resolution is not None
        assert service.stored_publications.inline_recovery.owner_death_resolution(
            envelope.publication_id
        ) == snapshot.resolution
        assert progress_events[:4] == [
            ("fence", 1), ("publication", 1),
            ("fence", 2), ("publication", 2),
        ]
        assert injected_failures == [failed_domain]
        scopes = [
            scope for handler, scope in calls
            if handler == control.INSTALL_OWNER_DEATH_FENCE_HANDLER
        ]
        assert protocol.OwnerDeathFenceScope.PUBLICATION_EXACT in scopes
        assert protocol.OwnerDeathFenceScope.OWNER_WIDE_SWEEP in scopes
        assert graph_aborted.is_set()
    finally:
        thread = service._owner_death_progress_thread
        service.stop()
        assert thread is None or not thread.is_alive()
        assert not service.is_running


@pytest.mark.unit
def test_owner_death_freeze_supersedes_normal_graph_and_inline_reports(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    incarnation, envelope, _edge = _fixture(service)
    request = protocol.ReportWorkerDeath(
        "owner-wins-forward-race", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    )
    death = service.report_worker_death(request)
    assert death.death is not None

    key = InlinePublicationOwnerDeathKey(envelope.publication_id)
    arbitration = service.stored_publications.owner_death_arbitration
    assert arbitration.decision(
        key, PublicationArbitrationOperation.NORMAL_FORWARD
    ) is PublicationArbitrationDecision.SUPERSEDED

    graph = service.commit_contained_graph(
        protocol.CommitContainedGraph(envelope.to_graph_manifest())
    )
    assert not graph.accepted
    assert graph.error_kind is (
        protocol.StoredPublicationRPCErrorKind.INVALID_STATE
    )
    assert "SUPERSEDED" in graph.error

    prepared = service.report_inline_publication_recovery(
        protocol.ReportInlinePublicationPrepared(envelope.publication_id)
    )
    assert prepared.ack is None
    assert prepared.error_kind is (
        protocol.StoredPublicationRPCErrorKind.INVALID_STATE
    )
    assert "SUPERSEDED" in prepared.error

    node_death = protocol.NodeDeathRecord(
        "publisher-after-owner", incarnation.node_id,
        incarnation.node_pid, incarnation.node_registration_epoch, 99, -9,
        protocol.NodeDeathReason.PROCESS_EXIT, "test publisher exit",
    )
    pull = service.get_inline_node_loss(protocol.GetInlineNodeLoss(
        envelope.publication_id, incarnation.worker_id, node_death
    ))
    assert pull.disposition is protocol.InlineNodeLossDisposition.REJECTED
    assert pull.error_kind is protocol.StoredPublicationRPCErrorKind.INVALID_STATE
    assert "SUPERSEDED" in pull.error
    progress = service.progress_inline_node_loss(
        protocol.ProgressInlineNodeLoss(envelope.publication_id, node_death)
    )
    assert progress.ack is None
    assert "SUPERSEDED" in progress.error
    service.stop()


@pytest.mark.unit
def test_terminal_registry_ack_failure_remains_active_and_replays(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)

    def rpc(_address, handler, request):
        if handler == control.INSTALL_OWNER_DEATH_FENCE_HANDLER:
            return protocol.InstallOwnerDeathFenceReply(
                request, protocol.OwnerDeathFenceDisposition.FENCED
            )
        if handler == control.RELEASE_CONTAINED_REFERENCE_HANDLER:
            return protocol.ReleaseContainedReferenceReply(
                request.object_id, request.owner_worker_id, request.hold,
                True, True,
            )
        if handler == control.FINALIZE_PUBLICATION_OWNER_DEATH_HANDLER:
            from miniray.publication_owner_death import (
                PublicationOwnerDeathTerminal,
            )
            return protocol.FinalizePublicationOwnerDeathReply(
                request,
                protocol.PublicationOwnerDeathFinalizeDisposition
                .OWNER_DEATH_CLEANED,
                PublicationOwnerDeathTerminal.OWNER_DEATH_CLEANED,
            )
        raise AssertionError(handler)

    service = control.GCSLite(stored_hold_rpc=rpc)
    incarnation, envelope, _edge = _fixture(service)
    service.report_worker_death(protocol.ReportWorkerDeath(
        "registry-ack-failure", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    ))
    authority = service.stored_publications.inline_recovery
    original = authority.acknowledge_owner_death_cleaned
    failures = 0

    def fail_once(resolution):
        nonlocal failures
        failures += 1
        if failures == 1:
            raise RuntimeError("injected registry terminal failure")
        return original(resolution)

    monkeypatch.setattr(
        authority, "acknowledge_owner_death_cleaned", fail_once
    )
    request = protocol.DrainPublicationOwnerDeaths("registry-ack-drain")
    with pytest.raises(RuntimeError, match="registry terminal failure"):
        for _ in range(4):
            service.drain_publication_owner_deaths(request)

    key = InlinePublicationOwnerDeathKey(envelope.publication_id)
    snapshot = service.stored_publications.owner_death.snapshot(key)
    assert snapshot.resolution is not None
    assert authority.owner_death_resolution(envelope.publication_id) is None
    assert service.stored_publications.active_owner_death_count() == 1
    assert service.stored_publications.has_active_operations()

    replay = service.drain_publication_owner_deaths(request)
    assert replay.clean
    assert authority.owner_death_resolution(
        envelope.publication_id
    ) == snapshot.resolution
    assert service.stored_publications.owner_death_arbitration.snapshot(
        key

@pytest.mark.unit
def test_preterminal_stored_owner_death_builds_intent_backed_saga(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    job = JobID.random()
    driver = TaskID.for_driver(job)
    task = TaskID.derive(job, driver, 10)
    attempt = AttemptID(task, 0)
    outer = ObjectID.for_task(task)
    owner = WorkerID.random()
    executor = WorkerID.random()
    node = NodeID.random()
    registration = service.register_node(protocol.RegisterNode(
        node, 4201, ("127.0.0.1", 39201), ResourceVector({"CPU": 1})
    ))
    incarnation = protocol.WorkerIncarnation(
        node, registration.node_pid, registration.registration_epoch,
        owner, 5201,
    )
    assert service.register_worker_incarnation(
        protocol.RegisterWorkerIncarnation(incarnation)
    ).accepted
    child = ObjectID.for_task(TaskID.derive(job, task, 11))
    transfer = PreparedContainedTransfer(
        child, executor, ("127.0.0.1", 39202),
        OwnedContainedSource(executor),
        ContainedReferenceHold(outer, executor, "stored-owner-death"),
        ContainedReferenceHold(outer, owner, "stored-owner-death"),
    )
    publication = StoredPublicationID(
        LeaseID.random(), task, attempt, outer
    )
    manifest = StoredPublicationManifest(
        publication, (transfer,), 5, hashlib.sha256(b"bytes").hexdigest(),
        hashlib.sha256(b"manifest").hexdigest(),
    )
    header = StoredPublicationHeader(
        publication, job, executor, owner, node
    )
    intent = StoredPublicationIntent(
        header, manifest, registration.node_pid,
        registration.registration_epoch,
    )
    assert service.report_stored_publication_recovery(
        protocol.ReportStoredPublicationIntent(intent)
    ).accepted
    assert service.prepare_contained_graph(
        protocol.PrepareContainedGraph(manifest.graph_manifest(owner))
    ).accepted

    reply = service.report_worker_death(protocol.ReportWorkerDeath(
        "stored-owner-exit", incarnation, -9,
        protocol.WorkerDeathReason.PROCESS_EXIT,
    ))

    assert reply.death is not None
    snapshots = service.stored_publications.owner_death_snapshots(owner)
    assert len(snapshots) == 1
    request = snapshots[0].request
    assert request.key.publication_id == publication
    assert request.envelope is None
    assert request.recovery.intent == intent
    assert request.replicas == ()
    assert snapshots[0].phase is PublicationOwnerDeathPhase.NODE_FENCE
    arbitration = service.stored_publications.owner_death_arbitration.snapshot(
        StoredPublicationOwnerDeathKey(publication)
    )
    assert arbitration.phase is PublicationOwnerDeathArbitrationPhase.CLAIMED

    report = service.report_stored_publication_recovery(
        protocol.ReportStoredPublicationManifest(manifest)
    )
    assert report.ack is None
    assert "SUPERSEDED" in report.error
    graph = service.commit_contained_graph(
        protocol.CommitContainedGraph(manifest.graph_manifest(owner))
    )
    assert not graph.accepted
    assert "SUPERSEDED" in graph.error

    node_death = protocol.NodeDeathRecord(
        "stored-publisher-after-owner", node, registration.node_pid,
        registration.registration_epoch, 99, -9,
        protocol.NodeDeathReason.PROCESS_EXIT, "test publisher exit",
    )
    pull = service.get_stored_node_loss(protocol.GetStoredNodeLoss(
        publication, owner, node_death
    ))
    assert pull.disposition is protocol.StoredNodeLossDisposition.REJECTED
    assert "SUPERSEDED" in pull.error
    progress = service.progress_stored_node_loss(
        protocol.ProgressStoredNodeLoss(publication, owner, node_death)
    )
    assert progress.snapshot is None
    assert "SUPERSEDED" in progress.error
    service.stop()


def _cross_node_owned_publications(service: control.GCSLite):
    victim = service.register_node(protocol.RegisterNode(
        NodeID.random(), 4301, ("127.0.0.1", 39301),
        ResourceVector({"CPU": 1}),
    ))
    publisher = service.register_node(protocol.RegisterNode(
        NodeID.random(), 4302, ("127.0.0.1", 39302),
        ResourceVector({"CPU": 1}),
    ))
    owner = WorkerID.random()
    executor = WorkerID.random()
    incarnation = protocol.WorkerIncarnation(
        victim.node_id, victim.node_pid, victim.registration_epoch, owner, 5301
    )
    assert service.register_worker_incarnation(
        protocol.RegisterWorkerIncarnation(incarnation)
    ).accepted
    job = JobID.random()
    driver = TaskID.for_driver(job)

    inline_task = TaskID.derive(job, driver, 20)
    inline_attempt = AttemptID(inline_task, 0)
    inline_outer = ObjectID.for_task(inline_task)
    inline_edge = ContainedReferenceEdge(
        inline_outer,
        ObjectID.for_task(TaskID.derive(job, inline_task, 21)), executor,
        ("127.0.0.1", 39303), "node-exit-inline",
    )
    inline_id = InlinePublicationID(
        LeaseID.random(), inline_task, inline_attempt, inline_outer
    )
    inline_envelope = InlinePublicationEnvelope.create(
        inline_id, job, executor, owner, publisher.node_id,
        b"node-exit-inline", (inline_edge,),
    )
    assert service.report_inline_publication_recovery(
        protocol.ReportInlinePublicationIntent(InlinePublicationIntent(
            inline_envelope.manifest, publisher.node_pid, publisher.registration_epoch
        ))
    ).ack is not None
    assert service.prepare_contained_graph(protocol.PrepareContainedGraph(
        inline_envelope.to_graph_manifest()
    )).accepted

    stored_task = TaskID.derive(job, driver, 30)
    stored_attempt = AttemptID(stored_task, 0)
    stored_outer = ObjectID.for_task(stored_task)
    stored_id = StoredPublicationID(
        LeaseID.random(), stored_task, stored_attempt, stored_outer
    )
    transfer = PreparedContainedTransfer(
        ObjectID.for_task(TaskID.derive(job, stored_task, 31)), executor,
        ("127.0.0.1", 39304), OwnedContainedSource(executor),
        ContainedReferenceHold(stored_outer, executor, "node-exit-stored"),
        ContainedReferenceHold(stored_outer, owner, "node-exit-stored"),
    )
    manifest = StoredPublicationManifest(
        stored_id, (transfer,), 6, hashlib.sha256(b"stored").hexdigest(),
        hashlib.sha256(b"node-exit-stored-manifest").hexdigest(),
    )
    header = StoredPublicationHeader(
        stored_id, job, executor, owner, publisher.node_id
    )
    assert service.report_stored_publication_recovery(
        protocol.ReportStoredPublicationIntent(StoredPublicationIntent(
            header, manifest, publisher.node_pid, publisher.registration_epoch
        ))
    ).accepted
    assert service.prepare_contained_graph(protocol.PrepareContainedGraph(
        manifest.graph_manifest(owner)
    )).accepted
    return victim, publisher, incarnation, inline_id, stored_id


@pytest.mark.unit
def test_node_exit_fanout_admits_inline_and_stored_owner_cleanup(
    monkeypatch,
) -> None:
    monkeypatch.setattr(control, "TCPServer", _Server)
    service = control.GCSLite()
    victim, publisher, incarnation, inline_id, stored_id = (
        _cross_node_owned_publications(service)
    )
    request = protocol.ReportNodeDeath(
        "owner-node-exit", victim.node_id, victim.node_pid,
        victim.registration_epoch, -9, protocol.NodeDeathReason.PROCESS_EXIT,
        "owner Node exited",
    )

    first = service.report_node_death(request)
    snapshots = service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    )
    replay = service.report_node_death(request)

    assert first.disposition is protocol.NodeDeathDisposition.APPLIED
    assert replay.disposition is protocol.NodeDeathDisposition.ALREADY_DEAD
    assert len(snapshots) == 2
    assert {snapshot.request.key for snapshot in snapshots} == {
        InlinePublicationOwnerDeathKey(inline_id),
        StoredPublicationOwnerDeathKey(stored_id),
    }
    assert all(
        snapshot.request.owner_death.reason
        is protocol.WorkerDeathReason.NODE_EXIT
        for snapshot in snapshots
    )
    assert all(
        snapshot.request.publishing_node_id == publisher.node_id
        for snapshot in snapshots
    )
    assert service.stored_publications.owner_death_snapshots(
        incarnation.worker_id
    ) == snapshots
    service.stop()
