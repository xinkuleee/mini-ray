"""Pure Node/GCS/Core composition: GCS metadata cannot resurrect payloads.

"Fullflow" here means the real component methods joined by synchronous RPC
callbacks.  No process, listener, background thread, or real wait is started.
The remote child-owner effect is represented by the Node fixture's pin set.
"""

from __future__ import annotations

from dataclasses import fields, is_dataclass
import queue
import threading

import pytest

from miniray import protocol
from miniray.contained_cycle import ContainedGraphTransactionState
from miniray.control import StoredPublicationControlAdapter
from miniray.core import (
    CoreWorker, ObjectRef, _InlineNodeLossObligation, _ObjectWaiter, _PendingTask,
)
from miniray.inline_recovery import (
    InlineRecoveryAction, InlineRecoveryOwnerDecision, InlineRecoveryResolution,
)
from miniray.inline_recovery_runtime import (
    InlineRecoveryHoldDisposition, InlineRecoveryHoldReceipt,
)
from miniray.ownership import ObjectCollectionState, ObjectOwnerTable, ObjectState
from miniray.recovery import RecoveryManager, TaskState
from miniray.trace import MemoryEventSink
from miniray.transport import TransportTimeout
from tests.unit.test_inline_publication_node_server import _Harness


pytestmark = pytest.mark.unit


def _no_payload(value: object, payload: bytes) -> None:
    """Walk protocol values without serializing or invoking user reducers."""

    if isinstance(value, bytes):
        assert value != payload
    elif is_dataclass(value) and not isinstance(value, type):
        for field in fields(value):
            assert field.name not in ("inline_data", "envelope", "descriptor")
            _no_payload(getattr(value, field.name), payload)
    elif isinstance(value, (tuple, list, set, frozenset)):
        for item in value:
            _no_payload(item, payload)
    elif isinstance(value, dict):
        for key, item in value.items():
            _no_payload(key, payload)
            _no_payload(item, payload)


class _Flow:
    def __init__(self) -> None:
        self.node = _Harness(edge_count=2)
        h = self.node
        self.calls: list[str] = []
        self.gcs_available = True
        self.drop_progress_reply = False

        def release(effect):
            reply = self.release_child(effect.request)
            return InlineRecoveryHoldReceipt(
                effect,
                InlineRecoveryHoldDisposition.RELEASED if reply.released
                else InlineRecoveryHoldDisposition.ALREADY_RELEASED,
                release_reply=reply,
            )

        self.gcs = StoredPublicationControlAdapter(
            graph=h.graph, inline_recovery=h.recovery, resolve_inline_hold=release,
        )
        h.node._stored_gcs_rpc = self.gcs_rpc
        h.node._inline_publications._prepare_graph = self.gcs.mutate_graph
        h.node._inline_publications._abort_graph = self.gcs.mutate_graph

        core = object.__new__(CoreWorker)
        self.core = core
        core.job_id = h.envelope.manifest.job_id
        core.worker_id = h.owner
        core.node_id = h.node_id
        core.node_address = ("127.0.0.1", 33201)
        core.gcs_address = ("127.0.0.1", 33200)
        core.event_sink = MemoryEventSink()
        core._state_lock = threading.RLock()
        core._completion = threading.Condition(core._state_lock)
        core._owner_table = ObjectOwnerTable()
        core._recovery = RecoveryManager()
        core._objects = {}
        core._stored_descriptors = {}
        core._submissions = queue.Queue()
        core._ready_tasks = queue.Queue()
        core._protocol_unresolved = {}
        core._object_gc_obligations = {}
        core._orphan_contained_edges = set()
        core._dead_nodes = {}
        core._accepting = True
        core._accepted_task_count = 1
        self.gc_checks: list[object] = []
        core._enqueue_inline_gc_check = self.gc_checks.append
        core._rpc = lambda _address, handler, request: self.gcs_rpc(handler, request)
        core._borrow_rpc = lambda _address, _handler, request: self.release_child(request)
        execution = h.publication_id
        spec = protocol.TaskSpec(
            core.job_id, execution.task_id, execution.attempt_id,
            protocol.FunctionKey(core.job_id, __name__, "producer", "v1"),
            (), 1, h.record.request.resources, core.worker_id, max_retries=1,
        )
        self.pending = _PendingTask(execution.outer_object_id, spec)
        core.owner_table.register(
            execution.outer_object_id, current_attempt=execution.attempt_id,
            producer_task_spec=spec, local_token="outer",
        )
        core._recovery.register_task(spec, max_retries=1)
        core._objects[execution.outer_object_id] = _ObjectWaiter(threading.Event())

    def release_child(self, request):
        assert isinstance(request, protocol.ReleaseContainedReference)
        matched = {
            pin for pin in self.node.pins
            if pin.edge.contained_object_id == request.object_id
            and pin.edge.contained_owner_worker_id == request.owner_worker_id
            and pin.edge.incoming_hold(self.node.owner) == request.hold
        }
        self.node.pins.difference_update(matched)
        self.calls.append("child_release")
        return protocol.ReleaseContainedReferenceReply(
            request.object_id, request.owner_worker_id, request.hold,
            True, bool(matched),
        )

    def gcs_rpc(self, handler, request):
        self.calls.append(handler)
        _no_payload(request, self.node.envelope.descriptor.inline_data)
        if not self.gcs_available:
            raise TransportTimeout("GCS unavailable")
        if isinstance(request, (
            protocol.ReportInlinePublicationIntent,
            protocol.ReportInlinePublicationPrepared,
            protocol.ReportInlinePublicationTerminal,
            protocol.AcknowledgeInlinePublicationAdopted,
            protocol.AcknowledgeInlinePublicationRetired,
            protocol.AcknowledgeInlinePublicationAborted,
        )):
            reply = self.gcs.report_inline_recovery(request)
        elif isinstance(request, (
            protocol.CommitContainedGraph, protocol.ReleaseContainedGraphContainer,
        )):
            reply = self.gcs.mutate_graph(request)
        elif isinstance(request, protocol.GetInlineNodeLoss):
            reply = self.gcs.get_inline_node_loss(request)
        elif isinstance(request, protocol.DecideInlineNodeLoss):
            reply = self.gcs.decide_inline_node_loss(request)
        elif isinstance(request, protocol.ProgressInlineNodeLoss):
            reply = self.gcs.progress_inline_node_loss(request)
            if self.drop_progress_reply:
                self.drop_progress_reply = False
                raise TransportTimeout("cleanup succeeded; reply lost")
        elif isinstance(request, protocol.AcknowledgeInlineNodeLoss):
            reply = self.gcs.acknowledge_inline_node_loss(request)
        else:
            raise AssertionError("unexpected GCS request: {}".format(handler))
        _no_payload(reply, self.node.envelope.descriptor.inline_data)
        return reply

    def prepare(self) -> None:
        assert self.node.open().accepted
        assert self.node.prepare().accepted
        assert len(self.node.pins) == 2

    def die(self):
        h = self.node
        death = protocol.NodeDeathRecord(
            "inline-fullflow-node-exit", h.node_id, h.node._node_pid,
            h.node._registration_epoch, 1, -9,
            protocol.NodeDeathReason.PROCESS_EXIT, "synthetic process exit",
        )
        work = self.gcs.inline_recovery.freeze_node_death(death)[0]
        self.core._dead_nodes[h.node_id] = death
        return work, _InlineNodeLossObligation(h.publication_id, death)


def test_local_complete_survives_gcs_outage_then_owner_ack_precedes_outbox() -> None:
    f = _Flow()
    f.prepare()
    f.calls.clear()
    f.gcs_available = False
    completed = f.node.complete()
    assert completed.accepted and completed.released
    assert f.calls == []
    assert f.node.node._ledger.available == f.node.node._ledger.total
    assert not f.node.node._drive_terminal_inline_publications()
    assert not f.node.node._inline_publications_clean_locked()
    assert f.node.node._handle_get_inline_worker_lease_outcome(
        f.node.outcome_request(), f.node.publication_id
    ).inline_publication == completed.inline_publication

    f.gcs_available = True
    reply = protocol.TaskReply(
        f.pending.task_id, f.pending.spec.attempt_id, f.node.executor,
        protocol.TaskReplyStatus.SUCCEEDED,
        (completed.inline_publication.descriptor,),
        inline_publication=completed.inline_publication,
    )
    assert f.core._publish_reply(f.pending, reply)
    snapshot = f.gcs.inline_recovery.snapshot(f.node.publication_id)
    assert snapshot.resolution is InlineRecoveryResolution.ADOPTED
    assert not f.node.record.inline_publication_terminal_acked
    assert f.node.node._drive_terminal_inline_publications()
    assert f.node.record.inline_publication_terminal_acked
    assert f.node.node._inline_publications_clean_locked()
    assert not f.gcs.inline_recovery.has_active_operations()
    assert f.core._finish_pending_task(f.pending)
    assert f.core.owner_table.release_local_reference(f.pending.object_id, "outer")
    f.core._reference_released(f.pending.object_id)
    assert not f.node.pins
    assert f.core.owner_table.collection_state(
        f.pending.object_id
    ) is ObjectCollectionState.COLLECTED
    assert not f.gcs.has_active_operations()
    _no_payload(f.gcs.inline_recovery._records, f.node.envelope.descriptor.inline_data)


@pytest.mark.parametrize("reported", (False, True))
def test_death_keeps_only_an_owner_received_envelope(reported: bool) -> None:
    f = _Flow()
    f.prepare()
    completion = f.node.complete()
    assert completion.accepted
    f.core._retain_completed_inline_result(f.pending, completion.inline_publication)
    if reported:
        assert f.node.node._drive_terminal_inline_publications()
    work, obligation = f.die()
    assert work.action is (
        InlineRecoveryAction.POSTCOMPLETE_RESOLVE if reported
        else InlineRecoveryAction.COMPLETION_UNKNOWN
    )
    assert f.core._execute(f.pending, f.pending.spec, inline_node_loss=obligation)
    snapshot = f.gcs.inline_recovery.snapshot(f.node.publication_id)
    assert snapshot.owner_decision is InlineRecoveryOwnerDecision.KEEP
    assert snapshot.resolution is InlineRecoveryResolution.ADOPTED
    assert f.core.owner_table.snapshot(f.pending.object_id).state is ObjectState.READY_INLINE
    assert f.core._recovery.task_record(f.pending.task_id).state is TaskState.SUCCEEDED
    assert len(f.node.pins) == 2
    assert f.gcs.graph.snapshot().manifests[0].state is ContainedGraphTransactionState.COMMITTED
    assert not f.gcs.inline_recovery.has_active_operations()
    assert not f.gcs.inline_node_loss.has_active_operations()
    assert f.core._finish_pending_task(f.pending)
    assert f.core.owner_table.release_local_reference(f.pending.object_id, "outer")
    f.core._reference_released(f.pending.object_id)
    assert not f.node.pins
    assert not f.gcs.has_active_operations()
    _no_payload(work, f.node.envelope.descriptor.inline_data)


@pytest.mark.parametrize("reported", (False, True))
@pytest.mark.parametrize("lost_ack", (False, True))
def test_unreceived_bytes_cleanup_is_lost_not_gcs_payload_recovery(
    reported: bool, lost_ack: bool,
) -> None:
    f = _Flow()
    f.prepare()
    assert f.node.complete().accepted
    if reported:
        assert f.node.node._drive_terminal_inline_publications()
    work, obligation = f.die()
    f.drop_progress_reply = lost_ack
    terminal = f.core._execute(f.pending, f.pending.spec, inline_node_loss=obligation)
    if lost_ack:
        assert not terminal
        assert f.core.owner_table.snapshot(f.pending.object_id).state is ObjectState.PENDING
        queued = f.core._submissions.get_nowait()
        terminal = f.core._execute(
            f.pending, f.pending.spec, inline_node_loss=queued.ready.inline_node_loss
        )
    snapshot = f.gcs.inline_recovery.snapshot(f.node.publication_id)
    assert snapshot.owner_decision is InlineRecoveryOwnerDecision.DROP
    assert snapshot.resolution is InlineRecoveryResolution.LOST
    assert not f.node.pins
    assert f.gcs.graph.snapshot().manifests[0].state is ContainedGraphTransactionState.ABORTED
    assert not f.gcs.has_active_operations()
    assert "commit_contained_graph" not in f.calls
    result = f.core.owner_table.snapshot(f.pending.object_id)
    assert result.inline_data is None
    assert not f.core._inline_task_progress(f.pending, f.node.publication_id).published
    if reported:
        assert terminal
        assert result.state is ObjectState.LOST
        assert f.core._recovery.task_record(f.pending.task_id).state is TaskState.SUCCEEDED
        with pytest.raises(TimeoutError):
            f.core.get(ObjectRef(f.pending.object_id, f.core.worker_id), timeout=0)
        assert result.current_attempt == f.pending.spec.attempt_id
        assert f.core._finish_pending_task(f.pending)
        assert not f.core._task_finish_barriers
    else:
        assert not terminal
        assert result.state is ObjectState.PENDING
        assert result.current_attempt == f.pending.spec.attempt_id.next()
        assert f.core._accepted_task_count == 1
        assert not f.core._finish_pending_task(f.pending)
    _no_payload(work, f.node.envelope.descriptor.inline_data)
