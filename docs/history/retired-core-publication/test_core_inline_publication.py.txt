def _graph_reply(
    request: protocol.ContainedGraphRequest,
    disposition: ContainedGraphManifestDisposition,
) -> protocol.ContainedGraphReply:
    released = (
        request.manifest.ordered_edges
        if isinstance(request, protocol.ReleaseContainedGraphContainer)
        else ()
    )
    return protocol.ContainedGraphReply(
        request,
        ContainedGraphManifestReceipt(
            request.manifest, ContainedGraphTransactionState.COMMITTED,
            disposition, released,
        ),
    )


def _normal_adoption_reply(
    request: protocol.AcknowledgeInlinePublicationAdopted,
    envelope: InlinePublicationEnvelope,
) -> protocol.InlinePublicationRecoveryReply:
    authority = InlinePublicationRecoveryAuthority()
    authority.report_intent(InlinePublicationIntent(envelope.manifest, 12345, 1))
    authority.report_prepared(envelope.publication_id)
    authority.report_terminal(InlinePublicationTerminal(
        envelope.publication_id, protocol.TaskReplyStatus.SUCCEEDED
    ))
    acknowledgement = authority.acknowledge_adopted(
        envelope.publication_id, envelope.manifest.manifest_digest,
        envelope.manifest.outer_owner_worker_id,
    )
    return protocol.InlinePublicationRecoveryReply(
        request, acknowledgement
    )


def _inline_authority(
    envelope: InlinePublicationEnvelope, *, terminal: bool, prepared: bool = False,
) -> InlinePublicationRecoveryAuthority:
    authority = InlinePublicationRecoveryAuthority()
    authority.report_intent(InlinePublicationIntent(envelope.manifest, 12345, 1))
    if prepared or terminal:
        authority.report_prepared(envelope.publication_id)
    if terminal:
        authority.report_terminal(InlinePublicationTerminal(
            envelope.publication_id, protocol.TaskReplyStatus.SUCCEEDED
        ))
    return authority


def _decision_reply(
    authority: InlinePublicationRecoveryAuthority,
    request: protocol.DecideInlineNodeLoss,
) -> protocol.DecideInlineNodeLossReply:
    assert isinstance(request, protocol.DecideInlineNodeLoss)
    acknowledgement = authority.decide_owner_resolution(
        request.work, request.owner_worker_id, request.decision
    )
    return protocol.DecideInlineNodeLossReply(request, acknowledgement)


def _protect_local_dependency(core: CoreWorker, pending: _PendingTask):
    """Give the attempt a real submitted hold without any runtime startup."""

    dependency_task = TaskID.derive(core.job_id, core.driver_task_id, 99)
    dependency_id = ObjectID.for_task(dependency_task)
    core.owner_table.register(
        dependency_id, current_attempt=AttemptID(dependency_task, 0),
        local_token="dependency-handle",
    )
    hold = protocol.TaskReferenceHold(
        protocol.TaskReferenceHoldKind.SUBMITTED, core.worker_id,
        pending.task_id, pending.spec.attempt_id,
    )
    assert core.owner_table.add_submitted_reference(dependency_id, hold)
    return replace(
        pending, protected_dependencies=(dependency_id,), dependency_hold=hold
    ), dependency_id, hold

def _commit_rpc(events: list[str], envelope: InlinePublicationEnvelope):
    def rpc(_address, handler, request):
        if handler == "commit_contained_graph":
            assert isinstance(request, protocol.CommitContainedGraph)
            events.append("commit_graph")
            return _graph_reply(
                request, ContainedGraphManifestDisposition.APPLIED
            )
        assert handler == "acknowledge_inline_publication_adopted"
        events.append("ack")
        return _normal_adoption_reply(request, envelope)

    return rpc


def test_publish_reply_commits_graph_then_atomic_owner_then_wakes(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    core, pending, envelope, reply, edge, _executor = _fixture()
    events: list[str] = []
    original_commit = core.owner_table.commit_inline_owner_publication
    original_wake = core._wake_object

    def commit(plan):
        before = core.owner_table.snapshot(pending.object_id)
        assert before.state is ObjectState.PENDING
        assert before.inline_data is None
        assert not before.outgoing_contained_edges
        events.append("owner")
        receipt = original_commit(plan)
        after = core.owner_table.snapshot(pending.object_id)
        assert after.state is ObjectState.READY_INLINE
        assert after.inline_data == envelope.descriptor.inline_data
        assert after.outgoing_contained_edges == frozenset({edge})
        return receipt

    def wake(object_id):
        snapshot = core.owner_table.snapshot(object_id)
        assert snapshot.inline_publication is not None
        assert snapshot.inline_publication.graph_manifest == (
            envelope.to_graph_manifest()
        )
        events.append("wake")
        original_wake(object_id)

    monkeypatch.setattr(core, "_rpc", _commit_rpc(events, envelope))
    monkeypatch.setattr(
        core.owner_table, "commit_inline_owner_publication", commit
    )
    monkeypatch.setattr(core, "_wake_object", wake)

    assert core._publish_reply(
        pending, reply, expected_node_id=core.node_id
    )
    assert events == ["commit_graph", "owner", "ack", "wake"]
    assert core._objects[pending.object_id].event.is_set()
    assert not core._protocol_unresolved
    assert not core._stored_descriptors
    assert core._recovery.task_record(pending.task_id).state is TaskState.SUCCEEDED

def test_lost_graph_commit_ack_retains_exact_obligation_and_replays(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    core, pending, envelope, reply, _edge, _executor = _fixture()
    requests: list[protocol.CommitContainedGraph] = []
    owner_commits: list[object] = []
    original_commit = core.owner_table.commit_inline_owner_publication

    def rpc(_address, handler, request):
        if handler == "commit_contained_graph":
            assert isinstance(request, protocol.CommitContainedGraph)
            requests.append(request)
            if len(requests) == 1:
                raise TransportTimeout("graph COMMIT applied but ACK was lost")
            return _graph_reply(
                request, ContainedGraphManifestDisposition.ALREADY_COMMITTED
            )
        assert handler == "acknowledge_inline_publication_adopted"
        return _normal_adoption_reply(request, envelope)

    def commit(plan):
        owner_commits.append(plan)
        return original_commit(plan)

    monkeypatch.setattr(core, "_rpc", rpc)
    monkeypatch.setattr(
        core.owner_table, "commit_inline_owner_publication", commit
    )

    assert not core._publish_reply(
        pending, reply, expected_node_id=core.node_id
    )
    unresolved = core._protocol_unresolved[pending.task_key]
    assert isinstance(unresolved.obligation, _InlineAdoptionObligation)
    assert unresolved.obligation.envelope == envelope
    assert not unresolved.obligation.graph_committed
    assert core.owner_table.snapshot(pending.object_id).state is ObjectState.PENDING
    assert not core._objects[pending.object_id].event.is_set()
    assert owner_commits == []

    queued = core._submissions.get_nowait()
    assert isinstance(queued, _DelayedReadyTask)
    assert queued.ready.inline_adoption == unresolved.obligation
    assert core._execute(
        pending, pending.spec, inline_adoption=queued.ready.inline_adoption
    )

    assert len(requests) == 2 and requests[0] == requests[1]
    assert len(owner_commits) == 1
    assert core.owner_table.snapshot(pending.object_id).state is (
        ObjectState.READY_INLINE
    )
    assert core._objects[pending.object_id].event.is_set()
    assert not core._protocol_unresolved

@pytest.mark.parametrize(
    "node_dies_at_reply", [False, True], ids=["live-node", "node-dies-at-reply"]
)
def test_completed_outcome_envelope_recovers_without_push_reexecution(
    monkeypatch: pytest.MonkeyPatch, node_dies_at_reply: bool,
) -> None:
    core, pending, envelope, _reply, _edge, executor = _fixture()
    grant = protocol.GrantWorkerLease(
        envelope.publication_id.lease_id, pending.task_id,
        pending.spec.attempt_id, core.node_id, executor,
        ("127.0.0.1", 39803), AllocationToken("inline-outcome"),
    )
    push = protocol.PushTask(grant.lease_id, executor, pending.spec)
    state = _PushRequestState(
        push, grant, core.node_address, grant.worker_address, 1, True
    )
    outcome = protocol.GetWorkerLeaseOutcomeReply(
        grant.lease_id, pending.task_id, pending.spec.attempt_id, executor,
        core.worker_id, pending.output_ids, core.node_id, True, True,
        state=protocol.LeaseExecutionState.COMPLETED,
        completion_status=protocol.TaskReplyStatus.SUCCEEDED,
        inline_publication=envelope,
    )
    events: list[str] = []
    authority = _inline_authority(envelope, terminal=False, prepared=True)
    death = _death(envelope)
    work = None

    def rpc(_address, handler, request):
        nonlocal work
        if handler == "get_worker_lease_outcome":
            events.append("outcome")
            assert isinstance(request, protocol.GetWorkerLeaseOutcome)
            if node_dies_at_reply:
                # The response is already in Core's hands when the death
                # notification lands. GCS never received Complete or its bytes.
                work = authority.freeze_node_death(death)[0]
                core._dead_nodes[death.node_id] = death
            return outcome
        if handler == "get_inline_node_loss":
            assert node_dies_at_reply and work is not None
            assert core._inline_result_custody[envelope.publication_id].envelope == envelope
            events.append("get")
            return protocol.GetInlineNodeLossReply(
                request, protocol.InlineNodeLossDisposition.WORK, work=work
            )
        if handler == "decide_inline_node_loss":
            assert request.decision is InlineRecoveryOwnerDecision.KEEP
            events.append("keep")
            return _decision_reply(authority, request)
        if handler == "commit_contained_graph":
            events.append("graph")
            return _graph_reply(
                request, ContainedGraphManifestDisposition.ALREADY_COMMITTED
            )
        if handler == "acknowledge_inline_node_loss":
            assert work is not None
            events.append("ack")
            return protocol.AcknowledgeInlineNodeLossReply(
                request, authority.acknowledge_adoption(work)
            )
        assert handler == "acknowledge_inline_publication_adopted"
        events.append("ack")
        return _normal_adoption_reply(request, envelope)

    original_commit = core.owner_table.commit_inline_owner_publication
    original_wake = core._wake_object

    def commit(plan):
        events.append("owner")
        return original_commit(plan)

    def wake(object_id):
        events.append("wake")
        original_wake(object_id)

    monkeypatch.setattr(core, "_rpc", rpc)
    monkeypatch.setattr(
        core.owner_table, "commit_inline_owner_publication", commit
    )
    monkeypatch.setattr(core, "_wake_object", wake)
    core._mark_protocol_unresolved(
        pending, "push_replay_wait", target_node_id=core.node_id
    )

    assert core._resolve_ambiguous_push_outcome(pending, state)
    assert events == (
        ["outcome", "get", "keep", "graph", "owner", "ack", "wake"]
        if node_dies_at_reply else
        ["outcome", "graph", "owner", "ack", "wake"]
    )
    snapshot = core.owner_table.snapshot(pending.object_id)
    assert snapshot.state is ObjectState.READY_INLINE
    assert snapshot.inline_data == envelope.descriptor.inline_data
    assert core._recovery.task_record(pending.task_id).state is TaskState.SUCCEEDED
    assert not core._protocol_unresolved
    assert core._submissions.empty()

def test_normal_adoption_ack_loss_replays_only_ack_before_ready(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    core, pending, envelope, reply, _edge, _executor = _fixture()
    authority = _inline_authority(envelope, terminal=True)
    events: list[str] = []
    acknowledgements = []
    original_commit = core.owner_table.commit_inline_owner_publication
    original_wake = core._wake_object

    def rpc(_address, handler, request):
        if handler == "commit_contained_graph":
            events.append("graph")
            return _graph_reply(
                request, ContainedGraphManifestDisposition.APPLIED
            )
        assert handler == "acknowledge_inline_publication_adopted"
        events.append("ack")
        ack = authority.acknowledge_adopted(
            request.publication_id, request.manifest_digest,
            request.owner_worker_id,
        )
        acknowledgements.append(ack)
        if len(acknowledgements) == 1:
            raise TransportTimeout("adoption ACK was lost after apply")
        return protocol.InlinePublicationRecoveryReply(request, ack)

    def commit(plan):
        events.append("owner")
        return original_commit(plan)

    def wake(object_id):
        events.append("wake")
        original_wake(object_id)

    monkeypatch.setattr(core, "_rpc", rpc)
    monkeypatch.setattr(
        core.owner_table, "commit_inline_owner_publication", commit
    )
    monkeypatch.setattr(core, "_wake_object", wake)

    assert not core._publish_reply(
        pending, reply, expected_node_id=core.node_id
    )
    unresolved = core._protocol_unresolved[pending.task_key]
    obligation = unresolved.obligation
    assert isinstance(obligation, _InlineAdoptionObligation)
    assert obligation.graph_committed and obligation.owner_committed
    assert not obligation.adoption_acknowledged
    assert not core._objects[pending.object_id].event.is_set()
    assert events == ["graph", "owner", "ack"]

    queued = core._submissions.get_nowait()
    assert core._execute(
        pending, pending.spec, inline_adoption=queued.ready.inline_adoption
    )
    assert events == ["graph", "owner", "ack", "ack", "wake"]
    assert acknowledgements[1].disposition.value == "ALREADY_RECORDED"
    assert core._objects[pending.object_id].event.is_set()
    assert pending.task_key not in core._protocol_unresolved

def test_death_frozen_stale_retirement_ack_loss_replays_only_ack(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    core, pending, envelope, _reply, edge, _executor = _fixture()
    authority = _inline_authority(envelope, terminal=True)
    death = _death(envelope)
    work = authority.freeze_node_death(death)[0]
    core._retain_completed_inline_result(pending, envelope)
    next_attempt = pending.spec.attempt_id.next()
    events = []
    retire_calls = 0

    def borrow_rpc(_address, _handler, request):
        events.append("child")
        return protocol.ReleaseContainedReferenceReply(
            request.object_id, request.owner_worker_id, request.hold, True, True
        )

    def rpc(_address, handler, request):
        nonlocal retire_calls
        if handler == "get_inline_node_loss":
            events.append("get")
            return protocol.GetInlineNodeLossReply(
                request, protocol.InlineNodeLossDisposition.WORK, work=work
            )
        if handler == "decide_inline_node_loss":
            assert request.decision is InlineRecoveryOwnerDecision.KEEP
            events.append("keep")
            return _decision_reply(authority, request)
        if handler == "commit_contained_graph":
            events.append("commit")
            # KEEP was legitimate when selected; a newer attempt fences this
            # owner only after graph COMMIT. This must retire, not switch DROP.
            assert core.owner_table.advance_task_outputs(
                pending.execution, next_attempt
            )
            core._recovery.record_task_failure(
                pending.task_id, pending.spec.attempt_id,
                protocol.TaskReplyStatus.SYSTEM_ERROR,
            )
            return _graph_reply(
                request, ContainedGraphManifestDisposition.APPLIED
            )
        if handler == "release_contained_graph_container":
            events.append("release_graph")
            return _graph_reply(
                request, ContainedGraphManifestDisposition.RELEASED
            )
        assert handler == "acknowledge_inline_node_loss"
        events.append("retire")
        retire_calls += 1
        ack = authority.acknowledge_retirement(work)
        if retire_calls == 1:
            raise TransportTimeout("RETIRED applied but ACK was lost")
        return protocol.AcknowledgeInlineNodeLossReply(request, ack)

    monkeypatch.setattr(core, "_rpc", rpc)
    monkeypatch.setattr(core, "_borrow_rpc", borrow_rpc)
    initial = _InlineNodeLossObligation(envelope.publication_id, death)

    assert not core._drive_inline_node_loss(pending, initial)
    queued = core._submissions.get_nowait()
    replay = queued.ready.inline_node_loss
    assert replay is not None
    assert replay.adoption.graph_released
    assert replay.adoption.released_edges == (edge,)
    assert events == [
        "get", "keep", "commit", "child", "release_graph", "retire"
    ]

    assert not core._drive_inline_node_loss(pending, replay)
    assert events == [
        "get", "keep", "commit", "child", "release_graph", "retire",
        "get", "keep", "retire",
    ]
    assert pending.task_key not in core._protocol_unresolved
