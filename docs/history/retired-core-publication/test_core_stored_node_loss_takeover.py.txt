def _core(*, max_retries=1):
    core = object.__new__(CoreWorker)
    core.job_id = JobID.random()
    core.driver_task_id = TaskID.for_driver(core.job_id)
    core.worker_id = WorkerID.random()
    core.node_id = NodeID.random()
    core.node_address = ("127.0.0.1", 32001)
    core.gcs_address = ("127.0.0.1", 32002)
    core._owner_table = ObjectOwnerTable()
    core._recovery = RecoveryManager()
    core._objects = {}
    core._stored_descriptors = {}
    core._state_lock = threading.RLock()
    core._completion = threading.Condition(core._state_lock)
    core._submissions = queue.Queue()
    core._ready_tasks = queue.Queue()
    core._protocol_unresolved = {}
    core._node_death_attempts = {}
    core._dead_nodes = {}
    core._object_gc_obligations = {}
    core._inline_gc_obligations = core._object_gc_obligations
    core._stored_publication_adoptions = {}
    core._stored_publication_adoption_routes = {}
    core._stored_node_loss_drivers = {}
    core._stored_node_loss_objects = {}
    core._owner_protocol_open = True
    core._accepting = True
    core._finishing_tasks = set()
    core._active_task_finishes = set()
    core._foreign_guard_release_retries = {}
    core._finished_tasks = set()

    task = TaskID.derive(core.job_id, core.driver_task_id, 1)
    attempt = AttemptID(task, 0)
    outer = ObjectID.for_task(task)
    spec = protocol.TaskSpec(
        core.job_id, task, attempt,
        protocol.FunctionKey(core.job_id, __name__, "producer", "v1"),
        (), 1, ResourceVector(), core.worker_id, max_retries=max_retries,
    )
    pending = _PendingTask(outer, spec)
    core.owner_table.register_task_outputs(spec)
    core._recovery.register_task(spec, max_retries=max_retries)
    core._objects[outer] = _ObjectWaiter(threading.Event())
    node = NodeID.random()
    death = protocol.NodeDeathRecord(
        "stored-takeover-death", node, 4444, 3, 5, -9,
        protocol.NodeDeathReason.PROCESS_EXIT, "publishing node exited",
    )
    publication = StoredPublicationID(
        LeaseID.random(), task, attempt, outer
    )
    return core, pending, publication, death


def test_absent_candidate_is_the_only_takeover_state_that_starts_generic_retry(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    core, pending, publication, death = _core()
    calls = []

    def rpc(_address, _handler, request):
        return protocol.GetStoredNodeLossReply(
            request, protocol.StoredNodeLossDisposition.ABSENT
        )

    monkeypatch.setattr(core, "_rpc", rpc)
    monkeypatch.setattr(
        core, "_retry_system_failure",
        lambda candidate, _error: calls.append(candidate) or False,
    )
    obligation = _StoredNodeLossObligation(publication, death)
    core._mark_protocol_unresolved(
        pending, "stored_node_loss", obligation,
        target_node_id=death.node_id, stored_candidate=publication,
    )

    assert not core._drive_stored_node_loss(pending, obligation)
    assert calls == [pending]
    assert pending.task_id not in core._protocol_unresolved

    core, pending, publication, death = _core()
    core._dead_nodes[death.node_id] = death

def test_candidate_survives_explicit_phase_overwrite_after_prior_clear() -> None:
    core, pending, publication, death = _core()
    core._protocol_unresolved.clear()

    core._mark_protocol_unresolved(
        pending, "lease_granted", target_node_id=death.node_id,
        stored_candidate=publication,
    )
    core._mark_protocol_unresolved(
        pending, "push_send", target_node_id=death.node_id,
        stored_candidate=publication,
    )

    unresolved = core._protocol_unresolved[pending.task_id]
    assert unresolved.stored_candidate == publication
    assert unresolved.inline_candidate is not None
    assert unresolved.inline_candidate.lease_id == publication.lease_id
    assert unresolved.inline_candidate.task_id == publication.task_id
    assert unresolved.inline_candidate.attempt_id == publication.attempt_id
    assert (
        unresolved.inline_candidate.outer_object_id
        == publication.outer_object_id
    )
    core._dead_nodes[death.node_id] = death
    assert core._consume_node_death_at_lane(pending, death.node_id) is False
    takeover = core._protocol_unresolved[pending.task_id]
    assert isinstance(takeover.obligation, core_module._InlineNodeLossObligation)
    assert takeover.stored_candidate == publication
    assert takeover.inline_candidate == unresolved.inline_candidate

def test_fenced_newer_attempt_still_finishes_old_lane_once() -> None:
    core, pending, publication, death = _core(max_retries=1)
    dependency_task = TaskID.derive(
        core.job_id, core.driver_task_id, 2
    )
    dependency = ObjectID.for_task(dependency_task)
    dependency_attempt = AttemptID(dependency_task, 0)
    core.owner_table.register(
        dependency, current_attempt=dependency_attempt
    )
    hold = protocol.TaskReferenceHold(
        protocol.TaskReferenceHoldKind.SUBMITTED, core.worker_id,
        pending.task_id, pending.spec.attempt_id,
    )
    core.owner_table.add_submitted_reference(dependency, hold)
    old_pending = replace(
        pending, protected_dependencies=(dependency,), dependency_hold=hold
    )
    core._accepted_task_count = 1

    child_task = TaskID.derive(core.job_id, core.driver_task_id, 3)
    child = ObjectID.for_task(child_task)
    executor = WorkerID.random()
    transfer = PreparedContainedTransfer(
        child, executor, ("127.0.0.1", 32005),
        OwnedContainedSource(executor),
        ContainedReferenceHold(pending.object_id, executor, "pin"),
        ContainedReferenceHold(pending.object_id, core.worker_id, "pin"),
    )
    payload = b"fenced stored takeover"
    checksum = hashlib.sha256(payload).hexdigest()
    manifest = StoredPublicationManifest(
        publication, (transfer,), len(payload), checksum,
        hashlib.sha256(b"fenced manifest").hexdigest(),
    )
    header = StoredPublicationHeader(
        publication, core.job_id, executor, core.worker_id, death.node_id
    )
    descriptor = protocol.ResultDescriptor(
        pending.object_id, protocol.ResultStorage.OBJECT_STORE, len(payload),
        core.worker_id, death.node_id, checksum,
    )
    intent = StoredPublicationIntent(
        header, manifest, death.node_pid, death.registration_epoch
    )
    recovery = StoredPublicationRecoveryRecord(
        manifest, descriptor, (True,),
        StoredPublicationLeaseTerminal(
            publication, protocol.TaskReplyStatus.SUCCEEDED
        ),
        intent,
    )
    request = StoredNodeLossRequest(
        death, NodeIncarnationWitness.from_intent(intent), recovery,
        StoredNodeLossGraphWitness(
            manifest.graph_manifest(core.worker_id),
            StoredNodeLossGraphState.PREPARED,
        ),
    )
    saga = StoredNodeLossRecoveryCoordinator()
    saga.begin(request)
    effect = saga.next_effect(publication)
    assert isinstance(effect, RetireStoredOwnerEffect)
    owner_effect = effect

    next_attempt = pending.spec.attempt_id.next()
    assert core.owner_table.advance_task_outputs(
        pending.execution, next_attempt
    )
    retirement = core.owner_table.retire_stored_owner(effect)
    assert retirement.disposition is StoredOwnerRetirementDisposition.FENCED
    saga.acknowledge(effect, retirement)
    while (effect := saga.next_effect(publication)) is not None:
        if isinstance(effect, AbortStoredGraphEffect):
            receipt = ContainedGraphManifestReceipt(
                effect.manifest, ContainedGraphTransactionState.ABORTED,
                ContainedGraphManifestDisposition.APPLIED,
            )
        elif isinstance(effect, ReleaseStoredHoldEffect):
            receipt = StoredHoldResolutionReceipt(
                effect, StoredHoldResolutionDisposition.RELEASED
            )
        else:
            assert isinstance(effect, ProveStoredReplicaGoneEffect)
            receipt = StoredReplicaResolutionReceipt(
                effect,
                StoredReplicaResolutionDisposition.DEAD_NODE_REPLICA_GONE,
            )
        saga.acknowledge(effect, receipt)
    resolution = saga.resolution(publication)
    assert resolution is not None
    obligation = _StoredNodeLossObligation(
        publication, death, True, owner_effect,
        retirement, resolution
    )
    new_publication = StoredPublicationID(
        LeaseID.random(), pending.task_id, next_attempt, pending.object_id
    )
    core._stored_node_loss_objects[pending.object_id] = new_publication
    core._mark_protocol_unresolved(
        old_pending, "stored_node_loss", obligation,
        target_node_id=death.node_id, stored_candidate=publication,
    )

    assert core._complete_stored_node_loss(
        old_pending, obligation, resolution
    )
    assert core._finish_pending_task(old_pending)
    assert core._accepted_task_count == 0
    assert hold not in core.owner_table.snapshot(dependency).submitted_tokens
    current = core.owner_table.snapshot(pending.object_id)
    assert current.current_attempt == next_attempt
    assert current.state is ObjectState.PENDING
    assert core._stored_node_loss_objects[pending.object_id] == new_publication

    assert core._finish_pending_task(old_pending)
    assert core._accepted_task_count == 0
    current = core.owner_table.snapshot(pending.object_id)
    assert current.current_attempt == next_attempt
    assert current.state is ObjectState.PENDING
    assert core._stored_node_loss_objects[pending.object_id] == new_publication


def test_owner_retirement_post_commit_exception_replays_exact_tombstone(
    monkeypatch: pytest.MonkeyPatch,

    core, pending, publication, death = _core()

        ),
    )
    saga = StoredNodeLossRecoveryCoordinator()
    owner_snapshot = saga.begin(request)
    owner_effect = owner_snapshot.next_effect
    assert isinstance(owner_effect, RetireStoredOwnerEffect)
    pull = protocol.GetStoredNodeLoss(
        publication, core.worker_id, death
    )
    reply = protocol.GetStoredNodeLossReply(
        pull, protocol.StoredNodeLossDisposition.OWNER_EFFECT,
        snapshot=owner_snapshot, owner_effect=owner_effect,
    )
    acknowledgements = []

    def rpc(_address, handler, wire):
        if handler == "get_stored_node_loss":
            assert wire == pull
            return reply
        assert handler == "acknowledge_stored_owner_retirement"
        acknowledgements.append(wire)
        acknowledged = saga.acknowledge(wire.effect, wire.receipt)
        return protocol.AcknowledgeStoredOwnerRetirementReply(
            wire, snapshot=acknowledged
        )

    monkeypatch.setattr(core, "_rpc", rpc)
    real_retire = core.owner_table.retire_stored_owner
    calls = 0

    def commit_then_raise(effect):
        nonlocal calls
        calls += 1
        receipt = real_retire(effect)
        if calls == 1:
            raise RuntimeError("injected post-commit owner failure")
        return receipt

    monkeypatch.setattr(
        core.owner_table, "retire_stored_owner", commit_then_raise
    )
    obligation = _StoredNodeLossObligation(publication, death, True)
    core._stored_node_loss_drivers[publication] = obligation
    core._stored_node_loss_objects[pending.object_id] = publication

    assert not core._drive_stored_node_loss(pending, obligation)
    first_scheduled = core._submissions.get_nowait()
    assert first_scheduled.event.obligation.owner_effect is None
    assert acknowledgements == []
    first_receipt = real_retire(owner_effect)

    assert not core._drive_stored_node_loss(
        pending, first_scheduled.event.obligation
    )
    assert calls == 2
    assert len(acknowledgements) == 1
    assert acknowledgements[0].receipt == first_receipt
